def rawA (k : Fin 3) : Matrix (Fin 3) (Fin 3) ℤ :=
  ![Pi.single (k.succAbove 0) 1,Pi.single (k.succAbove 1) 1,fun _ => 1]

def rawC (k : Fin 3) : Matrix (Fin 3) (Fin 3) ℤ := (rawA k).adjugate.transpose

def rawG (k2 k3 : Fin 3) (Q : ℝ) (c2 c3 : Fin 3 → ℝ) : AddSubgroup IntegerTriple :=
  mixedCongruence3 (rawA k2) (rawA k3)
    (fun i => Nat.ceil (-c2 i*Real.log Q/Real.log 2))
    (fun i => Nat.ceil (-c3 i*Real.log Q/Real.log 3))

def rawCG (k2 k3 : Fin 3) (Q : ℝ) (c2 c3 : Fin 3 → ℝ) : AddSubgroup IntegerTriple :=
  mixedCongruence3 (rawC k2) (rawC k3)
    (fun i => Nat.ceil (-c2 i*Real.log Q/Real.log 2))
    (fun i => Nat.ceil (-c3 i*Real.log Q/Real.log 3))

def rawReal (k : Fin 3) : RealTriple ≃ₗ[ℝ] RealTriple :=
  LinearEquiv.ofBijective (Matrix.toLin' ((rawA k).map (Int.castRingHom ℝ))) (by
    have h : IsUnit ((rawA k).map (Int.castRingHom ℝ)) := by
      change IsUnit (retainedRealMatrix3 k)
      apply (Matrix.isUnit_iff_isUnit_det _).mpr
      rw [retainedRealMatrix3_det]
      split_ifs <;> norm_num
    exact ⟨Matrix.mulVec_injective_iff_isUnit.mpr h,Matrix.mulVec_surjective_iff_isUnit.mpr h⟩)

def rawCReal (k : Fin 3) : RealTriple ≃ₗ[ℝ] RealTriple :=
  LinearEquiv.ofBijective (Matrix.toLin' ((rawC k).map (Int.castRingHom ℝ))) (by
    have h : IsUnit ((rawC k).map (Int.castRingHom ℝ)) := by
      change IsUnit (cofactorRealMatrix3 k)
      apply (Matrix.isUnit_iff_isUnit_det _).mpr
      rw [cofactorRealMatrix3_det]
      exact isUnit_one
    exact ⟨Matrix.mulVec_injective_iff_isUnit.mpr h,Matrix.mulVec_surjective_iff_isUnit.mpr h⟩)

def rawE (k : Fin 3) (Q : ℝ) (c : Fin 3 → ℝ) (hQ : 1 < Q) : RealTriple ≃ₗ[ℝ] RealTriple :=
  archCoordinates3 (rawReal k) (fun i => Q^c i) (fun i => Real.rpow_pos_of_pos (lt_trans zero_lt_one hQ) _)

def rawCE (k : Fin 3) (Q : ℝ) (c : Fin 3 → ℝ) (hQ : 1 < Q) : RealTriple ≃ₗ[ℝ] RealTriple :=
  archCoordinates3 (rawCReal k) (fun i => Q^c i) (fun i => Real.rpow_pos_of_pos (lt_trans zero_lt_one hQ) _)

def rawMin (B : Basis (Fin 3) ℝ RealTriple) (e : RealTriple ≃ₗ[ℝ] RealTriple) (j : Fin 3) : ℝ :=
  sInf {r : ℝ | ∃ u : Fin (j.val+1) → RealTriple, LinearIndependent ℝ u ∧
    (∀ i, u i ∈ lattice3 B) ∧ ∀ i, coordinateGauge3 e (u i) ≤ r}

def rawPair (c : Fin 3 → ℝ) (i : Fin 3) : ℝ := (c (i.succAbove 0)+c (i.succAbove 1))/2

def rawProducts (l : Fin 3 → ℝ) : Fin 3 → ℝ := ![l 0*l 1,l 0*l 2,l 1*l 2]

def rawWedge {R : Type*} [CommRing R] (x y : Fin 3 → R) : Fin 3 → R :=
  ![x 1*y 2-x 2*y 1,x 2*y 0-x 0*y 2,x 0*y 1-x 1*y 0]

def rawFrame {R : Type*} [CommRing R] (u : Fin 3 → Fin 3 → R) : Fin 3 → Fin 3 → R :=
  ![rawWedge (u 0) (u 1),rawWedge (u 0) (u 2),rawWedge (u 1) (u 2)]

    (hQ : 1 < Q) (hd : 0 < delta)
    (c0 c2 c3 : Fin 3 → ℝ)
    (hc0 : ∀ i, -1 ≤ c0 i ∧ c0 i ≤ 1)
    (hc2 : ∀ i, -1 ≤ c2 i ∧ c2 i ≤ 0) (hc3 : ∀ i, -1 ≤ c3 i ∧ c3 i ≤ 0)
    (hb : ∑ i, (c0 i+c2 i+c3 i) = 0)
    (B : Basis (Fin 3) ℝ RealTriple) (u : Fin 3 → rawG k2 k3 Q c2 c3)
    (hB : lattice3 B = (rawG k2 k3 Q c2 c3).toIntSubmodule.map integerTripleReal3)
    (hu : LinearIndependent ℝ (fun j => integerTripleReal3 (u j : IntegerTriple)))
    (hatt : ∀ j, coordinateGauge3 (rawE k0 Q c0 hQ) (integerTripleReal3 (u j : IntegerTriple)) =
      rawMin B (rawE k0 Q c0 hQ) j)
    (hgap : rawMin B (rawE k0 Q c0 hQ) 0 <
      rawMin B (rawE k0 Q c0 hQ) 1*Q^(-delta))
    (D : Basis (Fin 3) ℝ RealTriple)
    (v : Fin 3 → rawCG k2 k3 (Q^2) (rawPair c2) (rawPair c3))
    (hD : lattice3 D = (rawCG k2 k3 (Q^2) (rawPair c2) (rawPair c3)).toIntSubmodule.map integerTripleReal3)
    (hv : LinearIndependent ℝ (fun j => integerTripleReal3 (v j : IntegerTriple)))
    (hav : ∀ j, coordinateGauge3
      (rawCE k0 (Q^2) (rawPair c0) (pairWeights3_Q_squared Q hQ))
      (integerTripleReal3 (v j : IntegerTriple)) = rawMin D
      (rawCE k0 (Q^2) (rawPair c0) (pairWeights3_Q_squared Q hQ)) j)
    (hcut : (1119744 : ℝ)^2 < Q^(delta/2)) :
    rawMin D (rawCE k0 (Q^2) (rawPair c0) (pairWeights3_Q_squared Q hQ)) 1 <
      rawMin D (rawCE k0 (Q^2) (rawPair c0) (pairWeights3_Q_squared Q hQ)) 2*(Q^2)^(-delta/4) ∧
    Submodule.span ℚ {fun i => ((u 0 : IntegerTriple) i : ℚ)} =
      Submodule.span ℚ {dualFrameVector3 (fun j => (v j : IntegerTriple)) 2} := by
  exact JSP000334.source_exterior_square_gap_plane3 k0 k2 k3 Q delta hQ hd c0 c2 c3 hc0 hc2 hc3 hb B u hB hu hatt hgap D v hD hv hav hcut
