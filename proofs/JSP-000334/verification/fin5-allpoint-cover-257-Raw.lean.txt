def Gauge (e : (Fin 5 → ℝ) ≃ₗ[ℝ] (Fin 5 → ℝ)) (x : Fin 5 → ℝ) : ℝ := ‖e x‖
def LatticeSpan (b : Basis (Fin 5) ℝ (Fin 5 → ℝ)) : Submodule ℤ (Fin 5 → ℝ) := Submodule.span ℤ (Set.range b)
def Radius (b : Basis (Fin 5) ℝ (Fin 5 → ℝ)) (e : (Fin 5 → ℝ) ≃ₗ[ℝ] (Fin 5 → ℝ)) (k : Fin 5) : ℝ :=
  sInf {r : ℝ | ∃ u : Fin (k.val+1) → Fin 5 → ℝ,LinearIndependent ℝ u ∧ (∀ i,u i∈LatticeSpan b) ∧ ∀ i,Gauge e (u i)≤r}
def Flag (u : Fin 5 → Fin 5 → ℝ) (k : Fin 5) : Submodule ℝ (Fin 5 → ℝ) :=
  Submodule.span ℝ (Set.range (fun i : Fin k.val => u (Fin.castLE (by omega) i)))
def RationalFlag (j : Fin 4) (u : Fin 5 → Fin 5 → ℤ) : Submodule ℚ (Fin 5 → ℚ) :=
  Submodule.span ℚ (Set.range (fun i : Fin (j.val+1) => fun k => (u (Fin.castLE (by omega) i) k : ℚ)))
def Row (A : Matrix (Fin 5) (Fin 5) ℚ) (i : Fin 5) : (Fin 5 → ℤ) →+ ℚ where
  toFun x := (A *ᵥ (fun j => (x j : ℚ))) i
  map_zero' := by change (A *ᵥ (0 : Fin 5 → ℚ)) i=0; rw [Matrix.mulVec_zero]; rfl
  map_add' x y := by
    simp only [Pi.add_apply,Int.cast_add]
    change (A *ᵥ ((fun j => (x j : ℚ))+(fun j => (y j : ℚ)))) i=_
    rw [Matrix.mulVec_add];rfl
def Local (p : ℕ) [Fact p.Prime] (A : Matrix (Fin 5) (Fin 5) ℚ) (e : Fin 5 → ℕ) : AddSubgroup (Fin 5 → ℤ) :=
  ⨅ i,(normBall p (e i)).comap (Row A i)
def Group (A2 A3 : Matrix (Fin 5) (Fin 5) ℚ) (Q : ℝ) (c2 c3 : Fin 5 → ℝ) : AddSubgroup (Fin 5 → ℤ) :=
  Local 2 A2 (fun i => Nat.ceil (-c2 i*Real.log Q/Real.log 2)) ⊓ Local 3 A3 (fun i => Nat.ceil (-c3 i*Real.log Q/Real.log 3))
def Minimum (A2 A3 : Matrix (Fin 5) (Fin 5) ℚ) (L : Matrix (Fin 5) (Fin 5) ℝ) (hL : IsUnit L)
    (Q : ℝ) (hQ : 1<Q) (c0 c2 c3 : Fin 5 → ℝ) (j : Fin 5) : ℝ :=
  sInf {r : ℝ | ∃ u : Fin (j.val+1) → Fin 5 → ℝ,LinearIndependent ℝ u ∧
    (∀ i,u i∈integerRealImage (Group A2 A3 Q c2 c3)) ∧
    ∀ i,Gauge (originalCoordinates5 L hL Q hQ c0) (u i)≤r}
def Functional (f : Module.Dual ℚ (Fin 5 → ℚ)) : Module.Dual ℝ (Fin 5 → ℝ) :=
  ∑ i : Fin 5,(f (Pi.single i 1) : ℝ) • LinearMap.proj i
