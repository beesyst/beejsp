theorem absolute_ratio10 (lam : Fin 10 → ℝ) (hp : ∀ i, 0 < lam i)
    (hm : Monotone lam) (c Q delta : ℝ) (hc : 0 < c) (hQ : 1 < Q)
    (hprod : c ≤ ∏ i, lam i) (hsmall : lam 8 < Q^(-delta)) :
    lam 8 / lam 9 < Q^(-10*delta)/c := by
  have hnine : (∏ j : Fin 9, lam j.castSucc) ≤ (lam 8)^9 := by
    calc
      _ ≤ ∏ _j : Fin 9, lam 8 := Finset.prod_le_prod
        (fun j _ => (hp _).le) (fun j _ => hm (by change j.val ≤ 8; omega))
      _ = _ := by simp
  have htotal : c ≤ (lam 8)^9 * lam 9 := by
    apply hprod.trans
    rw [Fin.prod_univ_castSucc]
    exact mul_le_mul_of_nonneg_right hnine (hp 9).le
  have hmul : c * lam 8 ≤ (lam 8)^10 * lam 9 := by
    nlinarith [mul_le_mul_of_nonneg_right htotal (hp 8).le]
  have hpow : (lam 8)^10 < Q^(-10*delta) := by
    have hh := pow_lt_pow_left₀ hsmall (hp 8).le (by decide : (10 : ℕ) ≠ 0)
    convert hh using 1
    rw [← Real.rpow_natCast, ← Real.rpow_mul (zero_lt_one.trans hQ).le]
    congr 1
    ring
  apply (div_lt_div_iff₀ (hp 9) hc).mpr
  simpa only [mul_comm c] using hmul.trans_lt (mul_lt_mul_of_pos_right hpow (hp 9))

theorem absolute_gap_cutoff10 (c delta : ℝ) (hc : 0 < c) (hd : 0 < delta) :
    ∃ T : ℝ, 1 < T ∧ ∀ (Q : ℝ), T < Q →
      ∀ (lam : Fin 10 → ℝ), (∀ i, 0 < lam i) → Monotone lam →
      c ≤ (∏ i, lam i) → lam 8 < Q^(-delta) → lam 8 < lam 9*Q^(-delta) := by
  obtain ⟨T,hT,hcut⟩ := compound_gap_cutoff42 (1/c) (18*delta) (by positivity)
  refine ⟨T,hT,?_⟩
  intro Q hTQ lam hp hm hprod hsmall
  have hQ : 1 < Q := hT.trans hTQ
  have hq : 0 < Q := zero_lt_one.trans hQ
  have hcut' : 1/c < Q^(9*delta) := by
    convert hcut Q hTQ using 1; congr 1; ring
  have hr := absolute_ratio10 lam hp hm c Q delta hc hQ hprod hsmall
  have hle : Q^(-10*delta)/c ≤ Q^(-delta) := by
    calc
      _ = (1/c)*Q^(-10*delta) := by ring
      _ ≤ Q^(9*delta)*Q^(-10*delta) :=
        mul_le_mul_of_nonneg_right hcut'.le (Real.rpow_pos_of_pos hq _).le
      _ = _ := by rw [← Real.rpow_add hq]; congr 1; ring
  have hh := (div_lt_iff₀ (hp 9)).mp (hr.trans_le hle)
  simpa only [mul_comm] using hh

def normalizedWidth10 (Q A : ℝ) : ℝ := Real.log A/Real.log Q

def normalizedFinite10 (p : ℕ) (Q : ℝ) (f : ℤ) : ℝ :=
  (f : ℝ)*Real.log (p : ℝ)/Real.log Q

theorem normalized_width_exact10 (Q A : ℝ) (hQ : 1 < Q) (hA : 0 < A) :
    Q^(normalizedWidth10 Q A) = A := by
  rw [normalizedWidth10,Real.rpow_def_of_pos (zero_lt_one.trans hQ)]
  have he : Real.log Q*(Real.log A/Real.log Q) = Real.log A := by
    field_simp [(Real.log_pos hQ).ne']
  rw [he,Real.exp_log hA]

theorem normalized_finite_exact10 (p : ℕ) (hp : 1 < p) (Q : ℝ) (hQ : 1 < Q) (f : ℤ) :
    Q^(normalizedFinite10 p Q f) = (p : ℝ)^f := by
  have hpR : (1 : ℝ) < p := by exact_mod_cast hp
  rw [normalizedFinite10,Real.rpow_def_of_pos (zero_lt_one.trans hQ),
    ← Real.rpow_intCast (p : ℝ) f,
    Real.rpow_def_of_pos (zero_lt_one.trans hpR)]
  congr 1
  field_simp [(Real.log_pos hQ).ne']
  ring

theorem normalized_threshold_exact10 (p : ℕ) (hp : 1 < p) (Q : ℝ) (hQ : 1 < Q)
    (f : ℤ) :
    Nat.ceil (JSP000334.qThreshold p Q (normalizedFinite10 p Q f)) = (-f).toNat := by
  have hpR : (1 : ℝ) < p := by exact_mod_cast hp
  have he : JSP000334.qThreshold p Q (normalizedFinite10 p Q f) = ((-f : ℤ) : ℝ) := by
    unfold JSP000334.qThreshold normalizedFinite10
    push_cast
    field_simp [(Real.log_pos hQ).ne',(Real.log_pos hpR).ne']
    ring
  rw [he,Nat.ceil_intCast]

theorem normalized_balance10 (Q : ℝ) (A : Fin 10 → ℝ)
    (hA : ∀ i, 0 < A i) (f2 f3 : Fin 10 → ℤ)
    (hb : (∏ i, A i*(2 : ℝ)^(f2 i)*(3 : ℝ)^(f3 i)) = 1) :
    (∑ i, (normalizedWidth10 Q (A i)+normalizedFinite10 2 Q (f2 i)+
      normalizedFinite10 3 Q (f3 i))) = 0 := by

    mul_pos (mul_pos (hA i) (by positivity)) (by positivity)

  rw [Real.log_prod _ _ (fun i _ => (hpos i).ne'),Real.log_one] at h
  have he (i : Fin 10) : Real.log (A i*(2 : ℝ)^(f2 i)*(3 : ℝ)^(f3 i)) =
      Real.log (A i)+(f2 i : ℝ)*Real.log 2+(f3 i : ℝ)*Real.log 3 := by
    rw [Real.log_mul (mul_pos (hA i) (by positivity)).ne' (by positivity : (3 : ℝ)^f3 i ≠ 0),
      Real.log_mul (hA i).ne' (by positivity : (2 : ℝ)^f2 i ≠ 0),Real.log_zpow,Real.log_zpow]
  simp_rw [he] at h
  simp only [normalizedWidth10,normalizedFinite10,← add_div,← Finset.sum_div,Nat.cast_ofNat,h,zero_div]

theorem complement_product10 (a b d : Fin 10 → ℝ)
    (ha : ∀ i, 0 < a i) (hb : ∀ i, 0 < b i) (hd : ∀ i, 0 < d i)
    (hbal : (∏ i, a i)*(∏ i, b i)*(∏ i, d i) = 1) (s : MixedIndex10) :
    (∏ k : Fin 9, a ((s 0).succAbove k)) *
      (∏ k : Fin 9, b ((s 1).succAbove k)) *
      (∏ k : Fin 9, d ((s 2).succAbove k)) = (a (s 0)*b (s 1)*d (s 2))⁻¹ := by
  rw [inv_eq_one_div,eq_div_iff (mul_pos (mul_pos (ha _) (hb _)) (hd _)).ne']
  rw [Fin.prod_univ_succAbove a (s 0),Fin.prod_univ_succAbove b (s 1),
    Fin.prod_univ_succAbove d (s 2)] at hbal
  nlinarith only [hbal]

theorem absolute_selected_budget10 (Q delta N : ℝ) (hQ : 1 < Q)
    (hsmall : N ≤ Q^(-9*delta))
    (a b d : Fin 10 → ℝ) (ha : ∀ i, 0 < a i) (hb : ∀ i, 0 < b i) (hd : ∀ i, 0 < d i)
    (hbal : (∏ i, a i)*(∏ i, b i)*(∏ i, d i) = 1)
    (s : MixedIndex10) (hsel : 1 ≤ a (s 0)*b (s 1)*d (s 2)*Q^(delta/2)) :
    (362880*N*(∏ k : Fin 9, a ((s 0).succAbove k)))*
      (∏ k : Fin 9, b ((s 1).succAbove k))*
      (∏ k : Fin 9, d ((s 2).succAbove k)) ≤ 362880*Q^(-17*delta/2) := by
  have hq : 0 < Q := zero_lt_one.trans hQ
  have hpos := mul_pos (mul_pos (ha (s 0)) (hb (s 1))) (hd (s 2))
  have hinv : (a (s 0)*b (s 1)*d (s 2))⁻¹ ≤ Q^(delta/2) := by
    rw [inv_eq_one_div]

      rw [← complement_product10 a b d ha hb hd hbal s]

      mul_le_mul (mul_le_mul_of_nonneg_left hsmall (by norm_num)) hinv
        (inv_nonneg.mpr hpos.le) (by positivity)
    _ = _ := by rw [mul_assoc,← Real.rpow_add hq]; congr 2; ring

end
