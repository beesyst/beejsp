def Two : ℚˣ := Units.mk0 2 (by norm_num)

def Group : Subgroup ℚˣ := Subgroup.closure ({-1,Two,Three} : Set ℚˣ)
def Primitive (z : Fin 3 → ℤ) : Prop := ∃ c : Fin 3 → ℤ,∑ i,c i*z i=1
def Height (z : Fin 3 → ℤ) : ℚ := max |(z 0 : ℚ)| (max |(z 1 : ℚ)| |(z 2 : ℚ)|)
def Product (q : ℚ) : ℚ := |q| *padicNorm 2 q*padicNorm 3 q
def Local (v : Fin 3) (q : ℚ) : ℚ := ![|q|,padicNorm 2 q,padicNorm 3 q] v
def Forms (i : Fin 3) : Fin 3 → ((Fin 3 → ℚ) →ₗ[ℚ] ℚ) :=
  ![LinearMap.proj (i.succAbove 0),LinearMap.proj (i.succAbove 1),
    LinearMap.proj (R:=ℚ) (φ:=fun _ : Fin 3 => ℚ) 0+LinearMap.proj (R:=ℚ) (φ:=fun _ : Fin 3 => ℚ) 1+LinearMap.proj (R:=ℚ) (φ:=fun _ : Fin 3 => ℚ) 2]
def Retained (choice : Fin 3 → Fin 3) (y : Fin 3 → ℚ) : ℚ :=
  ∏ v : Fin 3,∏ j : Fin 3,Local v (Forms (choice v) j y)
def Raw_weightedIntegerDiagonal3 (t : Fin 3 → ℤ) : Matrix (Fin 3) (Fin 3) ℚ :=
  Matrix.diagonal (fun i => (t i : ℚ))

def Raw_weightedBalancedValues3 (H K : ℝ) (a : Fin 3 → Fin 3 → ℝ) (tau i : Fin 3) : ℝ :=
  if tau=0 then a tau i*(H/K)^((1 : ℝ)/3) else a tau i

def Raw_weightedDiagonalK3 (t : Fin 3 → ℤ) (L A : ℚ) (g : ℕ) : ℚ :=
  Product ((t 0 : ℚ)/g)*Product ((t 1 : ℚ)/g)*
    Product ((t 2 : ℚ)/g)*Product (L*A/g)

def Raw_weightedProductConstant3 (k : Fin 3 → ℚ) (a : ℚ) : ℚ :=
  Product (k 0)*Product (k 1)*Product (k 2)*Product a

def Raw_weightedProductBound3 (k : Fin 3 → ℚ) (a : ℚ) : ℚ :=
  max 1 (max (Product (k 0))⁻¹
    (max (Product (k 1))⁻¹ (max (Product (k 2))⁻¹ (Product a)⁻¹)))

noncomputable def Raw_weightedProductCutoff3 (k : Fin 3 → ℚ) (a : ℚ) : ℝ :=
  max 3 (max (3*(Raw_weightedProductBound3 k a : ℝ))
    (max ((Raw_weightedProductConstant3 k a : ℝ)⁻¹) ((Raw_weightedProductConstant3 k a : ℝ)^4)))

def Raw_weightedFixedProductTriples3 (k : Fin 3 → ℚ) (a : ℚ) (y : IntegerTriple) : Prop :=
  Primitive y ∧ (∀ i,Product (y i : ℚ)=Product (k i)) ∧
    Product ((y 0 : ℚ)+y 1+y 2)=Product a
