theorem original_dual_pairing10 (u : Fin 10 → Fin 10 → ℤ)
    (hu : LinearIndependent ℚ (fun j i => (u j i : ℚ))) (j : Fin 10) :
    (∑ i, (u j i : ℚ)*lastDual10 (fun j i => (u j i : ℚ)) hu i) =
      if j = (9 : Fin 10) then 1 else 0 := by
  have hf := functional_coordinates_dual10
    (rationalFrameDual42 (fun j i => (u j i : ℚ)) hu (9 : Fin 10))
    (fun i => (u j i : ℚ))
  rw [rationalFrameDual42_pairing] at hf
  simpa only [lastDual10,eq_comm] using hf.symm

def sourceDualCoordinate10 (u : Fin 10 → Fin 10 → ℤ)
    (hu : LinearIndependent ℚ (fun j i => (u j i : ℚ))) (k : Fin 10) : ℚ :=
  rationalFrameDual42 (fun j i => (u j i : ℚ)) hu (9 : Fin 10) (Pi.single k 1)

def sourceDualProduct10 (A2 A3 : Matrix (Fin 10) (Fin 10) ℚ)
    (L : Matrix (Fin 10) (Fin 10) ℝ) (u : Fin 10 → Fin 10 → ℤ)
    (hu : LinearIndependent ℚ (fun j i => (u j i : ℚ))) (s : Fin 3 → Fin 10) : ℝ :=
  |(L⁻¹.transpose *ᵥ (fun k => (lastDual10 (fun j i => (u j i : ℚ)) hu k : ℝ))) (s 0)| *
    ‖((A2.map (algebraMap ℚ ℚ_[2]))⁻¹.transpose *ᵥ
      (fun k => (lastDual10 (fun j i => (u j i : ℚ)) hu k : ℚ_[2]))) (s 1)‖ *
    ‖((A3.map (algebraMap ℚ ℚ_[3]))⁻¹.transpose *ᵥ
      (fun k => (lastDual10 (fun j i => (u j i : ℚ)) hu k : ℚ_[3]))) (s 2)‖

theorem exact_source_product_nonzero10 (A2 A3 : Matrix (Fin 10) (Fin 10) ℚ)
    (L : Matrix (Fin 10) (Fin 10) ℝ) (u : Fin 10 → Fin 10 → ℤ)
    (hu : LinearIndependent ℚ (fun j i => (u j i : ℚ))) (s : Fin 3 → Fin 10) :
    sourceDualProduct10 A2 A3 L u hu s ≠ 0 ↔
      mixedProduct10 A2 A3 L (integerLastDual10 u) s ≠ 0 := by
  have hp : sourceDualProduct10 A2 A3 L u hu s ≠ 0 ↔
      (L⁻¹.transpose *ᵥ (fun k => (lastDual10 (fun j i => (u j i : ℚ)) hu k : ℝ))) (s 0) ≠ 0 ∧
      ((A2.map (algebraMap ℚ ℚ_[2]))⁻¹.transpose *ᵥ
        (fun k => (lastDual10 (fun j i => (u j i : ℚ)) hu k : ℚ_[2]))) (s 1) ≠ 0 ∧
      ((A3.map (algebraMap ℚ ℚ_[3]))⁻¹.transpose *ᵥ
        (fun k => (lastDual10 (fun j i => (u j i : ℚ)) hu k : ℚ_[3]))) (s 2) ≠ 0 := by
    simp only [sourceDualProduct10,mul_ne_zero_iff,abs_ne_zero,norm_ne_zero_iff,and_assoc]
  constructor
  · intro h
    obtain ⟨h0,h2,h3⟩ := hp.mp h
    exact (mixedProduct10_ne_zero_iff A2 A3 L (integerLastDual10 u) s).mpr
      ⟨(original_real_nonzero10 L u hu (s 0)).mpr h0,
       (original_p2_nonzero10 A2 u hu (s 1)).mpr h2,
       (original_p3_nonzero10 A3 u hu (s 2)).mpr h3⟩
  · intro h
    obtain ⟨h0,h2,h3⟩ := (mixedProduct10_ne_zero_iff A2 A3 L (integerLastDual10 u) s).mp h
    exact hp.mpr ⟨(original_real_nonzero10 L u hu (s 0)).mp h0,
      (original_p2_nonzero10 A2 u hu (s 1)).mp h2,
      (original_p3_nonzero10 A3 u hu (s 2)).mp h3⟩

end

theorem source_data_to_rational10
    {A2 A3 : Matrix (Fin 10) (Fin 10) ℚ} {L : Matrix (Fin 10) (Fin 10) ℝ}
    {hL : IsUnit L} {Q : ℝ} {hQ : 1 < Q} {delta : ℝ} {S0 : Finset MixedIndex10}
    (D : OriginalData10 A2 A3 L hL Q hQ delta S0) (s : MixedIndex10)
    (hn : sourceDualProduct10 A2 A3 L D.u (integer_frame_rational_independent10 D.u D.independent) s ≠ 0) :
    dataRationalDualProduct10 D s ≠ 0 := by
  apply (data_rational_product_nonzero_iff D s).mpr
  exact (exact_source_product_nonzero10 A2 A3 L D.u
    (integer_frame_rational_independent10 D.u D.independent) s).mp hn

theorem source_counterexample_samples_contract10
    (A2 A3 : Matrix (Fin 10) (Fin 10) ℚ) (L : Matrix (Fin 10) (Fin 10) ℝ)
    (h2 : A2.det = 1) (h3 : A3.det = 1) (hLd : L.det = 1) (hL : IsUnit L)
    (hAlg : ∀ i j, IsAlgebraic ℚ (L i j)) (delta : ℝ) (hd : 0 < delta) (S0 : Finset MixedIndex10) :
    ∃ K : IntermediateField ℚ ℝ, FiniteDimensional ℚ K ∧
      ∃ D0 E0 alpha C_upper T : ℝ, 1 ≤ D0 ∧ 1 ≤ E0 ∧ 0 < alpha ∧ 0 < C_upper ∧ 1 < T ∧
      ∀ (_hfail : ∀ B : ℝ, ∃ Q : ℝ, ∃ hQ : 1 < Q, B < Q ∧
        ∃ D : OriginalData10 A2 A3 L hL Q hQ delta S0,
          ∃ s ∈ S0, sourceDualProduct10 A2 A3 L D.u (integer_frame_rational_independent10 D.u D.independent) s ≠ 0),
      ∀ B1 : ℝ, ∃ U : Set ℝ, ¬ BddAbove U ∧ ∃ hQU : ∀ q ∈ U, 1 < q,
        ∃ D : (q : U) → OriginalData10 A2 A3 L hL q.val (hQU q.val q.property) delta S0,
          (∀ q : U, ∃ s ∈ S0, dataRationalDualProduct10 (D q) s ≠ 0) ∧
      ∃ V : Set ℝ, ∃ hVU : V ⊆ U, ¬ BddAbove V ∧
        (∀ q : V, max T B1 < q.val) ∧
      ∃ e : ℝ, 0 < e ∧ e < 1 ∧ ∃ c : Fin 3 → Fin 10 → ℝ,
        (∀ tau i, |c tau i| ≤ 1) ∧ (∑ tau, ∑ i, c tau i) = 0 ∧
        (∀ i, c 1 i ≤ 0 ∧ c 2 i ≤ 0) ∧
      ∃ eta : ℝ, 0 < eta ∧ 4000*eta < e ∧ ∃ m : ℕ, ∃ hm : 0 < m,
        4*(eta^2)⁻¹*Real.log (60*(Module.finrank ℚ K : ℝ)) ≤ (m : ℝ) ∧
        (∀ rho : ℝ, 0 < rho → ∀ B : ℝ,
          Nonempty (OriginalAuxiliarySample10 A2 A3 L (fun i => 5*c 1 i) (fun i => 5*c 2 i)
            V (fun q => (D ⟨q.val,hVU q.property⟩).u) c e eta D0 E0 B rho m hm)) ∧
        ∀ (rho B : ℝ)
          (S : OriginalAuxiliarySample10 A2 A3 L (fun i => 5*c 1 i) (fun i => 5*c 2 i)
            V (fun q => (D ⟨q.val,hVU q.property⟩).u) c e eta D0 E0 B rho m hm) (h : Fin m),
          ((S.data.t h).val^5)^alpha ≤ (normalHeight10 (S.data.M h) : ℝ) ∧
          (normalHeight10 (S.data.M h) : ℝ) ≤ C_upper*((S.data.t h).val^5)^9 ∧
          ∃ s ∈ S0, mixedProduct10 A2 A3 L (S.data.M h) s ≠ 0 ∧
            mixedProduct10 A2 A3 L (S.data.M h) s ≤ 362880*((S.data.t h).val^5)^(-17*delta/10) := by
  obtain ⟨K,hK,D0,E0,alpha,C_upper,T,hD,hE,ha,hC,hT,hinputs⟩ :=
    original_rational_counterexample_samples10 A2 A3 L h2 h3 hLd hL hAlg delta hd S0
  refine ⟨K,hK,D0,E0,alpha,C_upper,T,hD,hE,ha,hC,hT,?_⟩
  intro hfail B1
  have hconverted : ∀ B : ℝ, ∃ Q : ℝ, ∃ hQ : 1 < Q, B < Q ∧
      ∃ D : OriginalData10 A2 A3 L hL Q hQ delta S0, ∃ s ∈ S0, dataRationalDualProduct10 D s ≠ 0 := by
    intro B
    obtain ⟨Q,hQ,hBQ,D,s,hs,hn⟩ := hfail B
    exact ⟨Q,hQ,hBQ,D,s,hs,source_data_to_rational10 D s hn⟩
  exact hinputs hconverted B1

end
