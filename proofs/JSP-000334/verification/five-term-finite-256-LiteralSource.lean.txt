theorem five_term_weighted_degenerate_pair4 (a b c d : ℚ)
    (ha : a≠0) (hb : b≠0) (hc : c≠0) (hd : d≠0)
    (hs : a+b+c+d=0) (hn : ¬Nondegenerate (![a,b,c,d] : Fin 4 → ℚ)) :
    a+b=0 ∨ a+c=0 ∨ a+d=0 := by
  classical
  unfold Nondegenerate at hn
  push_neg at hn
  obtain ⟨I,hI,hproper,hzero⟩ := hn
  fin_cases I <;> norm_num [Finset.sum_insert,Matrix.cons_val_succ,Matrix.cons_val_zero] at hzero hI
  all_goals first
    | exact False.elim (ha (by linarith))
    | exact False.elim (hb (by linarith))
    | exact False.elim (hc (by linarith))
    | exact False.elim (hd (by linarith))
    | exact Or.inl (by linarith)
    | exact Or.inr (Or.inl (by linarith))
    | exact Or.inr (Or.inr (by linarith))
    | exact False.elim (hproper (by decide))

theorem five_term_nondegenerate_coarsen4 (x : Fin 5 → ℚ) (hx : Nondegenerate x)
    (f : Fin 5 → Fin 4) (hf : Function.Surjective f) :
    Nondegenerate (fun j => ∑ i : Fin 5,if f i=j then x i else 0) := by
  classical
  intro I hI hproper
  let J : Finset (Fin 5) := Finset.univ.filter (fun i => f i∈I)
  have hJ : J.Nonempty := by
    obtain ⟨j,hj⟩ := hI
    obtain ⟨i,rfl⟩ := hf j
    exact ⟨i,by simpa [J] using hj⟩
  have hJproper : J≠Finset.univ := by
    intro he
    apply hproper
    apply Finset.eq_univ_of_forall
    intro j
    obtain ⟨i,rfl⟩ := hf j
    have hi : i∈J := by rw [he];exact Finset.mem_univ i
    simpa [J] using hi
  have heq : (∑ j∈I,∑ i : Fin 5,if f i=j then x i else 0)=∑ i∈J,x i := by
    rw [Finset.sum_comm]
    simp [J,Finset.sum_filter,Finset.sum_ite_eq']
  rw [heq]
  exact hx J hJ hJproper

theorem five_term_proper_annihilator4 (P : Submodule ℚ (Fin 4 → ℚ)) (hP : P≠⊤) :
    ∃ a0 a1 a2 a3 : ℚ,(a0≠0 ∨ a1≠0 ∨ a2≠0 ∨ a3≠0) ∧
      ∀ y∈P,a0*y 0+a1*y 1+a2*y 2+a3*y 3=0 := by
  obtain ⟨f,hf,hker⟩ := P.exists_le_ker_of_lt_top (lt_top_iff_ne_top.mpr hP)
  have heval (y : Fin 4 → ℚ) : f y=f ![1,0,0,0]*y 0+f ![0,1,0,0]*y 1+f ![0,0,1,0]*y 2+f ![0,0,0,1]*y 3 := by
    have hy : y=y 0 • ![(1 : ℚ),0,0,0]+y 1 • ![0,1,0,0]+y 2 • ![0,0,1,0]+y 3 • ![0,0,0,1] := by
      ext i
      fin_cases i <;> simp
    conv_lhs => rw [hy]
    simp only [map_add,map_smul,smul_eq_mul]
    ring
  refine ⟨f ![1,0,0,0],f ![0,1,0,0],f ![0,0,1,0],f ![0,0,0,1],?_,?_⟩
  · by_contra hn
    push_neg at hn
    apply hf
    apply LinearMap.ext
    intro y
    simpa [hn.1,hn.2.1,hn.2.2.1,hn.2.2.2] using heval y
  · intro y hy
    have hk : f y=0 := hker hy
    rw [heval] at hk
    exact hk

theorem five_term_relation_facts4 {x : Fin 5 → ℚ}
    (hx : x∈normalizedRelations fixedSUnitGroup 4) :
    x 0=1 ∧ x 0+x 1+x 2+x 3+x 4=0 ∧ (∀ i,x i≠0) ∧
      x 4=-(x 0+x 1+x 2+x 3) := by
  have hs : x 0+x 1+x 2+x 3+x 4=0 := by simpa [Fin.sum_univ_succ,add_assoc] using hx.2.2.1

  rw [← hu]
  exact Units.ne_zero _

theorem five_term_weighted_triple_projection4 (S : Set (Fin 5 → ℚ))
    (hs : S⊆normalizedRelations fixedSUnitGroup 4)
    (A B C D : ℚ) (hA : A≠0) (hB : B≠0) (hC : C≠0) (hD : D≠0)
    (i j k : Fin 5) (recover : ℚ → ℚ → ℚ → Fin 5 → ℚ)
    (he : ∀ x∈S,A+B*x i+C*x j+D*x k=0)
    (hn : ∀ x∈S,Nondegenerate (![A,B*x i,C*x j,D*x k] : Fin 4 → ℚ))
    (hr : ∀ x∈S,recover (x i) (x j) (x k)=x) : S.Finite := by
  let f : (fixedSUnitGroup × fixedSUnitGroup × fixedSUnitGroup) → Fin 5 → ℚ :=
    fun p => recover ((p.1 : ℚˣ) : ℚ) ((p.2.1 : ℚˣ) : ℚ) ((p.2.2 : ℚˣ) : ℚ)
  apply ((weighted_four_term_finite A B C D hA hB hC hD).image f).subset
  intro x hx
  obtain ⟨u,hu⟩ := (hs hx).2.1 i
  obtain ⟨v,hv⟩ := (hs hx).2.1 j
  obtain ⟨w,hw⟩ := (hs hx).2.1 k
  refine ⟨(u,v,w),?_,?_⟩
  · change A+B*((u : ℚˣ) : ℚ)+C*((v : ℚˣ) : ℚ)+D*((w : ℚˣ) : ℚ)=0 ∧ _
    rw [hu,hv,hw]
    exact ⟨he x hx,hn x hx⟩
  · change recover ((u : ℚˣ) : ℚ) ((v : ℚˣ) : ℚ) ((w : ℚˣ) : ℚ)=x
    rw [hu,hv,hw]
    exact hr x hx

theorem five_term_weighted_pair_projection4 (S : Set (Fin 5 → ℚ))
    (hs : S⊆normalizedRelations fixedSUnitGroup 4)
    (A B C : ℚ) (hA : A≠0) (hB : B≠0) (hC : C≠0)
    (i j : Fin 5) (recover : ℚ → ℚ → Fin 5 → ℚ)
    (he : ∀ x∈S,A+B*x i+C*x j=0)
    (hr : ∀ x∈S,recover (x i) (x j)=x) : S.Finite := by
  let f : (fixedSUnitGroup × fixedSUnitGroup) → Fin 5 → ℚ :=
    fun p => recover ((p.1 : ℚˣ) : ℚ) ((p.2 : ℚˣ) : ℚ)
  apply ((weighted_fixed_sunit_pair_finite A B C hA hB hC).image f).subset
  intro x hx
  obtain ⟨u,hu⟩ := (hs hx).2.1 i
  obtain ⟨v,hv⟩ := (hs hx).2.1 j
  refine ⟨(u,v),?_,?_⟩
  · change A+B*((u : ℚˣ) : ℚ)+C*((v : ℚˣ) : ℚ)=0
    rw [hu,hv]
    exact he x hx
  · change recover ((u : ℚˣ) : ℚ) ((v : ℚˣ) : ℚ)=x
    rw [hu,hv]
    exact hr x hx

theorem five_term_finite_over_pairs4 (S : Set (Fin 5 → ℚ))
    (f : (Fin 5 → ℚ) → fixedSUnitGroup × fixedSUnitGroup)
    (A B C : ℚ) (hA : A≠0) (hB : B≠0) (hC : C≠0)
    (he : ∀ x∈S,A+B*(((f x).1 : ℚˣ) : ℚ)+C*(((f x).2 : ℚˣ) : ℚ)=0)
    (hf : ∀ p∈weightedFixedSUnitPairs A B C,{x∈S | f x=p}.Finite) : S.Finite := by
  have hU : (⋃ p∈weightedFixedSUnitPairs A B C,{x∈S | f x=p}).Finite :=
    (weighted_fixed_sunit_pair_finite A B C hA hB hC).biUnion hf
  apply hU.subset
  intro x hx
  exact Set.mem_iUnion.mpr ⟨f x,Set.mem_iUnion.mpr ⟨he x hx,hx,rfl⟩⟩
