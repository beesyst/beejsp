def signedQuad246 {R : Type*} [CommRing R] (u : Fin 5 → Fin 5 → R) (j : Fin 5) : Fin 5 → R :=

theorem signedQuad_adjugate246 {R : Type*} [CommRing R]
    (u : Matrix (Fin 5) (Fin 5) R) (j i : Fin 5) :
    signedQuad246 u j i = u.adjugate i j := by
  let M : Matrix (Fin 5) (Fin 5) R :=
    Matrix.of (Fin.snoc (fun k : Fin 4 => u (j.succAbove k)) (0 : Fin 5 → R))
  have he : M.submatrix (Fin.last 4).succAbove i.succAbove =
      u.submatrix j.succAbove i.succAbove := by
    ext k l
    simp only [Matrix.submatrix_apply, M, Matrix.of_apply, Fin.succAbove_last, Fin.snoc_castSucc]
  change (-1 : R)^j.val * M.adjugate i (Fin.last 4) = _
  rw [Matrix.adjugate_fin_succ_eq_det_submatrix, he,
    Matrix.adjugate_fin_succ_eq_det_submatrix]
  have hs : (-1 : R)^j.val * (-1)^(i.val+4) = (-1)^(i.val+j.val) := by
    simp only [pow_add, show (-1 : R)^4 = 1 by ring, mul_one]
    ring
  rw [← mul_assoc]
  simpa only [Fin.val_last, add_comm] using
    congrArg (fun z : R => z * (u.submatrix j.succAbove i.succAbove).det) hs

def fourProducts246 (a : Fin 5 → ℝ) (j : Fin 5) : ℝ :=
  ∏ k : Fin 4, a (j.rev.succAbove k)

theorem fourProducts_pos246 (a : Fin 5 → ℝ) (ha : ∀ i, 0 < a i) :
    ∀ j, 0 < fourProducts246 a j := fun _ => Finset.prod_pos (fun _ _ => ha _)

theorem fourProducts_mono246 (a : Fin 5 → ℝ) (ha : ∀ i, 0 < a i) (hm : Monotone a) :
    Monotone (fourProducts246 a) := by
  have hi : ∀ j k : Fin 5, j ≤ k → ∀ t : Fin 4, j.rev.succAbove t ≤ k.rev.succAbove t := by decide
  intro j k hjk
  exact Finset.prod_le_prod (fun t _ => (ha _).le) (fun t _ => hm (hi j k hjk t))

    ring
  · change a 0 * (a 2 * (a 3 * (a 4 * 1))) = _

theorem product_reverse4246 (mu nu : Fin 5 → ℝ) (c C : ℝ)
    (hmu : ∀ k, 0 < mu k) (hnu : ∀ k, 0 < nu k)

    (hC : 0 < C) (j : Fin 5) :
    (c / (331776 * C ^ 4)) * mu j ≤ nu j := by
  have hdecomp (f : Fin 5 → ℝ) :
      f j * (∏ k ∈ Finset.univ.erase j, f k) = ∏ k, f k :=
    Finset.mul_prod_erase _ _ (Finset.mem_univ j)
  have hstep : c * mu j ≤ 331776 * nu j * (∏ k, mu k) := by
    calc
      c * mu j ≤ (∏ k, nu k) * mu j := mul_le_mul_of_nonneg_right hlower (hmu j).le
      _ = (nu j * (∏ k ∈ Finset.univ.erase j, nu k)) * mu j := by rw [hdecomp]
      _ ≤ (nu j * (331776 * ∏ k ∈ Finset.univ.erase j, mu k)) * mu j :=
        mul_le_mul_of_nonneg_right
          (mul_le_mul_of_nonneg_left (prod_erase_upper4246 mu nu hnu hupper j) (hnu j).le) (hmu j).le
      _ = 331776 * nu j * (∏ k, mu k) := by rw [← hdecomp mu]; ring
  have hfinal : c * mu j ≤ nu j * (331776 * C ^ 4) := by

      _ ≤ 331776 * nu j * (∏ k, mu k) := hstep
