def Primitive (z : Fin 2 → ℤ) : Prop := ∃ c : Fin 2 → ℤ,∑ i,c i*z i=1
def Height (z : Fin 2 → ℤ) : ℚ := max |(z 0 : ℚ)| |(z 1 : ℚ)|
def Place (tau : Fin 3) (q : ℚ) : ℚ := ![|q|,padicNorm 2 q,padicNorm 3 q] tau
def Abs (tau : Fin 3) (q : ℚ) : ℝ := if tau=0 then |(q : ℝ)| else if tau=1 then padicNorm 2 q else padicNorm 3 q

def Product (B C u v : ℚ) : ℚ :=
  ∏ tau : Fin 3,(min (Place tau u) (Place tau v)/max (Place tau u) (Place tau v))*
    (Place tau (B*u+C*v)/max (Place tau u) (Place tau v))
def Retained (B C u v : ℚ) : ℚ := ∏ tau : Fin 3,min (Place tau u) (Place tau v)*Place tau (B*u+C*v)

def MatrixInt (d : ℕ) (b c : ℤ) (k : Fin 2) : Matrix (Fin 2) (Fin 2) ℤ :=
  if k=0 then !![(d : ℤ),0;b,c] else !![0,(d : ℤ);b,c]
noncomputable def Form (L : Matrix (Fin 2) (Fin 2) ℤ) (z : Fin 2 → ℤ) (i : Fin 2) : ℚ :=
  ∑ j : Fin 2,(L i j : ℚ)*(z j : ℚ)
noncomputable def Norms (d : ℕ) (b c : ℤ) (k : Fin 3 → Fin 2) (z : Fin 2 → ℤ) (tau : Fin 3) (i : Fin 2) : ℝ :=
  Abs tau (Form (MatrixInt d b c (k tau)) z i)
noncomputable def Raw (a : Fin 3 → Fin 2 → ℝ) (Q : ℝ) (tau : Fin 3) (i : Fin 2) : ℝ := Real.log (a tau i)/Real.log Q
noncomputable def Alpha (a : Fin 3 → Fin 2 → ℝ) (Q : ℝ) : ℝ :=
  -(∑ tau : Fin 3,∑ i : Fin 2,Real.log (a tau i))/(6*Real.log Q)
noncomputable def Weights (a : Fin 3 → Fin 2 → ℝ) (Q : ℝ) (tau : Fin 3) (i : Fin 2) : ℝ := Raw a Q tau i+Alpha a Q
noncomputable def Index (c : Fin 3 → Fin 2 → ℝ) (tau : Fin 3) (i : Fin 2) : ℤ := ⌈1152*c tau i⌉
noncomputable def GridRaw (z : Fin 3 → Fin 2 → ℤ) (tau : Fin 3) (i : Fin 2) : ℝ := (z tau i : ℝ)/1152
noncomputable def Candidate (z : Fin 3 → Fin 2 → ℤ) (tau : Fin 3) (i : Fin 2) : ℝ :=
  if tau=0 then GridRaw z tau i-(∑ t : Fin 3,∑ j : Fin 2,GridRaw z t j)/2 else GridRaw z tau i
noncomputable def Approx (c : Fin 3 → Fin 2 → ℝ) : Fin 3 → Fin 2 → ℝ := Candidate (Index c)
noncomputable def Candidates : Set (Fin 3 → Fin 2 → ℝ) :=
  (Candidate '' {z : Fin 3 → Fin 2 → ℤ | ∀ tau i,-1152≤z tau i ∧ z tau i≤1152}) ∩
    {c | (∑ tau : Fin 3,∑ i : Fin 2,c tau i)=0 ∧ ∀ tau i,|c tau i|≤1}
def Normal (z : Fin 2 → ℤ) : Fin 2 → ℤ := ![-z 1,z 0]
def Eligible (B C : ℚ) (z : Fin 2 → ℤ) : Prop := Primitive z ∧ z 0 ≠ 0 ∧ z 1 ≠ 0 ∧ B*(z 0 : ℚ)+C*(z 1 : ℚ) ≠ 0 ∧
  (Product B C (z 0 : ℚ) (z 1 : ℚ) : ℝ)<Real.rpow (Height z : ℝ) (-(5 : ℝ)/2)
noncomputable def Span (z : Fin 2 → ℤ) : Submodule ℚ (Fin 2 → ℚ) := Submodule.span ℚ {(fun i => (z i : ℚ))}

theorem source_binary_parameter_selection : ∀ (D E delta : ℝ) (hD : 1≤D) (hE : 1≤E)
    (hd : 0<delta) (hd1 : delta<1),

    ∃ eta theta : ℝ,∃ m : ℕ,∃ hm : 0<m,
    0<eta ∧ eta<1/2 ∧ 0<theta ∧ theta<1/12 ∧ theta≤2*eta*m ∧ 40*eta<delta ∧
    4*(eta^2)⁻¹*Real.log 12≤(m : ℝ) ∧
    ∀ U : Set ℝ,¬BddAbove U → ∀ cut : ℝ,
    ∃ Q : Fin m → ℝ,∃ r : Fin m → ℕ,
    (∀ h,Q h ∈ U ∧ cut<Q h ∧ 1<Q h) ∧ (∀ h,0<r h) ∧
    (∀ (h : Fin m) (hh : h.val+1<m),(r ⟨h.val+1,hh⟩ : ℝ)≤rothInductionOmega m theta*(r h : ℝ)) ∧
    (∀ h,(r ⟨0,hm⟩ : ℝ)*Real.log (Q ⟨0,hm⟩)≤(r h : ℝ)*Real.log (Q h) ∧
      (r h : ℝ)*Real.log (Q h)≤(1+eta)*(r ⟨0,hm⟩ : ℝ)*Real.log (Q ⟨0,hm⟩)) ∧
    (∀ h,2*E<Real.rpow (Q h) eta) ∧
    ∀ H : Fin m → ℝ,(∀ h,H h=(Q h)^((1 : ℝ)/4)) →
    (∀ h,Real.rpow (H ⟨0,hm⟩) ((r ⟨0,hm⟩ : ℝ)*1)≤H h^r h) ∧
    (∀ h,(2 : ℝ)^(3*m)≤Real.rpow (H h) (rothInductionOmega m theta*1)) ∧
    D^(∑ h,r h)≤Real.rpow (H ⟨0,hm⟩) (rothInductionOmega m theta*(r ⟨0,hm⟩ : ℝ)*1) :=
  @JSP000334.binary_parameter_selection
