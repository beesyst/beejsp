theorem integer_matrix_index_dvd234 {n : ℕ} (G : AddSubgroup (Fin n → ℤ))
    (U : Matrix (Fin n) (Fin n) ℤ) (hU : U.det ≠ 0) (hm : ∀ j, U.col j ∈ G) :
    G.index ∣ U.det.natAbs := by
  classical
  let N := Submodule.span ℤ (Set.range U.col)
  obtain ⟨hf,hi,hd⟩ := integer_span5_index_det (Pi.basisFun ℤ (Fin n)) U.col
    (Matrix.linearIndependent_cols_of_det_ne_zero hU)
  have he : (Pi.basisFun ℤ (Fin n)).toMatrix U.col = U := by ext i j; rfl
  have hiMatrix := congrArg (fun M : Matrix (Fin n) (Fin n) ℤ => M.det.natAbs) he
  have hi := hi.trans hiMatrix
  letI : N.toAddSubgroup.FiniteIndex := ⟨by rw [hi]; exact Int.natAbs_ne_zero.mpr hU⟩
  have hle : N.toAddSubgroup ≤ G := by
    change N ≤ G.toIntSubmodule
    exact Submodule.span_le.mpr (by rintro _ ⟨j,rfl⟩; exact hm j)
  have hiN : N.toAddSubgroup.index = U.det.natAbs := hi
  rw [← hiN]
  exact AddSubgroup.index_dvd_of_le hle

theorem lattice10_index_dvd234 {n : ℕ} (p : ℕ) [Fact p.Prime]
    (A : Matrix (Fin n) (Fin n) ℚ) (D : ℤ) (B : Matrix (Fin n) (Fin n) ℤ)
    (hB : B.det ≠ 0)
    (he : B.map (Int.castRingHom ℚ) = (D : ℚ) • A⁻¹) (hA : IsUnit A)
    (e : Fin n → ℕ) :
    (lattice10 p A e).index ∣ B.det.natAbs * p ^ (∑ i, e i) := by
  classical
  let U := B * Matrix.diagonal (fun i => (p : ℤ) ^ e i)
  have hd : U.det = B.det * (p : ℤ) ^ (∑ i, e i) := by
    rw [Matrix.det_mul, Matrix.det_diagonal, Finset.prod_pow_eq_pow_sum]
  have hp : p ≠ 0 := (Fact.out : p.Prime).ne_zero
  have hU : U.det ≠ 0 := by rw [hd]; exact mul_ne_zero hB (pow_ne_zero _ (by exact_mod_cast hp))
  have hm (j : Fin n) : U.col j ∈ lattice10 p A e := by
    apply (lattice10_spec p A e _).mpr
    have hcol : (fun k => (U.col j k : ℚ)) =
        (p : ℚ) ^ e j • (B.map (Int.castRingHom ℚ)).col j := by
      ext k
      simp [U,Matrix.mul_diagonal,Matrix.col_apply,smul_eq_mul,mul_comm]
    have hmul : A * B.map (Int.castRingHom ℚ) = (D : ℚ) • (1 : Matrix (Fin n) (Fin n) ℚ) := by
      rw [he,Matrix.mul_smul,Matrix.mul_nonsing_inv A ((Matrix.isUnit_iff_isUnit_det A).mp hA)]
    have hact : A *ᵥ (B.map (Int.castRingHom ℚ)).col j =
        ((D : ℚ) • (1 : Matrix (Fin n) (Fin n) ℚ)).col j := by
      change (A * B.map (Int.castRingHom ℚ)).col j = _
      rw [hmul]
    intro i
    rw [hcol,Matrix.mulVec_smul,hact]
    by_cases hij : i = j
    · subst i
      simp only [Pi.smul_apply,smul_eq_mul,Matrix.col_apply,Matrix.smul_apply,Matrix.one_apply_eq,mul_one]
      rw [show (p : ℚ) ^ e j * (D : ℚ) = (((p : ℤ) ^ e j * D : ℤ) : ℚ) by push_cast; ring]
      apply padicNorm.dvd_iff_norm_le.mp
      exact_mod_cast (dvd_mul_right ((p : ℤ) ^ e j) D)
    · simp [Matrix.col_apply,Matrix.one_apply,hij,Ne.symm hij]
  have h := integer_matrix_index_dvd234 (lattice10 p A e) U hU hm
  rw [hd,Int.natAbs_mul,Int.natAbs_pow,Int.natAbs_natCast] at h
  exact h

def dualSupport234 (A : Matrix (Fin 10) (Fin 10) ℚ) (v : Fin 10 → ℤ) : Finset (Fin 10) :=
  Finset.univ.filter (fun i => (A⁻¹.transpose *ᵥ (fun j => (v j : ℚ))) i ≠ 0)

theorem dualSupport_nonempty234 (A : Matrix (Fin 10) (Fin 10) ℚ) (hA : IsUnit A)
    (v : Fin 10 → ℤ) (hv : v ≠ 0) : (dualSupport234 A v).Nonempty := by
  by_contra he
  have hz : A⁻¹.transpose *ᵥ (fun j => (v j : ℚ)) = 0 := by
    funext i
    by_contra hi
    change (A⁻¹.transpose *ᵥ (fun j => (v j : ℚ))) i ≠ 0 at hi
    exact he ⟨i,by simp [dualSupport234,hi]⟩
  have hr := congrArg (Matrix.mulVec A.transpose) hz
  rw [Matrix.mulVec_mulVec,← Matrix.transpose_mul,
    Matrix.nonsing_inv_mul A ((Matrix.isUnit_iff_isUnit_det A).mp hA),
    Matrix.transpose_one,Matrix.one_mulVec,Matrix.mulVec_zero] at hr
  apply hv
  funext i
  have hi := congrFun hr i
  simp only [Pi.zero_apply] at hi
  change v i = 0
  exact_mod_cast hi

def complementaryFloor234 (p : ℕ) (q : ℝ) (c : Fin 10 → ℝ) (i : Fin 10) : ℤ :=
  Int.floor ((Real.log q/Real.log p)*(∑ j : Fin 9, c (i.succAbove j)))

def floorMin234 (p : ℕ) (A : Matrix (Fin 10) (Fin 10) ℚ) (hA : IsUnit A)
    (q : ℝ) (c : Fin 10 → ℝ) (v : Fin 10 → ℤ) (hv : v ≠ 0) : ℤ :=
  ((dualSupport234 A v).image (complementaryFloor234 p q c)).min'
    ((dualSupport_nonempty234 A hA v hv).image _)

theorem floorMin_le234 (p : ℕ) (A : Matrix (Fin 10) (Fin 10) ℚ) (hA : IsUnit A)
    (q : ℝ) (c : Fin 10 → ℝ) (v : Fin 10 → ℤ) (hv : v ≠ 0)
    (i : Fin 10) (hi : i ∈ dualSupport234 A v) :
    floorMin234 p A hA q c v hv ≤ complementaryFloor234 p q c i := by
  exact Finset.min'_le _ _ (Finset.mem_image.mpr ⟨i,hi,rfl⟩)

theorem floorMin_nonpos234 (p : ℕ) (hp : 1 < p) (A : Matrix (Fin 10) (Fin 10) ℚ)
    (hA : IsUnit A) (q : ℝ) (hq : 1 < q) (c : Fin 10 → ℝ)
    (hc : ∀ i, c i ≤ 0) (v : Fin 10 → ℤ) (hv : v ≠ 0) :
    floorMin234 p A hA q c v hv ≤ 0 := by
  obtain ⟨i,hi⟩ := dualSupport_nonempty234 A hA v hv
  apply (floorMin_le234 p A hA q c v hv i hi).trans
  apply Int.floor_nonpos
  exact mul_nonpos_of_nonneg_of_nonpos
    (div_nonneg (Real.log_pos hq).le (Real.log_pos (by exact_mod_cast hp)).le)
    (Finset.sum_nonpos (fun j _ => hc (i.succAbove j)))

theorem rounding_margin234 (x : Fin 9 → ℝ) (hx : ∀ j, 0 ≤ x j) :
    (∑ j, Nat.ceil (x j)) ≤ (9-Int.floor (-(∑ j, x j))).toNat := by
  have hg : Int.floor (-(∑ j, x j)) ≤ 0 :=
    Int.floor_nonpos (neg_nonpos.mpr (Finset.sum_nonneg (fun j _ => hx j)))
  have hn : 0 ≤ 9-Int.floor (-(∑ j, x j)) := by omega
  have he : (((9-Int.floor (-(∑ j, x j))).toNat : ℕ) : ℝ) =
      9-(Int.floor (-(∑ j, x j)) : ℝ) := by
    rw [← Int.cast_natCast,Int.toNat_of_nonneg hn]
    push_cast
    rfl
  have hsum := Finset.sum_le_sum (s := Finset.univ)
    (fun j _ => (Nat.ceil_lt_add_one (hx j)).le)
  simp only [Finset.sum_add_distrib,Finset.sum_const,Finset.card_univ,Fintype.card_fin,nsmul_eq_mul,mul_one] at hsum
  have hf := Int.floor_le (-(∑ j, x j))
  have hh : (∑ j : Fin 9, (Nat.ceil (x j) : ℝ)) ≤
      (((9-Int.floor (-(∑ j, x j))).toNat : ℕ) : ℝ) := by
    rw [he]
    norm_num at hsum
    linarith
  exact_mod_cast hh

theorem rational_pairing234 (A : Matrix (Fin 10) (Fin 10) ℚ) (hA : IsUnit A)
    (x v : Fin 10 → ℚ) :
    (A *ᵥ x) ⬝ᵥ (A⁻¹.transpose *ᵥ v) = x ⬝ᵥ v := by
  rw [dotProduct_comm,Matrix.dotProduct_mulVec,← Matrix.mulVec_transpose,
    Matrix.mulVec_mulVec,← Matrix.transpose_mul,
    Matrix.nonsing_inv_mul A ((Matrix.isUnit_iff_isUnit_det A).mp hA),
    Matrix.transpose_one,Matrix.one_mulVec,dotProduct_comm]

theorem cleared_pairing234 (A : Matrix (Fin 10) (Fin 10) ℚ) (hA : IsUnit A)
    (d f : ℤ) (M N : Matrix (Fin 10) (Fin 10) ℤ)
    (hM : M.map (Int.castRingHom ℚ) = (d : ℚ) • A)
    (hN : N.map (Int.castRingHom ℚ) = (f : ℚ) • A⁻¹.transpose)
    (x v : Fin 10 → ℤ) :
    (∑ i, (M *ᵥ x) i * (N *ᵥ v) i) = d*f*(∑ i, x i*v i) := by
  have h := rational_pairing234 A hA (fun i => (x i : ℚ)) (fun i => (v i : ℚ))
  have hx : (fun i => ((M *ᵥ x) i : ℚ)) = (d : ℚ) • (A *ᵥ (fun i => (x i : ℚ))) := by
    ext i
    simpa only [hM,Matrix.smul_mulVec_assoc,Pi.smul_apply,smul_eq_mul] using
      (Int.castRingHom ℚ).map_mulVec M x i
  have hv : (fun i => ((N *ᵥ v) i : ℚ)) = (f : ℚ) • (A⁻¹.transpose *ᵥ (fun i => (v i : ℚ))) := by
    ext i
    simpa only [hN,Matrix.smul_mulVec_assoc,Pi.smul_apply,smul_eq_mul] using
      (Int.castRingHom ℚ).map_mulVec N v i
  have hh : (fun i => ((M *ᵥ x) i : ℚ)) ⬝ᵥ (fun i => ((N *ᵥ v) i : ℚ)) =
      (d : ℚ)*(f : ℚ)*((fun i => (x i : ℚ)) ⬝ᵥ (fun i => (v i : ℚ))) := by
    rw [hx,hv,smul_dotProduct,dotProduct_smul,h]
    simp only [smul_eq_mul]
    ring
  simp only [dotProduct] at hh
  exact_mod_cast hh

    (A : Matrix (Fin 10) (Fin 10) ℚ) (d : ℤ) (M : Matrix (Fin 10) (Fin 10) ℤ)

    have hh := (Int.castRingHom ℚ).map_mulVec M x i

    exact hh
  suffices hn : padicNorm p ((M *ᵥ x) i : ℚ) ≤ (p : ℚ)^(-(e i : ℤ)) by
    simpa only [Nat.cast_pow] using (padicNorm.dvd_iff_norm_le (p := p)).mpr hn
  rw [he,padicNorm.mul]
  have hb := (lattice10_spec p A e x).mp hx i
  exact (mul_le_mul_of_nonneg_right (padicNorm.of_int d) (padicNorm.nonneg _)).trans (by simpa using hb)
