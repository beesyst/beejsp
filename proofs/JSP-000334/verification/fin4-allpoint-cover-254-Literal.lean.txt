theorem allpoint_rank_scalar4 (a b c d g t q C : ℝ)
    (ha : 0<a) (hb : 0<b) (hc : 0<c) (hd : 0<d) (ht : 0<t) (hq : 1≤q)
    (hab : a≤b) (hbc : b≤c) (hcd : c≤d) (hag : a≤g) (hgt : g≤t)
    (hp : C≤a*b*c*d) (hcut : t^4*q^12<C) :
    (c*q<d ∧ g<d) ∨ (b*q<c ∧ g<c) ∨ (a*q<b ∧ g<b) := by
  have hq0 : 0<q := lt_of_lt_of_le zero_lt_one hq
  have hbig : t*q^3<d := by
    by_contra hh
    have hdle : d≤t*q^3 := le_of_not_gt hh
    have hp1 : a*b*c*d≤d^4 := by
      have h1 := mul_le_mul (hab.trans (hbc.trans hcd)) (hbc.trans hcd) hb.le hd.le
      have h2 := mul_le_mul hcd le_rfl hd.le hd.le
      have h3 := mul_le_mul h1 h2 (mul_pos hc hd).le (mul_nonneg hd.le hd.le)
      nlinarith only [h3]
    have hpow := pow_le_pow_left₀ hd.le hdle 4
    have he : (t*q^3)^4=t^4*q^12 := by ring
    rw [he] at hpow
    exact (not_lt_of_ge (hp.trans (hp1.trans hpow))) hcut
  have hq2 : 1≤q^2 := by nlinarith
  have hq3 : 1≤q^3 := by nlinarith [mul_le_mul_of_nonneg_right hq2 hq0.le]
  have htd : t<d := (by nlinarith : t≤t*q^3).trans_lt hbig
  by_cases h3 : c*q<d
  · exact Or.inl ⟨h3,hgt.trans_lt htd⟩
  have hdq : d≤c*q := le_of_not_gt h3
  have htc : t*q^2<c := by
    have hh := hbig.trans_le hdq
    have he : t*q^3=(t*q^2)*q := by ring
    rw [he] at hh
    exact (mul_lt_mul_right hq0).mp hh
  have hgc : g<c := hgt.trans_lt ((by nlinarith : t≤t*q^2).trans_lt htc)
  by_cases h2 : b*q<c
  · exact Or.inr (Or.inl ⟨h2,hgc⟩)
  have hcq : c≤b*q := le_of_not_gt h2
  have htb : t*q<b := by
    have hh := htc.trans_le hcq
    have he : t*q^2=(t*q)*q := by ring
    rw [he] at hh
    exact (mul_lt_mul_right hq0).mp hh
  have hgb : g<b := hgt.trans_lt ((by nlinarith : t≤t*q).trans_lt htb)
  have haq : a*q<b := (mul_le_mul_of_nonneg_right (hag.trans hgt) hq0.le).trans_lt htb
  exact Or.inr (Or.inr ⟨haq,hgb⟩)

theorem allpoint_rational_flag4 (w : Fin 4 → Fin 4 → ℤ)
    (hw : LinearIndependent ℝ (fun j => integerReal (w j))) (k : Fin 4) (z : Fin 4 → ℤ)
    (hz : integerReal z ∈ minimaFlag4 (fun j => integerReal (w j)) k) :
    (fun i => (z i : ℚ)) ∈ originalFirstFlag4 k w := by
  classical
  let v : Fin 4 → Fin 4 → ℚ := fun j i => (w j i : ℚ)
  have hQ : LinearIndependent ℚ v := integral_frame_rational_independent w hw
  let b := rationalFrameBasis42 v hQ
  have hb (i : Fin 4) : b i=v i := rationalFrameBasis42_apply v hQ i
  have hzero (j : Fin 4) (hj : k≤j) : b.repr (fun i => (z i : ℚ)) j=0 := by
    let f := rationalFrameDual42 v hQ j
    have hk : minimaFlag4 (fun i => integerReal (w i)) k ≤
        LinearMap.ker (rationalFunctionalReal42 f) := by
      apply Submodule.span_le.mpr
      rintro _ ⟨i,rfl⟩
      have hij : (Fin.castLE (by omega) i : Fin 4)≠j := by
        intro he
        have hh := congrArg Fin.val he
        change i.val=j.val at hh
        have hi := i.isLt
        change k.val≤j.val at hj
        omega
      change rationalFunctionalReal42 f (integerReal (w (Fin.castLE (by omega) i)))=0
      have he := rationalFunctionalReal42_cast f (v (Fin.castLE (by omega) i))
      change rationalFunctionalReal42 f (integerReal (w (Fin.castLE (by omega) i)))=_ at he
      rw [he]
      simp [f,rationalFrameDual42_pairing,hij]
    have hh := hk hz
    change rationalFunctionalReal42 f (integerReal z)=0 at hh
    have he := rationalFunctionalReal42_cast f (fun i => (z i : ℚ))
    change rationalFunctionalReal42 f (integerReal z)=_ at he
    rw [he] at hh
    have hf : f (fun i => (z i : ℚ))=0 := by exact_mod_cast hh
    rw [← Basis.dualBasis_apply]
    exact hf
  rw [← b.sum_repr (fun i => (z i : ℚ))]
  apply Submodule.sum_mem
  intro i _
  by_cases hi : i.val<k.val
  · apply Submodule.smul_mem
    rw [hb]
    apply Submodule.subset_span
    exact ⟨⟨i.val,hi⟩,rfl⟩
  · rw [hzero i (by omega),zero_smul]
    exact Submodule.zero_mem _

def retainedIntegerMatrix4 (k : Fin 4) : Matrix (Fin 4) (Fin 4) ℤ :=
  fun i j => if i=k then 1 else if i=j then 1 else 0

theorem retained_integer_matrix_det4 (k : Fin 4) : (retainedIntegerMatrix4 k).det=1 := by
  have he : retainedIntegerMatrix4 k =
      Matrix.updateRow (1 : Matrix (Fin 4) (Fin 4) ℤ) k
        (∑ j : Fin 4,(1 : ℤ) • (1 : Matrix (Fin 4) (Fin 4) ℤ) j) := by
    ext i j
    by_cases hi : i=k <;> simp [retainedIntegerMatrix4,Matrix.updateRow,hi,Matrix.one_apply]
  rw [he,Matrix.det_updateRow_sum]
  simp

theorem retained_integer_matrix_apply4 (k i : Fin 4) (z : Fin 4 → ℤ) :
    (retainedIntegerMatrix4 k *ᵥ z) i=if i=k then ∑ j,z j else z i := by
  by_cases hi : i=k
  · simp [Matrix.mulVec,dotProduct,retainedIntegerMatrix4,hi]
  · simp [Matrix.mulVec,dotProduct,retainedIntegerMatrix4,hi]

def PrimitiveFour (z : Fin 4 → ℤ) : Prop := ∃ c : Fin 4 → ℤ,∑ i,c i*z i=1

theorem primitive_four_integer_reduction (w : Fin 4 → ℤ) (hw0 : w 0≠0) :
    ∃ g : ℕ,0<g ∧ (g : ℤ)∣w 0 ∧ ∃ z : Fin 4 → ℤ,
      PrimitiveFour z ∧ ∀ i,w i=(g : ℤ)*z i := by
  let k := Int.gcd (w 2) (w 3)
  let l := Int.gcd (w 1) (k : ℤ)
  let g := Int.gcd (w 0) (l : ℤ)
  have hg : 0<g := Int.gcd_pos_of_ne_zero_left _ hw0
  have hg0 : (g : ℤ)∣w 0 := Int.gcd_dvd_left
  have hgl : (g : ℤ)∣(l : ℤ) := Int.gcd_dvd_right
  have hl1 : (l : ℤ)∣w 1 := Int.gcd_dvd_left
  have hlk : (l : ℤ)∣(k : ℤ) := Int.gcd_dvd_right
  have hk2 : (k : ℤ)∣w 2 := Int.gcd_dvd_left
  have hk3 : (k : ℤ)∣w 3 := Int.gcd_dvd_right
  obtain ⟨z0,h0⟩ := hg0
  obtain ⟨z1,h1⟩ := hgl.trans hl1
  obtain ⟨z2,h2⟩ := (hgl.trans hlk).trans hk2
  obtain ⟨z3,h3⟩ := (hgl.trans hlk).trans hk3
  let z : Fin 4 → ℤ := ![z0,z1,z2,z3]
  have hscale : ∀ i,w i=(g : ℤ)*z i := by intro i; fin_cases i <;> assumption
  let a := Int.gcdA (w 0) (l : ℤ)
  let b := Int.gcdB (w 0) (l : ℤ)
  let c := Int.gcdA (w 1) (k : ℤ)
  let d := Int.gcdB (w 1) (k : ℤ)
  let e := Int.gcdA (w 2) (w 3)
  let f := Int.gcdB (w 2) (w 3)
  have hb0 : (g : ℤ)=w 0*a+(l : ℤ)*b := Int.gcd_eq_gcd_ab _ _
  have hb1 : (l : ℤ)=w 1*c+(k : ℤ)*d := Int.gcd_eq_gcd_ab _ _
  have hb2 : (k : ℤ)=w 2*e+w 3*f := Int.gcd_eq_gcd_ab _ _
  have heq : (g : ℤ)*(a*z0+(b*c)*z1+(b*d*e)*z2+(b*d*f)*z3)=(g : ℤ)*1 := by
    calc
      _=w 0*a+(l : ℤ)*b := by rw [hb1,hb2,h0,h1,h2,h3]; ring
      _=_ := by rw [← hb0]; ring
  have hgn : (g : ℤ)≠0 := by exact_mod_cast ne_of_gt hg
  refine ⟨g,hg,Int.gcd_dvd_left,z,?_,hscale⟩
  refine ⟨![a,b*c,b*d*e,b*d*f],?_⟩
  simpa [z,Fin.sum_univ_succ,add_assoc] using mul_left_cancel₀ hgn heq

def fourRetainedForm (k i : Fin 4) (y : Fin 4 → ℚ) : ℚ := if i=k then ∑ j,y j else y i

def localFourHeight (tau : Fin 3) (y : Fin 4 → ℚ) : ℚ :=
  max (rationalPlaceNorm tau (y 0)) (max (rationalPlaceNorm tau (y 1))
    (max (rationalPlaceNorm tau (y 2)) (rationalPlaceNorm tau (y 3))))

def fourHeight (y : Fin 4 → ℚ) : ℚ := ∏ tau : Fin 3,localFourHeight tau y

def archFourHeight (z : Fin 4 → ℤ) : ℚ :=
  max |(z 0 : ℚ)| (max |(z 1 : ℚ)| (max |(z 2 : ℚ)| |(z 3 : ℚ)|))

def fourRetainedProduct (choice : Fin 3 → Fin 4) (y : Fin 4 → ℚ) : ℚ :=
  ∏ tau : Fin 3,∏ i : Fin 4,rationalPlaceNorm tau (fourRetainedForm (choice tau) i y)

theorem four_local_coordinate_le (tau : Fin 3) (y : Fin 4 → ℚ) (i : Fin 4) :
    rationalPlaceNorm tau (y i)≤localFourHeight tau y := by
  fin_cases i
  · exact le_max_left _ _
  · exact (le_max_left _ _).trans (le_max_right _ _)
  · exact (le_max_left _ _).trans ((le_max_right _ _).trans (le_max_right _ _))
  · exact (le_max_right _ _).trans ((le_max_right _ _).trans (le_max_right _ _))

theorem primitive_four_local_height (z : Fin 4 → ℤ) (hz : PrimitiveFour z)
    (p : ℕ) [Fact p.Prime] :
    max (padicNorm p (z 0 : ℚ)) (max (padicNorm p (z 1 : ℚ))
      (max (padicNorm p (z 2 : ℚ)) (padicNorm p (z 3 : ℚ))))=1 := by
  obtain ⟨c,hc⟩ := hz
  obtain ⟨i,hi⟩ : ∃ i : Fin 4,¬(p : ℤ)∣z i := by
    by_contra hn
    push_neg at hn
    have hd : (p : ℤ)∣∑ i,c i*z i := Finset.dvd_sum (fun i _ => dvd_mul_of_dvd_right (hn i) _)
    rw [hc] at hd
    exact (Fact.out : p.Prime).not_dvd_one (Int.natCast_dvd_natCast.mp hd)
  have he := (padicNorm.int_eq_one_iff (p := p) (z i)).mpr hi
  apply le_antisymm
  · exact max_le (padicNorm.of_int _) (max_le (padicNorm.of_int _)
      (max_le (padicNorm.of_int _) (padicNorm.of_int _)))
  · rw [← he]
    fin_cases i
    · exact le_max_left _ _
    · exact (le_max_left _ _).trans (le_max_right _ _)
    · exact (le_max_left _ _).trans ((le_max_right _ _).trans (le_max_right _ _))
    · exact (le_max_right _ _).trans ((le_max_right _ _).trans (le_max_right _ _))

theorem primitive_four_height (z : Fin 4 → ℤ) (hz : PrimitiveFour z) :
    fourHeight (fun i => (z i : ℚ))=archFourHeight z := by
  simp only [fourHeight,Fin.prod_univ_three]
  change archFourHeight z *
    (max (padicNorm 2 (z 0 : ℚ)) (max (padicNorm 2 (z 1 : ℚ))
      (max (padicNorm 2 (z 2 : ℚ)) (padicNorm 2 (z 3 : ℚ))))) *
    (max (padicNorm 3 (z 0 : ℚ)) (max (padicNorm 3 (z 1 : ℚ))
      (max (padicNorm 3 (z 2 : ℚ)) (padicNorm 3 (z 3 : ℚ)))))=_
  rw [primitive_four_local_height z hz 2,primitive_four_local_height z hz 3]
  ring

theorem exists_four_maximal_coordinate (tau : Fin 3) (y : Fin 4 → ℚ) :
    ∃ k : Fin 4,rationalPlaceNorm tau (y k)=localFourHeight tau y := by
  let f := fun i => rationalPlaceNorm tau (y i)
  have hex : ∃ k : Fin 4,f k=max (f 0) (max (f 1) (max (f 2) (f 3))) := by
    rcases le_total (f 2) (f 3) with h|h
    · rw [max_eq_right h]
      rcases le_total (f 1) (f 3) with h|h
      · rw [max_eq_right h]
        rcases le_total (f 0) (f 3) with h|h
        · exact ⟨3,(max_eq_right h).symm⟩
        · exact ⟨0,(max_eq_left h).symm⟩
      · rw [max_eq_left h]
        rcases le_total (f 0) (f 1) with h|h
        · exact ⟨1,(max_eq_right h).symm⟩
        · exact ⟨0,(max_eq_left h).symm⟩
    · rw [max_eq_left h]
      rcases le_total (f 1) (f 2) with h|h
      · rw [max_eq_right h]
        rcases le_total (f 0) (f 2) with h|h
        · exact ⟨2,(max_eq_right h).symm⟩
        · exact ⟨0,(max_eq_left h).symm⟩
      · rw [max_eq_left h]
        rcases le_total (f 0) (f 1) with h|h
        · exact ⟨1,(max_eq_right h).symm⟩
        · exact ⟨0,(max_eq_left h).symm⟩
  exact hex

    rw [← hchoice tau]
    generalize choice tau=k
    fin_cases k <;> simp [fourRetainedForm,Fin.prod_univ_succ] <;> ring
  unfold fourRetainedProduct fourHeight
  rw [← Finset.prod_mul_distrib]
  simp_rw [hlocal]
  rw [Finset.prod_mul_distrib,Finset.prod_comm]
  simp_rw [rationalPlaceNorm_product]

def sourceWeightConditions4 (c : Fin 3 → Fin 4 → ℝ) : Prop :=
  (∀ i, |c 0 i| ≤ 1) ∧ (∀ tau, tau ≠ 0 → ∀ i, -1 ≤ c tau i ∧ c tau i ≤ 0) ∧
    (∑ tau : Fin 3, ∑ i : Fin 4, c tau i) = 0

noncomputable def finiteWeightIndex4 (n : ℕ) (c : Fin 3 → Fin 4 → ℝ) (tau : Fin 3) (i : Fin 4) : ℤ :=
  if tau = 0 then ⌊(n : ℝ)*((1-4/(n : ℝ))*c tau i)⌋
  else ⌈(n : ℝ)*((1-4/(n : ℝ))*c tau i)⌉

noncomputable def finiteWeightRaw4 (n : ℕ) (z : Fin 3 → Fin 4 → ℤ) (tau : Fin 3) (i : Fin 4) : ℝ :=
  (z tau i : ℝ)/(n : ℝ)

noncomputable def finiteWeightCandidate4 (n : ℕ) (z : Fin 3 → Fin 4 → ℤ) (tau : Fin 3) (i : Fin 4) : ℝ :=
  if tau = 0 then finiteWeightRaw4 n z tau i - (∑ t : Fin 3, ∑ j : Fin 4, finiteWeightRaw4 n z t j)/4
  else finiteWeightRaw4 n z tau i

noncomputable def finiteWeightApprox4 (n : ℕ) (c : Fin 3 → Fin 4 → ℝ) : Fin 3 → Fin 4 → ℝ :=
  finiteWeightCandidate4 n (finiteWeightIndex4 n c)

theorem finiteWeight_floor_bounds4 (n : ℕ) (hn : 0 < n) (x : ℝ) :
    x-1/(n : ℝ) ≤ (⌊(n : ℝ)*x⌋ : ℝ)/(n : ℝ) ∧
      (⌊(n : ℝ)*x⌋ : ℝ)/(n : ℝ) ≤ x := by
  have hp : (0 : ℝ) < n := by exact_mod_cast hn
  have he : (1/(n : ℝ))*(n : ℝ) = 1 := one_div_mul_cancel hp.ne'
  constructor
  · apply (le_div_iff₀ hp).mpr
    nlinarith [Int.lt_floor_add_one ((n : ℝ)*x)]
  · apply (div_le_iff₀ hp).mpr
    nlinarith [Int.floor_le ((n : ℝ)*x)]

theorem finiteWeight_ceil_bounds4 (n : ℕ) (hn : 0 < n) (x : ℝ) :
    x ≤ (⌈(n : ℝ)*x⌉ : ℝ)/(n : ℝ) ∧
      (⌈(n : ℝ)*x⌉ : ℝ)/(n : ℝ) ≤ x+1/(n : ℝ) := by
  have hp : (0 : ℝ) < n := by exact_mod_cast hn
  have he : (1/(n : ℝ))*(n : ℝ) = 1 := one_div_mul_cancel hp.ne'
  constructor
  · apply (le_div_iff₀ hp).mpr
    nlinarith [Int.le_ceil ((n : ℝ)*x)]
  · apply (div_le_iff₀ hp).mpr
    nlinarith [Int.ceil_lt_add_one ((n : ℝ)*x)]

theorem finiteWeightCandidate4_balance (n : ℕ) (z : Fin 3 → Fin 4 → ℤ) :
    (∑ tau : Fin 3, ∑ i : Fin 4, finiteWeightCandidate4 n z tau i) = 0 := by
  simp [finiteWeightCandidate4, Fin.sum_univ_succ]
  ring

theorem finiteWeight_raw_errors4 (n : ℕ) (hn : 0 < n) (c : Fin 3 → Fin 4 → ℝ) :
    (∀ i, (1-4/(n : ℝ))*c 0 i-1/(n : ℝ) ≤ finiteWeightRaw4 n (finiteWeightIndex4 n c) 0 i ∧
      finiteWeightRaw4 n (finiteWeightIndex4 n c) 0 i ≤ (1-4/(n : ℝ))*c 0 i) ∧
    (∀ tau, tau ≠ 0 → ∀ i,
      (1-4/(n : ℝ))*c tau i ≤ finiteWeightRaw4 n (finiteWeightIndex4 n c) tau i ∧
      finiteWeightRaw4 n (finiteWeightIndex4 n c) tau i ≤ (1-4/(n : ℝ))*c tau i+1/(n : ℝ)) := by
  constructor
  · intro i
    simpa only [finiteWeightRaw4, finiteWeightIndex4, if_pos rfl] using
      finiteWeight_floor_bounds4 n hn ((1-4/(n : ℝ))*c 0 i)
  · intro tau ht i
    simpa only [finiteWeightRaw4, finiteWeightIndex4, if_neg ht] using
      finiteWeight_ceil_bounds4 n hn ((1-4/(n : ℝ))*c tau i)

theorem finiteWeight_raw_sum4 (n : ℕ) (hn : 0 < n) (c : Fin 3 → Fin 4 → ℝ)
    (hb : (∑ tau : Fin 3, ∑ i : Fin 4, c tau i) = 0) :

  have h2 := hp 2 (by decide)
  have hscaled : (1-4/(n : ℝ))*(∑ tau : Fin 3, ∑ i : Fin 4, c tau i) = 0 := by rw [hb, mul_zero]
  simp [Fin.sum_univ_succ] at hb hscaled ⊢
  simp only [div_eq_mul_inv, one_mul] at *
  constructor <;> nlinarith [(h0 0).1,(h0 0).2,(h0 1).1,(h0 1).2,(h0 2).1,(h0 2).2,(h0 3).1,(h0 3).2,
    (h1 0).1,(h1 0).2,(h1 1).1,(h1 1).2,(h1 2).1,(h1 2).2,(h1 3).1,(h1 3).2,
    (h2 0).1,(h2 0).2,(h2 1).1,(h2 1).2,(h2 2).1,(h2 2).2,(h2 3).1,(h2 3).2]

theorem finiteWeightApprox4_bounds (n : ℕ) (hn : 16 ≤ n)
    (c : Fin 3 → Fin 4 → ℝ) (hc : sourceWeightConditions4 c) :
    sourceWeightConditions4 (finiteWeightApprox4 n c) ∧
      (∀ i, c 0 i-7/(n : ℝ) ≤ finiteWeightApprox4 n c 0 i) ∧
      (∀ tau, tau ≠ 0 → ∀ i, c tau i ≤ finiteWeightApprox4 n c tau i) := by

  have h4 : 4/(n : ℝ) ≤ 1/4 := by
    calc
      _ = 4*(1/(n : ℝ)) := by ring
      _ ≤ 4*(1/16) := mul_le_mul_of_nonneg_left he' (by norm_num)
      _ = _ := by norm_num
  obtain ⟨h0,hf,hb⟩ := hc
  obtain ⟨ha0,haf⟩ := finiteWeight_raw_errors4 n hnp c
  obtain ⟨hBl,hBu⟩ := finiteWeight_raw_sum4 n hnp c hb
  have hreal (i : Fin 4) : |finiteWeightApprox4 n c 0 i| ≤ 1 ∧
      c 0 i-7/(n : ℝ) ≤ finiteWeightApprox4 n c 0 i := by
    have hc0 := abs_le.mp (h0 i)
    have hscaled := mul_le_mul_of_nonneg_left hc0.1 (by linarith : 0 ≤ 1-4/(n : ℝ))
    have hscaled' := mul_le_mul_of_nonneg_left hc0.2 (by linarith : 0 ≤ 1-4/(n : ℝ))
    change |finiteWeightRaw4 n (finiteWeightIndex4 n c) 0 i -
      (∑ t : Fin 3, ∑ j : Fin 4, finiteWeightRaw4 n (finiteWeightIndex4 n c) t j)/4| ≤ 1 ∧ _
    change _ ∧ c 0 i-7/(n : ℝ) ≤ finiteWeightRaw4 n (finiteWeightIndex4 n c) 0 i -
      (∑ t : Fin 3, ∑ j : Fin 4, finiteWeightRaw4 n (finiteWeightIndex4 n c) t j)/4
    have hsmall := mul_le_mul_of_nonneg_left hc0.2 (by positivity : 0 ≤ 4/(n : ℝ))
    simp only [div_eq_mul_inv, one_mul] at *
    refine ⟨abs_le.mpr ⟨?_,?_⟩,?_⟩ <;> nlinarith [(ha0 i).1,(ha0 i).2]
  have hfinite (tau : Fin 3) (ht : tau ≠ 0) (i : Fin 4) :
      -1 ≤ finiteWeightApprox4 n c tau i ∧ finiteWeightApprox4 n c tau i ≤ 0 ∧
      c tau i ≤ finiteWeightApprox4 n c tau i := by
    have hci := hf tau ht i
    have hscaled := mul_le_mul_of_nonneg_left hci.1 (by linarith : 0 ≤ 1-4/(n : ℝ))
    have hscaled' : (1-4/(n : ℝ))*c tau i ≤ 0 := mul_nonpos_of_nonneg_of_nonpos (by linarith) hci.2
    have hceil : ⌈(n : ℝ)*((1-4/(n : ℝ))*c tau i)⌉ ≤ (0 : ℤ) :=
      Int.ceil_le.mpr (by simpa using mul_nonpos_of_nonneg_of_nonpos hp.le hscaled')
    have hceilR : (⌈(n : ℝ)*((1-4/(n : ℝ))*c tau i)⌉ : ℝ) ≤ 0 := by exact_mod_cast hceil
    have hnonpos : finiteWeightApprox4 n c tau i ≤ 0 := by
      simp only [finiteWeightApprox4, finiteWeightCandidate4, finiteWeightRaw4, finiteWeightIndex4, if_neg ht]
      exact div_nonpos_of_nonpos_of_nonneg hceilR hp.le
    have heq : finiteWeightApprox4 n c tau i = finiteWeightRaw4 n (finiteWeightIndex4 n c) tau i := by
      simp only [finiteWeightApprox4, finiteWeightCandidate4, if_neg ht]
    rw [heq] at hnonpos ⊢
    have hsmall := mul_nonpos_of_nonneg_of_nonpos (by positivity : 0 ≤ 4/(n : ℝ)) hci.2
    simp only [div_eq_mul_inv, one_mul] at *
    refine ⟨?_,hnonpos,?_⟩ <;> nlinarith [(haf tau ht i).1]
  refine ⟨⟨fun i => (hreal i).1,fun tau ht i => ⟨(hfinite tau ht i).1,(hfinite tau ht i).2.1⟩,
    finiteWeightCandidate4_balance n _⟩,fun i => (hreal i).2,fun tau ht i => (hfinite tau ht i).2.2⟩

noncomputable def finiteWeightCandidates4 (n : ℕ) : Set (Fin 3 → Fin 4 → ℝ) :=
  (finiteWeightCandidate4 n '' {z : Fin 3 → Fin 4 → ℤ | ∀ tau i, -(n : ℤ) ≤ z tau i ∧ z tau i ≤ n}) ∩

theorem finiteWeightCandidates4_finite (n : ℕ) : (finiteWeightCandidates4 n).Finite := by

theorem finiteWeightIndex4_bounded (n : ℕ) (hn : 16 ≤ n) (c : Fin 3 → Fin 4 → ℝ)
    (hc : sourceWeightConditions4 c) :
    ∀ tau i, -(n : ℤ) ≤ finiteWeightIndex4 n c tau i ∧ finiteWeightIndex4 n c tau i ≤ n := by
  have hp : (0 : ℝ) < n := by exact_mod_cast (show 0 < n by omega)
  have hnR : (16 : ℝ) ≤ n := by exact_mod_cast hn
  have hfactor : 0 ≤ 1-4/(n : ℝ) := by
    apply sub_nonneg.mpr
    exact (div_le_one hp).mpr (by linarith)
  have hfactor' : 1-4/(n : ℝ) ≤ 1 := by
    have : 0 < 4/(n : ℝ) := by positivity
    linarith
  intro tau i
  have hci : -1 ≤ c tau i ∧ c tau i ≤ 1 := by
    by_cases ht : tau = 0
    · subst tau; exact abs_le.mp (hc.1 i)
    · exact ⟨(hc.2.1 tau ht i).1, (hc.2.1 tau ht i).2.trans (by norm_num)⟩
  have hscaled : -(n : ℝ) ≤ (n : ℝ)*((1-4/(n : ℝ))*c tau i) ∧
      (n : ℝ)*((1-4/(n : ℝ))*c tau i) ≤ n := by
    have h1 := mul_le_mul_of_nonneg_left hci.1 hfactor
    have h2 := mul_le_mul_of_nonneg_left hci.2 hfactor
    constructor
    · have h : -1 ≤ (1-4/(n : ℝ))*c tau i := by nlinarith
      nlinarith [mul_le_mul_of_nonneg_left h hp.le]
    · have h : (1-4/(n : ℝ))*c tau i ≤ 1 := by nlinarith
      nlinarith [mul_le_mul_of_nonneg_left h hp.le]
  unfold finiteWeightIndex4
  split_ifs with ht
  · constructor
    · exact Int.le_floor.mpr (by simpa using hscaled.1)
    · exact_mod_cast ((Int.floor_le _).trans hscaled.2)
  · constructor
    · exact_mod_cast (hscaled.1.trans (Int.le_ceil _))
    · exact Int.ceil_le.mpr (by simpa using hscaled.2)

theorem finiteWeightApprox4_mem (n : ℕ) (hn : 16 ≤ n) (c : Fin 3 → Fin 4 → ℝ)

noncomputable def sourceLogWeights4 (Q : ℝ) (W : Fin 3 → Fin 4 → ℝ) (tau : Fin 3) (i : Fin 4) : ℝ :=
  Real.log (W tau i)/Real.log Q

theorem sourceLogWeights4_rpow (Q : ℝ) (hQ : 1 < Q) (W : Fin 3 → Fin 4 → ℝ)
    (hW : ∀ tau i, 0 < W tau i) (tau : Fin 3) (i : Fin 4) :
    Q^(sourceLogWeights4 Q W tau i) = W tau i := by
  rw [Real.rpow_def_of_pos (lt_trans zero_lt_one hQ)]
  unfold sourceLogWeights4
  rw [mul_div_cancel₀ _ (Real.log_pos hQ).ne', Real.exp_log (hW tau i)]

theorem sourceLogWeights4_conditions (Q : ℝ) (hQ : 1 < Q) (W : Fin 3 → Fin 4 → ℝ)
    (hW : ∀ tau i, 0 < W tau i)
    (hcut : ∀ tau i, (W tau i)⁻¹ < Q)
    (hcut0 : ∀ i, W 0 i < Q)
    (hfinite : ∀ tau, tau ≠ 0 → ∀ i, W tau i ≤ 1)
    (hprod : (∏ tau : Fin 3, ∏ i : Fin 4, W tau i) = 1) :
    sourceWeightConditions4 (sourceLogWeights4 Q W) := by
  have hl := Real.log_pos hQ
  have hlo (tau : Fin 3) (i : Fin 4) : -1 ≤ sourceLogWeights4 Q W tau i := by
    have hh := Real.log_lt_log (inv_pos.mpr (hW tau i)) (hcut tau i)
    rw [Real.log_inv] at hh
    apply (le_div_iff₀ hl).mpr
    linarith
  refine ⟨?_,?_,?_⟩
  · intro i
    refine abs_le.mpr ⟨hlo 0 i,?_⟩
    apply (div_le_iff₀ hl).mpr
    have hh := Real.log_lt_log (hW 0 i) (hcut0 i)
    linarith
  · intro tau ht i
    refine ⟨hlo tau i,?_⟩
    apply div_nonpos_of_nonpos_of_nonneg _ hl.le
    exact Real.log_nonpos (hW tau i).le (hfinite tau ht i)
  · have hlogs : (∑ tau : Fin 3, ∑ i : Fin 4, Real.log (W tau i)) = 0 := by
      have heq : Real.log (∏ tau : Fin 3, ∏ i : Fin 4, W tau i) =
          (∑ tau : Fin 3, ∑ i : Fin 4, Real.log (W tau i)) := by
        rw [Real.log_prod Finset.univ (fun tau => ∏ i : Fin 4, W tau i) (fun tau _ => (Finset.prod_pos (fun i _ => hW tau i)).ne')]
        apply Finset.sum_congr rfl
        intro tau _
        exact Real.log_prod Finset.univ (W tau) (fun i _ => (hW tau i).ne')
      rw [hprod,Real.log_one] at heq
      exact heq.symm
    unfold sourceLogWeights4
    simp only [← Finset.sum_div]
    rw [hlogs,zero_div]

def allpoint_balanced_values4 (Q : ℝ) (a : Fin 3 → Fin 4 → ℝ) (tau : Fin 3) (i : Fin 4) : ℝ :=
  if tau = 0 then a tau i*Q^((1 : ℝ)/32) else a tau i

theorem allpoint_balanced_weights4 (H : ℝ) (hH : 4 ≤ H)
    (a : Fin 3 → Fin 4 → ℝ) (ha : ∀ tau i, 0 < a tau i)
    (hreal : ∀ i, a 0 i ≤ 4*H)
    (hfinite : ∀ tau, tau ≠ 0 → ∀ i, a tau i ≤ 1)
    (hinv : ∀ tau i, (a tau i)⁻¹ ≤ 4*H)
    (hprod : (∏ tau, ∏ i, a tau i)*H = 1) :
    let Q := H^8
    let W := allpoint_balanced_values4 Q a
    1 < Q ∧ sourceWeightConditions4 (sourceLogWeights4 Q W) ∧
      (∀ tau i, Q^(sourceLogWeights4 Q W tau i) = W tau i) ∧
      (∀ i, a 0 i = Q^(sourceLogWeights4 Q W 0 i-(1 : ℝ)/32)) ∧
      (∀ tau, tau ≠ 0 → ∀ i, a tau i = Q^(sourceLogWeights4 Q W tau i)) := by
  intro Q W
  have hH0 : 0 < H := by linarith
  have hQ : 1 < Q := by dsimp [Q]; nlinarith [sq_nonneg (H^2-1)]
  have hQ0 := lt_trans zero_lt_one hQ
  let r := Q^((1 : ℝ)/32)
  have hr : 0 < r := Real.rpow_pos_of_pos hQ0 _
  have hr1 : 1 ≤ r := Real.one_le_rpow hQ.le (by norm_num)
  have hrH : r ≤ H := by
    have he : (H^8)^((1 : ℝ)/32) = H^((1 : ℝ)/4) := by
      rw [← Real.rpow_natCast,← Real.rpow_mul hH0.le]
      congr 1
      norm_num
    change (H^8)^((1 : ℝ)/32) ≤ H
    rw [he]
    simpa using Real.rpow_le_rpow_of_exponent_le (show 1 ≤ H by linarith) (by norm_num : (1 : ℝ)/4 ≤ 1)
  have hrpow : r^4 = H := by
    change ((H^8)^((1 : ℝ)/32))^4 = H
    rw [← Real.rpow_mul_natCast (pow_pos hH0 8).le,← Real.rpow_natCast,← Real.rpow_mul hH0.le]
    norm_num
  have hW : ∀ tau i, 0 < W tau i := by
    intro tau i
    dsimp [W,allpoint_balanced_values4]
    split_ifs
    · exact mul_pos (ha tau i) (Real.rpow_pos_of_pos hQ0 _)
    · exact ha tau i
  have hH2 : 4*H^2 < H^4 := by nlinarith [sq_nonneg (H^2-4)]
  have h48 : H^4≤H^8 := by nlinarith [sq_nonneg (H^4-1)]
  have h3H : 4*H < Q := by change 4*H<H^8; nlinarith
  have h3H2 : 4*H^2 < Q := hH2.trans_le h48
  have hcut : ∀ tau i, (W tau i)⁻¹ < Q := by
    intro tau i
    by_cases ht : tau = 0
    · subst tau
      change (a 0 i*r)⁻¹ < Q
      rw [mul_inv_rev]
      have hri : r⁻¹ ≤ 1 := (inv_le_one₀ hr).mpr hr1
      have hh := mul_le_mul_of_nonneg_right hri (inv_pos.mpr (ha 0 i)).le
      exact (by nlinarith [hinv 0 i] : r⁻¹*(a 0 i)⁻¹ ≤ 4*H).trans_lt h3H
    · simpa only [W,allpoint_balanced_values4,if_neg ht] using (hinv tau i).trans_lt h3H
  have hcut0 : ∀ i, W 0 i < Q := by
    intro i
    change a 0 i*r < Q
    have hh := mul_le_mul (hreal i) hrH hr.le (by linarith : 0 ≤ 4*H)
    exact (by nlinarith : a 0 i*r ≤ 4*H^2).trans_lt h3H2
  have hWfin : ∀ tau, tau ≠ 0 → ∀ i, W tau i ≤ 1 := by
    intro tau ht i
    simpa only [W,allpoint_balanced_values4,if_neg ht] using hfinite tau ht i
  have hWprod : (∏ tau, ∏ i, W tau i) = 1 := by
    have hfactor (tau : Fin 3) : (∏ i : Fin 4,W tau i)=
        if tau=0 then (∏ i : Fin 4,a tau i)*r^4 else ∏ i : Fin 4,a tau i := by
      by_cases ht : tau=0
      · subst tau
        simp only [W,allpoint_balanced_values4,if_pos rfl,ite_true]
        rw [Finset.prod_mul_distrib]
        simp only [Finset.prod_const,Finset.card_univ,Fintype.card_fin]
        rfl
      · simp only [W,allpoint_balanced_values4,if_neg ht]
    simp_rw [hfactor]
    simp only [Fin.prod_univ_three,Fin.isValue,Fin.reduceEq,ite_true,ite_false]
    rw [hrpow]
    have hp := hprod
    simp only [Fin.prod_univ_three] at hp
    nlinarith only [hp]
  have heq := sourceLogWeights4_rpow Q hQ W hW
  refine ⟨hQ,sourceLogWeights4_conditions Q hQ W hW hcut hcut0 hWfin hWprod,heq,?_,?_⟩
  · intro i
    rw [Real.rpow_sub hQ0,heq]
    change a 0 i = (a 0 i*r)/r
    exact (mul_div_cancel_right₀ _ hr.ne').symm
  · intro tau ht i
    rw [heq]
    simp only [W,allpoint_balanced_values4,if_neg ht]

end

def allpoint_no_finite_cover4 (E : (Fin 4 → ℤ) → Prop) : Prop :=
  ¬ ∃ F : Set (Submodule ℚ (Fin 4 → ℚ)), F.Finite ∧ (∀ P ∈ F, P ≠ ⊤) ∧
    ∀ z, E z → ∃ P ∈ F, (fun i => (z i : ℚ)) ∈ P

theorem allpoint_avoid_finite4 (E : (Fin 4 → ℤ) → Prop) (hno : allpoint_no_finite_cover4 E)
    (F : Finset (Submodule ℚ (Fin 4 → ℚ))) (hF : ∀ P ∈ F, P ≠ ⊤) :
    ∃ z, E z ∧ ∀ P ∈ F, (fun i => (z i : ℚ)) ∉ P := by
  classical
  by_contra h
  push_neg at h
  exact hno ⟨(F : Set (Submodule ℚ (Fin 4 → ℚ))),F.finite_toSet,hF,h⟩

theorem allpoint_general_position_step4 (E : (Fin 4 → ℤ) → Prop) (hno : allpoint_no_finite_cover4 E)
    (n : ℕ) (v : Fin n → (Fin 4 → ℤ)) :
    ∃ z, E z ∧ max 4 (max (n : ℝ) ((∑ i, (archFourHeight (v i) : ℝ))+1)) < (archFourHeight z : ℝ) ∧
      ∀ i j k : Fin n, (fun l => (z l : ℚ)) ∉ Submodule.span ℚ
        {fun l => (v i l : ℚ),fun l => (v j l : ℚ),fun l => (v k l : ℚ)} := by
  classical
  let M := max 4 (max (n : ℝ) ((∑ i, (archFourHeight (v i) : ℝ))+1))
  let pairs : Finset (Submodule ℚ (Fin 4 → ℚ)) := Finset.univ.image
    (fun ij : Fin n × Fin n × Fin n => Submodule.span ℚ {fun k => (v ij.1 k : ℚ),fun k => (v ij.2.1 k : ℚ),fun k => (v ij.2.2 k : ℚ)})
  let low := (allpoint_bounded_height_finite4 M).toFinset.image
    (fun z => Submodule.span ℚ {fun k => (z k : ℚ),fun k => (z k : ℚ),fun k => (z k : ℚ)})
  have hproper : ∀ P ∈ pairs ∪ low, P ≠ ⊤ := by
    intro P hP
    rcases Finset.mem_union.mp hP with hp | hp
    · obtain ⟨ij,_,rfl⟩ := Finset.mem_image.mp hp
      exact allpoint_triple_proper4 _ _ _
    · obtain ⟨z,_,rfl⟩ := Finset.mem_image.mp hp
      exact allpoint_triple_proper4 _ _ _
  obtain ⟨z,hz,havoid⟩ := allpoint_avoid_finite4 E hno (pairs ∪ low) hproper
  refine ⟨z,hz,?_,?_⟩
  · by_contra h
    have hb : (archFourHeight z : ℝ) ≤ M := le_of_not_gt h
    have hlow : Submodule.span ℚ {fun k => (z k : ℚ),fun k => (z k : ℚ),fun k => (z k : ℚ)} ∈ low := by
      apply Finset.mem_image.mpr
      exact ⟨z,by simpa using hb,rfl⟩
    apply havoid _ (Finset.mem_union_right _ hlow)
    exact Submodule.subset_span (by simp)
  · intro i j k
    apply havoid
    apply Finset.mem_union_left
    exact Finset.mem_image.mpr ⟨(i,j,k),Finset.mem_univ _,rfl⟩

theorem allpoint_general_position_sequence4 (E : (Fin 4 → ℤ) → Prop) (hno : allpoint_no_finite_cover4 E) :
    ∃ z : ℕ → (Fin 4 → ℤ), (∀ n, E (z n)) ∧
      (∀ n, 4 < (archFourHeight (z n) : ℝ) ∧ (n : ℝ) < (archFourHeight (z n) : ℝ)) ∧
      StrictMono (fun n => (archFourHeight (z n) : ℝ)) ∧
      ∀ n, ∀ i j k : Fin n, (fun k => (z n k : ℚ)) ∉ Submodule.span ℚ
        {fun l => (z i.val l : ℚ),fun l => (z j.val l : ℚ),fun l => (z k.val l : ℚ)} := by
  classical
  choose next hE hheight havoid using allpoint_general_position_step4 E hno
  let step (n : ℕ) (prev : ∀ m, m < n → (Fin 4 → ℤ)) : (Fin 4 → ℤ) := next n (fun i => prev i.val i.isLt)
  let z (n : ℕ) : (Fin 4 → ℤ) := Nat.strongRecOn n step
  have hrec (n : ℕ) : z n = next n (fun i => z i.val) := by
    change Nat.lt_wfRel.wf.fix step n = _
    rw [WellFounded.fix_eq]
    rfl
  have hspec (n : ℕ) : E (z n) ∧
      max 4 (max (n : ℝ) ((∑ i : Fin n, (archFourHeight (z i.val) : ℝ))+1)) < (archFourHeight (z n) : ℝ) ∧
      ∀ i j k : Fin n, (fun k => (z n k : ℚ)) ∉ Submodule.span ℚ
        {fun l => (z i.val l : ℚ),fun l => (z j.val l : ℚ),fun l => (z k.val l : ℚ)} := by
    have h := And.intro (hE n (fun i => z i.val))
      (And.intro (hheight n (fun i => z i.val)) (havoid n (fun i => z i.val)))
    rw [← hrec n] at h
    exact h
  refine ⟨z,fun n => (hspec n).1,?_,?_,fun n => (hspec n).2.2⟩
  · intro n
    exact ⟨lt_of_le_of_lt (le_max_left _ _) (hspec n).2.1,
      lt_of_le_of_lt ((le_max_left _ _).trans (le_max_right _ _)) (hspec n).2.1⟩
  · intro i j hij
    have hnon (l : Fin j) : 0 ≤ (archFourHeight (z l.val) : ℝ) := by
      unfold archFourHeight
      positivity
    have hh := Finset.single_le_sum (fun l _ => hnon l) (Finset.mem_univ (⟨i,hij⟩ : Fin j))
    have hb : (archFourHeight (z i) : ℝ) < (∑ l : Fin j, (archFourHeight (z l.val) : ℝ))+1 := by linarith
    exact hb.trans (lt_of_le_of_lt ((le_max_right _ _).trans (le_max_right _ _)) (hspec j).2.1)



theorem allpoint_general_position_intersection4 (z : ℕ → Fin 4 → ℤ)
    (hH : ∀ n,4<(archFourHeight (z n) : ℝ))
    (havoid : ∀ n,∀ i j k : Fin n,(fun l => (z n l : ℚ))∉Submodule.span ℚ
      {fun l => (z i.val l : ℚ),fun l => (z j.val l : ℚ),fun l => (z k.val l : ℚ)})
    (P : Submodule ℚ (Fin 4 → ℚ)) (hP : P≠⊤) :
    {n | (fun i => (z n i : ℚ))∈P}.Finite := by
  classical
  let v (n : ℕ) : Fin 4 → ℚ := fun i => (z n i : ℚ)
  have hn (n : ℕ) : v n≠0 := by
    intro h
    have hh := hH n
    have hc (i : Fin 4) : (z n i : ℚ)=0 := by simpa only [v,Pi.zero_apply] using congrFun h i
    norm_num [archFourHeight,hc] at hh
  have hbad (i j k l : ℕ) (hij : i<j) (hjk : j<k) (hkl : k<l)
      (hi : v i∈P) (hj : v j∈P) (hk : v k∈P) (hl : v l∈P) : False := by
    have hnotj : v j∉Submodule.span ℚ {v i} := by
      simpa only [v,Set.insert_eq_of_mem (Set.mem_singleton _)] using
        havoid j ⟨i,hij⟩ ⟨i,hij⟩ ⟨i,hij⟩
    have hpair : LinearIndependent ℚ (![v j,v i] : Fin 2 → Fin 4 → ℚ) := by
      apply linearIndependent_fin2.mpr
      refine ⟨hn i,?_⟩
      intro a ha
      exact hnotj (Submodule.mem_span_singleton.mpr ⟨a,ha⟩)
    have hr2 : Set.range (![v j,v i] : Fin 2 → Fin 4 → ℚ)={v j,v i} := by
      ext x
      simp [Fin.exists_fin_two,or_comm,eq_comm]
    have hnotk : v k∉Submodule.span ℚ (Set.range (![v j,v i] : Fin 2 → Fin 4 → ℚ)) := by
      rw [hr2]
      simpa only [v,Set.insert_idem] using
        havoid k ⟨j,hjk⟩ ⟨j,hjk⟩ ⟨i,lt_trans hij hjk⟩
    have hthree : LinearIndependent ℚ (![v k,v j,v i] : Fin 3 → Fin 4 → ℚ) :=
      linearIndependent_fin_cons.mpr ⟨hpair,hnotk⟩
    have hr3 : Set.range (![v k,v j,v i] : Fin 3 → Fin 4 → ℚ)={v k,v j,v i} := by
      ext x
      simp [Fin.exists_fin_succ,or_comm,or_left_comm,or_assoc,eq_comm]
    have hnotl : v l∉Submodule.span ℚ (Set.range (![v k,v j,v i] : Fin 3 → Fin 4 → ℚ)) := by
      rw [hr3]
      exact havoid l ⟨k,hkl⟩ ⟨j,lt_trans hjk hkl⟩ ⟨i,lt_trans hij (lt_trans hjk hkl)⟩
    have hfour : LinearIndependent ℚ (![v l,v k,v j,v i] : Fin 4 → Fin 4 → ℚ) :=
      linearIndependent_fin_cons.mpr ⟨hthree,hnotl⟩
    have htop := hfour.span_eq_top_of_card_eq_finrank (by simp)
    have hle : Submodule.span ℚ (Set.range (![v l,v k,v j,v i] : Fin 4 → Fin 4 → ℚ))≤P := by
      apply Submodule.span_le.mpr
      rintro _ ⟨a,rfl⟩
      fin_cases a
      · exact hl
      · exact hk
      · exact hj
      · exact hi
    rw [htop] at hle
    exact hP (top_unique hle)
  let S : Set ℕ := {n | v n∈P}
  have hbound : BddAbove S := by
    apply Classical.byContradiction
    intro h
    obtain ⟨i,hi,_⟩ := not_bddAbove_iff.mp h 0
    obtain ⟨j,hj,hij⟩ := not_bddAbove_iff.mp h i
    obtain ⟨k,hk,hjk⟩ := not_bddAbove_iff.mp h j
    obtain ⟨l,hl,hkl⟩ := not_bddAbove_iff.mp h k
    exact hbad i j k l hij hjk hkl hi hj hk hl
  obtain ⟨N,hN⟩ := hbound
  exact (Set.finite_Iic N).subset (fun n hn => hN hn)
