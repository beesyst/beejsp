def Two : ℚˣ := Units.mk0 2 (by norm_num)
def Three : ℚˣ := Units.mk0 3 (by norm_num)
def Group : Subgroup ℚˣ := Subgroup.closure ({-1,Two,Three} : Set ℚˣ)
def Nondeg {n : ℕ} (x : Fin n → ℚ) : Prop :=
  ∀ I : Finset (Fin n),I.Nonempty → I≠Finset.univ → ∑ i∈I,x i≠0
def Product (x : ℚ) : ℚ := |x| *padicNorm 2 x*padicNorm 3 x
def Raw_WeightedFourTermSolutions (A B C D : ℚ) : Set (Group × Group × Group) :=
  {p | A+B*((p.1 : ℚˣ) : ℚ)+C*((p.2.1 : ℚˣ) : ℚ)+D*((p.2.2 : ℚˣ) : ℚ)=0 ∧
    Nondeg ![A,B*((p.1 : ℚˣ) : ℚ),C*((p.2.1 : ℚˣ) : ℚ),D*((p.2.2 : ℚˣ) : ℚ)]}

def Raw_weightedRetainedMatrix3 (B C D : ℚ) : Matrix (Fin 3) (Fin 3) ℚ :=
  !![B,C,D;0,1,0;0,0,1]

def Raw_weightedNormalizedMatrix3 (B C D : ℚ) : Matrix (Fin 3) (Fin 3) ℚ :=
  !![1,C/B,D/B;0,1,0;0,0,1]

theorem source_weighted_retained_det3 : ∀ (B C D : ℚ),
 (Raw_weightedRetainedMatrix3 B C D).det=B :=
  @JSP000334.weighted_retained_det3

theorem source_weighted_normalized_det3 : ∀ (B C D : ℚ),

  @JSP000334.weighted_normalized_det3

theorem source_weighted_normalized_apply3 : ∀ (B C D : ℚ) (hB : B≠0) (z : Fin 3 → ℚ),

    (Raw_weightedNormalizedMatrix3 B C D *ᵥ z) 0=(B*z 0+C*z 1+D*z 2)/B :=
  @JSP000334.weighted_normalized_apply3

theorem source_weighted_scaled_relation3 : ∀ (A B C D : ℚ) (u v w s : Group)
    (z : Fin 3 → ℤ)
    (he : A+B*((u : ℚˣ) : ℚ)+C*((v : ℚˣ) : ℚ)+D*((w : ℚˣ) : ℚ)=0)
    (hz : ∀ i,(z i : ℚ)=((s : ℚˣ) : ℚ)*(![((u : ℚˣ) : ℚ),((v : ℚˣ) : ℚ),((w : ℚˣ) : ℚ)] i)),

    B*(z 0 : ℚ)+C*(z 1 : ℚ)+D*(z 2 : ℚ)=-A*((s : ℚˣ) : ℚ) :=
  @JSP000334.weighted_scaled_relation3

theorem source_weighted_three_place_product3 : ∀ (A B C D : ℚ) (u v w s : Group)
    (z : Fin 3 → ℤ)
    (he : A+B*((u : ℚˣ) : ℚ)+C*((v : ℚˣ) : ℚ)+D*((w : ℚˣ) : ℚ)=0)
    (hz : ∀ i,(z i : ℚ)=((s : ℚˣ) : ℚ)*(![((u : ℚˣ) : ℚ),((v : ℚˣ) : ℚ),((w : ℚˣ) : ℚ)] i)),

    Product (B*(z 0 : ℚ)+C*(z 1 : ℚ)+D*(z 2 : ℚ))=Product A :=
  @JSP000334.weighted_three_place_product3

theorem source_weighted_coordinate_product3 : ∀ (s u : Group) (m : ℤ)
    (hm : (m : ℚ)=((s : ℚˣ) : ℚ)*((u : ℚˣ) : ℚ)),
