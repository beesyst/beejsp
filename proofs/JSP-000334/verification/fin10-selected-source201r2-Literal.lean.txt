theorem expanded_three_factor (x0 x2 x3 b0 b2 b3 : ℝ)
    (hx0 : 0 ≤ x0) (hx2 : 0 ≤ x2) (hx3 : 0 ≤ x3)
    (h0 : x0 ≤ b0) (h2 : x2 ≤ b2) (h3 : x3 ≤ b3) :
    x0*x2*x3 ≤ b0*b2*b3 := by
  calc
    x0*x2*x3 ≤ (b0*x2)*x3 := mul_le_mul_of_nonneg_right
      (mul_le_mul_of_nonneg_right h0 hx2) hx3
    _ ≤ (b0*b2)*x3 := mul_le_mul_of_nonneg_right
      (mul_le_mul_of_nonneg_left h2 (hx0.trans h0)) hx3
    _ ≤ (b0*b2)*b3 := mul_le_mul_of_nonneg_left h3
      (mul_nonneg (hx0.trans h0) (hx2.trans h2))
