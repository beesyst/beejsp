theorem spanning_five245P (v : Fin 10 → Fin 5 → ℝ)
    (hs : Submodule.span ℝ (Set.range v) = ⊤) :
    ∃ a : Fin 5 → Fin 10, LinearIndependent ℝ (fun i => v (a i)) := by
  classical
  obtain ⟨t,ht,hspan,hli⟩ := exists_linearIndependent ℝ (Set.range v)
  have hfinite : t.Finite := (Set.finite_range v).subset ht
  letI : Fintype t := hfinite.fintype
  let b : Basis t ℝ (Fin 5 → ℝ) := Basis.mk hli (by rw [Subtype.range_val,hspan,hs])
  have hc : Fintype.card t = 5 := by
    have h := Module.finrank_eq_card_basis b
    simpa using h.symm
  let e : t ≃ Fin 5 := Fintype.equivFinOfCardEq hc
  have hex (x : t) : ∃ j, v j = (x : Fin 5 → ℝ) := ht x.property
  choose a ha using hex
  refine ⟨fun i => a (e.symm i),?_⟩
  have hind := hli.comp e.symm e.symm.injective
  convert hind using 1
  funext i
  exact ha (e.symm i)

theorem projected_frame245P (p : (Fin 10 → ℝ) →ₗ[ℝ] (Fin 5 → ℝ))
    (hp : Function.Surjective p) (w : Fin 10 → Fin 10 → ℝ)
    (hw : LinearIndependent ℝ w) :
    ∃ a : Fin 5 → Fin 10, LinearIndependent ℝ (fun i => p (w (a i))) := by
  have hs : Submodule.span ℝ (Set.range w) = ⊤ := by
    simpa only [coe_basisOfPiSpaceOfLinearIndependent] using
      (basisOfPiSpaceOfLinearIndependent hw).span_eq
  apply spanning_five245P (fun j => p (w j))
  rw [show Set.range (fun j => p (w j)) = p '' Set.range w from Set.range_comp p w,
    ← Submodule.map_span,hs,Submodule.map_top,LinearMap.range_eq_top.mpr hp]

end CompoundFeasibility.LiteralBlock245P

def blockIndex245P : Fin 5 ⊕ Fin 5 ≃ Fin 10 := finSumFinEquiv

def block245P {R : Type*} [CommRing R] (A : Matrix (Fin 5) (Fin 5) R) :
    Matrix (Fin 10) (Fin 10) R :=
  Matrix.reindex blockIndex245P blockIndex245P (Matrix.fromBlocks A 0 0 1)

def weight0245P (c : Fin 5 → ℝ) (i : Fin 10) : ℝ :=
  Sum.elim (fun j => (c j-2)/3) (fun _ => 2/3) (blockIndex245P.symm i)

def weightp245P (c : Fin 5 → ℝ) (i : Fin 10) : ℝ :=
  Sum.elim (fun j => c j/3) (fun _ => 0) (blockIndex245P.symm i)

theorem weight_bounds245P (c0 c2 c3 : Fin 5 → ℝ)
    (h0 : ∀ i, |c0 i| ≤ 1) (h2 : ∀ i, -1 ≤ c2 i ∧ c2 i ≤ 0)
    (h3 : ∀ i, -1 ≤ c3 i ∧ c3 i ≤ 0) :
    (∀ i, |weight0245P c0 i| ≤ 1) ∧
    (∀ i, -1 ≤ weightp245P c2 i ∧ weightp245P c2 i ≤ 0) ∧
    (∀ i, -1 ≤ weightp245P c3 i ∧ weightp245P c3 i ≤ 0) := by
  have hreal (i : Fin 10) : |weight0245P c0 i| ≤ 1 := by
    unfold weight0245P
    cases h : blockIndex245P.symm i with
    | inl j => simp only [Sum.elim_inl]; exact abs_le.mpr ⟨by linarith [(abs_le.mp (h0 j)).1],by linarith [(abs_le.mp (h0 j)).2]⟩
    | inr j => apply abs_le.mpr; constructor <;> norm_num
  have hfinite (c : Fin 5 → ℝ) (hc : ∀ i, -1 ≤ c i ∧ c i ≤ 0) (i : Fin 10) :
      -1 ≤ weightp245P c i ∧ weightp245P c i ≤ 0 := by
    unfold weightp245P
    cases h : blockIndex245P.symm i with
    | inl j => simp only [Sum.elim_inl]; constructor <;> linarith [(hc j).1,(hc j).2]
    | inr j => norm_num
  exact ⟨hreal,hfinite c2 h2,hfinite c3 h3⟩

theorem weight_balance245P (c0 c2 c3 : Fin 5 → ℝ)
    (hb : (∑ i, (c0 i+c2 i+c3 i)) = 0) :
    (∑ i, (weight0245P c0 i+weightp245P c2 i+weightp245P c3 i)) = 0 := by
  rw [← Equiv.sum_comp blockIndex245P]
  simp only [weight0245P,weightp245P,Equiv.symm_apply_apply,Fintype.sum_sum_type,
    Sum.elim_inl,Sum.elim_inr]
  simp only [Finset.sum_add_distrib,Finset.sum_sub_distrib,Finset.sum_div,
    Finset.sum_const,Finset.card_univ,Fintype.card_fin] at hb
  simp only [Finset.sum_add_distrib,Finset.sum_sub_distrib,← Finset.sum_div,
    Finset.sum_const,Finset.card_univ,Fintype.card_fin,nsmul_eq_mul]
  linarith

end CompoundFeasibility.LiteralBlock245P

theorem actual_frame_projection (B2 B3 : Matrix (Fin 10) (Fin 10) ℚ)
    (B0 : Matrix (Fin 10) (Fin 10) ℝ) (hB0 : IsUnit B0)
    (p : ℝ) (hp : 1 < p) (d0 d2 d3 : Fin 10 → ℝ) :
    ∃ w : Fin 10 → originalSourceGroup10 B2 B3 p d2 d3,
      LinearIndependent ℝ (fun i => integerReal (w i : Fin 10 → ℤ)) ∧
      (∀ i, coordinateGauge10 (originalCoordinates10 B0 hB0 p hp d0)
        (integerReal (w i : Fin 10 → ℤ)) = sourceMinima10 B2 B3 B0 hB0 p hp d0 d2 d3 i) ∧
      ∃ a : Fin 5 → Fin 10, LinearIndependent ℝ
        (fun i j => integerReal (w (a i) : Fin 10 → ℤ) (blockIndex245P (Sum.inl j))) := by
  let pr : (Fin 10 → ℝ) →ₗ[ℝ] (Fin 5 → ℝ) :=
    LinearMap.pi (fun j => LinearMap.proj (blockIndex245P (Sum.inl j)))
  have hpr : Function.Surjective pr := by
    intro x
    refine ⟨fun j => Sum.elim x (fun _ => 0) (blockIndex245P.symm j),?_⟩
    ext j
    simp [pr]
  obtain ⟨w,hw,_,_,hatt,_⟩ := exists_sourceMinima10_attaining B2 B3 B0 hB0 p hp d0 d2 d3
  obtain ⟨a,ha⟩ := projected_frame245P pr hpr (fun j => integerReal (w j : Fin 10 → ℤ)) hw
  exact ⟨w,hw,hatt,a,ha⟩

theorem normalized_source_weights (c0 c2 c3 : Fin 5 → ℝ)
    (h0 : ∀ i, |c0 i| ≤ 1) (h2 : ∀ i, -1 ≤ c2 i ∧ c2 i ≤ 0)
    (h3 : ∀ i, -1 ≤ c3 i ∧ c3 i ≤ 0) (hb : (∑ i, (c0 i+c2 i+c3 i)) = 0) :
    (∀ i, |weight0245P c0 i| ≤ 1) ∧
    (∀ i, -1 ≤ weightp245P c2 i ∧ weightp245P c2 i ≤ 0) ∧
    (∀ i, -1 ≤ weightp245P c3 i ∧ weightp245P c3 i ≤ 0) ∧
    (∑ i, (weight0245P c0 i+weightp245P c2 i+weightp245P c3 i)) = 0 := by
  obtain ⟨h0,h2,h3⟩ := weight_bounds245P c0 c2 c3 h0 h2 h3
  exact ⟨h0,h2,h3,weight_balance245P c0 c2 c3 hb⟩
