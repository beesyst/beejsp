    (P : Submodule K (Fin 5 → K)) (b : Basis (Fin d) K P)

  fun i k => (A *ᵥ (b k : Fin 5 → K)) i

theorem rowCoordinates5_span {K : Type*} [Field K] {d : ℕ}
    (P : Submodule K (Fin 5 → K)) (b : Basis (Fin d) K P)
    (A : Matrix (Fin 5) (Fin 5) K) (hA : IsUnit A) :
    Submodule.span K (Set.range (rowCoordinates5 P b A).row) = ⊤ := by
  let e := LinearEquiv.ofBijective (Matrix.toLin' A)
    ⟨Matrix.mulVec_injective_iff_isUnit.mpr hA,Matrix.mulVec_surjective_iff_isUnit.mpr hA⟩
  have hli : LinearIndependent K (fun k => (A *ᵥ (b k : Fin 5 → K))) :=
    (b.linearIndependent.map' P.subtype P.ker_subtype).map' e.toLinearMap e.ker
  have hcol : LinearIndependent K (rowCoordinates5 P b A).transpose.row := hli
  have hrank : (rowCoordinates5 P b A).rank = d := by
    rw [← Matrix.rank_transpose]
    simpa only [Fintype.card_fin] using hcol.rank_matrix
  apply Submodule.eq_top_of_finrank_eq
  rw [← Matrix.rank_eq_finrank_span_row,hrank,Module.finrank_pi]
  simp

variable {K V : Type*} [Field K] [AddCommGroup V] [Module K V]

def rowPrefix5 (rows : Fin 5 → V) (n : ℕ) : Set V :=
  rows '' {i : Fin 5 | i.val < n}

def greedyPivot5 (rows : Fin 5 → V) (i : Fin 5) : Prop :=
  rows i ∉ Submodule.span K (rowPrefix5 rows i.val)

def greedyPrefix5 (rows : Fin 5 → V) (n : ℕ) : Set (Fin 5) :=
  {i | i.val < n ∧ greedyPivot5 (K := K) rows i}

theorem rowPrefix5_succ (rows : Fin 5 → V) (n : ℕ) (hn : n < 5) :
    rowPrefix5 rows (n+1) = insert (rows ⟨n,hn⟩) (rowPrefix5 rows n) := by
  have hs : {i : Fin 5 | i.val < n+1} = insert ⟨n,hn⟩ {i : Fin 5 | i.val < n} := by
    ext i
    simp only [Set.mem_setOf_eq,Set.mem_insert_iff,Fin.ext_iff]
    omega
  unfold rowPrefix5
  rw [hs,Set.image_insert_eq]

theorem greedyPrefix5_succ (rows : Fin 5 → V) (n : ℕ) (hn : n < 5) :
    greedyPrefix5 (K := K) rows (n+1) =
      if greedyPivot5 (K := K) rows ⟨n,hn⟩ then
        insert ⟨n,hn⟩ (greedyPrefix5 (K := K) rows n)
      else greedyPrefix5 (K := K) rows n := by
  classical
  ext k
  by_cases hki : k = ⟨n,hn⟩
  · subst k
    by_cases hp : greedyPivot5 (K := K) rows ⟨n,hn⟩ <;> simp [greedyPrefix5,hp]
  · have hv : k.val < n+1 ↔ k.val < n := by
      have hne : k.val ≠ n := fun h => hki (Fin.ext h)
      omega
    by_cases hp : greedyPivot5 (K := K) rows ⟨n,hn⟩ <;> simp [greedyPrefix5,hp,hki,hv]

theorem greedyPrefix5_independent_span (rows : Fin 5 → V) :
    ∀ n : ℕ, n ≤ 5 →
      LinearIndepOn K rows (greedyPrefix5 (K := K) rows n) ∧
      Submodule.span K (rows '' greedyPrefix5 (K := K) rows n) =
        Submodule.span K (rowPrefix5 rows n) := by
  classical
  intro n
  induction n with
  | zero => simp [greedyPrefix5,rowPrefix5]
  | succ n ih =>
    intro hn
    have hnlt : n < 5 := by omega
    obtain ⟨hli,hspan⟩ := ih (by omega)
    rw [greedyPrefix5_succ rows n hnlt,rowPrefix5_succ rows n hnlt]
    have hnot : (⟨n,hnlt⟩ : Fin 5) ∉ greedyPrefix5 (K := K) rows n := by
      simp [greedyPrefix5]
    by_cases hp : greedyPivot5 (K := K) rows ⟨n,hnlt⟩
    · rw [if_pos hp,Set.image_insert_eq]
      have hout : rows ⟨n,hnlt⟩ ∉ Submodule.span K (rows '' greedyPrefix5 (K := K) rows n) := by
        rw [hspan]
        exact hp
      exact ⟨(linearIndepOn_insert hnot).mpr ⟨hli,hout⟩,by rw [Submodule.span_insert,hspan,Submodule.span_insert]⟩
    · rw [if_neg hp]
      have hmem : rows ⟨n,hnlt⟩ ∈ Submodule.span K (rowPrefix5 rows n) := not_not.mp hp
      exact ⟨hli,by rw [Submodule.span_insert_eq_span hmem];exact hspan⟩

def greedyFinset5 (rows : Fin 5 → V) : Finset (Fin 5) :=
  Finset.univ.filter (greedyPivot5 (K := K) rows)

theorem greedyFinset5_eq_prefix (rows : Fin 5 → V) :
    (greedyFinset5 (K := K) rows : Set (Fin 5)) = greedyPrefix5 (K := K) rows 5 := by
  ext i
  simp [greedyFinset5,greedyPrefix5,i.isLt]

theorem greedyFinset5_independent_span (rows : Fin 5 → V)
    (hfull : Submodule.span K (Set.range rows) = ⊤) :
    LinearIndepOn K rows (greedyFinset5 (K := K) rows : Set (Fin 5)) ∧
      Submodule.span K (rows '' (greedyFinset5 (K := K) rows : Set (Fin 5))) = ⊤ := by
  rw [greedyFinset5_eq_prefix]
  obtain ⟨hi,hs⟩ := greedyPrefix5_independent_span (K := K) rows 5 (by omega)
  have hall : rowPrefix5 rows 5 = Set.range rows := by
    simp [rowPrefix5,show {i : Fin 5 | i.val < 5} = Set.univ by ext i; simp [i.isLt]]
  exact ⟨hi,hs.trans (hall ▸ hfull)⟩

theorem greedyFinset5_card [FiniteDimensional K V] (rows : Fin 5 → V)
    (hfull : Submodule.span K (Set.range rows) = ⊤) :
    (greedyFinset5 (K := K) rows).card = Module.finrank K V := by
  obtain ⟨hli,hspan⟩ := greedyFinset5_independent_span (K := K) rows hfull
  have hrange : Set.range (fun i : greedyFinset5 (K := K) rows => rows i.val) =
      rows '' (greedyFinset5 (K := K) rows : Set (Fin 5)) := by
    ext x
    constructor
    · rintro ⟨i,rfl⟩
      exact ⟨i.val,i.property,rfl⟩
    · rintro ⟨i,hi,rfl⟩
      exact ⟨⟨i,hi⟩,rfl⟩
  have hi : LinearIndependent K (fun i : greedyFinset5 (K := K) rows => rows i.val) := hli
  let b : Basis (greedyFinset5 (K := K) rows) K V :=
    Basis.mk hi (by rw [hrange,hspan])
  simpa using (Module.finrank_eq_card_basis b).symm

def greedyIndex5 [FiniteDimensional K V] (rows : Fin 5 → V)
    (hfull : Submodule.span K (Set.range rows) = ⊤) (d : ℕ) (hdim : Module.finrank K V = d) :
    Fin d ↪o Fin 5 :=
  (greedyFinset5 (K := K) rows).orderEmbOfFin ((greedyFinset5_card (K := K) rows hfull).trans hdim)

theorem greedyIndex5_pivot [FiniteDimensional K V] (rows : Fin 5 → V)
    (hfull : Submodule.span K (Set.range rows) = ⊤) (d : ℕ) (hdim : Module.finrank K V = d)
    (k : Fin d) : greedyPivot5 (K := K) rows (greedyIndex5 rows hfull d hdim k) := by
  have h := (greedyFinset5 (K := K) rows).orderEmbOfFin_mem
    ((greedyFinset5_card (K := K) rows hfull).trans hdim) k
  exact (Finset.mem_filter.mp h).2

theorem greedyIndex5_independent [FiniteDimensional K V] (rows : Fin 5 → V)
    (hfull : Submodule.span K (Set.range rows) = ⊤) (d : ℕ) (hdim : Module.finrank K V = d) :
    LinearIndependent K (fun k => rows (greedyIndex5 rows hfull d hdim k)) := by
  have hli := (greedyFinset5_independent_span (K := K) rows hfull).1
  let f : Fin d → greedyFinset5 (K := K) rows := fun k =>
    ⟨greedyIndex5 rows hfull d hdim k,
      (greedyFinset5 (K := K) rows).orderEmbOfFin_mem ((greedyFinset5_card (K := K) rows hfull).trans hdim) k⟩
  have hf : Function.Injective f := fun k l h =>
    (greedyIndex5 rows hfull d hdim).injective (congrArg Subtype.val h)
  exact hli.comp f hf

end

theorem greedyIndex5_prefix_set (rows : Fin 5 → V)
    (hfull : Submodule.span K (Set.range rows) = ⊤) (d : ℕ) (hdim : Module.finrank K V = d)
    (k : Fin d) :
    greedyPrefix5 (K := K) rows (greedyIndex5 rows hfull d hdim k).val =
      Set.range (fun a : Fin k.val => greedyIndex5 rows hfull d hdim (Fin.castLE (by omega) a)) := by
  let E := greedyIndex5 rows hfull d hdim
  have hrange : Set.range E = (greedyFinset5 (K := K) rows : Set (Fin 5)) :=
    (greedyFinset5 (K := K) rows).range_orderEmbOfFin ((greedyFinset5_card (K := K) rows hfull).trans hdim)
  ext i
  constructor
  · intro hi
    have himem : i ∈ (greedyFinset5 (K := K) rows : Set (Fin 5)) :=
      Finset.mem_filter.mpr ⟨Finset.mem_univ _,hi.2⟩
    rw [← hrange] at himem
    obtain ⟨a,rfl⟩ := himem
    have ha : a < k := E.lt_iff_lt.mp hi.1
    refine ⟨⟨a.val,ha⟩,?_⟩
    congr 1
  · rintro ⟨a,rfl⟩
    exact ⟨E.strictMono (show Fin.castLE (by omega) a < k from a.isLt),
      greedyIndex5_pivot rows hfull d hdim _⟩

theorem greedyIndex5_prefix_rank (rows : Fin 5 → V)
    (hfull : Submodule.span K (Set.range rows) = ⊤) (d : ℕ) (hdim : Module.finrank K V = d)
    (k : Fin d) :
    Module.finrank K (Submodule.span K (rowPrefix5 rows (greedyIndex5 rows hfull d hdim k).val)) = k.val := by
  have hs := (greedyPrefix5_independent_span (K := K) rows
    (greedyIndex5 rows hfull d hdim k).val (by omega)).2
  rw [← hs,greedyIndex5_prefix_set rows hfull d hdim k,← Set.range_comp]
  have hli := (greedyIndex5_independent rows hfull d hdim).comp
    (fun a : Fin k.val => Fin.castLE (by omega) a) (Fin.castLE_injective (by omega))
  simpa only [Fintype.card_fin] using finrank_span_eq_card hli

theorem greedyIndex5_first_rank_increase (rows : Fin 5 → V)
    (hfull : Submodule.span K (Set.range rows) = ⊤) (d : ℕ) (hdim : Module.finrank K V = d)
    (k : Fin d) :
    Module.finrank K (Submodule.span K (rowPrefix5 rows
      ((greedyIndex5 rows hfull d hdim k).val+1))) = k.val+1 ∧
    ∀ i : Fin 5, i < greedyIndex5 rows hfull d hdim k →
      Module.finrank K (Submodule.span K (rowPrefix5 rows (i.val+1))) ≤ k.val := by
  let E := greedyIndex5 rows hfull d hdim
  have hp := greedyIndex5_pivot rows hfull d hdim k
  have hprev := greedyIndex5_prefix_rank rows hfull d hdim k
  have hnonzero : rows (E k) ≠ 0 := by
    intro hz
    apply hp
    rw [hz]
    exact Submodule.zero_mem _
  have hdisjoint := Submodule.disjoint_span_singleton_of_not_mem hp
  have hdimstep := Submodule.finrank_sup_add_finrank_inf_eq
    (Submodule.span K (rowPrefix5 rows (E k).val)) (K ∙ rows (E k))
  rw [hdisjoint.eq_bot,finrank_bot,add_zero,finrank_span_singleton hnonzero,hprev] at hdimstep
  constructor
  · rw [rowPrefix5_succ rows (E k).val (E k).isLt,Submodule.span_insert,sup_comm]
    exact hdimstep
  · intro i hi
    calc
      _ ≤ Module.finrank K (Submodule.span K (rowPrefix5 rows (E k).val)) := by
        apply Submodule.finrank_mono
        apply Submodule.span_mono
        rintro x ⟨a,ha,rfl⟩
        have hle : i.val+1 ≤ (E k).val := by
          change i.val < (E k).val at hi
          omega
        exact ⟨a,lt_of_lt_of_le ha hle,rfl⟩
      _ = _ := hprev

end

theorem rationalRealSpan5_mem_rational (P : Submodule ℚ (Fin 5 → ℚ)) (x : Fin 5 → ℚ) :
    (fun i => (x i : ℝ)) ∈ rationalRealSpan5 P ↔ x ∈ P := by
  constructor
  · intro hx
    obtain ⟨g,hg⟩ := P.subtype.exists_leftInverse_of_injective P.ker_subtype
    let f : (Fin 5 → ℚ) →ₗ[ℚ] (Fin 5 → ℚ) := P.subtype.comp g
    let M : Matrix (Fin 5) (Fin 5) ℚ := LinearMap.toMatrix' f
    let fR := Matrix.toLin' (M.map (algebraMap ℚ ℝ))
    have hfixQ : ∀ y ∈ P, M *ᵥ y = y := by
      intro y hy
      have h := congrArg (fun z : P => (z : Fin 5 → ℚ)) (LinearMap.congr_fun hg ⟨y,hy⟩)
      change (Matrix.toLin' (LinearMap.toMatrix' f)) y = y
      rw [Matrix.toLin'_toMatrix']
      exact h
    have hle : rationalRealSpan5 P ≤ LinearMap.ker (fR-LinearMap.id) := by
      apply Submodule.span_le.mpr
      rintro _ ⟨y,hy,rfl⟩
      change fR (fun i => (y i : ℝ)) - (fun i => (y i : ℝ)) = 0
      apply sub_eq_zero.mpr
      ext i
      have h := congrFun (hfixQ y hy) i
      change ∑ k, (M i k : ℝ)*(y k : ℝ) = (y i : ℝ)
      simp only [Matrix.mulVec,dotProduct] at h
      exact_mod_cast h
    have hfixR : fR (fun i => (x i : ℝ)) = (fun i => (x i : ℝ)) :=
      sub_eq_zero.mp (hle hx)
    have hfix : M *ᵥ x = x := by
      ext i
      have h := congrFun hfixR i
      change (∑ k, (M i k : ℝ)*(x k : ℝ)) = (x i : ℝ) at h
      simp only [Matrix.mulVec,dotProduct]
      exact_mod_cast h
    have hm : M *ᵥ x ∈ P := by
      change (Matrix.toLin' (LinearMap.toMatrix' f)) x ∈ P
      rw [Matrix.toLin'_toMatrix']
      exact (g x).property
    rwa [hfix] at hm
  · intro hx
    exact Submodule.subset_span ⟨x,hx,rfl⟩

end

theorem source_nonroth_selected_opposite_growth5

              sourceMinima5 A2 A3 L hL Q (hq Q (hWV hQW)) c0 c2 c3 (RestrictedGeometry5.ambientPrefixIndex5 P hPbounds.2 k)) ∧

def rawOriginalBody250 (L : Matrix (Fin 5) (Fin 5) ℝ) (Q : ℝ) (c₀ : Fin 5 → ℝ) : Set (Fin 5 → ℝ) :=
  {x | ∀ s, |(L *ᵥ x) s| ≤ Q ^ c₀ s}

def rawOriginalGroup250 (A₂ A₃ : Matrix (Fin 5) (Fin 5) ℚ) (Q : ℝ) (c₂ c₃ : Fin 5 → ℝ) : AddSubgroup (Fin 5 → ℤ) :=
  (⨅ i, (normBall 2 (Nat.ceil (JSP000334.qThreshold 2 Q (c₂ i)))).comap (row5 A₂ i)) ⊓
  (⨅ i, (normBall 3 (Nat.ceil (JSP000334.qThreshold 3 Q (c₃ i)))).comap (row5 A₃ i))
def rawExponent250 (P : Submodule ℚ (Fin 5 → ℚ)) (hP : Module.finrank ℚ P ≤ 4)

    (h₂ : A₂.det = 1) (h₃ : A₃.det = 1) (hL : L.det = 1) (c₀ c₂ c₃ : Fin 5 → ℝ) : ℝ :=

      (selectedIndex5 (rationalRowCoordinates5 P A₃) c₃ (rationalRowCoordinates5_span P A₃ h₃) k)))

def rawBody250 {d : ℕ} (P : Submodule ℚ (Fin 5 → ℚ))

def rawLocal250 {d : ℕ} (p : ℕ) [Fact p.Prime]

def rawGroup250 {d : ℕ} (A₂ A₃ : Matrix (Fin 5) (Fin 5) ℚ)

  rawLocal250 2 A₂ Q c₂ s₂ ⊓ rawLocal250 3 A₃ Q c₃ s₃

def rawOppositePoint250 (P : Submodule ℚ (Fin 5 → ℚ)) (hP : Module.finrank ℚ P ≤ 4)

  let s₀ := realSelection5 P hP L hL c₀
  let s₂ := finiteSelection5 P A₂ h₂ c₂
  let s₃ := finiteSelection5 P A₃ h₃ c₃
  let G := rawLattice250 P A₂ A₃ Q c₂ c₃ s₂ s₃

  Module.finrank ℤ G = Module.finrank ℚ P ∧
  0 < ZLattice.covolume G (restrictedMeasure5 P) ∧
  ZLattice.covolume G (restrictedMeasure5 P) ≤
    ZLattice.covolume (rawRestrictedLattice250 P A₂ A₃ Q c₂ c₃) (restrictedMeasure5 P) ∧
  (∀ z : Fin 5 → ℤ, z ∈ rawGroup250 A₂ A₃ Q c₂ c₃ s₂ s₃ ↔
    (∀ k, (padicNorm 2 ((A₂ *ᵥ (fun i => (z i : ℚ))) (s₂ k)) : ℝ) ≤ Q^(c₂ (s₂ k))) ∧
    (∀ k, (padicNorm 3 ((A₃ *ᵥ (fun i => (z i : ℚ))) (s₃ k)) : ℝ) ≤ Q^(c₃ (s₃ k)))) ∧
  (restrictedMeasure5 P).real (rawRestrictedBody250 P L Q c₀) /
    ZLattice.covolume (rawRestrictedLattice250 P A₂ A₃ Q c₂ c₃) (restrictedMeasure5 P) ≤
      (restrictedMeasure5 P).real C / ZLattice.covolume G (restrictedMeasure5 P) ∧
  (restrictedMeasure5 P).real C / ZLattice.covolume G (restrictedMeasure5 P) ≤
    K * Q^(rawExponent250 P hP A₂ A₃ L h₂ h₃ hL c₀ c₂ c₃) ∧
  (restrictedMeasure5 P).real (rawRestrictedBody250 P L Q c₀) /
    ZLattice.covolume (rawRestrictedLattice250 P A₂ A₃ Q c₂ c₃) (restrictedMeasure5 P) ≤
      K * Q^(rawExponent250 P hP A₂ A₃ L h₂ h₃ hL c₀ c₂ c₃)
