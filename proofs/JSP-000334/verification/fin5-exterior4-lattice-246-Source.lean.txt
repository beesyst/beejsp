theorem raw_two_place_source (A2 A3 : Matrix (Fin 5) (Fin 5) ℚ)
    (Q : ℝ) (c2 c3 : Fin 5 → ℝ) (x : Fin 4 → Fin 5 → ℤ)
    (hx2 : ∀ j i, padicNorm 2 ((A2 *ᵥ (fun k => (x j k : ℚ))) i) ≤
      (2 : ℚ)^(-(Nat.ceil (JSP000334.qThreshold 2 Q (c2 i)) : ℤ)))
    (hx3 : ∀ j i, padicNorm 3 ((A3 *ᵥ (fun k => (x j k : ℚ))) i) ≤
      (3 : ℚ)^(-(Nat.ceil (JSP000334.qThreshold 3 Q (c3 i)) : ℤ))) :
    (∀ i, padicNorm 2 ((A2.adjugate.transpose *ᵥ
      (fun k => (((Matrix.of (Fin.snoc x 0)).adjugate k 4 : ℤ) : ℚ))) i) ≤
      (2 : ℚ)^(-(Nat.ceil (JSP000334.qThreshold 2 (Q^4)
        ((∑ r : Fin 4, c2 (i.succAbove r))/4)) : ℤ))) ∧
    (∀ i, padicNorm 3 ((A3.adjugate.transpose *ᵥ
      (fun k => (((Matrix.of (Fin.snoc x 0)).adjugate k 4 : ℤ) : ℚ))) i) ≤
      (3 : ℚ)^(-(Nat.ceil (JSP000334.qThreshold 3 (Q^4)
        ((∑ r : Fin 4, c3 (i.succAbove r))/4)) : ℤ))) := by
  let u : Fin 4 → originalSourceGroup5 A2 A3 Q c2 c3 := fun j =>
    ⟨x j, ⟨(lattice5_spec 2 A2 _ (x j)).mpr (hx2 j),
      (lattice5_spec 3 A3 _ (x j)).mpr (hx3 j)⟩⟩
  have hm := wedge_source_group4246 A2 A3 Q c2 c3 u
  exact ⟨(lattice5_spec 2 (compound4246 A2) _ _).mp hm.1,
    (lattice5_spec 3 (compound4246 A3) _ _).mp hm.2⟩
