theorem allpoint_source_family4 (Omega : Set ℝ) (hOmega : ¬BddAbove Omega)
    (x : Omega → Fin 5 → ℚ) (hx : ∀ q,x q∈normalizedRelations fixedSUnitGroup 4)

    (hz : ∀ q,PrimitiveFour (z q))
    (hscale : ∀ q,∀ i : Fin 4,(z q i : ℚ)=((s q : ℚˣ) : ℚ)*x q i.castSucc)
    (hH : ∀ q,4≤(archFourHeight (z q) : ℝ))

    ∃ P : Submodule ℚ (Fin 4 → ℚ),P≠⊤ ∧ ∃ W : Set ℝ,W⊆Omega ∧ ¬BddAbove W ∧

  classical
  have hp (q : Omega) : ∃ choice : Fin 3 → Fin 4,∃ d∈finiteWeightCandidates4 448,
      sourceWeightConditions4 d ∧
      CompoundFeasibility.OriginalVariableGap4.originalSourceSolution4
        (retainedRationalMatrix4 (choice 1)) (retainedRationalMatrix4 (choice 2))
        (retainedRealMatrix4 (choice 0)) (d 0) (d 1) (d 2) q.val ((1 : ℝ)/64) (z q) := by
    simpa only [← hparam q] using allpoint_fixed_source_point4 (hx q) (z q) (s q) (hz q) (hscale q) (hH q)
  choose choice d hd hc hsol using hp
  let F : Set ((Fin 3 → Fin 4) × (Fin 3 → Fin 4 → ℝ)) := Set.univ ×ˢ finiteWeightCandidates4 448
  have hF : F.Finite := allpoint_joint_parameters_finite4
  let a (Q : ℝ) := if h : Q∈Omega then (choice ⟨Q,h⟩,d ⟨Q,h⟩) else (fun _ => 0,fun _ _ => 0)
  have ha : ∀ Q∈Omega,a Q∈F := by
    intro Q h
    simp only [a,dif_pos h,F,Set.mem_prod,Set.mem_univ,true_and]
    exact hd ⟨Q,h⟩
  obtain ⟨⟨k,c⟩,hkc,hT⟩ := allpoint_finite_parameter_fiber3 F hF Omega hOmega a ha
  let T : Set ℝ := {Q | Q∈Omega ∧ a Q=(k,c)}
  let embed (t : T) : Omega := ⟨t.val,t.property.1⟩
  have he (t : T) : choice (embed t)=k ∧ d (embed t)=c := by
    have hh := t.property.2
    simp only [a,dif_pos t.property.1] at hh
    exact Prod.mk.inj hh
  have hcfix : sourceWeightConditions4 c := hkc.2.2
  have hs (t : T) : CompoundFeasibility.OriginalVariableGap4.originalSourceSolution4
      (retainedRationalMatrix4 (k 1)) (retainedRationalMatrix4 (k 2))
      (retainedRealMatrix4 (k 0)) (c 0) (c 1) (c 2) t.val ((1 : ℝ)/64) (z (embed t)) := by
    have hh := hsol (embed t)
    simpa only [(he t).1,(he t).2] using hh
  have htQ : ∀ Q∈T,1<Q := fun Q h => (hs ⟨Q,h⟩).2.1
  have hbal : ∑ i : Fin 4,(c 0 i+c 1 i+c 2 i)=0 := by
    have hh := hcfix.2.2
    simpa [Fin.sum_univ_three,Finset.sum_add_distrib] using hh
  have hAlg : ∀ i j,IsAlgebraic ℚ (retainedRealMatrix4 (k 0) i j) := by
    intro i j
    exact isAlgebraic_int _
  obtain ⟨P,hP,W,hWT,hW,hm⟩ := allpoint_fixed_source_family4
    (retainedRationalMatrix4 (k 1)) (retainedRationalMatrix4 (k 2))
    (retainedRealMatrix4 (k 0)) (retained_rational_det4 _) (retained_rational_det4 _)
    (retained_real_det4 _) hAlg (c 0) (c 1) (c 2) hcfix.1
    (hcfix.2.1 1 (by decide)) (hcfix.2.1 2 (by decide)) hbal ((1 : ℝ)/64) (by norm_num)
    T hT htQ (fun t => z (embed t)) hs
  refine ⟨P,hP,W,fun Q h => (hWT h).1,hW,?_⟩
  intro q hqW
  exact hm ⟨q.val,hWT hqW⟩ hqW

theorem source_primitive_five_term_finite_cover4 :
    ∃ F : Set (Submodule ℚ (Fin 4 → ℚ)),F.Finite ∧ (∀ P∈F,P≠⊤) ∧
      ∀ z : Fin 4 → ℤ,allpointEligible4 z → ∃ P∈F,(fun i => (z i : ℚ))∈P := by
  classical
  by_contra hno
  change allpoint_no_finite_cover4 allpointEligible4 at hno
  obtain ⟨z,hE,hheight,hmono,havoid⟩ := allpoint_general_position_sequence4 allpointEligible4 hno
  choose x hx s hscale using fun n => (hE n).2
  let q (n : ℕ) : ℝ := (archFourHeight (z n) : ℝ)^8
  let Omega : Set ℝ := Set.range q
  have hOmega : ¬BddAbove Omega := by
    rintro ⟨M,hM⟩
    obtain ⟨n,hn⟩ := exists_nat_gt M
    have hb := hM (show q n∈Omega from Set.mem_range_self n)
    have hH := (hheight n).1
    have hHn := (hheight n).2
    have hsq : (archFourHeight (z n) : ℝ)≤(archFourHeight (z n) : ℝ)^2 := by nlinarith
    have hfour : (archFourHeight (z n) : ℝ)^2≤(archFourHeight (z n) : ℝ)^4 := by
      nlinarith [sq_nonneg ((archFourHeight (z n) : ℝ)^2-1)]
    have height : (archFourHeight (z n) : ℝ)^4≤(archFourHeight (z n) : ℝ)^8 := by
      nlinarith [sq_nonneg ((archFourHeight (z n) : ℝ)^4-1)]
    change (archFourHeight (z n) : ℝ)^8≤M at hb
    linarith
  have hex (t : Omega) : ∃ n : ℕ,q n=t.val := t.property
  choose original hparam using hex
  obtain ⟨P,hP,W,hWO,hW,hm⟩ := allpoint_source_family4 Omega hOmega
    (fun t => x (original t)) (fun t => hx (original t)) (fun t => z (original t))
    (fun t => s (original t)) (fun t => (hE (original t)).1) (fun t => hscale (original t))
    (fun t => (hheight (original t)).1.le) (fun t => (hparam t).symm)
  have hS := allpoint_general_position_intersection4 z (fun n => (hheight n).1) havoid P hP
  have hsub : W⊆q '' {n | (fun i => (z n i : ℚ))∈P} := by
    intro Q hQW
    let t : Omega := ⟨Q,hWO hQW⟩
    exact ⟨original t,hm t hQW,hparam t⟩
  exact hW ((hS.image q).bddAbove.mono hsub)

theorem allpoint_five_term_original_mem4 {x : Fin 5 → ℚ}
    (P : Submodule ℚ (Fin 4 → ℚ)) (w : Fin 5 → ℤ) (s : fixedSUnitGroup)
    (hscale : ∀ i,(w i : ℚ)=((s : ℚˣ) : ℚ)*x i)
    (hw : (fun i : Fin 4 => (w i.castSucc : ℚ))∈P) :
    (fun i : Fin 4 => x i.castSucc)∈P := by
  have hs : ((s : ℚˣ) : ℚ)≠0 := Units.ne_zero _
  have hm := P.smul_mem (((s : ℚˣ) : ℚ)⁻¹) hw
  have he : (((s : ℚˣ) : ℚ)⁻¹) • (fun i : Fin 4 => (w i.castSucc : ℚ))=
      (fun i : Fin 4 => x i.castSucc) := by
    ext i
    change ((s : ℚˣ) : ℚ)⁻¹*(w i.castSucc : ℚ)=x i.castSucc
    rw [hscale,inv_mul_cancel_left₀ hs]
  rw [he] at hm
  exact hm

  obtain ⟨P,hPF,hwP⟩ := hcover (fun i => w i.castSucc) ⟨hprim,x,hx,s,fun i => hscale i.castSucc⟩
  exact ⟨P,hPF,allpoint_five_term_original_mem4 P w s hscale hwP⟩

def Nondeg {m : ℕ} (x : Fin m → ℚ) : Prop :=
  ∀ I : Finset (Fin m),I.Nonempty → I≠Finset.univ → ∑ i∈I,x i≠0
def Relations (G : Subgroup ℚˣ) (m : ℕ) : Set (Fin (m+1) → ℚ) :=
  {x | x 0=1 ∧ (∀ i,∃ u : G,(u : ℚˣ).val=x i) ∧ (∑ i,x i=0) ∧ Nondeg x}
def SourceSolution (A2 A3 : Matrix (Fin 4) (Fin 4) ℚ)
    (L : Matrix (Fin 4) (Fin 4) ℝ) (c0 c2 c3 : Fin 4 → ℝ) (Q delta : ℝ) (x : Fin 4 → ℤ) : Prop :=
  x≠0 ∧ 1<Q ∧ 0<delta ∧
    (∀ i,|(L *ᵥ integerReal x) i|≤Q^(c0 i-delta)) ∧
    (∀ i,(padicNorm 2 ((A2 *ᵥ (fun j => (x j : ℚ))) i) : ℝ)≤Q^(c2 i)) ∧
    (∀ i,(padicNorm 3 ((A3 *ᵥ (fun j => (x j : ℚ))) i) : ℝ)≤Q^(c3 i))
def Raw_retainedIntegerMatrix4 (k : Fin 4) : Matrix (Fin 4) (Fin 4) ℤ :=
  fun i j => if i=k then 1 else if i=j then 1 else 0

def Raw_PrimitiveFour (z : Fin 4 → ℤ) : Prop := ∃ c : Fin 4 → ℤ,∑ i,c i*z i=1

def Raw_fourRetainedForm (k i : Fin 4) (y : Fin 4 → ℚ) : ℚ := if i=k then ∑ j,y j else y i

def Raw_localFourHeight (tau : Fin 3) (y : Fin 4 → ℚ) : ℚ :=
  max (rationalPlaceNorm tau (y 0)) (max (rationalPlaceNorm tau (y 1))
    (max (rationalPlaceNorm tau (y 2)) (rationalPlaceNorm tau (y 3))))

def Raw_fourHeight (y : Fin 4 → ℚ) : ℚ := ∏ tau : Fin 3,Raw_localFourHeight tau y

def Raw_archFourHeight (z : Fin 4 → ℤ) : ℚ :=
  max |(z 0 : ℚ)| (max |(z 1 : ℚ)| (max |(z 2 : ℚ)| |(z 3 : ℚ)|))

def Raw_fourRetainedProduct (choice : Fin 3 → Fin 4) (y : Fin 4 → ℚ) : ℚ :=
  ∏ tau : Fin 3,∏ i : Fin 4,rationalPlaceNorm tau (Raw_fourRetainedForm (choice tau) i y)

def Raw_sourceWeightConditions4 (c : Fin 3 → Fin 4 → ℝ) : Prop :=
  (∀ i, |c 0 i| ≤ 1) ∧ (∀ tau, tau ≠ 0 → ∀ i, -1 ≤ c tau i ∧ c tau i ≤ 0) ∧
    (∑ tau : Fin 3, ∑ i : Fin 4, c tau i) = 0

noncomputable def Raw_finiteWeightIndex4 (n : ℕ) (c : Fin 3 → Fin 4 → ℝ) (tau : Fin 3) (i : Fin 4) : ℤ :=
  if tau = 0 then ⌊(n : ℝ)*((1-4/(n : ℝ))*c tau i)⌋
  else ⌈(n : ℝ)*((1-4/(n : ℝ))*c tau i)⌉

noncomputable def Raw_finiteWeightRaw4 (n : ℕ) (z : Fin 3 → Fin 4 → ℤ) (tau : Fin 3) (i : Fin 4) : ℝ :=
  (z tau i : ℝ)/(n : ℝ)

noncomputable def Raw_finiteWeightCandidate4 (n : ℕ) (z : Fin 3 → Fin 4 → ℤ) (tau : Fin 3) (i : Fin 4) : ℝ :=
  if tau = 0 then Raw_finiteWeightRaw4 n z tau i - (∑ t : Fin 3, ∑ j : Fin 4, Raw_finiteWeightRaw4 n z t j)/4
  else Raw_finiteWeightRaw4 n z tau i

noncomputable def Raw_finiteWeightApprox4 (n : ℕ) (c : Fin 3 → Fin 4 → ℝ) : Fin 3 → Fin 4 → ℝ :=
  Raw_finiteWeightCandidate4 n (Raw_finiteWeightIndex4 n c)
