theorem allpoint_form_product3 {x : Fin 4 → ℚ}
    (hx : x ∈ normalizedRelations fixedSUnitGroup 3)
    (z : IntegerTriple) (s : fixedSUnitGroup)
    (hscale : ∀ i : Fin 3, (z i : ℚ) = ((s : ℚˣ) : ℚ)*x i.castSucc)
    (k i : Fin 3) :
    threePlaceProduct (threeCoordinateSumForms k i (fun j => (z j : ℚ))) = 1 := by
  have hu (j : Fin 4) : threePlaceProduct (x j) = 1 := by
    obtain ⟨u,he⟩ := hx.2.1 j
    rw [← he]
    exact fixed_sunit_product_formula u
  have hz (j : Fin 3) : threePlaceProduct (z j : ℚ) = 1 := by
    rw [hscale,threePlaceProduct_mul,fixed_sunit_product_formula,hu]
    norm_num
  have hsumx : x 0+x 1+x 2+x 3 = 0 := by
    simpa [Fin.sum_univ_succ,add_assoc] using hx.2.2.1
  have hs : (z 0 : ℚ)+z 1+z 2 = -(((s : ℚˣ) : ℚ)*x 3) := by
    have h0 := hscale 0
    have h1 := hscale 1
    have h2 := hscale 2
    change (z 0 : ℚ) = ((s : ℚˣ) : ℚ)*x 0 at h0
    change (z 1 : ℚ) = ((s : ℚˣ) : ℚ)*x 1 at h1
    change (z 2 : ℚ) = ((s : ℚˣ) : ℚ)*x 2 at h2
    have hh : ((s : ℚˣ) : ℚ)*(x 0+x 1+x 2+x 3) = 0 := by rw [hsumx,mul_zero]
    nlinarith
  have hsum : threePlaceProduct ((z 0 : ℚ)+z 1+z 2) = 1 := by
    rw [hs]
    have hneg (q : ℚ) : threePlaceProduct (-q) = threePlaceProduct q := by simp [threePlaceProduct]
    rw [hneg,threePlaceProduct_mul,fixed_sunit_product_formula,hu]
    norm_num
  fin_cases i
  · simpa [threeCoordinateSumForms] using hz (k.succAbove 0)
  · simpa [threeCoordinateSumForms] using hz (k.succAbove 1)
  · simpa [threeCoordinateSumForms] using hsum

theorem allpoint_form_arch_bound3 (z : IntegerTriple) (k i : Fin 3) :
    |threeCoordinateSumForms k i (fun j => (z j : ℚ))| ≤ 3*archTripleHeight z := by
  have hcoord (j : Fin 3) : |(z j : ℚ)| ≤ archTripleHeight z := by
    fin_cases j
    · exact le_max_left _ _
    · exact (le_max_left _ _).trans (le_max_right _ _)
    · exact (le_max_right _ _).trans (le_max_right _ _)
  have hH : 0 ≤ archTripleHeight z := (abs_nonneg (z 0 : ℚ)).trans (hcoord 0)
  fin_cases i
  · simpa [threeCoordinateSumForms] using (hcoord (k.succAbove 0)).trans (by linarith : archTripleHeight z ≤ 3*archTripleHeight z)
  · simpa [threeCoordinateSumForms] using (hcoord (k.succAbove 1)).trans (by linarith : archTripleHeight z ≤ 3*archTripleHeight z)
  · change |(z 0 : ℚ)+z 1+z 2| ≤ _
    have hb := (abs_add ((z 0 : ℚ)+z 1) (z 2 : ℚ)).trans
      (add_le_add_right (abs_add (z 0 : ℚ) (z 1 : ℚ)) |(z 2 : ℚ)|)
    linarith [hcoord 0,hcoord 1,hcoord 2]


theorem allpoint_local_inverse3 (m : ℤ) (hprod : threePlaceProduct (m : ℚ) = 1) (tau : Fin 3) :
    0 < rationalPlaceNorm tau (m : ℚ) ∧
    (rationalPlaceNorm tau (m : ℚ))⁻¹ ≤ |(m : ℚ)| ∧
    (tau ≠ 0 → rationalPlaceNorm tau (m : ℚ) ≤ 1) := by
  have hn : (m : ℚ) ≠ 0 := by
    intro h
    simp [h,threePlaceProduct] at hprod
  have hpos := rationalPlaceNorm_pos tau hn
  have h2 := padicNorm.of_int (p := 2) m
  have h3 := padicNorm.of_int (p := 3) m
  have hp2 : 0 < padicNorm 2 (m : ℚ) := by simpa [rationalPlaceNorm] using rationalPlaceNorm_pos 1 hn
  have hp3 : 0 < padicNorm 3 (m : ℚ) := by simpa [rationalPlaceNorm] using rationalPlaceNorm_pos 2 hn
  have ha : 0 < |(m : ℚ)| := abs_pos.mpr hn
  change |(m : ℚ)| *padicNorm 2 (m : ℚ)*padicNorm 3 (m : ℚ) = 1 at hprod
  have h23 : padicNorm 2 (m : ℚ)*padicNorm 3 (m : ℚ) ≤ 1 := by
    nlinarith [mul_le_mul h2 h3 hp3.le (by norm_num : (0 : ℚ) ≤ 1)]
  have hA : 1 ≤ |(m : ℚ)| := by
    nlinarith [mul_le_mul_of_nonneg_left h23 ha.le]
  have hb2 : 1 ≤ |(m : ℚ)| *padicNorm 2 (m : ℚ) := by
    nlinarith [mul_le_mul_of_nonneg_left h3 (mul_pos ha hp2).le]
  have hb3 : 1 ≤ |(m : ℚ)| *padicNorm 3 (m : ℚ) := by
    nlinarith [mul_le_mul_of_nonneg_left h2 (mul_pos ha hp3).le]
  refine ⟨hpos,?_,?_⟩
  · fin_cases tau
    · change |(m : ℚ)|⁻¹ ≤ |(m : ℚ)|
      apply (inv_le_iff_one_le_mul₀ ha).mpr
      nlinarith
    · change (padicNorm 2 (m : ℚ))⁻¹ ≤ |(m : ℚ)|
      exact (inv_le_iff_one_le_mul₀ hp2).mpr hb2
    · change (padicNorm 3 (m : ℚ))⁻¹ ≤ |(m : ℚ)|
      exact (inv_le_iff_one_le_mul₀ hp3).mpr hb3
  · intro ht
    fin_cases tau
    · exact (ht rfl).elim
    · exact h2
    · exact h3

def allpoint_balanced_values3 (Q : ℝ) (a : Fin 3 → Fin 3 → ℝ) (tau i : Fin 3) : ℝ :=
  if tau = 0 then a tau i*Q^((1 : ℝ)/12) else a tau i

theorem allpoint_balanced_weights3 (H : ℝ) (hH : 3 ≤ H)
    (a : Fin 3 → Fin 3 → ℝ) (ha : ∀ tau i, 0 < a tau i)
    (hreal : ∀ i, a 0 i ≤ 3*H)
    (hfinite : ∀ tau, tau ≠ 0 → ∀ i, a tau i ≤ 1)
    (hinv : ∀ tau i, (a tau i)⁻¹ ≤ 3*H)
    (hprod : (∏ tau, ∏ i, a tau i)*H = 1) :
    let Q := H^4
    let W := allpoint_balanced_values3 Q a
    1 < Q ∧ sourceWeightConditions3 (sourceLogWeights3 Q W) ∧
      (∀ tau i, Q^(sourceLogWeights3 Q W tau i) = W tau i) ∧
      (∀ i, a 0 i = Q^(sourceLogWeights3 Q W 0 i-(1 : ℝ)/12)) ∧
      (∀ tau, tau ≠ 0 → ∀ i, a tau i = Q^(sourceLogWeights3 Q W tau i)) := by
  intro Q W
  have hH0 : 0 < H := by linarith
  have hQ : 1 < Q := by dsimp [Q]; nlinarith [sq_nonneg (H^2-1)]
  have hQ0 := lt_trans zero_lt_one hQ
  let r := Q^((1 : ℝ)/12)
  have hr : 0 < r := Real.rpow_pos_of_pos hQ0 _
  have hr1 : 1 ≤ r := Real.one_le_rpow hQ.le (by norm_num)
  have hrH : r ≤ H := by
    have he : (H^4)^((1 : ℝ)/12) = H^((1 : ℝ)/3) := by
      rw [← Real.rpow_natCast,← Real.rpow_mul hH0.le]
      congr 1
      norm_num
    change (H^4)^((1 : ℝ)/12) ≤ H
    rw [he]
    simpa using Real.rpow_le_rpow_of_exponent_le (show 1 ≤ H by linarith) (by norm_num : (1 : ℝ)/3 ≤ 1)
  have hrpow : r^3 = H := by
    change ((H^4)^((1 : ℝ)/12))^3 = H
    rw [← Real.rpow_mul_natCast (pow_pos hH0 4).le,← Real.rpow_natCast,← Real.rpow_mul hH0.le]
    norm_num
  have hW : ∀ tau i, 0 < W tau i := by
    intro tau i
    dsimp [W,allpoint_balanced_values3]
    split_ifs
    · exact mul_pos (ha tau i) (Real.rpow_pos_of_pos hQ0 _)
    · exact ha tau i
  have h3H : 3*H < Q := by dsimp [Q]; nlinarith [sq_nonneg (H^2-3)]
  have h3H2 : 3*H^2 < Q := by dsimp [Q]; nlinarith [sq_nonneg (H^2-3)]
  have hcut : ∀ tau i, (W tau i)⁻¹ < Q := by
    intro tau i
    by_cases ht : tau = 0
    · subst tau
      change (a 0 i*r)⁻¹ < Q
      rw [mul_inv_rev]
      have hri : r⁻¹ ≤ 1 := (inv_le_one₀ hr).mpr hr1
      have hh := mul_le_mul_of_nonneg_right hri (inv_pos.mpr (ha 0 i)).le
      exact (by nlinarith [hinv 0 i] : r⁻¹*(a 0 i)⁻¹ ≤ 3*H).trans_lt h3H
    · simpa only [W,allpoint_balanced_values3,if_neg ht] using (hinv tau i).trans_lt h3H
  have hcut0 : ∀ i, W 0 i < Q := by
    intro i
    change a 0 i*r < Q
    have hh := mul_le_mul (hreal i) hrH hr.le (by linarith : 0 ≤ 3*H)
    exact (by nlinarith : a 0 i*r ≤ 3*H^2).trans_lt h3H2
  have hWfin : ∀ tau, tau ≠ 0 → ∀ i, W tau i ≤ 1 := by
    intro tau ht i
    simpa only [W,allpoint_balanced_values3,if_neg ht] using hfinite tau ht i
  have hWprod : (∏ tau, ∏ i, W tau i) = 1 := by
    simp only [W,allpoint_balanced_values3,Fin.prod_univ_three,Fin.isValue,Fin.reduceEq,ite_true,ite_false] at ⊢
    have hp := hprod
    simp only [Fin.prod_univ_three] at hp
    change (a 0 0*r*(a 0 1*r)*(a 0 2*r))*(a 1 0*a 1 1*a 1 2)*(a 2 0*a 2 1*a 2 2) = 1
    calc
      _ = (a 0 0*a 0 1*a 0 2)*(a 1 0*a 1 1*a 1 2)*(a 2 0*a 2 1*a 2 2)*r^3 := by ring
      _ = 1 := by rw [hrpow]; exact hp
  have heq := sourceLogWeights3_rpow Q hQ W hW
  refine ⟨hQ,sourceLogWeights3_conditions Q hQ W hW hcut hcut0 hWfin hWprod,heq,?_,?_⟩
  · intro i
    rw [Real.rpow_sub hQ0,heq]
    change a 0 i = (a 0 i*r)/r
    exact (mul_div_cancel_right₀ _ hr.ne').symm
  · intro tau ht i
    rw [heq]
    simp only [W,allpoint_balanced_values3,if_neg ht]

end

theorem allpoint_round_point3 (Q : ℝ) (hQ : 1 < Q)
    (k0 k2 k3 : Fin 3) (c : Fin 3 → Fin 3 → ℝ) (hc : sourceWeightConditions3 c)
    (z : IntegerTriple) (hz : z ∈ qSourceLattice3 k2 k3 Q (c 1) (c 2))
    (hshort : coordinateGauge3 (qSourceCoordinates3 k0 Q (c 0) hQ) (integerTripleReal3 z) ≤
      Q^(-(1 : ℝ)/12)) :
    let d := finiteWeightApprox3 168 c
    d ∈ finiteWeightCandidates3 168 ∧ sourceWeightConditions3 d ∧
    z ∈ qSourceLattice3 k2 k3 Q (d 1) (d 2) ∧
    coordinateGauge3 (qSourceCoordinates3 k0 Q (d 0) hQ) (integerTripleReal3 z) ≤ Q^(-(1 : ℝ)/24) := by
  intro d
  obtain ⟨hd,hreal,hfinite⟩ := finiteWeightApprox3_bounds 168 (by norm_num) c hc
  have hQ0 := lt_trans zero_lt_one hQ
  have hpoint := ((qSourceLattice3_spec k2 k3 Q (c 1) (c 2) hQ
    (fun i => (hc.2.1 1 (by decide) i).2) (fun i => (hc.2.1 2 (by decide) i).2)).1 z).mp hz
  refine ⟨finiteWeightApprox3_mem 168 (by norm_num) c hc,hd,?_,?_⟩
  · apply ((qSourceLattice3_spec k2 k3 Q (d 1) (d 2) hQ
      (fun i => (hd.2.1 1 (by decide) i).2) (fun i => (hd.2.1 2 (by decide) i).2)).1 z).mpr
    exact ⟨fun i => (hpoint.1 i).trans (Real.rpow_le_rpow_of_exponent_le hQ.le (hfinite 1 (by decide) i)),
      fun i => (hpoint.2 i).trans (Real.rpow_le_rpow_of_exponent_le hQ.le (hfinite 2 (by decide) i))⟩
  · apply (pi_norm_le_iff_of_nonempty _).mpr
    intro i
    change |retainedRealCoordinates3 k0 (integerTripleReal3 z) i / Q^(d 0 i)| ≤ _
    rw [abs_div,abs_of_pos (Real.rpow_pos_of_pos hQ0 _),div_le_iff₀ (Real.rpow_pos_of_pos hQ0 _)]
    have hh := qSource_arch_bound k0 Q (c 0) hQ (integerTripleReal3 z) i
    have hbound := hh.trans (mul_le_mul_of_nonneg_right hshort (Real.rpow_nonneg hQ0.le _))
    have he : Q^(-(1 : ℝ)/12)*Q^(c 0 i) = Q^(c 0 i-(1 : ℝ)/12) := by
      rw [← Real.rpow_add hQ0]; congr 1; ring
    rw [he] at hbound
    have hw : c 0 i-(1 : ℝ)/12 ≤ d 0 i-(1 : ℝ)/24 := by
      have hr := hreal i
      norm_num at hr
      linarith
    have hp := hbound.trans (Real.rpow_le_rpow_of_exponent_le hQ.le hw)
    have he' : Q^(d 0 i-(1 : ℝ)/24) = Q^(-(1 : ℝ)/24)*Q^(d 0 i) := by
      rw [← Real.rpow_add hQ0]; congr 1; ring
    rw [he'] at hp
    exact hp

theorem allpoint_source_weights3 {x : Fin 4 → ℚ}
    (hx : x ∈ normalizedRelations fixedSUnitGroup 3)
    (z : IntegerTriple) (s : fixedSUnitGroup) (hz : PrimitiveTriple z)
    (hscale : ∀ i : Fin 3, (z i : ℚ) = ((s : ℚˣ) : ℚ)*x i.castSucc)
    (hH : 3 ≤ (archTripleHeight z : ℝ)) :
    let Q := (archTripleHeight z : ℝ)^4
    ∃ choice : Fin 3 → Fin 3, ∃ c : Fin 3 → Fin 3 → ℝ,
      ∃ hQ : 1 < Q, sourceWeightConditions3 c ∧
      c = sourceLogWeights3 Q (allpoint_balanced_values3 Q (allpoint_norms3 choice z)) ∧
      z ∈ qSourceLattice3 (choice 1) (choice 2) Q (c 1) (c 2) ∧
      coordinateGauge3 (qSourceCoordinates3 (choice 0) Q (c 0) hQ) (integerTripleReal3 z) ≤ Q^(-(1 : ℝ)/12) := by
  intro Q
  obtain ⟨choice,hret⟩ := allpoint_primitive_product3 hx z s hz hscale
  let a := allpoint_norms3 choice z
  have hlocal (tau i : Fin 3) :
      0 < a tau i ∧ (a tau i)⁻¹ ≤ 3*(archTripleHeight z : ℝ) ∧
      (tau ≠ 0 → a tau i ≤ 1) := by
    let m := (retainedIntegerMatrix3 (choice tau) *ᵥ z) i
    have hp : threePlaceProduct (m : ℚ) = 1 := by
      rw [retainedIntegerMatrix3_apply]
      exact allpoint_form_product3 hx z s hscale _ _
    have hl := allpoint_local_inverse3 m hp tau
    have ha := allpoint_form_arch_bound3 z (choice tau) i
    have hm : (m : ℚ) = threeCoordinateSumForms (choice tau) i (fun j => (z j : ℚ)) :=
      retainedIntegerMatrix3_apply _ _ _
    rw [hm] at hl
    dsimp only [a,allpoint_norms3]
    refine ⟨by exact_mod_cast hl.1,?_,?_⟩
    · have hb := hl.2.1.trans ha
      exact_mod_cast hb
    · intro ht
      exact_mod_cast hl.2.2 ht
  have hprod : (∏ tau, ∏ i, a tau i)*(archTripleHeight z : ℝ) = 1 := by
    unfold tripleChoiceFormsProduct at hret
    dsimp only [a,allpoint_norms3]
    exact_mod_cast hret
  obtain ⟨hQ,hc,heq,hreal,hfinite⟩ := allpoint_balanced_weights3 (archTripleHeight z : ℝ) hH a
    (fun tau i => (hlocal tau i).1)
    (fun i => by
      change ((|threeCoordinateSumForms (choice 0) i (fun j => (z j : ℚ))| : ℚ) : ℝ) ≤ 3*(archTripleHeight z : ℝ)
      exact_mod_cast allpoint_form_arch_bound3 z (choice 0) i)
    (fun tau ht i => (hlocal tau i).2.2 ht)
    (fun tau i => (hlocal tau i).2.1) hprod
  let c := sourceLogWeights3 Q (allpoint_balanced_values3 Q a)
  refine ⟨choice,c,hQ,hc,rfl,?_,?_⟩
  · apply ((qSourceLattice3_spec (choice 1) (choice 2) Q (c 1) (c 2) hQ
      (fun i => (hc.2.1 1 (by decide) i).2) (fun i => (hc.2.1 2 (by decide) i).2)).1 z).mpr
    constructor
    · intro i
      change a 1 i ≤ Q^(c 1 i)
      exact (hfinite 1 (by decide) i).le
    · intro i
      change a 2 i ≤ Q^(c 2 i)
      exact (hfinite 2 (by decide) i).le
  · apply (pi_norm_le_iff_of_nonempty _).mpr
    intro i
    have hQ0 := lt_trans zero_lt_one hQ
    change |retainedRealCoordinates3 (choice 0) (integerTripleReal3 z) i / Q^(c 0 i)| ≤ _
    rw [abs_div,abs_of_pos (Real.rpow_pos_of_pos hQ0 _),← finiteWeight_real_local_value3,
      finiteWeight_rational_local_value3]
    have hnorm : rothLocalAbs 0 (threeCoordinateSumForms (choice 0) i (fun j => (z j : ℚ))) = a 0 i := by
      simp [a,allpoint_norms3,rothLocalAbs,rationalPlaceNorm]
    rw [hnorm,hreal]
    have hp : Q^(c 0 i-(1 : ℝ)/12) = Q^(-(1 : ℝ)/12)*Q^(c 0 i) := by
      rw [← Real.rpow_add hQ0]; congr 1; ring
    rw [hp,mul_div_cancel_right₀ _ (Real.rpow_pos_of_pos hQ0 _).ne']

theorem allpoint_rank_cutoff3 (epsilon : ℝ) (he : 0 < epsilon) :
    ∃ C : ℝ, 1 ≤ C ∧ ∀ Q : ℝ, C < Q →
      6 * (Q ^ (-epsilon)) ^ 3 * (Q ^ (epsilon / 8)) ^ 4 < 1 := by
  obtain ⟨C,hC⟩ := Filter.eventually_atTop.mp
    ((tendsto_rpow_atTop (show 0 < 5*epsilon/2 by linarith)).eventually
      (Filter.eventually_gt_atTop 6))
  refine ⟨max 1 C,le_max_left _ _,?_⟩
  intro Q hQ
  have hq : 0 < Q := lt_of_lt_of_le zero_lt_one ((le_max_left 1 C).trans hQ.le)
  have hh : 6 < Q^(5*epsilon/2) := hC Q (lt_of_le_of_lt (le_max_right 1 C) hQ).le
  have hp : 0 < Q^(-(5*epsilon/2)) := Real.rpow_pos_of_pos hq _
  have hmul := mul_lt_mul_of_pos_right hh hp
  rw [← Real.rpow_add hq,add_neg_cancel,Real.rpow_zero] at hmul
  have heq : (Q^(-epsilon))^3*(Q^(epsilon/8))^4 = Q^(-(5*epsilon/2)) := by
    rw [← Real.rpow_mul_natCast hq.le,← Real.rpow_mul_natCast hq.le,← Real.rpow_add hq]
    congr 1
    ring
  simpa only [mul_assoc,heq] using hmul

theorem allpoint_rank_scalar3 (a b c g t q : ℝ)
    (ha : 0 < a) (hb : 0 < b) (hc : 0 < c) (ht : 0 < t) (hq : 1 ≤ q)
    (hab : a ≤ b) (hbc : b ≤ c) (hag : a ≤ g) (hgt : g ≤ t)
    (hp : (1 : ℝ)/6 ≤ a*b*c) (hcut : 6*t^3*q^4 < 1) :
    (b < c/q ∧ g < c) ∨ (c/q ≤ b ∧ a < b/q ∧ g < b) := by
  have hq0 : 0 < q := lt_of_lt_of_le zero_lt_one hq
  have hprod : a*b*c ≤ t*c^2 := by
    have h1 := mul_le_mul_of_nonneg_right (hag.trans hgt) hb.le
    have h2 := mul_le_mul_of_nonneg_left hbc ht.le
    nlinarith [mul_le_mul_of_nonneg_right (h1.trans h2) hc.le]
  have hbig : t*q^2 < c := by
    by_contra h

theorem allpoint_general_position_step3 (E : IntegerTriple → Prop) (hno : allpoint_no_finite_cover3 E)

theorem allpoint_general_position_intersection3 (z : ℕ → IntegerTriple)

    {n | (fun i => (z n i : ℚ)) ∈ P}.Finite ∧ {n | (fun i => (z n i : ℚ)) ∈ P}.ncard ≤ 2 := by

    obtain ⟨k,hk,hjk⟩ := not_bddAbove_iff.mp h j

  refine ⟨hfinite,?_⟩
  apply Nat.le_of_not_gt
  intro hcard
  obtain ⟨a,b,c,ha,hb,hc,hab,hac,hbc⟩ := (Set.two_lt_ncard_iff hfinite).mp hcard
  have hord : (a < b ∧ b < c) ∨ (a < c ∧ c < b) ∨ (b < a ∧ a < c) ∨
      (b < c ∧ c < a) ∨ (c < a ∧ a < b) ∨ (c < b ∧ b < a) := by omega
  rcases hord with h | h | h | h | h | h
  · exact hbad a b c h.1 h.2 ha hb hc
  · exact hbad a c b h.1 h.2 ha hc hb

def allpointEligible3 (z : IntegerTriple) : Prop :=
  PrimitiveTriple z ∧ ∃ x : Fin 4 → ℚ, x ∈ normalizedRelations fixedSUnitGroup 3 ∧
    ∃ s : fixedSUnitGroup, ∀ i : Fin 3, (z i : ℚ) = ((s : ℚˣ) : ℚ)*x i.castSucc

theorem source_primitive_four_term_finite_cover3 :
    ∃ F : Set (Submodule ℚ (Fin 3 → ℚ)), F.Finite ∧ (∀ P ∈ F, P ≠ ⊤) ∧
      ∀ z : IntegerTriple, allpointEligible3 z → ∃ P ∈ F, (fun i => (z i : ℚ)) ∈ P := by
  classical
  by_contra hno
  change allpoint_no_finite_cover3 allpointEligible3 at hno
  obtain ⟨z,hE,hheight,hmono,havoid⟩ := allpoint_general_position_sequence3 allpointEligible3 hno
  choose x hx s hscale using (fun n => (hE n).2)
  let q (n : ℕ) : ℝ := (archTripleHeight (z n) : ℝ)^4
  let Omega : Set ℝ := Set.range q
  have hOmega : ¬ BddAbove Omega := by
    rintro ⟨M,hM⟩
    obtain ⟨n,hn⟩ := exists_nat_gt M
    have hb := hM (show q n ∈ Omega from Set.mem_range_self n)
    have hH := (hheight n).1
    have hHn := (hheight n).2
    have hsq : (archTripleHeight (z n) : ℝ) ≤ (archTripleHeight (z n) : ℝ)^2 := by nlinarith
    have hfour : (archTripleHeight (z n) : ℝ)^2 ≤ (archTripleHeight (z n) : ℝ)^4 := by
      nlinarith [sq_nonneg ((archTripleHeight (z n) : ℝ)^2-1)]
    change (archTripleHeight (z n) : ℝ)^4 ≤ M at hb
    linarith
  have hex (t : Omega) : ∃ n : ℕ, q n = t.val := t.property
  choose original hparam using hex
  obtain ⟨P,hP,W,hWO,hW,hm⟩ := allpoint_source_family3 Omega hOmega
    (fun t => x (original t)) (fun t => hx (original t)) (fun t => z (original t))
    (fun t => s (original t)) (fun t => (hE (original t)).1) (fun t => hscale (original t))
    (fun t => (hheight (original t)).1.le) (fun t => (hparam t).symm)
  have hS := (allpoint_general_position_intersection3 z (fun n => (hheight n).1) havoid P hP).1
  have hsub : W ⊆ q '' {n | (fun i => (z n i : ℚ)) ∈ P} := by
    intro Q hQW
    let t : Omega := ⟨Q,hWO hQW⟩
    exact ⟨original t,hm t hQW,hparam t⟩
  exact hW ((hS.image q).bddAbove.mono hsub)

theorem allpoint_four_term_recovery3 {x : Fin 4 → ℚ}
    (hx : x ∈ normalizedRelations fixedSUnitGroup 3) (w : Fin 4 → ℤ) (s : fixedSUnitGroup)
    (hscale : ∀ i, (w i : ℚ) = ((s : ℚˣ) : ℚ)*x i) :
    ∀ i, x i = (w i : ℚ)/(w 0 : ℚ) := by
  have h0 : (w 0 : ℚ) = ((s : ℚˣ) : ℚ) := by simpa only [hx.1,mul_one] using hscale 0
  intro i
  rw [hscale,h0]
  exact (mul_div_cancel_left₀ (x i) (Units.ne_zero (s : ℚˣ))).symm

theorem source_four_term_finite_cover3 :
    ∃ F : Set (Submodule ℚ (Fin 3 → ℚ)), F.Finite ∧ (∀ P ∈ F, P ≠ ⊤) ∧
      ∀ x ∈ normalizedRelations fixedSUnitGroup 3,
        ∃ w : Fin 4 → ℤ, ∃ s : fixedSUnitGroup,
          PrimitiveTriple (fun i => w i.castSucc) ∧ (∀ i, w i ≠ 0) ∧
          (∑ i, w i = 0) ∧ Nondegenerate (fun i => (w i : ℚ)) ∧
          (∀ i, (w i : ℚ) = ((s : ℚˣ) : ℚ)*x i) ∧
          (∀ i, x i = (w i : ℚ)/(w 0 : ℚ)) ∧ ∃ P ∈ F, (fun i : Fin 3 => (w i.castSucc : ℚ)) ∈ P := by
  obtain ⟨F,hF,hproper,hcover⟩ := source_primitive_four_term_finite_cover3
  refine ⟨F,hF,hproper,?_⟩
  intro x hx
  obtain ⟨w,s,hprim,hn,hs,hd,hscale⟩ := exists_primitive_four_term_relation hx
  refine ⟨w,s,hprim,hn,hs,hd,hscale,allpoint_four_term_recovery3 hx w s hscale,?_⟩
  exact hcover _ ⟨hprim,x,hx,s,fun i => hscale i.castSucc⟩
