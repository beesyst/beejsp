def firstProductIndex5 (j : Fin 4) (a : Fin (j.val+1)) : Fin 5 := ⟨a.val,by omega⟩

def tailProductIndex5 (j : Fin 4) (b : Fin (4-j.val)) : Fin 5 := ⟨j.val+1+b.val,by omega⟩

theorem prefixProduct5_pos (j : Fin 4) (lam : Fin 5 → ℝ) (hp : ∀ i, 0 < lam i) :
    0 < prefixProduct5 j lam := Finset.prod_pos (fun _ _ => hp _)

theorem tailProduct5_pos (j : Fin 4) (lam : Fin 5 → ℝ) (hp : ∀ i, 0 < lam i) :
    0 < tailProduct5 j lam := Finset.prod_pos (fun _ _ => hp _)

theorem prefix_tail_product5 (j : Fin 4) (lam : Fin 5 → ℝ) :
    prefixProduct5 j lam * tailProduct5 j lam = ∏ i, lam i := by
  have hd : (j.val+1)+(4-j.val) = 5 := by omega
  let f : Fin ((j.val+1)+(4-j.val)) → ℝ := fun i => lam ⟨i.val,by omega⟩
  have h : (∏ i : Fin 5, lam i) = ∏ i, f i := by
    apply Fintype.prod_equiv (finCongr hd.symm)
    intro i
    rfl
  rw [Fin.prod_univ_add] at h
  exact h.symm

theorem prefix_tail_power5 (j : Fin 4) (lam : Fin 5 → ℝ) (Q delta : ℝ)
    (hQ : 0 < Q) (hp : ∀ i, 0 < lam i) (hm : Monotone lam)
    (hg : lam j.castSucc < lam j.succ * Q^(-delta)) :
    (prefixProduct5 j lam)^(4-j.val) ≤ (tailProduct5 j lam)^(j.val+1) *
      Q^(-delta*((j.val+1 : ℕ) : ℝ)*((4-j.val : ℕ) : ℝ)) := by
  let r := Q^(-delta)
  have hr : 0 < r := Real.rpow_pos_of_pos hQ _
  have hpair (a : Fin (j.val+1)) (b : Fin (4-j.val)) :
      lam (firstProductIndex5 j a) ≤ lam (tailProductIndex5 j b) * r := by
    exact ((hm (show (firstProductIndex5 j a : Fin 5) ≤ j.castSucc by change a.val ≤ j.val; omega)).trans_lt hg).le.trans
      (mul_le_mul_of_nonneg_right (hm (show j.succ ≤ (tailProductIndex5 j b : Fin 5) by change j.val+1 ≤ j.val+1+b.val; omega)) hr.le)
  have ha (a : Fin (j.val+1)) :
      (lam (firstProductIndex5 j a))^(4-j.val) ≤ tailProduct5 j lam * r^(4-j.val) := by
    calc
      _ = ∏ _b : Fin (4-j.val), lam (firstProductIndex5 j a) := by simp
      _ ≤ ∏ b : Fin (4-j.val), lam (tailProductIndex5 j b) * r :=
        Finset.prod_le_prod (fun _ _ => (hp _).le) (fun b _ => hpair a b)
      _ = _ := by rw [Finset.prod_mul_distrib]; simp [tailProduct5,tailProductIndex5]
  have h := Finset.prod_le_prod (s := Finset.univ)
    (fun a _ => pow_nonneg (hp (firstProductIndex5 j a : Fin 5)).le (4-j.val)) (fun a _ => ha a)
  have he : (r^(4-j.val))^(j.val+1) = Q^(-delta*((j.val+1 : ℕ) : ℝ)*((4-j.val : ℕ) : ℝ)) := by
    dsimp [r]
    rw [← Real.rpow_natCast,← Real.rpow_natCast,← Real.rpow_mul hQ.le,← Real.rpow_mul hQ.le]
    congr 1
    ring
  simpa only [← Finset.prod_pow,prefixProduct5,Finset.prod_mul_distrib,
    Finset.prod_const,Finset.card_univ,Fintype.card_fin,he,firstProductIndex5] using h

theorem prefix_fifth_power5 (j : Fin 4) (lam : Fin 5 → ℝ) (Q delta : ℝ)
    (hQ : 0 < Q) (hp : ∀ i, 0 < lam i) (hm : Monotone lam)
    (hg : lam j.castSucc < lam j.succ * Q^(-delta)) :
    (prefixProduct5 j lam)^5 ≤ (∏ i, lam i)^(j.val+1) *
      Q^(-delta*((j.val+1 : ℕ) : ℝ)*(5-((j.val+1 : ℕ) : ℝ))) := by
  have ht : ((4-j.val : ℕ) : ℝ) = 5-((j.val+1 : ℕ) : ℝ) := by
    have h : (j.val+1)+(4-j.val) = 5 := by omega
    have hr := congrArg (fun n : ℕ => (n : ℝ)) h
    push_cast at hr ⊢
    linarith
  have h := mul_le_mul_of_nonneg_left (prefix_tail_power5 j lam Q delta hQ hp hm hg)
    (pow_nonneg (prefixProduct5_pos j lam hp).le (j.val+1))
  have hn : (j.val+1)+(4-j.val) = 5 := by omega
  calc
    _ = (prefixProduct5 j lam)^(j.val+1) * (prefixProduct5 j lam)^(4-j.val) := by rw [← pow_add,hn]
    _ ≤ _ := h
    _ = _ := by rw [← mul_assoc,← mul_pow,prefix_tail_product5,ht]

theorem source_prefix_product_decay5 (j : Fin 4) (lam : Fin 5 → ℝ)
    (Q delta C5 : ℝ) (hQ : 0 < Q) (hC : 0 < C5)
    (hp : ∀ i, 0 < lam i) (hm : Monotone lam)
    (hg : lam j.castSucc < lam j.succ * Q^(-delta)) (hprod : ∏ i, lam i ≤ C5) :
    prefixProduct5 j lam ≤ C5^(((j.val+1 : ℕ) : ℝ)/5) * Q^(-decayExponent5 j delta) := by
  have h := (prefix_fifth_power5 j lam Q delta hQ hp hm hg).trans
    (mul_le_mul_of_nonneg_right (pow_le_pow_left₀ (Finset.prod_pos (fun i _ => hp i)).le hprod _)
      (Real.rpow_pos_of_pos hQ _).le)
  have he : (C5^(((j.val+1 : ℕ) : ℝ)/5) * Q^(-decayExponent5 j delta))^5 =
      C5^(j.val+1) * Q^(-delta*((j.val+1 : ℕ) : ℝ)*(5-((j.val+1 : ℕ) : ℝ))) := by
    rw [mul_pow,← Real.rpow_natCast,← Real.rpow_mul hC.le,
      ← Real.rpow_natCast,← Real.rpow_mul hQ.le]
    have hc : (((j.val+1 : ℕ) : ℝ)/5)*5 = ((j.val+1 : ℕ) : ℝ) := by ring
    norm_num only [Nat.cast_ofNat]
    rw [hc,Real.rpow_natCast]
    congr 1
    congr 1
    unfold decayExponent5
    push_cast
    ring
  exact le_of_pow_le_pow_left₀ (by decide : 5 ≠ 0) (by positivity) (he.symm ▸ h)

theorem source_gap_geometric_mean5 (j : Fin 4) (lam : Fin 5 → ℝ)
    (Q delta : ℝ) (hQ : 0 < Q) (hp : ∀ i, 0 < lam i) (hm : Monotone lam)
    (hg : lam j.castSucc < lam j.succ * Q^(-delta)) :
    ∀ i : Fin 5, i.val < j.val+1 → lam i <
      (tailProduct5 j lam)^(1/(5-((j.val+1 : ℕ) : ℝ))) * Q^(-delta) := by
  have hn : 0 < 4-j.val := by omega
  have hnR : (0 : ℝ) < ((4-j.val : ℕ) : ℝ) := by exact_mod_cast hn
  have he : ((4-j.val : ℕ) : ℝ) = 5-((j.val+1 : ℕ) : ℝ) := by
    have h : (j.val+1)+(4-j.val) = 5 := by omega
    have hr := congrArg (fun n : ℕ => (n : ℝ)) h
    push_cast at hr ⊢
    linarith
  have ht : (lam j.succ)^(4-j.val) ≤ tailProduct5 j lam := by
    calc
      _ = ∏ _b : Fin (4-j.val), lam j.succ := by simp
      _ ≤ _ := Finset.prod_le_prod (fun _ _ => (hp _).le) (fun b _ => hm
        (show j.succ ≤ (tailProductIndex5 j b : Fin 5) by change j.val+1 ≤ j.val+1+b.val; omega))
  have hr := Real.rpow_le_rpow (pow_nonneg (hp _).le _) ht (by positivity : (0 : ℝ) ≤ 1/((4-j.val : ℕ) : ℝ))
  rw [← Real.rpow_natCast,← Real.rpow_mul (hp _).le] at hr
  have hc : ((4-j.val : ℕ) : ℝ) * (1/((4-j.val : ℕ) : ℝ)) = 1 := by exact mul_one_div_cancel hnR.ne'
  rw [hc,Real.rpow_one,he] at hr
  intro i hi
  exact ((hm (show i ≤ j.castSucc by change i.val ≤ j.val; omega)).trans_lt hg).trans_le
    (mul_le_mul_of_nonneg_right hr (Real.rpow_pos_of_pos hQ _).le)

theorem restrictedBasis5_map_addHaar
    (c : Basis (Fin d) ℤ (restrictedLattice5 P A₂ A₃ Q c₂ c₃)) :
    Measure.map (c.ofZLatticeBasis ℝ _).equivFun
      (c.ofZLatticeBasis ℝ _).addHaar = volume := by
  let b := c.ofZLatticeBasis ℝ (restrictedLattice5 P A₂ A₃ Q c₂ c₃)
  have heq : b.map b.equivFunL.toLinearEquiv = Pi.basisFun ℝ (Fin d) := by
    ext i j
    simp [Basis.map_apply,Basis.equivFun_apply,Pi.basisFun_apply,Finsupp.single_apply,Pi.single_apply,eq_comm]
  change Measure.map b.equivFunL b.addHaar = volume
  rw [Basis.map_addHaar b b.equivFunL,heq,Basis.addHaar_def,
    Basis.parallelepiped_basisFun,MeasureTheory.addHaarMeasure_eq_volume_pi]

theorem restrictedBasis5_measure_eq_covolume_smul
    (c : Basis (Fin d) ℤ (restrictedLattice5 P A₂ A₃ Q c₂ c₃)) :
    restrictedMeasure5 P =
      ENNReal.ofReal (ZLattice.covolume (restrictedLattice5 P A₂ A₃ Q c₂ c₃)
        (restrictedMeasure5 P)) • (c.ofZLatticeBasis ℝ _).addHaar := by
  let b := c.ofZLatticeBasis ℝ (restrictedLattice5 P A₂ A₃ Q c₂ c₃)
  let mu := restrictedMeasure5 P
  have hcv : ENNReal.ofReal (ZLattice.covolume (restrictedLattice5 P A₂ A₃ Q c₂ c₃) mu) =
      mu (_root_.parallelepiped b) := by
    rw [ZLattice.covolume_eq_measure_fundamentalDomain _ mu (ZLattice.isAddFundamentalDomain c mu)]
    change ENNReal.ofReal (mu (ZSpan.fundamentalDomain b)).toReal = _
    rw [ENNReal.ofReal_toReal (ZSpan.fundamentalDomain_isBounded b).measure_lt_top.ne,
      measure_congr (ZSpan.fundamentalDomain_ae_parallelepiped b mu)]
  have h := Measure.addHaarMeasure_unique mu b.parallelepiped
  rw [← Basis.addHaar_def] at h
  change mu = _
  rw [hcv]
  exact h

theorem restrictedBasis5_volume_ratio
    (c : Basis (Fin d) ℤ (restrictedLattice5 P A₂ A₃ Q c₂ c₃))
    (C : Set (rationalRealSpan5 P)) (hC : IsCompact C) :
    volume.real ((c.ofZLatticeBasis ℝ _).equivFun '' C) =
      (restrictedMeasure5 P).real C /
        ZLattice.covolume (restrictedLattice5 P A₂ A₃ Q c₂ c₃) (restrictedMeasure5 P) := by
  let b := c.ofZLatticeBasis ℝ (restrictedLattice5 P A₂ A₃ Q c₂ c₃)
  have himage : volume (b.equivFun '' C) = b.addHaar C := by
    rw [← restrictedBasis5_map_addHaar c]
    change (Measure.map b.equivFunL b.addHaar) (b.equivFunL '' C) = b.addHaar C
    rw [Measure.map_apply b.equivFunL.continuous.measurable (hC.image b.equivFunL.continuous).measurableSet,
      Set.preimage_image_eq _ b.equivFunL.injective]
  have hcv := restrictedLattice5_covolume_pos P A₂ A₃ Q c₂ c₃
  change (volume (b.equivFun '' C)).toReal = _
  rw [himage]
  apply (eq_div_iff hcv.ne').mpr
  have heq := congrArg (fun mu : Measure (rationalRealSpan5 P) => mu.real C)
    (restrictedBasis5_measure_eq_covolume_smul c)
  simpa [measureReal_def,Measure.smul_apply,ENNReal.toReal_mul,
    ENNReal.toReal_ofReal hcv.le,mul_comm,b] using heq.symm

theorem restrictedFrame5_det_lower {P : Submodule ℚ (Fin 5 → ℚ)}
    {A₂ A₃ : Matrix (Fin 5) (Fin 5) ℚ} {Q : ℝ} {c₂ c₃ : Fin 5 → ℝ} {d : ℕ}
    (c : Basis (Fin d) ℤ (restrictedLattice5 P A₂ A₃ Q c₂ c₃))
    (v : Fin d → rationalRealSpan5 P) (hv : LinearIndependent ℝ v)
    (hvL : ∀ i, v i ∈ restrictedLattice5 P A₂ A₃ Q c₂ c₃) :
    1 ≤ |(Matrix.of (fun i => (c.ofZLatticeBasis ℝ _).equivFun (v i))).det| := by
  classical
  let b := c.ofZLatticeBasis ℝ (restrictedLattice5 P A₂ A₃ Q c₂ c₃)
  let z : Matrix (Fin d) (Fin d) ℤ := fun i j => c.repr ⟨v i,hvL i⟩ j
  have hz : Matrix.of (fun i => b.equivFun (v i)) = z.map (Int.castRingHom ℝ) := by
    ext i j
    exact Basis.ofZLatticeBasis_repr_apply ℝ _ c ⟨v i,hvL i⟩ j
  have hind : LinearIndependent ℝ (fun i => b.equivFun (v i)) :=
    hv.map' b.equivFun.toLinearMap b.equivFun.ker
  let a := basisOfPiSpaceOfLinearIndependent hind
  have hne : (Matrix.of (fun i => b.equivFun (v i))).det ≠ 0 := by
    rw [← Pi.basisFun_det_apply]
    have ha : (fun i => b.equivFun (v i)) = (a : Fin d → Fin d → ℝ) :=
      (coe_basisOfPiSpaceOfLinearIndependent hind).symm
    rw [ha]
    exact isUnit_iff_ne_zero.mp ((Pi.basisFun ℝ (Fin d)).isUnit_det a)
  have hcast : (Matrix.of (fun i => b.equivFun (v i))).det = (z.det : ℝ) := by
    rw [hz]
    exact ((Int.castRingHom ℝ).map_det z).symm
  have hzne : z.det ≠ 0 := by
    intro h
    apply hne
    rw [hcast,h,Int.cast_zero]
  have hz1 : (1 : ℤ) ≤ |z.det| := by
    have hp : (0 : ℤ) < |z.det| := abs_pos.mpr hzne
    omega
  change 1 ≤ |(Matrix.of (fun i => b.equivFun (v i))).det|
  rw [hcast]
  exact_mod_cast hz1

def restrictedCrosspolytopeMap5
    (c : Basis (Fin d) ℤ (restrictedLattice5 P A₂ A₃ Q c₂ c₃))
    (v : Fin d → rationalRealSpan5 P) (mu : Fin d → ℝ) :
    (Fin d → ℝ) →ₗ[ℝ] (Fin d → ℝ) :=
  (Pi.basisFun ℝ (Fin d)).constr ℕ
    (fun i => (mu i)⁻¹ • (c.ofZLatticeBasis ℝ _).equivFun (v i))

theorem restrictedCrosspolytopeMap5_det
    (c : Basis (Fin d) ℤ (restrictedLattice5 P A₂ A₃ Q c₂ c₃))
    (v : Fin d → rationalRealSpan5 P) (mu : Fin d → ℝ) :
    LinearMap.det (restrictedCrosspolytopeMap5 c v mu) =
      (∏ i, (mu i)⁻¹) * (Matrix.of (fun i => (c.ofZLatticeBasis ℝ _).equivFun (v i))).det := by
  rw [restrictedCrosspolytopeMap5,← LinearMap.det_toMatrix (Pi.basisFun ℝ (Fin d)),
    ← Basis.toMatrix_eq_toMatrix_constr,← Basis.det_apply,Pi.basisFun_det_apply]
  exact Matrix.det_mul_column (fun i => (mu i)⁻¹) _

theorem restrictedCrosspolytopeMap5_pullback
    (c : Basis (Fin d) ℤ (restrictedLattice5 P A₂ A₃ Q c₂ c₃))
    (v : Fin d → rationalRealSpan5 P) (mu : Fin d → ℝ) (x : Fin d → ℝ) :
    (c.ofZLatticeBasis ℝ _).equivFun.symm (restrictedCrosspolytopeMap5 c v mu x) =

  simp only [restrictedCrosspolytopeMap5,Basis.constr_apply_fintype,

theorem restrictedCrosspolytope5_subset
    (L : Matrix (Fin 5) (Fin 5) ℝ) (hL : IsUnit L) (hQ : 1 < Q) (c₀ : Fin 5 → ℝ)
    (c : Basis (Fin d) ℤ (restrictedLattice5 P A₂ A₃ Q c₂ c₃))
    (v : Fin d → rationalRealSpan5 P) (mu : Fin d → ℝ) (hpos : ∀ i, 0 < mu i)
    (hatt : ∀ i, restrictedGauge5 P L hL Q hQ c₀ (v i) = mu i) :
    restrictedCrosspolytopeMap5 c v mu '' {x | ∑ i, |x i| ≤ 1} ⊆
      (c.ofZLatticeBasis ℝ _).equivFun '' restrictedBody5 P L Q c₀ := by
  rintro _ ⟨x,hx,rfl⟩
  rw [LinearEquiv.image_eq_preimage]
  change (c.ofZLatticeBasis ℝ _).equivFun.symm (restrictedCrosspolytopeMap5 c v mu x) ∈
    restrictedBody5 P L Q c₀
  rw [restrictedCrosspolytopeMap5_pullback,restrictedBody5_gauge P L hL Q hQ c₀]
  change ‖originalCoordinates5 L hL Q hQ c₀
    ((∑ i, (x i / mu i) • v i : rationalRealSpan5 P) : Fin 5 → ℝ)‖ ≤ 1
  simp only [Submodule.coe_sum,Submodule.coe_smul,map_sum,map_smul]
  calc
    _ ≤ ∑ i, ‖(x i / mu i) • originalCoordinates5 L hL Q hQ c₀ (v i : Fin 5 → ℝ)‖ :=
      norm_sum_le _ _
    _ = ∑ i, |x i| := by
      apply Finset.sum_congr rfl
      intro i _
      rw [norm_smul,Real.norm_eq_abs]
      change |x i / mu i| * restrictedGauge5 P L hL Q hQ c₀ (v i) = |x i|
      rw [hatt,abs_div,abs_of_pos (hpos i),div_mul_cancel₀ _ (hpos i).ne']
    _ ≤ 1 := hx

theorem source_restricted_minkowski_lower5 (P : Submodule ℚ (Fin 5 → ℚ))
    (hPlo : 1 ≤ Module.finrank ℚ P) (hP : Module.finrank ℚ P ≤ 4)
    (A₂ A₃ : Matrix (Fin 5) (Fin 5) ℚ) (L : Matrix (Fin 5) (Fin 5) ℝ) (hL : IsUnit L)
    (Q : ℝ) (hQ : 1 < Q) (c₀ c₂ c₃ : Fin 5 → ℝ) :
    (2 : ℝ)^(Module.finrank ℚ P) / (Nat.factorial (Module.finrank ℚ P) : ℝ) ≤
      (∏ i, restrictedMinima5 P A₂ A₃ L hL Q hQ c₀ c₂ c₃ i) *
        (restrictedMeasure5 P).real (restrictedBody5 P L Q c₀) /
          ZLattice.covolume (restrictedLattice5 P A₂ A₃ Q c₂ c₃) (restrictedMeasure5 P) := by
  obtain ⟨v,hv,hvL,hpos,_,hatt,_⟩ := exists_restrictedMinima5_attaining P hP A₂ A₃ L hL Q hQ c₀ c₂ c₃
  let dim : Fin 5 := ⟨Module.finrank ℚ P,by omega⟩
  let G := restrictedLattice5 P A₂ A₃ Q c₂ c₃
  let c : Basis (Fin dim.val) ℤ G :=
    Module.finBasisOfFinrankEq ℤ G (restrictedLattice5_rank P hP A₂ A₃ Q c₂ c₃)
  let mu := restrictedMinima5 P A₂ A₃ L hL Q hQ c₀ c₂ c₃
  let b := c.ofZLatticeBasis ℝ G
  let T := restrictedCrosspolytopeMap5 c v mu
  have hinc := restrictedCrosspolytope5_subset L hL hQ c₀ c v mu hpos hatt
  have hcpt : IsCompact (b.equivFun '' restrictedBody5 P L Q c₀) :=
    (restrictedBody5_compact_convex P L hL Q hQ c₀).1.image b.equivFunL.continuous
  have hm : volume.real (T '' {x | ∑ i, |x i| ≤ 1}) ≤
      volume.real (b.equivFun '' restrictedBody5 P L Q c₀) :=
    ENNReal.toReal_mono hcpt.measure_lt_top.ne (measure_mono hinc)
  have hvol := restrictedCrosspolytope5_volume dim (by change 0 < Module.finrank ℚ P; omega)
    c v mu hpos
  have hbody := restrictedBasis5_volume_ratio c (restrictedBody5 P L Q c₀)
    (restrictedBody5_compact_convex P L hL Q hQ c₀).1
  change volume.real (restrictedCrosspolytopeMap5 c v mu '' {x | ∑ i, |x i| ≤ 1}) ≤ _ at hm
  rw [hvol,hbody] at hm
  have hprod : 0 < ∏ i, mu i := Finset.prod_pos (fun i _ => hpos i)
  have hdet := restrictedFrame5_det_lower c v hv hvL
  have hconst : 0 ≤ (2 : ℝ)^dim.val / (Nat.factorial dim.val : ℝ) := by positivity
  have hlow : (2 : ℝ)^dim.val / (Nat.factorial dim.val : ℝ) ≤
      ((2 : ℝ)^dim.val / (Nat.factorial dim.val : ℝ)) *
        |(Matrix.of (fun i => b.equivFun (v i))).det| := by
    exact le_mul_of_one_le_right hconst hdet
  have hcancel := (div_le_iff₀ hprod).mp hm
  calc
    _ ≤ ((2 : ℝ)^dim.val / (Nat.factorial dim.val : ℝ)) *
        |(Matrix.of (fun i => b.equivFun (v i))).det| := hlow
    _ ≤ ((restrictedMeasure5 P).real (restrictedBody5 P L Q c₀) /
        ZLattice.covolume G (restrictedMeasure5 P)) * (∏ i, mu i) := hcancel
    _ = _ := by ring

def quantitativeOriginalPoint5 (P : Submodule ℚ (Fin 5 → ℚ))
    (hP : Module.finrank ℚ P ≤ 4) (j : Fin 4)

theorem power_growth_of_product_decay5 (j : Fin 4) (Q delta C5 p volume covol : ℝ)
    (hQ : 0 < Q) (hC5 : 0 < C5) (hv : 0 < volume) (hc : 0 < covol)
    (hdec : p ≤ C5^(((j.val+1 : ℕ) : ℝ)/5) * Q^(-decayExponent5 j delta))
    (hmink : (2 : ℝ)^(j.val+1) / (Nat.factorial (j.val+1) : ℝ) ≤ p*volume/covol) :
    growthConstant5 j C5 * Q^(decayExponent5 j delta) ≤ volume/covol := by
  have hpow : Q^(-decayExponent5 j delta) * Q^(decayExponent5 j delta) = 1 := by
    rw [← Real.rpow_add hQ]
    simp
  have hstep : (2 : ℝ)^(j.val+1) / (Nat.factorial (j.val+1) : ℝ) ≤
      (C5^(((j.val+1 : ℕ) : ℝ)/5) * Q^(-decayExponent5 j delta))*(volume/covol) := by
    calc
      _ ≤ p*(volume/covol) := by simpa [mul_div_assoc] using hmink
      _ ≤ _ := mul_le_mul_of_nonneg_right hdec (le_of_lt (div_pos hv hc))
  have h := mul_le_mul_of_nonneg_right hstep
    (le_of_lt (Real.rpow_pos_of_pos hQ (decayExponent5 j delta)))
  have hcancel :
      (C5^(((j.val+1 : ℕ) : ℝ)/5) * Q^(-decayExponent5 j delta)) *
        (volume/covol) * Q^(decayExponent5 j delta) =
      (volume/covol) * C5^(((j.val+1 : ℕ) : ℝ)/5) := by
    calc
      _ = (volume/covol) * C5^(((j.val+1 : ℕ) : ℝ)/5) *
          (Q^(-decayExponent5 j delta)*Q^(decayExponent5 j delta)) := by ring
      _ = _ := by rw [hpow, mul_one]
  rw [hcancel] at h
  unfold growthConstant5
  rw [div_mul_eq_mul_div]
  exact (div_le_iff₀ (Real.rpow_pos_of_pos hC5 _)).mpr h

theorem source_quantitative_original_point5 (P : Submodule ℚ (Fin 5 → ℚ))
    (hPlo : 1 ≤ Module.finrank ℚ P) (hP : Module.finrank ℚ P ≤ 4)

  have hMlower := source_restricted_minkowski_lower5 P hPlo hP A₂ A₃ L hL Q hQ c₀ c₂ c₃
  have hM : (2 : ℝ)^(j.val+1) / (Nat.factorial (j.val+1) : ℝ) ≤
      (∏ i, restrictedMinima5 P A₂ A₃ L hL Q hQ c₀ c₂ c₃ i) *
        (restrictedMeasure5 P).real (restrictedBody5 P L Q c₀) /
          ZLattice.covolume (restrictedLattice5 P A₂ A₃ Q c₂ c₃) (restrictedMeasure5 P) := by
    calc
      _ = (2 : ℝ)^(Module.finrank ℚ P) / (Nat.factorial (Module.finrank ℚ P) : ℝ) := by rw [hdim]

  exact ⟨h718,h719,heq,hdec,hReal.trans hdim,hInt.trans hdim,hVol,hFinite,hCov,hM,
