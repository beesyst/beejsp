theorem weighted_diagonal_gcd_divides3 (t z : Fin 3 → ℤ) (hz : PrimitiveTriple z)
    (g : ℤ) (hg : ∀ i,g∣t i*z i) : g∣t 0*t 1*t 2 := by
  obtain ⟨c,hc⟩ := hz
  have hd (i : Fin 3) : g∣(t 0*t 1*t 2)*z i := by
    have h0 : g∣(t 0*t 1*t 2)*z 0 := by
      have hh := dvd_mul_of_dvd_right (hg 0) (t 1*t 2)
      convert hh using 1 <;> ring
    have h1 : g∣(t 0*t 1*t 2)*z 1 := by
      have hh := dvd_mul_of_dvd_right (hg 1) (t 0*t 2)
      convert hh using 1 <;> ring
    have h2 : g∣(t 0*t 1*t 2)*z 2 := by
      have hh := dvd_mul_of_dvd_right (hg 2) (t 0*t 1)
      convert hh using 1 <;> ring
    fin_cases i <;> assumption
  have hs : g∣∑ i,c i*((t 0*t 1*t 2)*z i) :=
    Finset.dvd_sum (fun i _ => dvd_mul_of_dvd_right (hd i) (c i))
  have he : (∑ i,c i*((t 0*t 1*t 2)*z i))=t 0*t 1*t 2 := by
    calc
      _=(t 0*t 1*t 2)*(∑ i,c i*z i) := by
        rw [Finset.mul_sum]
        apply Finset.sum_congr rfl
        intro i _
        ring
      _=_ := by rw [hc,mul_one]
  rwa [he] at hs

theorem weighted_diagonal_gcd_values_finite3 (t : Fin 3 → ℤ) (ht : ∀ i,t i≠0) :
    {g : ℕ | 0<g ∧ (g : ℤ)∣t 0*t 1*t 2}.Finite := by
  have hn : t 0*t 1*t 2≠0 := mul_ne_zero (mul_ne_zero (ht 0) (ht 1)) (ht 2)
  apply (Set.finite_Iic (t 0*t 1*t 2).natAbs).subset
  intro g hg
  exact Nat.le_of_dvd (Int.natAbs_pos.mpr hn) (Int.natCast_dvd.mp hg.2)

def weightedBalancedValues3 (H K : ℝ) (a : Fin 3 → Fin 3 → ℝ) (tau i : Fin 3) : ℝ :=
  if tau=0 then a tau i*(H/K)^((1 : ℝ)/3) else a tau i

theorem weighted_balanced_weights3 (H K C : ℝ) (hH : 3≤H) (hK : 0<K)
    (hC : C≤H) (hKi : K⁻¹≤H) (hK4 : K^4≤H)
    (a : Fin 3 → Fin 3 → ℝ) (ha : ∀ tau i,0<a tau i)
    (hreal : ∀ i,a 0 i≤C*H)
    (hfinite : ∀ tau,tau≠0 → ∀ i,a tau i≤1)
    (hinv : ∀ tau i,(a tau i)⁻¹≤C*H)
    (hprod : (∏ tau,∏ i,a tau i)*H=K) :
    let Q := H^4
    let W := weightedBalancedValues3 H K a
    1<Q ∧ sourceWeightConditions3 (sourceLogWeights3 Q W) ∧
      (∀ tau i,Q^(sourceLogWeights3 Q W tau i)=W tau i) ∧
      (∀ i,a 0 i≤Q^(sourceLogWeights3 Q W 0 i-(1 : ℝ)/16)) ∧
      (∀ tau,tau≠0 → ∀ i,a tau i=Q^(sourceLogWeights3 Q W tau i)) := by
  intro Q W
  have hH0 : 0<H := by linarith
  have hH1 : 1≤H := by linarith
  have hQ : 1<Q := by dsimp [Q]; nlinarith [sq_nonneg (H^2-1)]
  have hQ0 := zero_lt_one.trans hQ
  let r := (H/K)^((1 : ℝ)/3)
  let q := H^((1 : ℝ)/4)
  have hr : 0<r := Real.rpow_pos_of_pos (div_pos hH0 hK) _
  have hq : 0<q := Real.rpow_pos_of_pos hH0 _
  have hrpow : r^3=H/K := by
    dsimp [r]
    rw [← Real.rpow_mul_natCast (div_pos hH0 hK).le]
    norm_num
  have hqpow : q^4=H := by
    dsimp [q]
    rw [← Real.rpow_mul_natCast hH0.le]
    norm_num
  have hq1 : 1≤q := Real.one_le_rpow hH1 (by norm_num)
  have hKq : K≤q := le_of_pow_le_pow_left (by decide : (4 : ℕ)≠0) hq.le (by rw [hqpow];exact hK4)
  have hqr : q≤r := by
    apply le_of_pow_le_pow_left (by decide : (3 : ℕ)≠0) hr.le
    rw [hrpow]
    apply (le_div_iff₀ hK).mpr
    have hh := mul_le_mul_of_nonneg_left hKq (pow_nonneg hq.le 3)
    nlinarith [hqpow]
  have hr1 : 1≤r := hq1.trans hqr
  have hrH : r≤H := by
    apply le_of_pow_le_pow_left (by decide : (3 : ℕ)≠0) hH0.le
    rw [hrpow]
    have hh : H/K≤H^2 := by
      rw [div_eq_mul_inv]
      nlinarith [mul_le_mul_of_nonneg_left hKi hH0.le]
    exact hh.trans (by nlinarith [mul_le_mul_of_nonneg_left hH1 (sq_nonneg H)])
  have hQq : Q^((1 : ℝ)/16)=q := by
    dsimp [Q,q]
    rw [← Real.rpow_natCast,← Real.rpow_mul hH0.le]
    congr 1
    norm_num
  have hW : ∀ tau i,0<W tau i := by
    intro tau i
    dsimp [W,weightedBalancedValues3]
    split_ifs
    · exact mul_pos (ha tau i) hr
    · exact ha tau i
  have hcut : ∀ tau i,(W tau i)⁻¹<Q := by
    intro tau i
    have hb : (a tau i)⁻¹≤H^2 := (hinv tau i).trans (by nlinarith [mul_le_mul_of_nonneg_right hC hH0.le])
    have hpow : H^2<Q := pow_lt_pow_right₀ (by linarith : 1<H) (by decide : (2 : ℕ)<4)
    by_cases ht : tau=0
    · subst tau
      change (a 0 i*r)⁻¹<Q
      rw [mul_inv_rev]
      have hri : r⁻¹≤1 := (inv_le_one₀ hr).mpr hr1
      exact (by nlinarith [mul_le_mul_of_nonneg_right hri (inv_pos.mpr (ha 0 i)).le] : r⁻¹*(a 0 i)⁻¹≤H^2).trans_lt hpow
    · simpa only [W,weightedBalancedValues3,if_neg ht] using hb.trans_lt hpow
  have hcut0 : ∀ i,W 0 i<Q := by
    intro i
    change a 0 i*r<Q
    have hb : a 0 i≤H^2 := (hreal i).trans (by nlinarith [mul_le_mul_of_nonneg_right hC hH0.le])
    have hh := mul_le_mul hb hrH hr.le (sq_nonneg H)
    exact (by nlinarith : a 0 i*r≤H^3).trans_lt (pow_lt_pow_right₀ (by linarith : 1<H) (by decide : (3 : ℕ)<4))
  have hWfin : ∀ tau,tau≠0 → ∀ i,W tau i≤1 := by
    intro tau ht i
    simpa only [W,weightedBalancedValues3,if_neg ht] using hfinite tau ht i
  have hWprod : (∏ tau,∏ i,W tau i)=1 := by
    simp only [W,weightedBalancedValues3,Fin.prod_univ_three,Fin.isValue,Fin.reduceEq,ite_true,ite_false]
    have hp := hprod
    simp only [Fin.prod_univ_three] at hp
    change (a 0 0*r*(a 0 1*r)*(a 0 2*r))*(a 1 0*a 1 1*a 1 2)*(a 2 0*a 2 1*a 2 2)=1
    calc
      _=(a 0 0*a 0 1*a 0 2)*(a 1 0*a 1 1*a 1 2)*(a 2 0*a 2 1*a 2 2)*r^3 := by ring
      _=1 := by rw [hrpow,← mul_div_assoc,hp,div_self hK.ne']
  have heq := sourceLogWeights3_rpow Q hQ W hW
  refine ⟨hQ,sourceLogWeights3_conditions Q hQ W hW hcut hcut0 hWfin hWprod,heq,?_,?_⟩
  · intro i
    rw [Real.rpow_sub hQ0,heq,hQq]
    change a 0 i≤(a 0 i*r)/q
    exact (le_div_iff₀ hq).mpr (mul_le_mul_of_nonneg_left hqr (ha 0 i).le)
  · intro tau ht i
    rw [heq]
    simp only [W,weightedBalancedValues3,if_neg ht]

end

      exact (by apply (inv_le_iff_one_le_mul₀ ha).mpr;nlinarith : |(m : ℚ)|⁻¹≤|(m : ℚ)|).trans harch

theorem weighted_primitive_triple_reduction3 (z : Fin 3 → ℤ) (hz0 : z 0 ≠ 0) :
    ∃ g : ℕ, 0 < g ∧ (g : ℤ) ∣ z 0 ∧ ∃ y : Fin 3 → ℤ,
      PrimitiveTriple y ∧ ∀ i, z i = (g : ℤ) * y i := by
  let k := Int.gcd (z 1) (z 2)
  let g := Int.gcd (z 0) (k : ℤ)
  have hg : 0 < g := Int.gcd_pos_of_ne_zero_left _ hz0
  have hg0 : (g : ℤ) ∣ z 0 := Int.gcd_dvd_left
  have hgk : (g : ℤ) ∣ (k : ℤ) := Int.gcd_dvd_right
  have hg1 : (g : ℤ) ∣ z 1 := hgk.trans (Int.gcd_dvd_left)
  have hg2 : (g : ℤ) ∣ z 2 := hgk.trans (Int.gcd_dvd_right)
  obtain ⟨y0, hy0⟩ := hg0
  obtain ⟨y1, hy1⟩ := hg1
  obtain ⟨y2, hy2⟩ := hg2
  let y : Fin 3 → ℤ := ![y0, y1, y2]
  have hscale (i : Fin 3) : z i = (g : ℤ) * y i := by
    fin_cases i <;> assumption
  let a := Int.gcdA (z 0) (k : ℤ)
  let b := Int.gcdB (z 0) (k : ℤ)
  let c := Int.gcdA (z 1) (z 2)
  let d := Int.gcdB (z 1) (z 2)
  have hgbez : (g : ℤ) = z 0 * a + (k : ℤ) * b := Int.gcd_eq_gcd_ab _ _
  have hkbez : (k : ℤ) = z 1 * c + z 2 * d := Int.gcd_eq_gcd_ab _ _
  have hcoeff : (g : ℤ) * (a * y0 + (b * c) * y1 + (b * d) * y2) = (g : ℤ) * 1 := by
    calc
      _ = z 0 * a + (k : ℤ) * b := by rw [hkbez, hy0, hy1, hy2]; ring
      _ = _ := by rw [← hgbez]; ring
  have hgneq : (g : ℤ) ≠ 0 := by exact_mod_cast (ne_of_gt hg)
  have hprimitive : PrimitiveTriple y := by
    refine ⟨![a, b * c, b * d], ?_⟩
    simpa [y, Fin.sum_univ_succ, Matrix.cons_val, Matrix.head_cons, Matrix.tail_cons,
      add_assoc] using mul_left_cancel₀ hgneq hcoeff
  exact ⟨g, hg, Int.gcd_dvd_left, y, hprimitive, hscale⟩


theorem weighted_primitive_sunit_normalized3 (x : Fin 3 → ℚ) (h0 : x 0=1)
    (u1 u2 : fixedSUnitGroup) (hu1 : ((u1 : ℚˣ) : ℚ)=x 1) (hu2 : ((u2 : ℚˣ) : ℚ)=x 2) :
    ∃ z : Fin 3 → ℤ,∃ s : fixedSUnitGroup,PrimitiveTriple z ∧ z≠0 ∧
      ∀ i : Fin 3,(z i : ℚ)=((s : ℚˣ) : ℚ)*x i := by
  obtain ⟨s1, hs1⟩ := sunit_denominator u1
  obtain ⟨s2, hs2⟩ := sunit_denominator u2
  have hd1 : ((s1 : ℚˣ) : ℚ) = ((x 1).den : ℚ) := by simpa [hu1] using hs1
  have hd2 : ((s2 : ℚˣ) : ℚ) = ((x 2).den : ℚ) := by simpa [hu2] using hs2

    have h := integralPair_scale (x 1) (1 : Fin 2)

    have h := integralPair_scale (x 2) (1 : Fin 2)
    exact h
  have hw (i : Fin 3) : (w i : ℚ) = ((s0 : ℚˣ) : ℚ) * x i := by

    · simp [w, h0]
    · change (((x 1).num * ((x 2).den : ℤ) : ℤ) : ℚ) = (D : ℚ) * x 1
      push_cast
      rw [hnum1]
      dsimp [D]
      push_cast
      ring
    · change (((x 2).num * ((x 1).den : ℤ) : ℤ) : ℚ) = (D : ℚ) * x 2
      push_cast
      rw [hnum2]
      dsimp [D]
      push_cast
      ring
  have hD : D ≠ 0 := mul_ne_zero (Rat.den_ne_zero _) (Rat.den_ne_zero _)
  have hw0 : w 0 ≠ 0 := by
    change (D : ℤ) ≠ 0
    exact_mod_cast hD

  have hgD : g ∣ D := Int.natCast_dvd_natCast.mp hgdvd

      simpa [hu1] using hp1
    · apply sunit_den_prime_support u2 p hp
      simpa [hu2] using hp2
  obtain ⟨ug, hug⟩ := positive_smooth23_sunit4 (ne_of_gt hg) hgsmooth
  let s := s0 / ug
  have hzscale (i : Fin 3) : (z i : ℚ) = ((s : ℚˣ) : ℚ) * x i := by
    have hc := congrArg ((↑) : ℤ → ℚ) (hscale i)
    push_cast at hc
    have hsval : ((s : ℚˣ) : ℚ) = ((s0 : ℚˣ) : ℚ) / (g : ℚ) := by simp [s, hug]
    rw [hsval]
    have hg0 : (g : ℚ) ≠ 0 := by exact_mod_cast (ne_of_gt hg)
    rw [div_mul_eq_mul_div]
    apply (eq_div_iff hg0).mpr
    have hwi := hw i
    rw [hc] at hwi
    nlinarith [hwi]
  have hnz : z ≠ 0 := by
    intro heq
    have hh := hzscale 0
    simp only [heq,Pi.zero_apply,Int.cast_zero,h0,mul_one] at hh

    (g : ℕ) (u : Fin 3 → fixedSUnitGroup) (s : fixedSUnitGroup)

      ((g : ℚ)/((s : ℚˣ) : ℚ)) • (fun i => (y i : ℚ)) := by
    ext i
    change (weightedIntegerDiagonal3 t *ᵥ (fun i => ((u i : ℚˣ) : ℚ))) i=_
    simp only [weightedIntegerDiagonal3,Matrix.mulVec_diagonal,Pi.smul_apply,smul_eq_mul]
    have hc := congrArg ((↑) : ℤ → ℚ) (he i)
    push_cast at hc
    rw [hz i] at hc
    apply (mul_left_cancel₀ hs)
    calc
      _=(t i : ℚ)*(((s : ℚˣ) : ℚ)*((u i : ℚˣ) : ℚ)) := by ring
      _=(g : ℚ)*(y i : ℚ) := hc
      _=_ := by field_simp
  change weightedDiagonalMap3 t (fun i => ((u i : ℚˣ) : ℚ))∈P
  rwa [heq]
