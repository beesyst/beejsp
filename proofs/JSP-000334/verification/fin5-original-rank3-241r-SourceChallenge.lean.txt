    (hz : ∀ s : Fin 10, s ≠ 9 → (∑ t, (b t : ℚ)*wedge237 (u (lo237 s)) (u (hi237 s)) t) = 0)

    radical241 b = Submodule.span ℚ (Set.range (fun j : Fin 3 => u (Fin.castLE (by decide) j))) ∧
    Module.finrank ℚ (radical241 b) = 3 ∧
    ∀ x, x ∈ radical241 b ↔ ∀ y, (∑ t, (b t : ℚ)*wedge237 x y t) = 0 := by
  have hr := radical_rank3_241 b u hu
    (fun s hs => by simpa only [pairForm_apply241] using hz s hs)
    (by simpa only [pairForm_apply241] using hc)
  refine ⟨hr.1,hr.2,?_⟩
  intro x
  change pairForm241 b x = 0 ↔ _
  constructor
  · intro hx y
    have h := congrArg (fun f : Module.Dual ℚ (Fin 5 → ℚ) => f y) hx
    simpa only [pairForm_apply241,LinearMap.zero_apply] using h
  · intro hx
    apply LinearMap.ext
    intro y
    exact hx y
