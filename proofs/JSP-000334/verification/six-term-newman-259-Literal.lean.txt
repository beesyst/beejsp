theorem six_term_annihilator_degenerate5 (x : Fin 5 → ℚ)
    (hx : ∀ i,x i≠0) (hs : ∑ i,x i=0) (hn : ¬Nondegenerate x) :
    x 0+x 1=0 ∨ x 0+x 2=0 ∨ x 0+x 3=0 ∨ x 0+x 4=0 ∨ x 0+x 1+x 2=0 ∨ x 0+x 1+x 3=0 ∨ x 0+x 1+x 4=0 ∨ x 0+x 2+x 3=0 ∨ x 0+x 2+x 4=0 ∨ x 0+x 3+x 4=0 := by
  classical
  unfold Nondegenerate at hn
  push_neg at hn
  obtain ⟨I,hI,hproper,hzero⟩ := hn
  have h0 := hx 0
  have h1 := hx 1
  have h2 := hx 2
  have h3 := hx 3
  have h4 := hx 4
  have hs' : x 0+x 1+x 2+x 3+x 4=0 := by
    simpa [Fin.sum_univ_succ,add_assoc] using hs
  clear hs
  have he : x=![x 0,x 1,x 2,x 3,x 4] := by
    ext i
    fin_cases i <;> rfl
  rw [he] at hzero
  fin_cases I <;> norm_num [Finset.sum_insert,Matrix.cons_val_succ,Matrix.cons_val_zero] at hzero hI
  all_goals first
    | exact False.elim (h0 (by linarith))
    | exact False.elim (h1 (by linarith))
    | exact False.elim (h2 (by linarith))
    | exact False.elim (h3 (by linarith))
    | exact False.elim (h4 (by linarith))
    | exact Or.inl (by linarith)
    | exact Or.inr (Or.inl (by linarith))
    | exact Or.inr (Or.inr (Or.inl (by linarith)))
    | exact Or.inr (Or.inr (Or.inr (Or.inl (by linarith))))
    | exact Or.inr (Or.inr (Or.inr (Or.inr (Or.inl (by linarith)))))
    | exact Or.inr (Or.inr (Or.inr (Or.inr (Or.inr (Or.inl (by linarith))))))
    | exact Or.inr (Or.inr (Or.inr (Or.inr (Or.inr (Or.inr (Or.inl (by linarith)))))))
    | exact Or.inr (Or.inr (Or.inr (Or.inr (Or.inr (Or.inr (Or.inr (Or.inl (by linarith))))))))

theorem six_term_nondegenerate_coarsen5 (x : Fin 6 → ℚ) (hx : Nondegenerate x)
    (f : Fin 6 → Fin 5) (hf : Function.Surjective f) :
    Nondegenerate (fun j => ∑ i : Fin 6,if f i=j then x i else 0) := by
  classical
  intro I hI hproper
  let J : Finset (Fin 6) := Finset.univ.filter (fun i => f i∈I)
  have hJ : J.Nonempty := by
    obtain ⟨j,hj⟩ := hI
    obtain ⟨i,rfl⟩ := hf j
    exact ⟨i,by simpa [J] using hj⟩
  have hJproper : J≠Finset.univ := by
    intro he
    apply hproper
    apply Finset.eq_univ_of_forall
    intro j
    obtain ⟨i,rfl⟩ := hf j
    have hi : i∈J := by rw [he];exact Finset.mem_univ i
    simpa [J] using hi
  have heq : (∑ j∈I,∑ i : Fin 6,if f i=j then x i else 0)=∑ i∈J,x i := by
    rw [Finset.sum_comm]
    simp [J,Finset.sum_filter,Finset.sum_ite_eq']
  rw [heq]
  exact hx J hJ hJproper

theorem six_term_weighted_triple_projection5 (S : Set (Fin 6 → ℚ))
    (hs : S⊆normalizedRelations fixedSUnitGroup 5)
    (A B C D : ℚ) (hA : A≠0) (hB : B≠0) (hC : C≠0) (hD : D≠0)
    (i j k : Fin 6) (recover : ℚ → ℚ → ℚ → Fin 6 → ℚ)
    (he : ∀ x∈S,A+B*x i+C*x j+D*x k=0)
    (hn : ∀ x∈S,Nondegenerate (![A,B*x i,C*x j,D*x k] : Fin 4 → ℚ))
    (hr : ∀ x∈S,recover (x i) (x j) (x k)=x) : S.Finite := by
  let f : (fixedSUnitGroup × fixedSUnitGroup × fixedSUnitGroup) → Fin 6 → ℚ :=
    fun p => recover ((p.1 : ℚˣ) : ℚ) ((p.2.1 : ℚˣ) : ℚ) ((p.2.2 : ℚˣ) : ℚ)
  apply ((weighted_four_term_finite A B C D hA hB hC hD).image f).subset
  intro x hx
  obtain ⟨u,hu⟩ := (hs hx).2.1 i
  obtain ⟨v,hv⟩ := (hs hx).2.1 j
  obtain ⟨w,hw⟩ := (hs hx).2.1 k
  refine ⟨(u,v,w),?_,?_⟩
  · change A+B*((u : ℚˣ) : ℚ)+C*((v : ℚˣ) : ℚ)+D*((w : ℚˣ) : ℚ)=0 ∧ _
    rw [hu,hv,hw]
    exact ⟨he x hx,hn x hx⟩
  · change recover ((u : ℚˣ) : ℚ) ((v : ℚˣ) : ℚ) ((w : ℚˣ) : ℚ)=x
    rw [hu,hv,hw]
    exact hr x hx

theorem six_term_weighted_pair_projection5 (S : Set (Fin 6 → ℚ))
    (hs : S⊆normalizedRelations fixedSUnitGroup 5)
    (A B C : ℚ) (hA : A≠0) (hB : B≠0) (hC : C≠0)
    (i j : Fin 6) (recover : ℚ → ℚ → Fin 6 → ℚ)
    (he : ∀ x∈S,A+B*x i+C*x j=0)
    (hr : ∀ x∈S,recover (x i) (x j)=x) : S.Finite := by
  let f : (fixedSUnitGroup × fixedSUnitGroup) → Fin 6 → ℚ :=
    fun p => recover ((p.1 : ℚˣ) : ℚ) ((p.2 : ℚˣ) : ℚ)
  apply ((weighted_fixed_sunit_pair_finite A B C hA hB hC).image f).subset
  intro x hx
  obtain ⟨u,hu⟩ := (hs hx).2.1 i
  obtain ⟨v,hv⟩ := (hs hx).2.1 j
  refine ⟨(u,v),?_,?_⟩
  · change A+B*((u : ℚˣ) : ℚ)+C*((v : ℚˣ) : ℚ)=0
    rw [hu,hv]
    exact he x hx
  · change recover ((u : ℚˣ) : ℚ) ((v : ℚˣ) : ℚ)=x
    rw [hu,hv]
    exact hr x hx

theorem six_term_finite_over_pairs5 (S : Set (Fin 6 → ℚ))
    (f : (Fin 6 → ℚ) → fixedSUnitGroup × fixedSUnitGroup)
    (A B C : ℚ) (hA : A≠0) (hB : B≠0) (hC : C≠0)
    (he : ∀ x∈S,A+B*(((f x).1 : ℚˣ) : ℚ)+C*(((f x).2 : ℚˣ) : ℚ)=0)
    (hf : ∀ p∈weightedFixedSUnitPairs A B C,{x∈S | f x=p}.Finite) : S.Finite := by
  have hU : (⋃ p∈weightedFixedSUnitPairs A B C,{x∈S | f x=p}).Finite :=
    (weighted_fixed_sunit_pair_finite A B C hA hB hC).biUnion hf
  apply hU.subset
  intro x hx
  exact Set.mem_iUnion.mpr ⟨f x,Set.mem_iUnion.mpr ⟨he x hx,hx,rfl⟩⟩


theorem six_term_weighted_quad_projection5 (S : Set (Fin 6 → ℚ))
    (hs : S⊆normalizedRelations fixedSUnitGroup 5)
    (A : ℚ) (B : Fin 4 → ℚ) (hA : A≠0) (hB : ∀ i,B i≠0)
    (idx : Fin 4 → Fin 6) (recover : (Fin 4 → ℚ) → Fin 6 → ℚ)
    (he : ∀ x∈S,A+∑ i,B i*x (idx i)=0)
    (hn : ∀ x∈S,Nondegenerate (Fin.cons A (fun i => B i*x (idx i))))
    (hr : ∀ x∈S,recover (fun i => x (idx i))=x) : S.Finite := by
  let f (u : Fin 4 → fixedSUnitGroup) := recover (fun i => ((u i : ℚˣ) : ℚ))
  apply ((weighted_five_term_finite A B hA hB).image f).subset
  intro x hx
  let u (i : Fin 4) := sixTermUnit5 x (idx i)
  have hu (i : Fin 4) : ((u i : ℚˣ) : ℚ)=x (idx i) := six_term_unit_value5 (hs hx) _
  refine ⟨u,?_,?_⟩
  · change A+∑ i,B i*((u i : ℚˣ) : ℚ)=0 ∧ _
    change A+∑ i,B i*((u i : ℚˣ) : ℚ)=0 ∧ Nondegenerate (Fin.cons A (fun i => B i*((u i : ℚˣ) : ℚ)))
    simp_rw [hu]
    exact ⟨he x hx,hn x hx⟩
  · change recover (fun i => ((u i : ℚˣ) : ℚ))=x
    simp_rw [hu]
    exact hr x hx

theorem six_term_proper_annihilator5 (P : Submodule ℚ (Fin 5 → ℚ)) (hP : P≠⊤) :
    ∃ b : Fin 5 → ℚ,(∃ i,b i≠0) ∧ ∀ y∈P,∑ i,b i*y i=0 := by
  obtain ⟨f,hf,hker⟩ := P.exists_le_ker_of_lt_top (lt_top_iff_ne_top.mpr hP)
  let b (i : Fin 5) := f (Pi.single i 1)
  have heval (y : Fin 5 → ℚ) : f y=∑ i,b i*y i := by
    have hy : y=∑ i,y i • Pi.single i 1 := by
      ext j
      simp [Finset.sum_apply,Pi.single_apply]
    conv_lhs => rw [hy]
    rw [map_sum]
    apply Finset.sum_congr rfl
    intro i _
    simp only [map_smul,smul_eq_mul,b,mul_comm]
  refine ⟨b,?_,?_⟩
  · by_contra hnone
    have hz (i : Fin 5) : b i=0 := by
      by_contra hi
      exact hnone ⟨i,hi⟩
    apply hf
    apply LinearMap.ext
    intro y
    change f y=0
    rw [heval]
    simp only [hz,zero_mul,Finset.sum_const_zero]
  · intro y hy
    exact (heval y).symm.trans (hker hy)
