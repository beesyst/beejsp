theorem coefficient_zero_of_unit10 {F : Type*} [Field F] [CharZero F]
    {m : ℕ} (L : Matrix (Fin 10) (Fin 10) F) (hL : IsUnit L)
    (r : Fin m → ℕ) (eta : ℝ) (P : MvPolynomial (Vars10 m) ℤ)
    (hi : matrixIndex10 L r eta P) (i : Fin 10) (J b : Vars10 m →₀ ℕ)
    (hb : rothSourceDerivativeWeight r J + coordinateWeight r i b < (1/10-eta)*m) :
    (expansion L J P).coeff b = 0 := by
  have hd : L.det ≠ 0 := ((Matrix.isUnit_iff_isUnit_det L).mp hL).ne_zero
  have hn : ∃ p, L i p ≠ 0 := by
    by_contra! hh
    exact hd (Matrix.det_eq_zero_of_row_eq_zero i hh)
  obtain ⟨p,hp⟩ := hn
  apply coeff_zero_below_index r L hd i p hp ((1/10-eta)*m) P
  · intro D hD
    exact hi i D hD (fun _ => p) (fun _ => hp)
  · exact hb

      (∀ tau, matrixIndex10 (fun i j => (⟨SourceForms10.forms10 A2 A3 L tau i j,hmem tau i j⟩ : K)) r eta P) ∧
      (∀ tau, matrixIndex10 (SourceForms10.forms10 A2 A3 L tau) r eta P) ∧

  obtain ⟨E,hE,hbound⟩ := source_coefficient_bounds A2 A3 L D hD

  intro eta heta m hm r hr

def logWeight {m : ℕ} (Q : Fin m → ℝ) (b : Vars10 m →₀ ℕ) (i : Fin 10) : ℝ :=
  ∑ h, (b (h,i) : ℝ) * Real.log (Q h)

theorem log_weight_transport {m : ℕ} (r : Fin m → ℕ) (hr : ∀ h, 0 < r h)
    (Q : Fin m → ℝ) (eta A : ℝ) (heta : 0 ≤ eta) (hA : 0 ≤ A)
    (hlog : ∀ h, A ≤ (r h : ℝ)*Real.log (Q h) ∧ (r h : ℝ)*Real.log (Q h) ≤ (1+eta)*A)
    (b : Vars10 m →₀ ℕ) (hdegree : ∀ h, (∑ i : Fin 10, b (h,i)) ≤ r h)
    (hbalance : ∀ i, |coordinateWeight r i b - (m : ℝ)/10| ≤ 54*eta*m) :
    ∀ i, |logWeight Q b i - A*m/10| ≤ 55*A*m*eta := by
  intro i
  have hnorm : coordinateWeight r i b ≤ (m : ℝ) := by
    unfold coordinateWeight
    calc
      _ ≤ ∑ _h : Fin m, (1 : ℝ) := by
        apply Finset.sum_le_sum
        intro h hh
        apply (div_le_one (by exact_mod_cast hr h)).mpr
        exact_mod_cast (Finset.single_le_sum (fun j _ => Nat.zero_le (b (h,j)))
          (Finset.mem_univ i)).trans (hdegree h)
      _ = _ := by simp
  have hterm (h : Fin m) : (b (h,i) : ℝ)*Real.log (Q h) =
      ((b (h,i) : ℝ)/r h)*((r h : ℝ)*Real.log (Q h)) := by
    have hn : (r h : ℝ) ≠ 0 := ne_of_gt (by exact_mod_cast hr h)
    field_simp [hn]
    ring
  have hlo : A*coordinateWeight r i b ≤ logWeight Q b i := by
    unfold coordinateWeight logWeight
    rw [Finset.mul_sum]
    apply Finset.sum_le_sum
    intro h hh
    rw [hterm]
    simpa only [mul_comm] using mul_le_mul_of_nonneg_left (hlog h).1
      (show 0 ≤ (b (h,i) : ℝ)/(r h : ℝ) from div_nonneg (Nat.cast_nonneg _) (Nat.cast_nonneg _))
  have hhi : logWeight Q b i ≤ (1+eta)*A*coordinateWeight r i b := by
    unfold coordinateWeight logWeight
    rw [Finset.mul_sum]
    apply Finset.sum_le_sum
    intro h hh
    rw [hterm]
    simpa only [mul_comm] using mul_le_mul_of_nonneg_left (hlog h).2
      (show 0 ≤ (b (h,i) : ℝ)/(r h : ℝ) from div_nonneg (Nat.cast_nonneg _) (Nat.cast_nonneg _))
  have hbal := abs_le.mp (hbalance i)
  have hs1 := mul_le_mul_of_nonneg_left hbal.1 hA
  have hs2 := mul_le_mul_of_nonneg_left hbal.2 hA
  have herr := mul_le_mul_of_nonneg_left hnorm (mul_nonneg heta hA)
  apply abs_le.mpr
  constructor <;> nlinarith

theorem monomial_bound {K : Type*} [NormedField K] {m : ℕ}
    (A : Matrix (Fin 10) (Fin 10) K) (v : Fin m → Fin 10 → ℤ)
    (Q : Fin m → ℝ) (hQ : ∀ h, 0 < Q h) (c : Fin 10 → ℝ)
    (hv : ∀ h i, ‖(A *ᵥ (fun j => (v h j : K))) i‖ ≤ (Q h)^(c i))
    (b : Vars10 m →₀ ℕ) :
    ‖∏ u : Vars10 m, ((A *ᵥ (fun i => (v u.1 i : K))) u.2)^b u‖ ≤
      Real.exp (∑ i : Fin 10, c i * logWeight Q b i) := by
  rw [norm_prod]
  calc
    _ ≤ ∏ u : Vars10 m, ((Q u.1)^(c u.2))^b u := by
      apply Finset.prod_le_prod (fun _ _ => norm_nonneg _)
      intro u hu
      rw [norm_pow]
      exact pow_le_pow_left₀ (norm_nonneg _) (hv u.1 u.2) _
    _ = _ := by
      have he (u : Vars10 m) : ((Q u.1)^(c u.2))^b u =
          Real.exp ((b u : ℝ)*(Real.log (Q u.1)*c u.2)) := by
        rw [Real.rpow_def_of_pos (hQ u.1), ← Real.exp_nat_mul]
      simp_rw [he]
      rw [← Real.exp_sum]
      congr 1
      unfold logWeight
      rw [Fintype.sum_prod_type, Finset.sum_comm]
      apply Finset.sum_congr rfl
      intro i hi
      rw [Finset.mul_sum]
      apply Finset.sum_congr rfl
      intro h hh
      ring

theorem expansion_support_card {K : Type*} [Field K] [CharZero K] {m : ℕ}
    (r : Fin m → ℕ) (A : Matrix (Fin 10) (Fin 10) K) (J : Vars10 m →₀ ℕ)
    (P : MvPolynomial (Vars10 m) ℤ) (hP : homogeneous10 r P) :
    (expansion A J P).support.card ≤ 1024^(∑ h, r h) := by
  let s : Fin m → ℕ := fun h => r h - ∑ i : Fin 10, J (h,i)
  have hs : (expansion A J P).support ⊆ Finset.univ.image (@exponent10 m s) := by
    intro b hb
    have hdeg (h : Fin m) : (∑ i : Fin 10, b (h,i)) = s h :=
      expansion_block_degree r A J P hP b (MvPolynomial.mem_support_iff.mp hb) h
    have hle (h : Fin m) (i : Fin 10) : b (h,i) ≤ s h := by
      rw [← hdeg h]
      exact Finset.single_le_sum (f := fun j : Fin 10 => b (h,j))
        (fun _ _ => Nat.zero_le _) (Finset.mem_univ i)
    refine Finset.mem_image.mpr ⟨fun h => ⟨fun i => ⟨b (h,i), by have := hle h i; omega⟩,hdeg h⟩,
      Finset.mem_univ _,?_⟩
    ext u
    rfl
  have hc : Fintype.card (RothAuxCoefficients s) ≤ 1024^(∑ h, s h) := by
    change rothAuxDimension s ≤ _
    rw [rothAuxDimension_eq]
    calc
      _ ≤ ∏ h, 1024^s h := by
        apply Finset.prod_le_prod (fun _ _ => Nat.zero_le _)
        intro h hh
        by_cases hz : s h = 0
        · simp [hz]
        · calc
            (s h+9).choose 9 ≤ 2^(s h+9) := (Nat.choose_le_two_pow _ _ (by omega)).le
            _ ≤ 2^(10*s h) := Nat.pow_le_pow_right (by decide) (by omega)
            _ = _ := by rw [pow_mul]; norm_num
      _ = _ := Finset.prod_pow_eq_pow_sum _ _ _
  exact (Finset.card_le_card hs).trans (Finset.card_image_le.trans
    (hc.trans (Nat.pow_le_pow_right (by decide) (Finset.sum_le_sum (fun _ _ => Nat.sub_le _ _)))))

theorem grid_form_norm {K : Type*} [NormedField K] {m : ℕ}
    (A : Matrix (Fin 10) (Fin 10) K) (w : Fin m → Fin 9 → Fin 10 → ℤ)
    (a : Fin m → Fin 9 → ℕ) (h : Fin m) (i : Fin 10) (U B : ℝ)
    (hU : 0 ≤ U) (ha : ∀ j, ‖(a h j : K)‖ ≤ U)
    (hw : ∀ j, ‖(A *ᵥ (fun k => (w h j k : K))) i‖ ≤ B) :
    ‖(A *ᵥ (fun k => (gridPoint w a h k : K))) i‖ ≤ 9*U*B := by
  rw [matrix_grid]
  calc
    _ ≤ ∑ j : Fin 9, ‖(a h j : K)*(A *ᵥ (fun k => (w h j k : K))) i‖ := norm_sum_le _ _
    _ ≤ ∑ _j : Fin 9, U*B := by
      apply Finset.sum_le_sum
      intro j hj
      rw [norm_mul]
      exact mul_le_mul (ha j) (hw j) (norm_nonneg _) hU
    _ = _ := by simp; ring

theorem real_multiplier {m : ℕ} (a : Fin m → Fin 9 → ℕ) (eta : ℝ) (heta : 0 < eta)

theorem sourceBound_product {m : ℕ} (r : Fin m → ℕ) (E eta : ℝ) (Q : Fin m → ℝ)
    (h0 : Fin m) (c : Fin 3 → Fin 10 → ℝ) (hc : (∑ tau, ∑ i, c tau i) = 0) :
    sourceBound r E eta Q h0 c 0 * sourceBound r E eta Q h0 c 1 *
      sourceBound r E eta Q h0 c 2 =
      (1060*E)^(3*(∑ h,r h))*Real.exp (-880*((r h0 : ℝ)*Real.log (Q h0))*m*eta) := by
  have hs : (∑ i,c 0 i)+(∑ i,c 1 i)+(∑ i,c 2 i)=0 := by
    simpa only [Fin.sum_univ_succ,Fin.sum_univ_zero,Fin.val_zero,
      Fin.val_one,Fin.val_succ,Fin.succ_zero_eq_one,Fin.succ_one_eq_two,add_zero,add_assoc] using hc
  let F : ℝ := (1060*E)^(∑ h,r h)
  let T : Fin 3 → ℝ := fun tau => ((r h0 : ℝ)*Real.log (Q h0))*m*
    ((∑ i,c tau i)/10+(if tau=0 then -2000*eta else 560*eta))
  have ht : T 0+T 1+T 2 = -880*((r h0 : ℝ)*Real.log (Q h0))*m*eta := by
    dsimp [T]
    linear_combination ((r h0 : ℝ)*Real.log (Q h0)*(m : ℝ)/10)*hs
  have hf : F*F*F = (1060*E)^(3*(∑ h,r h)) := by
    dsimp [F]
    rw [← pow_add,← pow_add]
    congr 1
    omega
  change (F*Real.exp (T 0))*(F*Real.exp (T 1))*(F*Real.exp (T 2)) = _
  calc
    _ = (F*F*F)*Real.exp (T 0+T 1+T 2) := by rw [Real.exp_add,Real.exp_add]; ring
    _ = _ := by rw [hf,ht]

def restriction {K : Type*} [Field K] {m : ℕ} (a : Fin 10 → K)
    (k : Fin m → Fin 10) (P : MvPolynomial (Vars10 m) K) :=
  MvPolynomial.eval₂Hom MvPolynomial.C (fun u : Vars10 m =>
    if u.2 = k u.1 then -(∑ j ∈ Finset.univ.erase (k u.1),
      MvPolynomial.C (a j / a (k u.1)) * MvPolynomial.X (u.1,j))
    else MvPolynomial.X u) P

def index {K : Type*} [Field K] {m : ℕ} (A : Matrix (Fin 10) (Fin 10) K)
    (r : Fin m → ℕ) (eta : ℝ) (P : MvPolynomial (Vars10 m) ℤ) : Prop :=
  ∀ i J, rothSourceDerivativeWeight r J < (1/10 - eta) * m →
    ∀ k : Fin m → Fin 10, (∀ h, A i (k h) ≠ 0) →
      restriction (A i) k (derivative10 J (P.map (Int.castRingHom K))) = 0

def expansion {K : Type*} [Field K] {m : ℕ} (A : Matrix (Fin 10) (Fin 10) K)
    (J : Vars10 m →₀ ℕ) (P : MvPolynomial (Vars10 m) ℤ) :=
  MvPolynomial.eval₂Hom MvPolynomial.C (fun u : Vars10 m =>
    ∑ j : Fin 10, MvPolynomial.C (A⁻¹ u.2 j) * MvPolynomial.X (u.1,j))
    (derivative10 J (P.map (Int.castRingHom K)))

def IntendedSourceAuxiliaryPolynomial10 : Prop :=
  ∀ (A2 A3 : Matrix (Fin 10) (Fin 10) ℚ) (L : Matrix (Fin 10) (Fin 10) ℝ),
  A2.det = 1 → A3.det = 1 → L.det = 1 → (∀ i j, IsAlgebraic ℚ (L i j)) →
  ∃ K : IntermediateField ℚ ℝ, FiniteDimensional ℚ K ∧
  (∀ i j, L i j ∈ K) ∧
  ∃ D E : ℝ, 1 ≤ D ∧ 1 ≤ E ∧
  ∀ eta : ℝ, 0 < eta → eta < 1/10 → ∀ m : ℕ,
  4 * eta⁻¹ ^ 2 * Real.log (60 * Module.finrank ℚ K) ≤ m →
  ∀ r : Fin m → ℕ, (∀ h, 0 < r h) →
  ∃ P : MvPolynomial (Vars10 m) ℤ,
  P ≠ 0 ∧ homogeneous10 r P ∧ (∀ h, blockDegree10 P h ≤ r h) ∧
  (∀ u, P.degreeOf u ≤ r u.1) ∧ (height10 P : ℝ) ≤ D ^ (∑ h, r h) ∧
  index L r eta P ∧ index (A2.map (algebraMap ℚ ℚ_[2])) r eta P ∧
  index (A3.map (algebraMap ℚ ℚ_[3])) r eta P ∧
  (∀ J b, ‖(expansion L J P).coeff b‖ ≤ E ^ (∑ h, r h)) ∧
  (∀ J b, ‖(expansion (A2.map (algebraMap ℚ ℚ_[2])) J P).coeff b‖ ≤ E ^ (∑ h, r h)) ∧
  (∀ J b, ‖(expansion (A3.map (algebraMap ℚ ℚ_[3])) J P).coeff b‖ ≤ E ^ (∑ h, r h))

theorem expansion_agrees {F : Type*} [Field F] {m : ℕ}
    (A : Matrix (Fin 10) (Fin 10) F) (J : Vars10 m →₀ ℕ)
    (P : MvPolynomial (Vars10 m) ℤ) :
    M198Contract.expansion A J P =
      CompoundFeasibility.Auxiliary10.SourceExpansion10.expansion A J P := rfl

theorem original_contract : M198Contract.IntendedSourceAuxiliaryPolynomial10 := by
  intro A2 A3 L h2 h3 hL ha
  obtain ⟨K,hK,hmem,D,E,hD,hE,hconstruct⟩ := M198RLiteral.CompoundFeasibility.Auxiliary10.source_auxiliary_polynomial10 A2 A3 L h2 h3 hL ha
  refine ⟨K,hK,?_,D,E,hD,hE,?_⟩
  · intro i j
    exact hmem 0 i j
  · intro eta heta hsmall m hm r hr
    have hm' : 4*(eta^2)⁻¹*Real.log (60*(Module.finrank ℚ K : ℝ)) ≤ (m : ℝ) := by
      simpa only [inv_pow] using hm
    obtain ⟨P,hn,hhom,hblock,hdegree,hheight,hfield,hreal,hp2,hp3,hclaims,hcfs,hgrid⟩ :=
      hconstruct eta heta m hm' r hr
    refine ⟨P,hn,hhom,fun h => (hblock h).le,hdegree,hheight,?_,?_,?_,?_,?_,?_⟩
    · exact hreal 0
    · exact hp2
    · exact hp3
    · intro J b
      rw [expansion_agrees]
      exact (hcfs J b).1
    · intro J b
      rw [show A2.map (algebraMap ℚ ℚ_[2]) = CompoundFeasibility.SourceForms10.finiteMatrix10 2 A2 from rfl,
        expansion_agrees]
      exact (hcfs J b).2.1
    · intro J b
      rw [show A3.map (algebraMap ℚ ℚ_[3]) = CompoundFeasibility.SourceForms10.finiteMatrix10 3 A3 from rfl,
        expansion_agrees]
      exact (hcfs J b).2.2
