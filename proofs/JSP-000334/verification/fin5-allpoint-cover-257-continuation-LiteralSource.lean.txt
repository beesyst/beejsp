def PrimitiveFive (z : Fin 5 → ℤ) : Prop := ∃ c : Fin 5 → ℤ,∑ i,c i*z i=1

theorem primitive_five_integer_reduction (w : Fin 5 → ℤ) (hw0 : w 0≠0) :
    ∃ g : ℕ,0<g ∧ (g : ℤ)∣w 0 ∧ ∃ z : Fin 5 → ℤ,
      PrimitiveFive z ∧ ∀ i,w i=(g : ℤ)*z i := by
  let k := Int.gcd (w 3) (w 4)
  let l := Int.gcd (w 2) (k : ℤ)
  let m := Int.gcd (w 1) (l : ℤ)
  let g := Int.gcd (w 0) (m : ℤ)
  have hg : 0<g := Int.gcd_pos_of_ne_zero_left _ hw0
  have hg0 : (g : ℤ)∣w 0 := Int.gcd_dvd_left
  have hgm : (g : ℤ)∣(m : ℤ) := Int.gcd_dvd_right
  have hm1 : (m : ℤ)∣w 1 := Int.gcd_dvd_left
  have hml : (m : ℤ)∣(l : ℤ) := Int.gcd_dvd_right
  have hl2 : (l : ℤ)∣w 2 := Int.gcd_dvd_left
  have hlk : (l : ℤ)∣(k : ℤ) := Int.gcd_dvd_right
  have hk3 : (k : ℤ)∣w 3 := Int.gcd_dvd_left
  have hk4 : (k : ℤ)∣w 4 := Int.gcd_dvd_right
  obtain ⟨z0,h0⟩ := hg0
  obtain ⟨z1,h1⟩ := hgm.trans hm1
  obtain ⟨z2,h2⟩ := (hgm.trans hml).trans hl2
  obtain ⟨z3,h3⟩ := ((hgm.trans hml).trans hlk).trans hk3
  obtain ⟨z4,h4⟩ := ((hgm.trans hml).trans hlk).trans hk4
  let z : Fin 5 → ℤ := ![z0,z1,z2,z3,z4]
  have hscale : ∀ i,w i=(g : ℤ)*z i := by intro i; fin_cases i <;> assumption
  let a := Int.gcdA (w 0) (m : ℤ)
  let b := Int.gcdB (w 0) (m : ℤ)
  let c := Int.gcdA (w 1) (l : ℤ)
  let d := Int.gcdB (w 1) (l : ℤ)
  let e := Int.gcdA (w 2) (k : ℤ)
  let f := Int.gcdB (w 2) (k : ℤ)
  let h := Int.gcdA (w 3) (w 4)
  let j := Int.gcdB (w 3) (w 4)
  have hb0 : (g : ℤ)=w 0*a+(m : ℤ)*b := Int.gcd_eq_gcd_ab _ _
  have hb1 : (m : ℤ)=w 1*c+(l : ℤ)*d := Int.gcd_eq_gcd_ab _ _
  have hb2 : (l : ℤ)=w 2*e+(k : ℤ)*f := Int.gcd_eq_gcd_ab _ _
  have hb3 : (k : ℤ)=w 3*h+w 4*j := Int.gcd_eq_gcd_ab _ _
  have heq : (g : ℤ)*(a*z0+(b*c)*z1+(b*d*e)*z2+(b*d*f*h)*z3+(b*d*f*j)*z4)=(g : ℤ)*1 := by
    calc
      _=w 0*a+(m : ℤ)*b := by rw [hb1,hb2,hb3,h0,h1,h2,h3,h4]; ring
      _=_ := by rw [← hb0]; ring
  have hgn : (g : ℤ)≠0 := by exact_mod_cast ne_of_gt hg
  refine ⟨g,hg,Int.gcd_dvd_left,z,?_,hscale⟩
  refine ⟨![a,b*c,b*d*e,b*d*f*h,b*d*f*j],?_⟩
  simpa [z,Fin.sum_univ_succ,add_assoc] using mul_left_cancel₀ hgn heq

def retainedIntegerMatrix5 (k : Fin 5) : Matrix (Fin 5) (Fin 5) ℤ :=
  fun i j => if i=k then 1 else if i=j then 1 else 0

theorem retained_integer_matrix_det5 (k : Fin 5) : (retainedIntegerMatrix5 k).det=1 := by
  have he : retainedIntegerMatrix5 k =
      Matrix.updateRow (1 : Matrix (Fin 5) (Fin 5) ℤ) k
        (∑ j : Fin 5,(1 : ℤ) • (1 : Matrix (Fin 5) (Fin 5) ℤ) j) := by
    ext i j
    by_cases hi : i=k <;> simp [retainedIntegerMatrix5,Matrix.updateRow,hi,Matrix.one_apply]
  rw [he,Matrix.det_updateRow_sum]
  simp

theorem retained_integer_matrix_apply5 (k i : Fin 5) (z : Fin 5 → ℤ) :
    (retainedIntegerMatrix5 k *ᵥ z) i=if i=k then ∑ j,z j else z i := by
  by_cases hi : i=k
  · simp [Matrix.mulVec,dotProduct,retainedIntegerMatrix5,hi]
  · simp [Matrix.mulVec,dotProduct,retainedIntegerMatrix5,hi]

def archFiveHeight (z : Fin 5 → ℤ) : ℚ :=
  max |(z 0 : ℚ)| (max |(z 1 : ℚ)| (max |(z 2 : ℚ)| (max |(z 3 : ℚ)| |(z 4 : ℚ)|)))

def fiveRetainedProduct (choice : Fin 3 → Fin 5) (y : Fin 5 → ℚ) : ℚ :=
  ∏ tau : Fin 3,∏ i : Fin 5,rationalPlaceNorm tau (fiveRetainedForm (choice tau) i y)

theorem five_local_coordinate_le (tau : Fin 3) (y : Fin 5 → ℚ) (i : Fin 5) :
    rationalPlaceNorm tau (y i)≤localFiveHeight tau y := by
  fin_cases i
  · exact le_max_left _ _
  · exact (le_max_left _ _).trans (le_max_right _ _)
  · exact (le_max_left _ _).trans ((le_max_right _ _).trans (le_max_right _ _))
  · exact (le_max_left _ _).trans ((le_max_right _ _).trans ((le_max_right _ _).trans (le_max_right _ _)))
  · exact (le_max_right _ _).trans ((le_max_right _ _).trans ((le_max_right _ _).trans (le_max_right _ _)))

theorem primitive_five_local_height (z : Fin 5 → ℤ) (hz : PrimitiveFive z)
    (p : ℕ) [Fact p.Prime] :
    max (padicNorm p (z 0 : ℚ)) (max (padicNorm p (z 1 : ℚ))
      (max (padicNorm p (z 2 : ℚ)) (max (padicNorm p (z 3 : ℚ)) (padicNorm p (z 4 : ℚ)))))=1 := by
  obtain ⟨c,hc⟩ := hz
  obtain ⟨i,hi⟩ : ∃ i : Fin 5,¬(p : ℤ)∣z i := by
    by_contra hn
    push_neg at hn
    have hd : (p : ℤ)∣∑ i,c i*z i := Finset.dvd_sum (fun i _ => dvd_mul_of_dvd_right (hn i) _)
    rw [hc] at hd
    exact (Fact.out : p.Prime).not_dvd_one (Int.natCast_dvd_natCast.mp hd)
  have he := (padicNorm.int_eq_one_iff (p := p) (z i)).mpr hi
  apply le_antisymm
  · exact max_le (padicNorm.of_int _) (max_le (padicNorm.of_int _)
      (max_le (padicNorm.of_int _) (max_le (padicNorm.of_int _) (padicNorm.of_int _))))
  · rw [← he]
    fin_cases i
    · exact le_max_left _ _
    · exact (le_max_left _ _).trans (le_max_right _ _)
    · exact (le_max_left _ _).trans ((le_max_right _ _).trans (le_max_right _ _))
    · exact (le_max_left _ _).trans ((le_max_right _ _).trans ((le_max_right _ _).trans (le_max_right _ _)))
    · exact (le_max_right _ _).trans ((le_max_right _ _).trans ((le_max_right _ _).trans (le_max_right _ _)))

theorem primitive_five_height (z : Fin 5 → ℤ) (hz : PrimitiveFive z) :
    fiveHeight (fun i => (z i : ℚ))=archFiveHeight z := by
  simp only [fiveHeight,Fin.prod_univ_three]
  change archFiveHeight z *
    (max (padicNorm 2 (z 0 : ℚ)) (max (padicNorm 2 (z 1 : ℚ))
      (max (padicNorm 2 (z 2 : ℚ)) (max (padicNorm 2 (z 3 : ℚ)) (padicNorm 2 (z 4 : ℚ)))))) *
    (max (padicNorm 3 (z 0 : ℚ)) (max (padicNorm 3 (z 1 : ℚ))
      (max (padicNorm 3 (z 2 : ℚ)) (max (padicNorm 3 (z 3 : ℚ)) (padicNorm 3 (z 4 : ℚ))))))=_
  rw [primitive_five_local_height z hz 2,primitive_five_local_height z hz 3]
  ring

theorem exists_five_maximal_coordinate (tau : Fin 3) (y : Fin 5 → ℚ) :
    ∃ k : Fin 5,rationalPlaceNorm tau (y k)=localFiveHeight tau y := by
  obtain ⟨k,_,hk⟩ := Finset.exists_max_image Finset.univ (fun i => rationalPlaceNorm tau (y i)) Finset.univ_nonempty
  refine ⟨k,le_antisymm (five_local_coordinate_le tau y k) ?_⟩
  exact max_le (hk 0 (Finset.mem_univ _)) (max_le (hk 1 (Finset.mem_univ _))
    (max_le (hk 2 (Finset.mem_univ _)) (max_le (hk 3 (Finset.mem_univ _)) (hk 4 (Finset.mem_univ _)))))

theorem five_retained_product_height (choice : Fin 3 → Fin 5) (y : Fin 5 → ℚ)
    (hchoice : ∀ tau,rationalPlaceNorm tau (y (choice tau))=localFiveHeight tau y) :
    fiveRetainedProduct choice y*fiveHeight y=
      (∏ i : Fin 5,threePlaceProduct (y i))*threePlaceProduct (∑ i,y i) := by
  have hlocal (tau : Fin 3) : (∏ i : Fin 5,rationalPlaceNorm tau (fiveRetainedForm (choice tau) i y))*
      localFiveHeight tau y=(∏ i : Fin 5,rationalPlaceNorm tau (y i))*rationalPlaceNorm tau (∑ i,y i) := by
    rw [← hchoice tau]
    generalize choice tau=k
    fin_cases k <;> simp [fiveRetainedForm,Fin.prod_univ_succ] <;> ring
  unfold fiveRetainedProduct fiveHeight
  rw [← Finset.prod_mul_distrib]
  simp_rw [hlocal]
  rw [Finset.prod_mul_distrib,Finset.prod_comm]
  simp_rw [rationalPlaceNorm_product]

theorem six_term_integer_product5 {x : Fin 6 → ℚ}

theorem allpoint_primitive_product5 {x : Fin 6 → ℚ}

theorem allpoint_form_arch_bound5 (z : Fin 5 → ℤ) (k i : Fin 5) :
    |fiveRetainedForm k i (fun j => (z j : ℚ))|≤5*archFiveHeight z := by
  have hc (j : Fin 5) : |(z j : ℚ)|≤archFiveHeight z := five_local_coordinate_le 0 (fun i => (z i : ℚ)) j
  have hH : 0≤archFiveHeight z := (abs_nonneg (z 0 : ℚ)).trans (hc 0)
  unfold fiveRetainedForm
  split_ifs
  · calc
      |∑ j : Fin 5,(z j : ℚ)|≤∑ j : Fin 5,|(z j : ℚ)| := Finset.abs_sum_le_sum_abs _ _
      _≤∑ _j : Fin 5,archFiveHeight z := Finset.sum_le_sum (fun j _ => hc j)
      _=5*archFiveHeight z := by simp [mul_comm]
  · exact (hc i).trans (by linarith)


def sourceWeightConditions5 (c : Fin 3 → Fin 5 → ℝ) : Prop :=
  (∀ i, |c 0 i| ≤ 1) ∧ (∀ tau, tau ≠ 0 → ∀ i, -1 ≤ c tau i ∧ c tau i ≤ 0) ∧
    (∑ tau : Fin 3, ∑ i : Fin 5, c tau i) = 0

noncomputable def finiteWeightIndex5 (n : ℕ) (c : Fin 3 → Fin 5 → ℝ) (tau : Fin 3) (i : Fin 5) : ℤ :=
  if tau = 0 then ⌊(n : ℝ)*((1-4/(n : ℝ))*c tau i)⌋
  else ⌈(n : ℝ)*((1-4/(n : ℝ))*c tau i)⌉

noncomputable def finiteWeightRaw5 (n : ℕ) (z : Fin 3 → Fin 5 → ℤ) (tau : Fin 3) (i : Fin 5) : ℝ :=
  (z tau i : ℝ)/(n : ℝ)

noncomputable def finiteWeightCandidate5 (n : ℕ) (z : Fin 3 → Fin 5 → ℤ) (tau : Fin 3) (i : Fin 5) : ℝ :=
  if tau = 0 then finiteWeightRaw5 n z tau i - (∑ t : Fin 3, ∑ j : Fin 5, finiteWeightRaw5 n z t j)/5
  else finiteWeightRaw5 n z tau i

noncomputable def finiteWeightApprox5 (n : ℕ) (c : Fin 3 → Fin 5 → ℝ) : Fin 3 → Fin 5 → ℝ :=
  finiteWeightCandidate5 n (finiteWeightIndex5 n c)

theorem finiteWeight_floor_bounds5 (n : ℕ) (hn : 0 < n) (x : ℝ) :
    x-1/(n : ℝ) ≤ (⌊(n : ℝ)*x⌋ : ℝ)/(n : ℝ) ∧
      (⌊(n : ℝ)*x⌋ : ℝ)/(n : ℝ) ≤ x := by
  have hp : (0 : ℝ) < n := by exact_mod_cast hn
  have he : (1/(n : ℝ))*(n : ℝ) = 1 := one_div_mul_cancel hp.ne'
  constructor
  · apply (le_div_iff₀ hp).mpr
    nlinarith [Int.lt_floor_add_one ((n : ℝ)*x)]
  · apply (div_le_iff₀ hp).mpr
    nlinarith [Int.floor_le ((n : ℝ)*x)]

theorem finiteWeight_ceil_bounds5 (n : ℕ) (hn : 0 < n) (x : ℝ) :
    x ≤ (⌈(n : ℝ)*x⌉ : ℝ)/(n : ℝ) ∧
      (⌈(n : ℝ)*x⌉ : ℝ)/(n : ℝ) ≤ x+1/(n : ℝ) := by
  have hp : (0 : ℝ) < n := by exact_mod_cast hn
  have he : (1/(n : ℝ))*(n : ℝ) = 1 := one_div_mul_cancel hp.ne'
  constructor
  · apply (le_div_iff₀ hp).mpr
    nlinarith [Int.le_ceil ((n : ℝ)*x)]
  · apply (div_le_iff₀ hp).mpr
    nlinarith [Int.ceil_lt_add_one ((n : ℝ)*x)]

theorem finiteWeightCandidate5_balance (n : ℕ) (z : Fin 3 → Fin 5 → ℤ) :
    (∑ tau : Fin 3, ∑ i : Fin 5, finiteWeightCandidate5 n z tau i) = 0 := by
  simp [finiteWeightCandidate5, Fin.sum_univ_succ]
  ring

theorem finiteWeight_raw_errors5 (n : ℕ) (hn : 0 < n) (c : Fin 3 → Fin 5 → ℝ) :
    (∀ i, (1-4/(n : ℝ))*c 0 i-1/(n : ℝ) ≤ finiteWeightRaw5 n (finiteWeightIndex5 n c) 0 i ∧
      finiteWeightRaw5 n (finiteWeightIndex5 n c) 0 i ≤ (1-4/(n : ℝ))*c 0 i) ∧
    (∀ tau, tau ≠ 0 → ∀ i,
      (1-4/(n : ℝ))*c tau i ≤ finiteWeightRaw5 n (finiteWeightIndex5 n c) tau i ∧
      finiteWeightRaw5 n (finiteWeightIndex5 n c) tau i ≤ (1-4/(n : ℝ))*c tau i+1/(n : ℝ)) := by
  constructor
  · intro i
    simpa only [finiteWeightRaw5, finiteWeightIndex5, if_pos rfl] using
      finiteWeight_floor_bounds5 n hn ((1-4/(n : ℝ))*c 0 i)
  · intro tau ht i
    simpa only [finiteWeightRaw5, finiteWeightIndex5, if_neg ht] using
      finiteWeight_ceil_bounds5 n hn ((1-4/(n : ℝ))*c tau i)

theorem finiteWeight_raw_sum5 (n : ℕ) (hn : 0 < n) (c : Fin 3 → Fin 5 → ℝ)
    (hb : (∑ tau : Fin 3, ∑ i : Fin 5, c tau i) = 0) :
    -5/(n : ℝ) ≤ (∑ tau : Fin 3, ∑ i : Fin 5, finiteWeightRaw5 n (finiteWeightIndex5 n c) tau i) ∧
    (∑ tau : Fin 3, ∑ i : Fin 5, finiteWeightRaw5 n (finiteWeightIndex5 n c) tau i) ≤ 10/(n : ℝ) := by
  obtain ⟨h0,hp⟩ := finiteWeight_raw_errors5 n hn c
  let R := finiteWeightRaw5 n (finiteWeightIndex5 n c)
  let a : ℝ := 1-4/(n : ℝ)
  have hlo : (∑ i : Fin 5,a*c 0 i)-5/(n : ℝ)≤∑ i : Fin 5,R 0 i := by
    have hh := Finset.sum_le_sum (fun i (_ : i∈(Finset.univ : Finset (Fin 5))) => (h0 i).1)
    simpa only [Finset.sum_sub_distrib,Finset.sum_const,Finset.card_univ,Fintype.card_fin,nsmul_eq_mul,mul_div_assoc,mul_one, Nat.cast_ofNat, mul_one_div] using hh
  have hhi : (∑ i : Fin 5,R 0 i)≤∑ i : Fin 5,a*c 0 i :=
    Finset.sum_le_sum (fun i _ => (h0 i).2)
  have hfl (tau : Fin 3) (ht : tau≠0) : (∑ i : Fin 5,a*c tau i)≤∑ i : Fin 5,R tau i :=
    Finset.sum_le_sum (fun i _ => (hp tau ht i).1)
  have hfu (tau : Fin 3) (ht : tau≠0) : (∑ i : Fin 5,R tau i)≤(∑ i : Fin 5,a*c tau i)+5/(n : ℝ) := by
    have hh := Finset.sum_le_sum (fun i (_ : i∈(Finset.univ : Finset (Fin 5))) => (hp tau ht i).2)
    simpa only [Finset.sum_add_distrib,Finset.sum_const,Finset.card_univ,Fintype.card_fin,nsmul_eq_mul,mul_div_assoc,mul_one, Nat.cast_ofNat, mul_one_div] using hh
  have hz : (∑ i : Fin 5,a*c 0 i)+(∑ i : Fin 5,a*c 1 i)+(∑ i : Fin 5,a*c 2 i)=0 := by
    have hh : a*(∑ tau : Fin 3,∑ i : Fin 5,c tau i)=0 := by rw [hb,mul_zero]
    simpa only [Fin.sum_univ_three,Finset.mul_sum,mul_add] using hh
  change -5/(n : ℝ)≤(∑ tau : Fin 3,∑ i : Fin 5,R tau i) ∧ (∑ tau : Fin 3,∑ i : Fin 5,R tau i)≤10/(n : ℝ)
  simp only [Fin.sum_univ_three]
  constructor
  · calc
      -5/(n : ℝ)=(∑ i : Fin 5,a*c 0 i)-5/(n : ℝ)+(∑ i : Fin 5,a*c 1 i)+(∑ i : Fin 5,a*c 2 i)  := by
        calc
          _=(∑ i : Fin 5,a*c 0 i)+(∑ i : Fin 5,a*c 1 i)+(∑ i : Fin 5,a*c 2 i)-5/(n : ℝ) := by rw [hz]; ring
          _=_ := by ring
      _≤_ := add_le_add (add_le_add hlo (hfl 1 (by decide))) (hfl 2 (by decide))
  · calc
      _≤(∑ i : Fin 5,a*c 0 i)+((∑ i : Fin 5,a*c 1 i)+5/(n : ℝ))+((∑ i : Fin 5,a*c 2 i)+5/(n : ℝ)) :=
        add_le_add (add_le_add hhi (hfu 1 (by decide))) (hfu 2 (by decide))
      _=10/(n : ℝ) := by
        calc
          _=((∑ i : Fin 5,a*c 0 i)+(∑ i : Fin 5,a*c 1 i)+(∑ i : Fin 5,a*c 2 i))+10/(n : ℝ) := by ring
          _=_ := by rw [hz]; ring

theorem finiteWeightApprox5_bounds (n : ℕ) (hn : 16 ≤ n)
    (c : Fin 3 → Fin 5 → ℝ) (hc : sourceWeightConditions5 c) :
    sourceWeightConditions5 (finiteWeightApprox5 n c) ∧
      (∀ i, c 0 i-7/(n : ℝ) ≤ finiteWeightApprox5 n c 0 i) ∧
      (∀ tau, tau ≠ 0 → ∀ i, c tau i ≤ finiteWeightApprox5 n c tau i) := by
  have hnp : 0 < n := by omega
  have hp : (0 : ℝ) < n := by exact_mod_cast hnp
  have he : 0 < 1/(n : ℝ) := by positivity
  have he' : 1/(n : ℝ) ≤ 1/16 := (div_le_div_iff₀ hp (by norm_num : (0 : ℝ) < 16)).mpr (by simpa only [one_mul] using (show (16 : ℝ) ≤ n by exact_mod_cast hn))
  have h5 : 4/(n : ℝ) ≤ 1/4 := by
    calc
      _ = 4*(1/(n : ℝ)) := by ring
      _ ≤ 4*(1/16) := mul_le_mul_of_nonneg_left he' (by norm_num)
      _ = _ := by norm_num
  obtain ⟨h0,hf,hb⟩ := hc
  obtain ⟨ha0,haf⟩ := finiteWeight_raw_errors5 n hnp c
  obtain ⟨hBl,hBu⟩ := finiteWeight_raw_sum5 n hnp c hb
  have hreal (i : Fin 5) : |finiteWeightApprox5 n c 0 i| ≤ 1 ∧
      c 0 i-7/(n : ℝ) ≤ finiteWeightApprox5 n c 0 i := by
    have hc0 := abs_le.mp (h0 i)
    have hscaled := mul_le_mul_of_nonneg_left hc0.1 (by linarith : 0 ≤ 1-4/(n : ℝ))
    have hscaled' := mul_le_mul_of_nonneg_left hc0.2 (by linarith : 0 ≤ 1-4/(n : ℝ))
    change |finiteWeightRaw5 n (finiteWeightIndex5 n c) 0 i -
      (∑ t : Fin 3, ∑ j : Fin 5, finiteWeightRaw5 n (finiteWeightIndex5 n c) t j)/5| ≤ 1 ∧ _
    change _ ∧ c 0 i-7/(n : ℝ) ≤ finiteWeightRaw5 n (finiteWeightIndex5 n c) 0 i -
      (∑ t : Fin 3, ∑ j : Fin 5, finiteWeightRaw5 n (finiteWeightIndex5 n c) t j)/5
    have hsmall := mul_le_mul_of_nonneg_left hc0.2 (by positivity : 0 ≤ 4/(n : ℝ))
    simp only [div_eq_mul_inv, one_mul] at *
    refine ⟨abs_le.mpr ⟨?_,?_⟩,?_⟩ <;> nlinarith [(ha0 i).1,(ha0 i).2]
  have hfinite (tau : Fin 3) (ht : tau ≠ 0) (i : Fin 5) :
      -1 ≤ finiteWeightApprox5 n c tau i ∧ finiteWeightApprox5 n c tau i ≤ 0 ∧
      c tau i ≤ finiteWeightApprox5 n c tau i := by
    have hci := hf tau ht i
    have hscaled := mul_le_mul_of_nonneg_left hci.1 (by linarith : 0 ≤ 1-4/(n : ℝ))
    have hscaled' : (1-4/(n : ℝ))*c tau i ≤ 0 := mul_nonpos_of_nonneg_of_nonpos (by linarith) hci.2
    have hceil : ⌈(n : ℝ)*((1-4/(n : ℝ))*c tau i)⌉ ≤ (0 : ℤ) :=
      Int.ceil_le.mpr (by simpa using mul_nonpos_of_nonneg_of_nonpos hp.le hscaled')
    have hceilR : (⌈(n : ℝ)*((1-4/(n : ℝ))*c tau i)⌉ : ℝ) ≤ 0 := by exact_mod_cast hceil
    have hnonpos : finiteWeightApprox5 n c tau i ≤ 0 := by
      simp only [finiteWeightApprox5, finiteWeightCandidate5, finiteWeightRaw5, finiteWeightIndex5, if_neg ht]
      exact div_nonpos_of_nonpos_of_nonneg hceilR hp.le
    have heq : finiteWeightApprox5 n c tau i = finiteWeightRaw5 n (finiteWeightIndex5 n c) tau i := by
      simp only [finiteWeightApprox5, finiteWeightCandidate5, if_neg ht]
    rw [heq] at hnonpos ⊢
    have hsmall := mul_nonpos_of_nonneg_of_nonpos (by positivity : 0 ≤ 4/(n : ℝ)) hci.2
    simp only [div_eq_mul_inv, one_mul] at *
    refine ⟨?_,hnonpos,?_⟩ <;> nlinarith [(haf tau ht i).1]
  refine ⟨⟨fun i => (hreal i).1,fun tau ht i => ⟨(hfinite tau ht i).1,(hfinite tau ht i).2.1⟩,
    finiteWeightCandidate5_balance n _⟩,fun i => (hreal i).2,fun tau ht i => (hfinite tau ht i).2.2⟩

noncomputable def finiteWeightCandidates5 (n : ℕ) : Set (Fin 3 → Fin 5 → ℝ) :=
  (finiteWeightCandidate5 n '' {z : Fin 3 → Fin 5 → ℤ | ∀ tau i, -(n : ℤ) ≤ z tau i ∧ z tau i ≤ n}) ∩
    {d | sourceWeightConditions5 d}

theorem finiteWeightCandidates5_finite (n : ℕ) : (finiteWeightCandidates5 n).Finite := by
  have hbox : {z : Fin 3 → Fin 5 → ℤ | ∀ tau i, -(n : ℤ) ≤ z tau i ∧ z tau i ≤ n}.Finite :=
    Set.Finite.pi' (fun _ => Set.Finite.pi' (fun _ => Set.finite_Icc _ _))
  exact (hbox.image (finiteWeightCandidate5 n)).subset Set.inter_subset_left

theorem finiteWeightIndex5_bounded (n : ℕ) (hn : 16 ≤ n) (c : Fin 3 → Fin 5 → ℝ)
    (hc : sourceWeightConditions5 c) :
    ∀ tau i, -(n : ℤ) ≤ finiteWeightIndex5 n c tau i ∧ finiteWeightIndex5 n c tau i ≤ n := by
  have hp : (0 : ℝ) < n := by exact_mod_cast (show 0 < n by omega)
  have hnR : (16 : ℝ) ≤ n := by exact_mod_cast hn
  have hfactor : 0 ≤ 1-4/(n : ℝ) := by
    apply sub_nonneg.mpr
    exact (div_le_one hp).mpr (by linarith)
  have hfactor' : 1-4/(n : ℝ) ≤ 1 := by
    have : 0 < 4/(n : ℝ) := by positivity
    linarith
  intro tau i
  have hci : -1 ≤ c tau i ∧ c tau i ≤ 1 := by
    by_cases ht : tau = 0
    · subst tau; exact abs_le.mp (hc.1 i)
    · exact ⟨(hc.2.1 tau ht i).1, (hc.2.1 tau ht i).2.trans (by norm_num)⟩
  have hscaled : -(n : ℝ) ≤ (n : ℝ)*((1-4/(n : ℝ))*c tau i) ∧
      (n : ℝ)*((1-4/(n : ℝ))*c tau i) ≤ n := by
    have h1 := mul_le_mul_of_nonneg_left hci.1 hfactor
    have h2 := mul_le_mul_of_nonneg_left hci.2 hfactor
    constructor
    · have h : -1 ≤ (1-4/(n : ℝ))*c tau i := by nlinarith
      nlinarith [mul_le_mul_of_nonneg_left h hp.le]
    · have h : (1-4/(n : ℝ))*c tau i ≤ 1 := by nlinarith
      nlinarith [mul_le_mul_of_nonneg_left h hp.le]
  unfold finiteWeightIndex5
  split_ifs with ht
  · constructor
    · exact Int.le_floor.mpr (by simpa using hscaled.1)

noncomputable def sourceLogWeights5 (Q : ℝ) (W : Fin 3 → Fin 5 → ℝ) (tau : Fin 3) (i : Fin 5) : ℝ :=
  Real.log (W tau i)/Real.log Q

theorem sourceLogWeights5_rpow (Q : ℝ) (hQ : 1 < Q) (W : Fin 3 → Fin 5 → ℝ)
    (hW : ∀ tau i, 0 < W tau i) (tau : Fin 3) (i : Fin 5) :
    Q^(sourceLogWeights5 Q W tau i) = W tau i := by
  rw [Real.rpow_def_of_pos (lt_trans zero_lt_one hQ)]
  unfold sourceLogWeights5
  rw [mul_div_cancel₀ _ (Real.log_pos hQ).ne', Real.exp_log (hW tau i)]

theorem sourceLogWeights5_conditions (Q : ℝ) (hQ : 1 < Q) (W : Fin 3 → Fin 5 → ℝ)
    (hW : ∀ tau i, 0 < W tau i)
    (hcut : ∀ tau i, (W tau i)⁻¹ < Q)
    (hcut0 : ∀ i, W 0 i < Q)
    (hfinite : ∀ tau, tau ≠ 0 → ∀ i, W tau i ≤ 1)
    (hprod : (∏ tau : Fin 3, ∏ i : Fin 5, W tau i) = 1) :
    sourceWeightConditions5 (sourceLogWeights5 Q W) := by
  have hl := Real.log_pos hQ
  have hlo (tau : Fin 3) (i : Fin 5) : -1 ≤ sourceLogWeights5 Q W tau i := by
    have hh := Real.log_lt_log (inv_pos.mpr (hW tau i)) (hcut tau i)
    rw [Real.log_inv] at hh
    apply (le_div_iff₀ hl).mpr
    linarith
  refine ⟨?_,?_,?_⟩
  · intro i
    refine abs_le.mpr ⟨hlo 0 i,?_⟩
    apply (div_le_iff₀ hl).mpr
    have hh := Real.log_lt_log (hW 0 i) (hcut0 i)
    linarith
  · intro tau ht i
    refine ⟨hlo tau i,?_⟩
    apply div_nonpos_of_nonpos_of_nonneg _ hl.le
    exact Real.log_nonpos (hW tau i).le (hfinite tau ht i)
  · have hlogs : (∑ tau : Fin 3, ∑ i : Fin 5, Real.log (W tau i)) = 0 := by
      have heq : Real.log (∏ tau : Fin 3, ∏ i : Fin 5, W tau i) =
          (∑ tau : Fin 3, ∑ i : Fin 5, Real.log (W tau i)) := by
        rw [Real.log_prod Finset.univ (fun tau => ∏ i : Fin 5, W tau i) (fun tau _ => (Finset.prod_pos (fun i _ => hW tau i)).ne')]
        apply Finset.sum_congr rfl
        intro tau _
        exact Real.log_prod Finset.univ (W tau) (fun i _ => (hW tau i).ne')
      rw [hprod,Real.log_one] at heq
      exact heq.symm
    unfold sourceLogWeights5
    simp only [← Finset.sum_div]
    rw [hlogs,zero_div]

def allpoint_balanced_values5 (Q : ℝ) (a : Fin 3 → Fin 5 → ℝ) (tau : Fin 3) (i : Fin 5) : ℝ :=
  if tau = 0 then a tau i*Q^((1 : ℝ)/40) else a tau i

theorem allpoint_balanced_weights5 (H : ℝ) (hH : 5 ≤ H)
    (a : Fin 3 → Fin 5 → ℝ) (ha : ∀ tau i, 0 < a tau i)
    (hreal : ∀ i, a 0 i ≤ 5*H)
    (hfinite : ∀ tau, tau ≠ 0 → ∀ i, a tau i ≤ 1)
    (hinv : ∀ tau i, (a tau i)⁻¹ ≤ 5*H)
    (hprod : (∏ tau, ∏ i, a tau i)*H = 1) :
    let Q := H^8
    let W := allpoint_balanced_values5 Q a
    1 < Q ∧ sourceWeightConditions5 (sourceLogWeights5 Q W) ∧
      (∀ tau i, Q^(sourceLogWeights5 Q W tau i) = W tau i) ∧
      (∀ i, a 0 i = Q^(sourceLogWeights5 Q W 0 i-(1 : ℝ)/40)) ∧
      (∀ tau, tau ≠ 0 → ∀ i, a tau i = Q^(sourceLogWeights5 Q W tau i)) := by
  intro Q W
  have hH0 : 0 < H := by linarith
  have hQ : 1 < Q := by dsimp [Q]; nlinarith [sq_nonneg (H^2-1)]
  have hQ0 := lt_trans zero_lt_one hQ
  let r := Q^((1 : ℝ)/40)
  have hr : 0 < r := Real.rpow_pos_of_pos hQ0 _
  have hr1 : 1 ≤ r := Real.one_le_rpow hQ.le (by norm_num)
  have hrH : r ≤ H := by
    have he : (H^8)^((1 : ℝ)/40) = H^((1 : ℝ)/5) := by
      rw [← Real.rpow_natCast,← Real.rpow_mul hH0.le]
      congr 1
      norm_num
    change (H^8)^((1 : ℝ)/40) ≤ H
    rw [he]
    simpa using Real.rpow_le_rpow_of_exponent_le (show 1 ≤ H by linarith) (by norm_num : (1 : ℝ)/5 ≤ 1)
  have hrpow : r^5 = H := by
    change ((H^8)^((1 : ℝ)/40))^5 = H
    rw [← Real.rpow_mul_natCast (pow_pos hH0 8).le,← Real.rpow_natCast,← Real.rpow_mul hH0.le]
    norm_num
  have hW : ∀ tau i, 0 < W tau i := by
    intro tau i
    dsimp [W,allpoint_balanced_values5]
    split_ifs
    · exact mul_pos (ha tau i) (Real.rpow_pos_of_pos hQ0 _)
    · exact ha tau i
  have hH2 : 5*H^2 < H^4 := by nlinarith [sq_nonneg (H^2-4)]
  have h58 : H^4≤H^8 := by nlinarith [sq_nonneg (H^4-1)]
  have h3H : 5*H < Q := by change 5*H<H^8; nlinarith
  have h3H2 : 5*H^2 < Q := hH2.trans_le h58
  have hcut : ∀ tau i, (W tau i)⁻¹ < Q := by
    intro tau i
    by_cases ht : tau = 0
    · subst tau
      change (a 0 i*r)⁻¹ < Q
      rw [mul_inv_rev]
      have hri : r⁻¹ ≤ 1 := (inv_le_one₀ hr).mpr hr1
      have hh := mul_le_mul_of_nonneg_right hri (inv_pos.mpr (ha 0 i)).le
      exact (by nlinarith [hinv 0 i] : r⁻¹*(a 0 i)⁻¹ ≤ 5*H).trans_lt h3H
    · simpa only [W,allpoint_balanced_values5,if_neg ht] using (hinv tau i).trans_lt h3H
  have hcut0 : ∀ i, W 0 i < Q := by
    intro i
    change a 0 i*r < Q
    have hh := mul_le_mul (hreal i) hrH hr.le (by linarith : 0 ≤ 5*H)
    exact (by nlinarith : a 0 i*r ≤ 5*H^2).trans_lt h3H2
  have hWfin : ∀ tau, tau ≠ 0 → ∀ i, W tau i ≤ 1 := by
    intro tau ht i
    simpa only [W,allpoint_balanced_values5,if_neg ht] using hfinite tau ht i
  have hWprod : (∏ tau, ∏ i, W tau i) = 1 := by
    have hfactor (tau : Fin 3) : (∏ i : Fin 5,W tau i)=
        if tau=0 then (∏ i : Fin 5,a tau i)*r^5 else ∏ i : Fin 5,a tau i := by
      by_cases ht : tau=0
      · subst tau
        simp only [W,allpoint_balanced_values5,if_pos rfl,ite_true]
        rw [Finset.prod_mul_distrib]
        simp only [Finset.prod_const,Finset.card_univ,Fintype.card_fin]
        rfl
      · simp only [W,allpoint_balanced_values5,if_neg ht]
    simp_rw [hfactor]
    simp only [Fin.prod_univ_three,Fin.isValue,Fin.reduceEq,ite_true,ite_false]
    rw [hrpow]
    have hp := hprod
    simp only [Fin.prod_univ_three] at hp
    nlinarith only [hp]
  have heq := sourceLogWeights5_rpow Q hQ W hW
  refine ⟨hQ,sourceLogWeights5_conditions Q hQ W hW hcut hcut0 hWfin hWprod,heq,?_,?_⟩
  · intro i
    rw [Real.rpow_sub hQ0,heq]
    change a 0 i = (a 0 i*r)/r
    exact (mul_div_cancel_right₀ _ hr.ne').symm
  · intro tau ht i
    rw [heq]
    simp only [W,allpoint_balanced_values5,if_neg ht]

end

theorem allpoint_general_position_intersection5 (z : ℕ → Fin 5 → ℤ)
    (hH : ∀ n,5<(archFiveHeight (z n) : ℝ))
    (havoid : ∀ n,∀ i j k l : Fin n,(fun a => (z n a : ℚ))∉Submodule.span ℚ
