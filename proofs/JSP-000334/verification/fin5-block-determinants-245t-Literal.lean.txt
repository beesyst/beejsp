theorem block_rat_det_one245T (A : Matrix (Fin 5) (Fin 5) ℚ) (hA : A.det = 1) :
    (block245P A).det = 1 := by
  let E : (Fin 10 → ℚ) ≃ₗ[ℚ] ((Fin 5 → ℚ) × (Fin 5 → ℚ)) :=
    { toFun := fun z => (first245Q z, second245Q z)
      invFun := fun z => join245Q z.1 z.2
      left_inv := by
        intro z
        funext j
        obtain ⟨s,rfl⟩ := blockIndex245P.surjective j
        cases s <;> simp [first245Q, second245Q, join245Q]
      right_inv := by
        intro z
        exact Prod.ext (first_join245Q z.1 z.2) (second_join245Q z.1 z.2)
      map_add' := fun _ _ => rfl
      map_smul' := fun _ _ => rfl }
  have hprod : E.toLinearMap ∘ₗ (block245P A).mulVecLin ∘ₗ E.symm.toLinearMap =
      LinearMap.prodMap A.mulVecLin (LinearMap.id : Module.End ℚ (Fin 5 → ℚ)) := by
    apply LinearMap.ext
    intro z
    apply Prod.ext
    · funext i
      change (block245P A *ᵥ join245Q z.1 z.2) (blockIndex245P (Sum.inl i)) = (A *ᵥ z.1) i
      rw [block_mulVec_left245Q, first_join245Q]
    · funext i
      change (block245P A *ᵥ join245Q z.1 z.2) (blockIndex245P (Sum.inr i)) = z.2 i
      rw [block_mulVec_right245Q, second_join245Q]
  have hconj := LinearMap.det_conj (block245P A).mulVecLin E
  rw [hprod] at hconj
  have hfirst : LinearMap.det A.mulVecLin = A.det := by
    rw [← Matrix.toLin'_apply']
    exact LinearMap.det_toLin' A
  have hfull : LinearMap.det (block245P A).mulVecLin = (block245P A).det := by
    rw [← Matrix.toLin'_apply']
    exact LinearMap.det_toLin' (block245P A)
  have hid : LinearMap.det (LinearMap.id : Module.End ℚ (Fin 5 → ℚ)) = 1 :=
    map_one LinearMap.det
  rw [LinearMap.det_prodMap, hfirst, hid, hA, one_mul, hfull] at hconj
  exact hconj.symm

theorem block_algebraic245T (L : Matrix (Fin 5) (Fin 5) ℝ)
    (hAlg : ∀ i j, IsAlgebraic ℚ (L i j)) :
    ∀ i j, IsAlgebraic ℚ (block245P L i j) := by
  intro i j
  simp only [block245P, Matrix.reindex_apply, Matrix.submatrix_apply]
  cases hi : blockIndex245P.symm i with
  | inl a =>
    cases hj : blockIndex245P.symm j with
    | inl b => exact hAlg a b
    | inr b => exact isAlgebraic_zero
  | inr a =>
    cases hj : blockIndex245P.symm j with
    | inl b => exact isAlgebraic_zero
    | inr b =>
      change IsAlgebraic ℚ (if a = b then (1 : ℝ) else 0)
      split_ifs
      · exact isAlgebraic_one
      · exact isAlgebraic_zero

theorem source_block_prerequisites245T
    (A2 A3 : Matrix (Fin 5) (Fin 5) ℚ) (L : Matrix (Fin 5) (Fin 5) ℝ)
    (h2d : A2.det = 1) (h3d : A3.det = 1) (hLd : L.det = 1) (hL : IsUnit L)
    (hAlg : ∀ i j, IsAlgebraic ℚ (L i j)) (c0 c2 c3 : Fin 5 → ℝ)

    (block245P A2).det = 1 ∧
    (block245P A3).det = 1 ∧
    (block245P L).det = 1 ∧
    IsUnit (block245P L) ∧
    (∀ i j, IsAlgebraic ℚ (block245P L i j)) ∧
    (∀ i, |weight0245P c0 i| ≤ 1) ∧
    (∀ i, -1 ≤ weightp245P c2 i ∧ weightp245P c2 i ≤ 0) ∧
    (∀ i, -1 ≤ weightp245P c3 i ∧ weightp245P c3 i ≤ 0) ∧
    (∑ i, (weight0245P c0 i+weightp245P c2 i+weightp245P c3 i)) = 0 := by
  obtain ⟨h0,h2,h3⟩ := weight_bounds245P c0 c2 c3 hc0 hc2 hc3
  exact ⟨block_rat_det_one245T A2 h2d, block_rat_det_one245T A3 h3d,
    block_real_det_one245T L hLd, block_unit245Q L hL, block_algebraic245T L hAlg,
    h0,h2,h3,weight_balance245P c0 c2 c3 hb⟩
