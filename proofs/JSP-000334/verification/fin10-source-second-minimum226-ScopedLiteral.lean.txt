    (h2 : z2 ∈ Submodule.span ℚ {fun i => (M i : ℚ)}) :
    ∃ a b : ℚ, z1 = a • (fun i => (M i : ℚ)) ∧ z2 = b • (fun i => (M i : ℚ)) := by
