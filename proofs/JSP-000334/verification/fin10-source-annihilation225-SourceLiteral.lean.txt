      (ZLattice.covolume_pos _ MeasureTheory.volume) A D.positive (ratReal222 z) ≤ 2) :
    ∀ j : Fin 9, pairing219 z (D.u j.castSucc) = 0 := by
  intro j
  obtain ⟨a, ha, har⟩ := source_integer_pairing225 D z hz j
  have hb := source_rational_pairing224 D z 2 hg j
  rw [har, abs_mul, abs_of_pos (ZLattice.covolume_pos _ MeasureTheory.volume)] at hb
  have hd := ZLattice.covolume_pos
    (integerRealImage (originalSourceGroup10 A2 A3 q c2 c3)) MeasureTheory.volume
  have hstrict : |(a : ℝ)| < 1 := by
    nlinarith [mul_pos hd (sub_pos.mpr hsmall)]
  have ha0 : a = 0 := by
    have hbounds := abs_lt.mp hstrict
    have hlo : (-1 : ℤ) < a := by exact_mod_cast hbounds.1
    have hhi : a < (1 : ℤ) := by exact_mod_cast hbounds.2

  rw [ha, ha0, Int.cast_zero, mul_zero]

theorem delta_cutoff225 (delta : ℝ) (hdelta : 0 < delta) :
    ∃ T : ℝ, 1 < T ∧ ∀ q : ℝ, T < q → 20 * q ^ (-delta) < 1 := by
  obtain ⟨B, hB⟩ := Filter.eventually_atTop.mp
    ((tendsto_rpow_neg_atTop hdelta).eventually
      (gt_mem_nhds (show (0 : ℝ) < 1 / 20 by norm_num)))
  refine ⟨max B 2, lt_of_lt_of_le (by norm_num) (le_max_right _ _), ?_⟩
  intro q hq
  have hb := hB q ((le_max_left B 2).trans hq.le)
  linarith
