theorem rational_place_2_lower (B : Matrix (Fin 10) (Fin 10) ℚ) :
    ∃ C : ℝ, 0 < C ∧ ∀ (M : Fin 10 → ℤ) i,
      (∑ k, (B i k : ℚ_[2])*(M k : ℚ_[2])) ≠ 0 →
      C ≤ ‖∑ k, (B i k : ℚ_[2])*(M k : ℚ_[2])‖ * max 1 (normalHeight10 M : ℝ) := by
  obtain ⟨C,hC,h⟩ := finite_form_lower10 2 (Or.inl rfl) B
  refine ⟨C,hC,?_⟩
  intro M i hn
  exact (h M i hn).1
