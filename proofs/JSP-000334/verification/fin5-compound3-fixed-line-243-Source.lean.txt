theorem actual_cube243 (U : Set ℝ) (hU : ¬ BddAbove U) (hq : ∀ Q ∈ U, 1<Q) :
    ∃ lift : ((fun Q : ℝ => Q^3) '' U) → U,
      (∀ p, (lift p).val^3=p.val) ∧
      ¬ BddAbove ((fun Q : ℝ => Q^3) '' U) ∧
      (∀ p ∈ ((fun Q : ℝ => Q^3) '' U), 1<p) ∧
      ∀ (V : Set ℝ), V ⊆ ((fun Q : ℝ => Q^3) '' U) → ¬ BddAbove V →
        ¬ BddAbove {Q | ∃ p : ((fun Q : ℝ => Q^3) '' U), p.val ∈ V ∧ (lift p).val=Q} := by
  obtain ⟨lift,hlift⟩ := exists_cube_section243 U
  exact ⟨lift,hlift,cubeDomain_unbounded243 U hU hq,cubeDomain_positive243 U hq,
    fun V hVP hV => cube_pullback_unbounded243 U hq lift hlift V hVP hV⟩
