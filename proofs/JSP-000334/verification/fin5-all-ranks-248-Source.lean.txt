  sInf {r : ℝ | ∃ v : Fin (j.val+1) → (Fin 5 → ℝ), LinearIndependent ℝ v ∧
    (∀ i, ∃ z : Fin 5 → ℤ,
      (∀ k, padicNorm 2 ((A2 *ᵥ (fun l => (z l : ℚ))) k) ≤

theorem raw_selected_gap_fixed_flag248
    (A2 A3 : Matrix (Fin 5) (Fin 5) ℚ) (L : Matrix (Fin 5) (Fin 5) ℝ)
    (h2d : A2.det = 1) (h3d : A3.det = 1) (hLd : L.det = 1) (hL : IsUnit L)
    (hAlg : ∀ i j, IsAlgebraic ℚ (L i j)) (c0 c2 c3 : Fin 5 → ℝ)
    (hc0 : ∀ i, |c0 i| ≤ 1) (hc2 : ∀ i, -1 ≤ c2 i ∧ c2 i ≤ 0)
    (hc3 : ∀ i, -1 ≤ c3 i ∧ c3 i ≤ 0) (hb : (∑ i, (c0 i+c2 i+c3 i)) = 0)
    (delta : ℝ) (hd : 0 < delta) (Omega : Set ℝ) (hO : ¬ BddAbove Omega)
    (hq : ∀ Q ∈ Omega, 1 < Q) (j : Fin 4)
    (hgap : ∀ (Q : ℝ) (hQO : Q ∈ Omega),
      rawMinima248 A2 A3 L Q c0 c2 c3 j.castSucc <
        rawMinima248 A2 A3 L Q c0 c2 c3 j.succ*Q^(-delta)) :
    ∃ P : Submodule ℚ (Fin 5 → ℚ), Module.finrank ℚ P = j.val+1 ∧
      ∃ V : Set ℝ, V ⊆ Omega ∧ ¬ BddAbove V ∧
        ∀ (Q : ℝ) (hQV : Q ∈ V),
          ∃ u : Fin 5 → Fin 5 → ℤ,
            LinearIndependent ℝ (fun i => integerReal (u i)) ∧
            (∀ a k, (padicNorm 2 ((A2 *ᵥ (fun l => (u a l : ℚ))) k) : ℝ) ≤ Q^(c2 k)) ∧
            (∀ a k, (padicNorm 3 ((A3 *ᵥ (fun l => (u a l : ℚ))) k) : ℝ) ≤ Q^(c3 k)) ∧
            (∀ i, rawGauge247 L Q c0
              (integerReal (u i)) = rawMinima248 A2 A3 L Q c0 c2 c3 i) ∧
            P = Submodule.span ℚ (Set.range (fun a : Fin (j.val+1) => fun i =>
              (u (Fin.castLE (by omega) a) i : ℚ))) := by
  have hs (Q : ℝ) (hQO : Q ∈ Omega) :
      sourceMinima5 A2 A3 L hL Q (hq Q hQO) c0 c2 c3 j.castSucc <
        sourceMinima5 A2 A3 L hL Q (hq Q hQO) c0 c2 c3 j.succ*Q^(-delta) := by
    rw [← rawMinima_eq248 A2 A3 L hL Q (hq Q hQO) c0 c2 c3 (fun k => (hc2 k).2) (fun k => (hc3 k).2) j.castSucc,
      ← rawMinima_eq248 A2 A3 L hL Q (hq Q hQO) c0 c2 c3 (fun k => (hc2 k).2) (fun k => (hc3 k).2) j.succ]
    exact hgap Q hQO
  obtain ⟨P,hP,V,hVO,hV,hframes⟩ := source_selected_gap_fixed_flag5
    A2 A3 L h2d h3d hLd hL hAlg c0 c2 c3 hc0 hc2 hc3 hb delta hd Omega hO hq j hs
  refine ⟨P,hP,V,hVO,hV,?_⟩
  intro Q hQV
  obtain ⟨u,hu,ha,hspan⟩ := hframes Q hQV
  refine ⟨fun a => (u a : Fin 5 → ℤ),hu,?_,?_,?_,hspan⟩
  · intro a k
    apply (raw_place_eq248 2 Q (c2 k) (hq Q (hVO hQV)) (hc2 k).2 _).mpr
    exact (lattice5_spec 2 A2 _ _).mp (u a).property.1 k
  · intro a k
    apply (raw_place_eq248 3 Q (c3 k) (hq Q (hVO hQV)) (hc3 k).2 _).mpr
    exact (lattice5_spec 3 A3 _ _).mp (u a).property.2 k
  · intro i
    rw [rawGauge_eq247 L hL Q (hq Q (hVO hQV)),
      rawMinima_eq248 A2 A3 L hL Q (hq Q (hVO hQV)) c0 c2 c3 (fun k => (hc2 k).2) (fun k => (hc3 k).2) i]
    exact ha i

theorem raw_normalized_nonroth_fixed_flag248

        ((∀ k, (padicNorm 2 ((A2 *ᵥ (fun l => (x l : ℚ))) k) : ℝ) ≤ Q^(c2 k)) ∧
        (∀ k, (padicNorm 3 ((A3 *ᵥ (fun l => (x l : ℚ))) k) : ℝ) ≤ Q^(c3 k))) ∧

              (integerReal (u i)) = rawMinima248 A2 A3 L Q c0 c2 c3 i) ∧
