def Nondeg {n : ℕ} (x : Fin n → ℚ) : Prop := ∀ I : Finset (Fin n),I.Nonempty → I≠Finset.univ → ∑ i∈I,x i≠0
def Relations (G : Subgroup ℚˣ) (m : ℕ) : Set (Fin (m+1) → ℚ) :=

def WeightedTerms (A : ℚ) (B : Fin 4 → ℚ) (u : Fin 4 → Group) : Fin 5 → ℚ :=

def WeightedFive (A : ℚ) (B : Fin 4 → ℚ) : Set (Fin 4 → Group) :=
  {u | A+∑ i,B i*((u i : ℚˣ) : ℚ)=0 ∧ Nondeg (WeightedTerms A B u)}
def Pairs (A B C : ℚ) : Set (Group × Group) := {p | A+B*((p.1 : ℚˣ) : ℚ)+C*((p.2 : ℚˣ) : ℚ)=0}
def WeightedFour (A B C D : ℚ) : Set (Group × Group × Group) :=
  {p | A+B*((p.1 : ℚˣ) : ℚ)+C*((p.2.1 : ℚˣ) : ℚ)+D*((p.2.2 : ℚˣ) : ℚ)=0 ∧
    Nondeg ![A,B*((p.1 : ℚˣ) : ℚ),C*((p.2.1 : ℚˣ) : ℚ),D*((p.2.2 : ℚˣ) : ℚ)]}
def FixedFinite : Prop := ∀ m : Fin 5,(Relations Group (m.val+1)).Finite
def Value (r : Exponents) : ℕ := 2^r.a+3^r.b+2^r.c*3^r.d
def Solutions (n : ℕ) : Set Exponents := {r | Value r=n}
