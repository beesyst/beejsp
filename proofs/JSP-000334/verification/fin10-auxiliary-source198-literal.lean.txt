def forms10 {n : ℕ} (A2 A3 : Matrix (Fin n) (Fin n) ℚ) (L : Matrix (Fin n) (Fin n) ℝ)
    (tau : Fin 3) : Matrix (Fin n) (Fin n) ℝ :=
  if tau = 0 then L else if tau = 1 then A2.map (algebraMap ℚ ℝ) else A3.map (algebraMap ℚ ℝ)

def finiteMatrix10 {n : ℕ} (p : ℕ) [Fact p.Prime] (A : Matrix (Fin n) (Fin n) ℚ) :
    Matrix (Fin n) (Fin n) ℚ_[p] := A.map (algebraMap ℚ ℚ_[p])

theorem source_det_one10 {n : ℕ} (A2 A3 : Matrix (Fin n) (Fin n) ℚ)
    (L : Matrix (Fin n) (Fin n) ℝ) (h2 : A2.det = 1) (h3 : A3.det = 1) (hL : L.det = 1) :
    ∀ tau, (forms10 A2 A3 L tau).det = 1 := by
  intro tau
  fin_cases tau
  · simp only [forms10, ↓reduceIte]
    exact hL
  · simp only [forms10, show (1 : Fin 3) ≠ 0 by decide, ↓reduceIte]
    exact OriginalDual10.mapped_det_one10 (K := ℝ) A2 h2
  · simp only [forms10, show (2 : Fin 3) ≠ 0 by decide, show (2 : Fin 3) ≠ 1 by decide, ↓reduceIte]
    exact OriginalDual10.mapped_det_one10 (K := ℝ) A3 h3

theorem source_unit10 {n : ℕ} (A2 A3 : Matrix (Fin n) (Fin n) ℚ)
    (L : Matrix (Fin n) (Fin n) ℝ) (h2 : A2.det = 1) (h3 : A3.det = 1) (hL : L.det = 1) :
    ∀ tau, IsUnit (forms10 A2 A3 L tau) := by
  intro tau
  apply (Matrix.isUnit_iff_isUnit_det _).mpr
  rw [source_det_one10 A2 A3 L h2 h3 hL tau]
  exact isUnit_one

theorem finite_det_one10 {n : ℕ} (p : ℕ) [Fact p.Prime] (A : Matrix (Fin n) (Fin n) ℚ)
    (hA : A.det = 1) : (finiteMatrix10 p A).det = 1 := by
  unfold finiteMatrix10
  exact OriginalDual10.mapped_det_one10 (K := ℚ_[p]) A hA

theorem source_algebraic10 {n : ℕ} (A2 A3 : Matrix (Fin n) (Fin n) ℚ)
    (L : Matrix (Fin n) (Fin n) ℝ) (ha : ∀ i j, IsAlgebraic ℚ (L i j)) :
    ∀ tau i j, IsAlgebraic ℚ (forms10 A2 A3 L tau i j) := by
  intro tau i j
  fin_cases tau
  · exact ha i j
  · exact isAlgebraic_algebraMap _
  · exact isAlgebraic_algebraMap _

theorem common_source_field10 (A2 A3 : Matrix (Fin 10) (Fin 10) ℚ)
    (L : Matrix (Fin 10) (Fin 10) ℝ) (ha : ∀ i j, IsAlgebraic ℚ (L i j)) :
    ∃ K : IntermediateField ℚ ℝ, FiniteDimensional ℚ K ∧
      ∀ tau i j, forms10 A2 A3 L tau i j ∈ K := by
  obtain ⟨K,hK,hmem⟩ := liouville10_common_field L ha
  refine ⟨K,hK,?_⟩
  intro tau i j
  fin_cases tau
  · exact hmem i j
  · exact K.algebraMap_mem _
  · exact K.algebraMap_mem _

theorem source_rows_nonzero10 {n : ℕ} (A2 A3 : Matrix (Fin n) (Fin n) ℚ)
    (L : Matrix (Fin n) (Fin n) ℝ) (h2 : A2.det = 1) (h3 : A3.det = 1) (hL : L.det = 1) :
    ∀ tau i, ∃ j, forms10 A2 A3 L tau i j ≠ 0 := by
  intro tau i
  by_contra! hz
  have he := Matrix.det_eq_zero_of_row_eq_zero i hz
  rw [source_det_one10 A2 A3 L h2 h3 hL tau] at he
  exact one_ne_zero he

def exponent10 {m : ℕ} {r : Fin m → ℕ} (b : RothAuxCoefficients r) : Vars10 m →₀ ℕ :=
  Finsupp.equivFunOnFinite.symm (fun u => ((b u.1).val u.2).val)

theorem exponent10_injective {m : ℕ} (r : Fin m → ℕ) : Function.Injective (@exponent10 m r) := by
  intro b c h
  funext g
  apply Subtype.ext
  funext j
  apply Fin.ext
  exact congrArg (fun e : Vars10 m →₀ ℕ => e (g,j)) h

def polynomial10 {m : ℕ} {r : Fin m → ℕ} {R : Type*} [CommRing R] (c : RothAuxCoefficients r → R) : MvPolynomial (Vars10 m) R :=
  ∑ b, MvPolynomial.monomial (exponent10 b) (c b)

theorem polynomial10_coeff {m : ℕ} {r : Fin m → ℕ} {R : Type*} [CommRing R] (c : RothAuxCoefficients r → R) (b : RothAuxCoefficients r) :
    (polynomial10 c).coeff (exponent10 b) = c b := by
  simp only [polynomial10,MvPolynomial.coeff_sum,MvPolynomial.coeff_monomial,(exponent10_injective r).eq_iff]
  simp

theorem polynomial10_homogeneous {m : ℕ} {r : Fin m → ℕ} {R : Type*} [CommRing R] (c : RothAuxCoefficients r → R) : homogeneous10 r (polynomial10 c) := by
  intro d hd h
  obtain ⟨b,_,hb⟩ := Finset.mem_biUnion.mp (MvPolynomial.support_sum hd)
  have hab := Finset.mem_singleton.mp (MvPolynomial.support_monomial_subset hb)
  subst d
  exact (b h).property


variable {R : Type*} [CommRing R]

def chart10 {m : ℕ} (a : Fin 10 → R) (k : Fin 10) (u : Vars10 m) : MvPolynomial (Vars10 m) R :=
  if u.2 = k then -(∑ j ∈ Finset.univ.erase k, MvPolynomial.C (a j) * MvPolynomial.X (u.1,j))
  else MvPolynomial.C (a k) * MvPolynomial.X u

def substitute10 {m : ℕ} (a : Fin 10 → R) (k : Fin 10) : MvPolynomial (Vars10 m) R →+* MvPolynomial (Vars10 m) R :=
  MvPolynomial.eval₂Hom MvPolynomial.C (chart10 a k)

def pivotDerivative10 {m : ℕ} (k : Fin 10) (i : Fin m → ℕ) : Vars10 m →₀ ℕ :=
  Finsupp.equivFunOnFinite.symm (fun u => if u.2 = k then i u.1 else 0)

theorem transformed_monomial10 {m : ℕ} (a : Fin 10 → R) (k : Fin 10) (J b : Vars10 m →₀ ℕ) :
    substitute10 a k (derivative10 J (MvPolynomial.monomial b 1)) =
      MvPolynomial.C (∏ u : Vars10 m, (Nat.choose (b u) (J u) : R)) *
        ∏ u : Vars10 m, chart10 a k u ^ (b-J) u := by
  rw [derivative10_monomial]
  change MvPolynomial.eval₂ MvPolynomial.C (chart10 a k) (MvPolynomial.monomial (b-J)
    (1 * ∏ u : Vars10 m, (Nat.choose (b u) (J u) : R))) = _
  rw [MvPolynomial.eval₂_monomial,one_mul]
  congr 1
  exact Finsupp.prod_fintype _ _ (fun _ => pow_zero _)

theorem exponent10_sum {m : ℕ} {r : Fin m → ℕ} (b : RothAuxCoefficients r) :
    (∑ u : Vars10 m, exponent10 b u) = ∑ h, r h := by
  rw [Fintype.sum_prod_type]
  exact Finset.sum_congr rfl (fun h _ => (b h).property)


theorem transformed_coeff_sum10 {m : ℕ} {ι : Type*} [Fintype ι]
    (e : ι → Vars10 m →₀ ℕ) (c : ι → R)
    (J d : Vars10 m →₀ ℕ)
    (a : Fin 10 → R) (k : Fin 10) :
    let S := substitute10 (m := m) a k;
    (S (derivative10 J (∑ b, MvPolynomial.monomial (e b) (c b)))).coeff d =
      ∑ b, c b * (S (derivative10 J
        (MvPolynomial.monomial (e b) 1))).coeff d := by
  dsimp only
  let S := substitute10 (m := m) a k
  have hadd : derivative10 J
      (∑ b, MvPolynomial.monomial (e b) (c b)) =
      ∑ b, derivative10 J (MvPolynomial.monomial (e b) (c b)) := by
    apply MvPolynomial.ext
    intro a
    simp only [derivative10_coeff, MvPolynomial.coeff_sum,
      Finset.sum_mul]
  rw [hadd, map_sum, MvPolynomial.coeff_sum]
  apply Finset.sum_congr rfl
  intro b _
  rw [derivative10_monomial, derivative10_monomial]
  simp only [one_mul]
  have hc : MvPolynomial.monomial (e b - J)
      (c b * ∏ u : Vars10 m, (Nat.choose (e b u) (J u) : R)) =
      MvPolynomial.C (c b) * MvPolynomial.monomial (e b - J)
        (∏ u : Vars10 m, (Nat.choose (e b u) (J u) : R)) := by
    rw [MvPolynomial.C_mul_monomial]
  rw [hc, map_mul]
  have hC : S (MvPolynomial.C (c b)) = MvPolynomial.C (c b) := by
    simp [S,substitute10]
  rw [hC, MvPolynomial.coeff_C_mul]


theorem rothAux_weight_pivot {m : ℕ} (d : Vars10 m →₀ ℕ) (k : Fin 10) :
    Finsupp.weight (fun u : Vars10 m => if u.2 = k then 1 else 0) d =
      ∑ h : Fin m, d (h,k) := by
  rw [Finsupp.weight_apply, Finsupp.sum_fintype _ _ (by simp)]
  simp [Fintype.sum_prod_type, nsmul_eq_mul]

theorem weighted_neg_ring10 {m : ℕ} (w : Vars10 m → ℕ)
    (P : MvPolynomial (Vars10 m) R) (n : ℕ)
    (hP : MvPolynomial.IsWeightedHomogeneous w P n) :
    MvPolynomial.IsWeightedHomogeneous w (-P) n := by
  intro d hd
  apply hP
  simpa using hd

theorem chart10_block_weight {m : ℕ} (a : Fin 10 → R) (k : Fin 10)
    (h : Fin m) (u : Vars10 m) :
    MvPolynomial.IsWeightedHomogeneous (fun v : Vars10 m => if v.1 = h then 1 else 0)
      (chart10 a k u) (if u.1 = h then 1 else 0) := by
  let w := fun v : Vars10 m => if v.1 = h then 1 else 0
  have hx (j : Fin 10) : MvPolynomial.IsWeightedHomogeneous w
      (MvPolynomial.C (a j) * MvPolynomial.X (u.1,j)) (if u.1 = h then 1 else 0) := by
    simpa only [zero_add] using
      (MvPolynomial.isWeightedHomogeneous_C w (a j)).mul (MvPolynomial.isWeightedHomogeneous_X R w (u.1,j))
  unfold chart10
  by_cases hu : u.2 = k
  · rw [if_pos hu]
    exact weighted_neg_ring10 w _ _
      (MvPolynomial.IsWeightedHomogeneous.sum _ _ _ (fun j _ => hx j))
  · rw [if_neg hu]
    simpa only [zero_add] using
      (MvPolynomial.isWeightedHomogeneous_C w (a k)).mul (MvPolynomial.isWeightedHomogeneous_X R w u)

theorem chart10_pivot_weight {m : ℕ} (a : Fin 10 → R) (k : Fin 10)
    (u : Vars10 m) :
    MvPolynomial.IsWeightedHomogeneous (fun v : Vars10 m => if v.2 = k then 1 else 0)
      (chart10 a k u) 0 := by
  let w := fun v : Vars10 m => if v.2 = k then 1 else 0
  have hx (j : Fin 10) (hj : j ≠ k) : MvPolynomial.IsWeightedHomogeneous w
      (MvPolynomial.C (a j) * MvPolynomial.X (u.1,j)) 0 := by
    simpa only [w, hj, if_false, zero_add] using
      (MvPolynomial.isWeightedHomogeneous_C w (a j)).mul (MvPolynomial.isWeightedHomogeneous_X R w (u.1,j))
  unfold chart10
  split_ifs with hu
  · exact weighted_neg_ring10 w _ 0
      (MvPolynomial.IsWeightedHomogeneous.sum _ _ 0 (fun j hj => hx j (Finset.mem_erase.mp hj).1))
  · simpa only [w, hu, if_false, zero_add] using
      (MvPolynomial.isWeightedHomogeneous_C w (a k)).mul (MvPolynomial.isWeightedHomogeneous_X R w u)

theorem rothAux_scaled_monomial_weighted {m : ℕ} (a : Fin 10 → R) (k : Fin 10)
    (J b : Vars10 m →₀ ℕ) (w : Vars10 m → ℕ) (v : Vars10 m → ℕ)
    (hv : ∀ u, MvPolynomial.IsWeightedHomogeneous w (chart10 a k u) (v u)) :
    MvPolynomial.IsWeightedHomogeneous w
      (substitute10 a k
        (derivative10 J (MvPolynomial.monomial b 1)))
      (∑ u : Vars10 m, (b - J) u * v u) := by
  rw [transformed_monomial10]
  have hp := MvPolynomial.IsWeightedHomogeneous.prod Finset.univ
    (fun u : Vars10 m => chart10 a k u ^ (b - J) u)
    (fun u => (b - J) u * v u) (fun u _ => by simpa only [nsmul_eq_mul] using (hv u).pow ((b - J) u))
  simpa only [zero_add] using (MvPolynomial.isWeightedHomogeneous_C w _).mul hp



theorem rothAux_transformed_polynomial {m : ℕ} {r : Fin m → ℕ}
    (a : Fin 10 → R) (k : Fin 10) (J : Vars10 m →₀ ℕ)
    (c : RothAuxCoefficients r → R) :
    substitute10 a k (derivative10 J (polynomial10 c)) =
      ∑ b, MvPolynomial.C (c b) * substitute10 a k
        (derivative10 J (MvPolynomial.monomial (exponent10 b) 1)) := by
  apply MvPolynomial.ext
  intro d
  rw [polynomial10, transformed_coeff_sum10]
  simp only [MvPolynomial.coeff_sum, MvPolynomial.coeff_C_mul]

theorem rothAux_pivot_monomial_block_weight {m : ℕ} {r : Fin m → ℕ}
    (a : Fin 10 → R) (k : Fin 10) (i : Fin m → ℕ) (b : RothAuxCoefficients r) (h : Fin m) :
    MvPolynomial.IsWeightedHomogeneous (fun u : Vars10 m => if u.1 = h then 1 else 0)
      (substitute10 a k (derivative10
        (pivotDerivative10 k i) (MvPolynomial.monomial (exponent10 b) 1)))
      (r h - i h) := by
  let J := pivotDerivative10 k i
  let w := fun u : Vars10 m => if u.1 = h then 1 else 0
  by_cases hle : J ≤ exponent10 b
  · have hw := rothAux_scaled_monomial_weighted a k J (exponent10 b) w w
      (chart10_block_weight a k h)
    have heq : (∑ u : Vars10 m, (exponent10 b - J) u * w u) =
        ∑ j : Fin 10, (exponent10 b - J) (h,j) := by
      have he := rothAux_weight_block (exponent10 b - J) h
      rw [Finsupp.weight_apply] at he
      rw [Finsupp.sum_fintype (exponent10 b - J)
        (fun u (n : ℕ) => n • w u) (by intro u; simp)] at he
      simpa only [nsmul_eq_mul, Nat.cast_id, w] using he
    have hdeg : (∑ j : Fin 10, (exponent10 b - J) (h,j)) = r h - i h := by
      change (∑ j : Fin 10, (exponent10 b (h,j) - J (h,j))) = _
      rw [Finset.sum_tsub_distrib _ (fun j _ => hle (h,j))]
      have hb : (∑ j : Fin 10, exponent10 b (h,j)) = r h := (b h).property
      rw [hb]
      congr 1
      simp [J, pivotDerivative10]
    rw [heq, hdeg] at hw
    exact hw
  · change ¬ ∀ u, J u ≤ exponent10 b u at hle
    push_neg at hle
    obtain ⟨u, hu⟩ := hle
    have hz : (∏ v : Vars10 m, (Nat.choose (exponent10 b v) (J v) : R)) = 0 := by
      apply Finset.prod_eq_zero (Finset.mem_univ u)
      simp [Nat.choose_eq_zero_of_lt hu]
    rw [transformed_monomial10, hz, map_zero, zero_mul]
    exact MvPolynomial.isWeightedHomogeneous_zero R w _

theorem rothAux_transformed_block_weight {m : ℕ} {r : Fin m → ℕ}

theorem rothAux_transformed_pivot_weight {m : ℕ} {r : Fin m → ℕ}

theorem rothAux_transformed_pivot_zero {m : ℕ} {r : Fin m → ℕ}
    (a : Fin 10 → R) (k : Fin 10) (J : Vars10 m →₀ ℕ)
    (c : RothAuxCoefficients r → R) (d : Vars10 m →₀ ℕ)
    (hd : (substitute10 a k
      (derivative10 J (polynomial10 c))).coeff d ≠ 0) (h : Fin m) :
    d (h,k) = 0 := by
  have hw := rothAux_transformed_pivot_weight a k J c hd
  rw [rothAux_weight_pivot] at hw
  have hle : d (h,k) ≤ ∑ g : Fin m, d (g,k) :=
    Finset.single_le_sum (fun g _ => Nat.zero_le (d (g,k))) (Finset.mem_univ h)
  omega

  have hsum (h : Fin m) : (∑ t : Fin 9, d (h,k.succAbove t)) = s h := by
    have he := Fin.sum_univ_succAbove (fun t => d (h,t)) k
    rw [hs h,hz h,zero_add] at he
    exact he.symm
  let j : (h : Fin m) → Block9 (s h) := fun h => ⟨fun t => ⟨d (h,k.succAbove t),by
    have hh : d (h,k.succAbove t) ≤ ∑ u : Fin 10, d (h,u) :=
      Finset.single_le_sum (s := Finset.univ) (f := fun u : Fin 10 => d (h,u)) (fun u _ => Nat.zero_le _) (Finset.mem_univ _)
    rw [hs h] at hh
    omega⟩,hsum h⟩
  refine ⟨j,?_⟩
  ext u
  rcases u with ⟨h,t⟩
  change Fin.insertNth (α := fun _ : Fin 10 => ℕ) k 0 (fun v : Fin 9 => ((j h).val v).val) t = d (h,t)

  · rw [Fin.insertNth_apply_succAbove]

theorem chart_mass10 {m : ℕ} (a : Fin 10 → R) (k : Fin 10)
    (A : ℝ) (hA : 1 ≤ A) (ha : ∀ j, ‖a j‖ ≤ A) (u : Vars10 m) :
    mass (chart10 a k u) ≤ 10 * A := by
  unfold chart10
  split_ifs
  · rw [mass_neg]
    calc
      _ ≤ ∑ j ∈ Finset.univ.erase k, mass (MvPolynomial.C (a j) * MvPolynomial.X (u.1,j)) := mass_sum _ _
      _ ≤ ∑ _j ∈ Finset.univ.erase k, A := by
        apply Finset.sum_le_sum
        intro j _
        rw [mass_C_mul_X]
        exact ha j
      _ ≤ 10 * A := by simp; linarith
  · rw [mass_C_mul_X]
    exact (ha k).trans (by linarith)

theorem binomial_norm10 {m : ℕ} {r : Fin m → ℕ} (b : RothAuxCoefficients r) (J : Vars10 m →₀ ℕ) :
    ‖∏ u : Vars10 m, (Nat.choose (exponent10 b u) (J u) : R)‖ ≤ (2 : ℝ)^(∑ h, r h) := by
  have hnat (n : ℕ) : ‖(n : R)‖ ≤ (n : ℝ) := by
    rw [← nsmul_one]
    exact (norm_nsmul_le (n := n) (a := (1 : R))).trans_eq (by simp)
  calc
    _ ≤ ∏ u : Vars10 m, ‖(Nat.choose (exponent10 b u) (J u) : R)‖ := Finset.norm_prod_le _ _
    _ ≤ ∏ u : Vars10 m, (2 : ℝ)^exponent10 b u := by
      apply Finset.prod_le_prod
      · intro u _; exact norm_nonneg _
      · intro u _
        apply (hnat _).trans
        have hc : (exponent10 b u).choose (J u) ≤ 2 ^ exponent10 b u := by
          rcases Nat.eq_zero_or_pos (exponent10 b u) with hz | hp
          · rw [hz]; cases J u <;> simp
          · exact (Nat.choose_le_two_pow _ _ hp).le
        exact_mod_cast hc
    _ = _ := by rw [Finset.prod_pow_eq_pow_sum,exponent10_sum]

theorem transformed_entry_norm10 {m : ℕ} {r : Fin m → ℕ}

theorem source_integral_scaling10 (a : Fin 3 → Fin 10 → Fin 10 → 𝕂) :
    ∃ M : ℤ, M ≠ 0 ∧ ∃ b : Fin 3 → Fin 10 → Fin 10 → 𝓞 𝕂,
      ∀ tau s j, (b tau s j : 𝕂) = (M : 𝕂) * a tau s j := by
  haveI : Algebra.IsAlgebraic ℚ 𝕂 := Algebra.IsAlgebraic.of_finite ℚ 𝕂
  haveI : Algebra.IsAlgebraic ℤ 𝕂 :=
    (IsFractionRing.comap_isAlgebraic_iff (A := ℤ) (K := ℚ) (C := 𝕂)).mpr inferInstance
  obtain ⟨M, hM, hi⟩ := Algebra.IsAlgebraic.exists_integral_multiples ℤ
    (Finset.univ.image (fun q : Fin 3 × Fin 10 × Fin 10 => a q.1 q.2.1 q.2.2))
  have hint : ∀ tau s j, IsIntegral ℤ ((M : 𝕂) * a tau s j) := by
    intro tau s j
    simpa only [Algebra.smul_def, eq_intCast] using
      hi (a tau s j) (Finset.mem_image.mpr ⟨(tau,s,j),Finset.mem_univ _,rfl⟩)
  exact ⟨M, hM, (fun tau s j => ⟨(M : 𝕂) * a tau s j, hint tau s j⟩), fun _ _ _ => rfl⟩

theorem algebraic_auxiliary10 (a : Fin 3 → Fin 10 → Fin 10 → 𝕂)
    (ha : ∀ tau s, ∃ j, a tau s j ≠ 0) :
    ∃ D : ℝ, 1 ≤ D ∧
      ∀ eta : ℝ, 0 < eta →
      ∀ m : ℕ, 4 * (eta^2)⁻¹ * Real.log (60 * (Module.finrank ℚ 𝕂 : ℝ)) ≤ (m : ℝ) →
      ∀ r : Fin m → ℕ, (∀ h, 0 < r h) →
      ∃ P : MvPolynomial (Vars10 m) ℤ,
        P ≠ 0 ∧ homogeneous10 r P ∧ (height10 P : ℝ) ≤ D^(∑ h, r h) ∧
        ∀ tau s (J : Vars10 m →₀ ℕ),
          rothSourceDerivativeWeight r J < (1 / 10 - eta) * m →
          ∀ k : Fin m → Fin 10, (∀ h, a tau s (k h) ≠ 0) →
          kernelRestriction10 (a tau s) k
            (derivative10 J (P.map (Int.castRingHom 𝕂))) = 0 := by
  obtain ⟨M, hM, b, hb⟩ := source_integral_scaling10 a
  have hMK : (M : 𝕂) ≠ 0 := by exact_mod_cast hM
  have hbn : ∀ tau s, ∃ j, b tau s j ≠ 0 := by
    intro tau s
    obtain ⟨j, hj⟩ := ha tau s
    refine ⟨j, ?_⟩
    intro hz
    exact (mul_ne_zero hMK hj) (by rw [← hb tau s j, hz]; rfl)
  obtain ⟨D, hD, hconstruction⟩ := integral_auxiliary10 b hbn
  refine ⟨D, hD, ?_⟩
  intro eta heta m hm r hr
  obtain ⟨P, hP, hhom, hheight, hderiv⟩ := hconstruction eta heta m hm r hr

    intro h
    rw [hb]
    exact mul_ne_zero hMK (hk h)

  simp only [hb] at hh
  rw [kernelRestriction10_scale (a tau s) (M : 𝕂) hMK] at hh
  exact hh

theorem monomial_mass {K : Type*} [NormedField K] {m : ℕ} {r : Fin m → ℕ}
    (A : Matrix (Fin 10) (Fin 10) K) (G : ℝ) (hG : 1 ≤ G)
    (hentry : ∀ i j, ‖A⁻¹ i j‖ ≤ G)
    (b : RothAuxCoefficients r) (J : Vars10 m →₀ ℕ) (c : ℤ) :
    mass (expansion A J (MvPolynomial.monomial (exponent10 b) c)) ≤
      (c.natAbs : ℝ) * (20 * G) ^ (∑ h, r h) := by
  let R := ∑ h, r h
  have hbase : 1 ≤ 10 * G := by linarith
  have hsum : (∑ u : Vars10 m, (exponent10 b - J) u) ≤ R := by
    change (∑ u : Vars10 m, (exponent10 b - J) u) ≤ ∑ h, r h
    rw [← exponent10_sum b]
    exact Finset.sum_le_sum (fun u _ => Nat.sub_le _ _)
  have hprod : mass (∏ u : Vars10 m, linearChart A⁻¹ u ^ (exponent10 b - J) u) ≤ (10 * G) ^ R := by
    calc
      _ ≤ ∏ u : Vars10 m, mass (linearChart A⁻¹ u ^ (exponent10 b - J) u) := mass_prod _ _
      _ ≤ ∏ u : Vars10 m, (10 * G) ^ (exponent10 b - J) u := by
        apply Finset.prod_le_prod
        · intro u _; exact mass_nonneg _
        · intro u _
          exact (mass_pow _ _).trans
            (pow_le_pow_left₀ (mass_nonneg _) (chart_mass A⁻¹ G hentry u) _)
      _ = (10 * G) ^ (∑ u : Vars10 m, (exponent10 b - J) u) := Finset.prod_pow_eq_pow_sum _ _ _
      _ ≤ _ := pow_le_pow_right₀ hbase hsum
  have hc : ‖(c : K)‖ ≤ (c.natAbs : ℝ) := by
    simpa only [zsmul_eq_mul, mul_one, norm_one, Int.norm_eq_abs, Int.cast_abs, Int.cast_natAbs] using (norm_zsmul_le (n := c) (a := (1 : K)))
  have he : expansion A J (MvPolynomial.monomial (exponent10 b) c) =
      MvPolynomial.C ((c : K) * ∏ u : Vars10 m, (Nat.choose (exponent10 b u) (J u) : K)) *
        ∏ u : Vars10 m, linearChart A⁻¹ u ^ (exponent10 b - J) u := by
    rw [expansion, MvPolynomial.map_monomial, derivative10_monomial, substitution,
      MvPolynomial.eval₂Hom_monomial, Finsupp.prod_fintype _ _ (fun _ => pow_zero _)]
    rfl
  rw [he]
  calc
    _ ≤ mass (MvPolynomial.C ((c : K) * ∏ u : Vars10 m,
        (Nat.choose (exponent10 b u) (J u) : K))) *
        mass (∏ u : Vars10 m, linearChart A⁻¹ u ^ (exponent10 b - J) u) := mass_mul _ _
    _ ≤ ((c.natAbs : ℝ) * (2 : ℝ)^R) * (10 * G)^R := by
      apply mul_le_mul
      · rw [MvPolynomial.C_apply, mass_monomial]
        exact (norm_mul_le _ _).trans
          (mul_le_mul hc (Aux10Height.binomial_norm10 b J) (norm_nonneg _) (Nat.cast_nonneg _))
      · exact hprod
      · exact mass_nonneg _
      · positivity
    _ = _ := by rw [mul_assoc, ← mul_pow]; congr 2; ring

theorem expansion_sum {K : Type*} [Field K] {m : ℕ}

theorem coefficient_bound {K : Type*} [NormedField K] {m : ℕ}
    (r : Fin m → ℕ) (hr : ∀ h, 0 < r h)
    (P : MvPolynomial (Vars10 m) ℤ) (hP : homogeneous10 r P)
    (D : ℝ) (hD : 1 ≤ D) (hheight : (height10 P : ℝ) ≤ D ^ (∑ h, r h))
    (A : Matrix (Fin 10) (Fin 10) K) (G : ℝ) (hG : 1 ≤ G)
    (hentry : ∀ i j, ‖A⁻¹ i j‖ ≤ G) (J b : Vars10 m →₀ ℕ) :
    ‖(expansion A J P).coeff b‖ ≤ (20480 * D * G) ^ (∑ h, r h) := by
  have hDpos : 0 ≤ D := by linarith
  have hGpos : 0 ≤ G := by linarith
  apply (coeff_le_mass _ _).trans
  rw [expansion_sum]
  calc
    _ ≤ ∑ d ∈ P.support, mass (expansion A J (MvPolynomial.monomial d (P.coeff d))) := mass_sum _ _
    _ ≤ ∑ _d ∈ P.support, D^(∑ h, r h) * (20 * G)^(∑ h, r h) := by
      apply Finset.sum_le_sum
      intro d hd
      obtain ⟨a,ha⟩ := support_index r P hP d hd
      have hc : ((P.coeff d).natAbs : ℝ) ≤ D^(∑ h, r h) := by
        have hh : (P.coeff d).natAbs ≤ height10 P := Finset.le_sup (f := fun e => (P.coeff e).natAbs) hd
        exact (by exact_mod_cast hh : ((P.coeff d).natAbs : ℝ) ≤ (height10 P : ℝ)).trans hheight
      have hm := monomial_mass A G hG hentry a J (P.coeff d)
      rw [ha] at hm
      exact hm.trans (mul_le_mul_of_nonneg_right hc (by positivity))
    _ = (P.support.card : ℝ) * (D^(∑ h, r h) * (20 * G)^(∑ h, r h)) := by simp
    _ ≤ (1024 : ℝ)^(∑ h, r h) * (D^(∑ h, r h) * (20 * G)^(∑ h, r h)) := by
      have hc : (P.support.card : ℝ) ≤ (1024 : ℝ)^(∑ h, r h) := by exact_mod_cast support_card r hr P hP
      exact mul_le_mul_of_nonneg_right hc (by positivity)
    _ = _ := by rw [← mul_pow, ← mul_pow]; congr 1; ring



def inverseBudget {K : Type*} [NormedField K] (A : Matrix (Fin 10) (Fin 10) K) : ℝ :=
  1 + ∑ i : Fin 10, ∑ j : Fin 10, ‖A⁻¹ i j‖

theorem inverseBudget_one {K : Type*} [NormedField K]
    (A : Matrix (Fin 10) (Fin 10) K) : 1 ≤ inverseBudget A := by
  unfold inverseBudget
  exact le_add_of_nonneg_right (Finset.sum_nonneg (fun _ _ => Finset.sum_nonneg (fun _ _ => norm_nonneg _)))

theorem inverseBudget_entry {K : Type*} [NormedField K]
    (A : Matrix (Fin 10) (Fin 10) K) (i j : Fin 10) : ‖A⁻¹ i j‖ ≤ inverseBudget A := by
  have hrow : ‖A⁻¹ i j‖ ≤ ∑ k : Fin 10, ‖A⁻¹ i k‖ :=
    Finset.single_le_sum (fun _ _ => norm_nonneg _) (Finset.mem_univ j)
  have hall : (∑ k : Fin 10, ‖A⁻¹ i k‖) ≤ ∑ h : Fin 10, ∑ k : Fin 10, ‖A⁻¹ h k‖ :=
    Finset.single_le_sum (fun _ _ => Finset.sum_nonneg (fun _ _ => norm_nonneg _)) (Finset.mem_univ i)
  unfold inverseBudget
  linarith

theorem source_coefficient_bounds (A2 A3 : Matrix (Fin 10) (Fin 10) ℚ)
    (L : Matrix (Fin 10) (Fin 10) ℝ) (D : ℝ) (hD : 1 ≤ D) :
    ∃ E : ℝ, 1 ≤ E ∧ ∀ (m : ℕ) (r : Fin m → ℕ), (∀ h, 0 < r h) →

    (max (inverseBudget (SourceForms10.finiteMatrix10 2 A2)) (inverseBudget (SourceForms10.finiteMatrix10 3 A3)))
  have h0 : inverseBudget L ≤ G := le_max_left _ _

  have hG : 1 ≤ G := (inverseBudget_one _).trans h0

  exact ⟨coefficient_bound r hr P hP D hD hheight _ G hG

theorem chart_pderiv {m : ℕ} (B : Matrix (Fin 10) (Fin 10) K)
    (h : Fin m) (i : Fin 10) (u : Vars10 m) :

  by_cases hg : g = h
  · subst g
    simp [linearChart, map_sum, MvPolynomial.pderiv_C_mul, MvPolynomial.pderiv_X,

theorem substitution_chain {m : ℕ} (B : Matrix (Fin 10) (Fin 10) K)
    (h : Fin m) (i : Fin 10) (P : MvPolynomial (Vars10 m) K) :
    MvPolynomial.pderiv (h,i) (substitution B P) =
      ∑ j : Fin 10, MvPolynomial.C (B j i) *
        substitution B (MvPolynomial.pderiv (h,j) P) := by
  classical
  induction P using MvPolynomial.induction_on with
  | C c => simp [substitution]
  | add P Q hP hQ => simp only [map_add, hP, hQ, mul_add, Finset.sum_add_distrib]
  | mul_X P u hP =>
    have hdelta : (∑ j : Fin 10, (MvPolynomial.C (B j i) * substitution B P) *
        (if (h,j) = u then 1 else 0)) =
        substitution B P * (if u.1 = h then MvPolynomial.C (B u.2 i) else 0) := by
      rcases u with ⟨g,l⟩
      by_cases hg : g = h
      · subst g
        simp [Prod.mk.injEq, mul_ite, Finset.sum_ite_eq, Finset.sum_ite_eq', mul_comm]
      · simp [Prod.mk.injEq, hg, Ne.symm hg]
    have hX : substitution B (MvPolynomial.X u) = linearChart B u := by
      simp [substitution]
    have hD (j : Fin 10) :
        substitution B (MvPolynomial.pderiv (h,j) (P * MvPolynomial.X u)) =
          substitution B (MvPolynomial.pderiv (h,j) P) * linearChart B u +
          substitution B P * (if (h,j) = u then 1 else 0) := by
      by_cases he : (h,j) = u <;>
        simp [MvPolynomial.pderiv_mul, hX, MvPolynomial.pderiv_X, Pi.single_apply, he, eq_comm] <;> ring
    rw [map_mul, MvPolynomial.pderiv_mul, hP, hX, chart_pderiv]
    simp_rw [hD, mul_add, ← mul_assoc]
    rw [Finset.sum_add_distrib, ← Finset.sum_mul, hdelta]

theorem kernel_factor_variable {m : ℕ} (L : Matrix (Fin 10) (Fin 10) K)
    (hL : (L).det ≠ 0) (i p : Fin 10) (hp : L i p ≠ 0)
    (u : Vars10 m) :
    MvPolynomial.eval₂Hom MvPolynomial.C (kernelImage L i)
      (kernelChart10 (L i) (fun _ => p) u) =
      kernelImage L i u := by
  classical
  have ha : (L i p : K) ≠ 0 := by exact_mod_cast hp
  have hcancel : (MvPolynomial.C ((L i p : K)⁻¹) : MvPolynomial (Vars10 m) K) *
      MvPolynomial.C (L i p : K) = 1 := by
    rw [← MvPolynomial.C_mul, inv_mul_cancel₀ ha, map_one]
  by_cases hu : u.2 = p
  · have hs := kernel_image_form L hL i u.1
    rw [← Finset.sum_erase_add Finset.univ _ (Finset.mem_univ p)] at hs
    have hr : -(∑ j ∈ Finset.univ.erase p,
        MvPolynomial.C (L i j : K) * kernelImage L i (u.1,j)) =
        MvPolynomial.C (L i p : K) * kernelImage L i (u.1,p) := by
      linear_combination -hs
    simp only [kernelChart10, hu, if_pos rfl, ite_true, map_neg, map_sum, map_mul,
      MvPolynomial.eval₂Hom_C, MvPolynomial.eval₂Hom_X']
    calc
      _ = MvPolynomial.C ((L i p : K)⁻¹) *
          -(∑ j ∈ Finset.univ.erase p,
            MvPolynomial.C (L i j : K) * kernelImage L i (u.1,j)) := by
        rw [mul_neg, Finset.mul_sum]
        congr 1
        apply Finset.sum_congr rfl
        intro j hj
        rw [div_eq_mul_inv, MvPolynomial.C_mul]
        ring
      _ = _ := by rw [hr, ← mul_assoc, hcancel, one_mul]; simp [← hu]
  · simp [kernelChart10, hu]

theorem zero_monomial {m : ℕ} (i : Fin 10)
    (b : Vars10 m →₀ ℕ) (c : K) :
    zeroCoordinate i (MvPolynomial.monomial b c) =
      if (∀ h, b (h,i) = 0) then MvPolynomial.monomial b c else 0 := by
  classical
  rw [zeroCoordinate, MvPolynomial.eval₂Hom_monomial]
  rw [Finsupp.prod_fintype _ _ (fun _ => by simp)]
  by_cases hb : ∀ h, b (h,i) = 0
  · rw [if_pos hb, MvPolynomial.monomial_eq]
    congr 1
    rw [Finsupp.prod_fintype _ _ (fun _ => by simp)]
    apply Finset.prod_congr rfl
    intro u _
    by_cases hu : u.2 = i
    · have he : u = (u.1,i) := Prod.ext rfl hu
      have hz : b u = 0 := (congrArg b he).trans (hb u.1)
      simp [hu, hz]
    · simp [hu]
  · rw [if_neg hb]
    push_neg at hb
    obtain ⟨h, hh⟩ := hb
    have hz : (∏ u : Vars10 m, (if u.2 = i then 0 else MvPolynomial.X u) ^ b u) =
        (0 : MvPolynomial (Vars10 m) K) := by
      apply Finset.prod_eq_zero (Finset.mem_univ (h,i))
      simp [hh]
    rw [hz, mul_zero]

theorem zero_coeff {m : ℕ} (i : Fin 10)
    (Q : MvPolynomial (Vars10 m) K) (b : Vars10 m →₀ ℕ) :
    (zeroCoordinate i Q).coeff b = if (∀ h, b (h,i) = 0) then Q.coeff b else 0 := by
  classical
  induction Q using MvPolynomial.induction_on' with
  | monomial a c =>
    rw [zero_monomial]
    by_cases ha : ∀ h, a (h,i) = 0
    · rw [if_pos ha]
      by_cases hb : ∀ h, b (h,i) = 0
      · simp [hb]
      · have hab : a ≠ b := by intro he; subst a; exact hb ha
        simp [hb, hab]
    · rw [if_neg ha]
      by_cases hb : ∀ h, b (h,i) = 0
      · have hab : a ≠ b := by intro he; subst a; exact ha hb
        simp [hb, hab]
      · simp [hb]
  | add P Q hP hQ =>
    rw [map_add, MvPolynomial.coeff_add, hP, hQ, MvPolynomial.coeff_add]
    split_ifs <;> simp

theorem pderiv_coeff {m : ℕ} (u : Vars10 m)
    (Q : MvPolynomial (Vars10 m) K) (b : Vars10 m →₀ ℕ) :
    (MvPolynomial.pderiv u Q).coeff b =
      (b u + 1 : ℕ) * Q.coeff (b + Finsupp.single u 1) := by
  classical
  induction Q using MvPolynomial.induction_on' with
  | add P Q hP hQ => simp only [map_add, MvPolynomial.coeff_add, hP, hQ]; ring
  | monomial a c =>
    by_cases ha : a u = 0
    · have hab : a ≠ b + Finsupp.single u 1 := by
        intro he
        have h := congrArg (fun d : Vars10 m →₀ ℕ => d u) he
        simp [ha, Finsupp.add_apply, Finsupp.single_apply] at h
      simp [MvPolynomial.pderiv_monomial, ha, MvPolynomial.coeff_monomial, hab]
    · have hle : Finsupp.single u 1 ≤ a := by rw [Finsupp.single_le_iff]; omega
      have heq : a - Finsupp.single u 1 = b ↔ a = b + Finsupp.single u 1 := by
        constructor
        · intro he
          rw [← he]
          exact (tsub_add_cancel_of_le hle).symm
        · intro he
          rw [he, add_tsub_cancel_right]
      simp only [MvPolynomial.pderiv_monomial, MvPolynomial.coeff_monomial, heq]
      split_ifs with he
      · subst a
        simp only [Finsupp.add_apply, Finsupp.single_eq_same, Nat.cast_add, Nat.cast_one]
        ring
      · simp

theorem coeff_zero_below_index {m : ℕ} (r : Fin m → ℕ)
    (L : Matrix (Fin 10) (Fin 10) K) (hL : (L).det ≠ 0)
    (i p : Fin 10) (hp : L i p ≠ 0) (T : ℝ)
    (P : MvPolynomial (Vars10 m) ℤ)
    (hindex : ∀ J : Vars10 m →₀ ℕ, rothSourceDerivativeWeight r J < T →
      kernelRestriction10 (L i) (fun _ => p) (derivative10 J (P.map (Int.castRingHom K))) = 0)
    (J b : Vars10 m →₀ ℕ)
    (hbad : rothSourceDerivativeWeight r J + coordinateWeight r i b < T) :
    (expansion L J P).coeff b = 0 := by
  classical
  have go : ∀ n : ℕ, ∀ J b : Vars10 m →₀ ℕ,
      (∑ h : Fin m, b (h,i)) = n →
      rothSourceDerivativeWeight r J + coordinateWeight r i b < T →
      (expansion L J P).coeff b = 0 := by
    intro n
    induction n using Nat.strong_induction_on with
    | h n ih =>
      intro J b hn hw
      by_cases hnzero : n = 0
      · have hbzero : ∀ h, b (h,i) = 0 := by
          intro h
          have hle : b (h,i) ≤ ∑ g : Fin m, b (g,i) :=
            Finset.single_le_sum (f := fun g : Fin m => b (g,i)) (fun _ _ => Nat.zero_le _) (Finset.mem_univ h)
          omega
        have hnonneg : 0 ≤ coordinateWeight r i b := by
          unfold coordinateWeight
          positivity
        have hJ : rothSourceDerivativeWeight r J < T := by linarith
        have hz := index_zero_slice r L hL i p hp T P hindex J hJ
        have he := congrArg (MvPolynomial.coeff b) hz
        rw [zero_coeff, if_pos hbzero, MvPolynomial.coeff_zero] at he
        exact he
      · have hex : ∃ h : Fin m, b (h,i) ≠ 0 := by
          by_contra hzero
          push_neg at hzero
          have hs : (∑ h : Fin m, b (h,i)) = 0 := Finset.sum_eq_zero (fun h _ => hzero h)
          omega
        obtain ⟨h, hh⟩ := hex
        let c := b - Finsupp.single (h,i) 1
        have hle : Finsupp.single (h,i) 1 ≤ b := by rw [Finsupp.single_le_iff]; omega
        have hceq : c + Finsupp.single (h,i) 1 = b := tsub_add_cancel_of_le hle
        have hcount : (∑ g : Fin m, c (g,i)) + 1 = n := by
          have hs := congrArg (fun d : Vars10 m →₀ ℕ => ∑ g : Fin m, d (g,i)) hceq
          simpa [Finsupp.add_apply, Finsupp.single_apply, Prod.mk.injEq,
            Finset.sum_add_distrib, Finset.sum_ite_eq, Finset.sum_ite_eq', hn] using hs
        have hcoord : coordinateWeight r i c + 1 / (r h : ℝ) =
            coordinateWeight r i b := by
          rw [← coordinate_weight_single r i h, ← coordinate_weight_add, hceq]
        have hz (j : Fin 10) : (expansion L (J + Finsupp.single (h,j) 1) P).coeff c = 0 := by
          apply ih (∑ g : Fin m, c (g,i)) (by omega) _ c rfl
          rw [rothSource_weight_add, rothSource_weight_single]
          linarith
        have hd : (MvPolynomial.pderiv (h,i) (expansion L J P)).coeff c = 0 := by
          rw [expansion_chain, MvPolynomial.coeff_sum]
          apply Finset.sum_eq_zero
          intro j hj
          rw [MvPolynomial.coeff_C_mul, MvPolynomial.coeff_smul, hz j, smul_zero, mul_zero]

theorem chart_block_weight {m : ℕ} (B : Matrix (Fin 10) (Fin 10) K)
    (h : Fin m) (u : Vars10 m) :
    MvPolynomial.IsWeightedHomogeneous (fun v : Vars10 m => if v.1 = h then 1 else 0)
      (linearChart B u) (if u.1 = h then 1 else 0) := by
  unfold linearChart
  apply MvPolynomial.IsWeightedHomogeneous.sum
  intro j hj
  simpa only [zero_add] using
    (MvPolynomial.isWeightedHomogeneous_C _ (B u.2 j)).mul
      (MvPolynomial.isWeightedHomogeneous_X K (fun v : Vars10 m => if v.1 = h then 1 else 0) (u.1,j))

theorem expansion_block_weight {m : ℕ} (r : Fin m → ℕ)
    (L : Matrix (Fin 10) (Fin 10) K) (J : Vars10 m →₀ ℕ)
    (P : MvPolynomial (Vars10 m) ℤ) (hP : homogeneous10 r P) (h : Fin m) :
    MvPolynomial.IsWeightedHomogeneous (fun v : Vars10 m => if v.1 = h then 1 else 0)
      (expansion L J P) (r h - ∑ j : Fin 10, J (h,j)) := by
  classical
  let Q := derivative10 J (P.map (Int.castRingHom K))
  let w := fun v : Vars10 m => if v.1 = h then 1 else 0
  have hdegree (d : Vars10 m →₀ ℕ) (hd : d ∈ Q.support) :
      (∑ j : Fin 10, d (h,j)) = r h - ∑ j : Fin 10, J (h,j) := by
    have hcoeff : P.coeff (d + J) ≠ 0 := by
      intro hz
      have hn := MvPolynomial.mem_support_iff.mp hd
      rw [derivative10_coeff, MvPolynomial.coeff_map, hz, map_zero, zero_mul] at hn
      exact hn rfl
    have hh := hP (d + J) (MvPolynomial.mem_support_iff.mpr hcoeff) h
    simp only [Finsupp.add_apply, Finset.sum_add_distrib] at hh
    omega
  change MvPolynomial.IsWeightedHomogeneous w
    (MvPolynomial.eval₂Hom MvPolynomial.C
      (linearChart (L)⁻¹) Q) _
  conv_lhs => arg 2; rw [MvPolynomial.as_sum Q]
  rw [map_sum]
  apply MvPolynomial.IsWeightedHomogeneous.sum
  intro d hd
  rw [MvPolynomial.eval₂Hom_monomial, Finsupp.prod_fintype _ _ (fun _ => by simp)]
  have hw : (∑ u : Vars10 m, d u * w u) = ∑ j : Fin 10, d (h,j) := by
    have he := rothAux_weight_block d h
    rw [Finsupp.weight_apply, Finsupp.sum_fintype d (fun u (n : ℕ) => n • w u) (by intro u; simp)] at he
    simpa only [nsmul_eq_mul, Nat.cast_id, w] using he
  have hprod := MvPolynomial.IsWeightedHomogeneous.prod Finset.univ
    (fun u : Vars10 m => linearChart (L)⁻¹ u ^ d u)
    (fun u => d u * w u)
    (fun u _ => by simpa only [nsmul_eq_mul] using
      (chart_block_weight (L)⁻¹ h u).pow (d u))
  simpa only [zero_add, hw, hdegree d hd] using
    (MvPolynomial.isWeightedHomogeneous_C w (Q.coeff d)).mul hprod

theorem expansion_block_degree {m : ℕ} (r : Fin m → ℕ)
    (L : Matrix (Fin 10) (Fin 10) K) (J : Vars10 m →₀ ℕ)
    (P : MvPolynomial (Vars10 m) ℤ) (hP : homogeneous10 r P)
    (b : Vars10 m →₀ ℕ) (hb : (expansion L J P).coeff b ≠ 0) (h : Fin m) :
    (∑ j : Fin 10, b (h,j)) = r h - ∑ j : Fin 10, J (h,j) := by
  have hw := expansion_block_weight r L J P hP h hb
  rwa [rothAux_weight_block] at hw

theorem total_coordinate_weight_le {m : ℕ} (r : Fin m → ℕ)
    (hr : ∀ h, 0 < r h) (L : Matrix (Fin 10) (Fin 10) K) (J : Vars10 m →₀ ℕ)
    (P : MvPolynomial (Vars10 m) ℤ) (hP : homogeneous10 r P)
    (b : Vars10 m →₀ ℕ) (hb : (expansion L J P).coeff b ≠ 0) :
    (∑ i : Fin 10, coordinateWeight r i b) ≤ (m : ℝ) := by
  unfold coordinateWeight
  rw [Finset.sum_comm]
  calc
    _ ≤ ∑ h : Fin m, (1 : ℝ) := by
      apply Finset.sum_le_sum
      intro h hh
      rw [← Finset.sum_div]
      have hrh : (0 : ℝ) < r h := by exact_mod_cast hr h
      apply (div_le_one hrh).mpr
      have hdeg := expansion_block_degree r L J P hP b hb h
      exact_mod_cast (show (∑ i : Fin 10, b (h,i)) ≤ r h by omega)
    _ = _ := by simp




theorem balance_from_weights (m : ℕ) (eta : ℝ) (heta : 0 ≤ eta)
    (x : Fin 10 → ℝ) (hsum : (∑ i, x i) ≤ m)
    (hlower : ∀ i, (m : ℝ)/10 - 3*eta*m ≤ x i) :
    ∀ i, |x i - (m : ℝ)/10| ≤ 54*eta*m := by
  intro i
  have hc : (Finset.univ.erase i : Finset (Fin 10)).card = 9 := by simp
  have ho := Finset.sum_le_sum (s := Finset.univ.erase i) (fun j _ => hlower j)
  have ho' : 9*((m : ℝ)/10 - 3*eta*m) ≤ ∑ j ∈ Finset.univ.erase i, x j := by
    simpa only [Finset.sum_const, hc, nsmul_eq_mul] using ho
  have ht : (∑ j ∈ Finset.univ.erase i, x j) + x i ≤ m := by
    rw [Finset.sum_erase_add _ _ (Finset.mem_univ i)]
    exact hsum
  have hn : 0 ≤ eta*(m : ℝ) := mul_nonneg heta (Nat.cast_nonneg _)
  exact abs_le.mpr ⟨by nlinarith [hlower i], by nlinarith⟩

theorem balanced_of_vanishing {m : ℕ} (r : Fin m → ℕ) (hr : ∀ h, 0 < r h)
    (eta : ℝ) (heta : 0 < eta) (A : Matrix (Fin 10) (Fin 10) K)
    (P : MvPolynomial (Vars10 m) ℤ) (hP : homogeneous10 r P)
    (hz : ∀ (i : Fin 10) (J b : Vars10 m →₀ ℕ),
      rothSourceDerivativeWeight r J + coordinateWeight r i b < (1/10-eta)*m →
      (expansion A J P).coeff b = 0)
    (J b : Vars10 m →₀ ℕ) (hJ : rothSourceDerivativeWeight r J ≤ 2*eta*m)
    (hb : (expansion A J P).coeff b ≠ 0) :
    ∀ i, |coordinateWeight r i b - (m : ℝ)/10| ≤ 54*eta*m := by
  have hlower (i : Fin 10) : (m : ℝ)/10 - 3*eta*m ≤ coordinateWeight r i b := by
    have ht : (1/10-eta)*m ≤ rothSourceDerivativeWeight r J + coordinateWeight r i b := by
      by_contra! hbad
      exact hb (hz i J b hbad)
    nlinarith
  exact balance_from_weights m eta heta.le (fun i => coordinateWeight r i b)
    (total_coordinate_weight_le r hr A J P hP b hb) hlower

def Claims {m : ℕ} (r : Fin m → ℕ) (eta : ℝ)
    (A : Matrix (Fin 10) (Fin 10) K) (P : MvPolynomial (Vars10 m) ℤ) : Prop :=
  (∀ J, substitution A (expansion A J P) = derivative10 J (P.map (Int.castRingHom K))) ∧
  (∀ (i : Fin 10) (J b : Vars10 m →₀ ℕ),
    rothSourceDerivativeWeight r J + coordinateWeight r i b < (1/10-eta)*m →
    (expansion A J P).coeff b = 0) ∧
  (∀ (J b : Vars10 m →₀ ℕ), (expansion A J P).coeff b ≠ 0 →
    ∀ h, (∑ j : Fin 10, b (h,j)) = r h - ∑ j : Fin 10, J (h,j)) ∧
  (∀ (J b : Vars10 m →₀ ℕ), (expansion A J P).coeff b ≠ 0 →
    (∑ i : Fin 10, coordinateWeight r i b) ≤ m) ∧
  (∀ (J b : Vars10 m →₀ ℕ), rothSourceDerivativeWeight r J ≤ 2*eta*m →
    (expansion A J P).coeff b ≠ 0 →
    ∀ i, |coordinateWeight r i b - (m : ℝ)/10| ≤ 54*eta*m)

theorem claims_of_charts {m : ℕ} (r : Fin m → ℕ) (hr : ∀ h, 0 < r h)
    (eta : ℝ) (heta : 0 < eta) (A : Matrix (Fin 10) (Fin 10) K) (hA : A.det ≠ 0)
    (P : MvPolynomial (Vars10 m) ℤ) (hP : homogeneous10 r P)
    (hindex : ∀ (i : Fin 10) (J : Vars10 m →₀ ℕ),
      rothSourceDerivativeWeight r J < (1/10-eta)*m →
      ∀ p : Fin 10, A i p ≠ 0 → kernelRestriction10 (A i) (fun _ => p)
        (derivative10 J (P.map (Int.castRingHom K))) = 0) : Claims r eta A P := by
  have hz (i : Fin 10) (J b : Vars10 m →₀ ℕ)
      (hb : rothSourceDerivativeWeight r J + coordinateWeight r i b < (1/10-eta)*m) :
      (expansion A J P).coeff b = 0 := by
    have hn : ∃ p, A i p ≠ 0 := by
      by_contra! hh
      exact hA (Matrix.det_eq_zero_of_row_eq_zero i hh)
    obtain ⟨p,hp⟩ := hn
    apply coeff_zero_below_index r A hA i p hp ((1/10-eta)*m) P
    · intro D hD
      exact hindex i D hD p hp
    · exact hb
  exact ⟨fun J => reconstruct A hA J P, hz,
    fun J b hb h => expansion_block_degree r A J P hP b hb h,
    fun J b hb => total_coordinate_weight_le r hr A J P hP b hb,
    fun J b hJ hb => balanced_of_vanishing r hr eta heta A P hP hz J b hJ hb⟩



theorem claims_of_matrix_index10 {F : Type*} [Field F] [CharZero F]
    {m : ℕ} (B : Matrix (Fin 10) (Fin 10) F) (hB : B.det = 1)
    (r : Fin m → ℕ) (hr : ∀ h, 0 < r h) (eta : ℝ) (heta : 0 < eta)
    (P : MvPolynomial (Vars10 m) ℤ) (hhom : homogeneous10 r P)
    (hi : matrixIndex10 B r eta P) : Claims r eta B P := by
  apply claims_of_charts r hr eta heta B (by rw [hB]; exact one_ne_zero) P hhom
  intro i J hJ k hk
  exact hi i J hJ (fun _ => k) (fun _ => hk)
