theorem omitted_exponent (p : ℝ) (hp : p ≠ 0) (f : Fin 10 → ℤ) (i : Fin 10) :
    (∏ k : Fin 9, p^(f (i.succAbove k))) = p^(∑ k : Fin 9, f (i.succAbove k)) := by
  have h (s : Finset (Fin 9)) :
      (∏ k ∈ s, p^(f (i.succAbove k))) = p^(∑ k ∈ s, f (i.succAbove k)) := by
    induction s using Finset.induction_on with
    | empty => simp
    | @insert a s ha ih => rw [Finset.prod_insert ha,Finset.sum_insert ha,ih,zpow_add₀ hp]
  exact h Finset.univ

theorem source_constant : (Nat.factorial 9 : ℝ) = 362880 := by norm_num [Nat.factorial]

    (A2 A3 : Matrix (Fin 10) (Fin 10) ℚ) (L : Matrix (Fin 10) (Fin 10) ℝ)
    (h2 : A2.det = 1) (h3 : A3.det = 1) (hLd : L.det = 1) (hL : IsUnit L)
    (hAlg : ∀ i j, IsAlgebraic ℚ (L i j)) (delta : ℝ) (hd : 0 < delta)
    (S0 : Finset (Fin 3 → Fin 10)) :
    ∃ Q0 : ℝ, 0 < Q0 ∧ ∀ (Q : ℝ) (hQ : 1 < Q), Q0 < Q →
      ∀ D : OriginalData10 A2 A3 L hL Q hQ delta S0,
      let G := originalLattice212 D
      let J := originalIndex212 D
      let d := ZLattice.covolume (integerRealImage G)
      let v := lastDual10 (fun j i => (D.u j i : ℚ))
        (integer_frame_rational_independent10 D.u D.independent)
      (∀ x, x ∈ G ↔
        (∀ i, padicNorm 2 ((A2 *ᵥ (fun k => (x k : ℚ))) i) ≤ (2 : ℚ)^D.f2 i) ∧
        (∀ i, padicNorm 3 ((A3 *ᵥ (fun k => (x k : ℚ))) i) ≤ (3 : ℚ)^D.f3 i)) ∧
      0 < J ∧ 0 < d ∧
      J * G.index = (Matrix.of D.u : Matrix (Fin 10) (Fin 10) ℤ).det.natAbs ∧
      |((Matrix.of D.u : Matrix (Fin 10) (Fin 10) ℤ).det : ℝ)| = (J : ℝ)*d ∧
      d = (G.index : ℝ) ∧ ((J*G.index : ℕ) : ℝ) = (J : ℝ)*d ∧
      (∀ i : Fin 10,
        ‖∑ k, ((J : ℝ)*d*(v k : ℝ))*L⁻¹.transpose i k‖ ≤
          (Nat.factorial 9 : ℝ)*(D.A i)⁻¹*d*Q^(-delta*(10-1)) ∧
        ‖∑ k, (((J*G.index : ℕ) : ℚ_[2])*(v k : ℚ_[2]))*
          (A2.map (algebraMap ℚ ℚ_[2]))⁻¹.transpose i k‖ ≤

        ‖((A2.map (algebraMap ℚ ℚ_[2]))⁻¹.transpose *ᵥ (fun k => (v k : ℚ_[2]))) (s 1)‖ *
