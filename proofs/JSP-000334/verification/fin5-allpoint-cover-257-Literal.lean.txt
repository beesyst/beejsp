theorem successiveRadius5_le_family (b : Basis (Fin 5) ℝ (Fin 5 → ℝ))
    (e : (Fin 5 → ℝ) ≃ₗ[ℝ] (Fin 5 → ℝ)) (k : Fin 5)
    (u : Fin (k.val + 1) → (Fin 5 → ℝ)) (hu : LinearIndependent ℝ u)
    (huL : ∀ i, u i ∈ lattice5Span b) (r : ℝ)
    (hr : ∀ i, coordinateGauge5 e (u i) ≤ r) :
    successiveRadius5 b e k ≤ r := by
  unfold successiveRadius5
  apply csInf_le (s := {s : ℝ | ∃ w : Fin (k.val + 1) → (Fin 5 → ℝ),
    LinearIndependent ℝ w ∧ (∀ i, w i ∈ lattice5Span b) ∧
      ∀ i, coordinateGauge5 e (w i) ≤ s}) _ ⟨u, hu, huL, hr⟩
  refine ⟨0, ?_⟩
  rintro s ⟨w, _, _, hw⟩
  exact (norm_nonneg (e (w 0))).trans (hw 0)

theorem lattice5Span_short_mem_minimaFlag (b : Basis (Fin 5) ℝ (Fin 5 → ℝ))
    (e : (Fin 5 → ℝ) ≃ₗ[ℝ] (Fin 5 → ℝ)) (v : Fin 5 → (Fin 5 → ℝ))
    (hv : LinearIndependent ℝ v) (hvL : ∀ i, v i ∈ lattice5Span b)
    (hvr : ∀ i, coordinateGauge5 e (v i) = successiveRadius5 b e i)
    (k : Fin 5) (hgap : ∀ i : Fin 5, i < k →
      successiveRadius5 b e i < successiveRadius5 b e k)
    (x : (Fin 5 → ℝ)) (hx : x ∈ lattice5Span b)
    (hxr : coordinateGauge5 e x < successiveRadius5 b e k) :
    x ∈ minimaFlag5 v k := by
  classical
  by_contra hxW
  let u : Fin k.val → (Fin 5 → ℝ) := fun i => v (Fin.castLE (by omega) i)
  have hu : LinearIndependent ℝ u := hv.comp _ (Fin.castLE_injective _)
  let w : Fin (k.val + 1) → (Fin 5 → ℝ) := Fin.snoc u x
  have hux : LinearIndependent ℝ w :=
    linearIndependent_fin_snoc.mpr ⟨hu, hxW⟩
  have hL : ∀ i : Fin (k.val + 1), w i ∈ lattice5Span b := by
    intro i
    refine Fin.lastCases ?_ (fun j => ?_) i
    · simpa [w] using hx
    · simpa [w, u] using hvL (Fin.castLE (by omega) j)
  have hlt : ∀ i : Fin (k.val + 1), coordinateGauge5 e (w i) < successiveRadius5 b e k := by
    intro i
    refine Fin.lastCases ?_ (fun j => ?_) i
    · simpa [w] using hxr
    · simpa [w, u, hvr] using hgap (Fin.castLE (by omega) j) (by exact j.isLt)
  obtain ⟨i, _, hi⟩ := Finset.exists_max_image Finset.univ
    (fun i => coordinateGauge5 e (w i)) (Finset.univ_nonempty)
  have hle := successiveRadius5_le_family b e k w hux hL _
    (fun j => hi j (Finset.mem_univ j))
  exact (not_lt_of_ge hle) (hlt i)
def allpoint_rational_functional5 (f : Module.Dual ℚ (Fin 5 → ℚ)) : Module.Dual ℝ (Fin 5 → ℝ) :=
  ∑ i : Fin 5, (f (Pi.single i 1) : ℝ) • LinearMap.proj i

theorem allpoint_rational_functional5_cast (f : Module.Dual ℚ (Fin 5 → ℚ)) (x : Fin 5 → ℚ) :
    allpoint_rational_functional5 f (fun i => (x i : ℝ)) = (f x : ℝ) := by
  rw [OriginalDual10.functional_coordinates_dual10 f x]
  simp only [allpoint_rational_functional5,LinearMap.sum_apply,LinearMap.smul_apply,
    LinearMap.proj_apply,smul_eq_mul,Rat.cast_sum,Rat.cast_mul]
  apply Finset.sum_congr rfl
  intro i _
  ring


theorem allpoint_rational_flag5 (w : Fin 5 → Fin 5 → ℤ)
    (hw : LinearIndependent ℝ (fun j => integerReal (w j))) (j : Fin 4) (z : Fin 5 → ℤ)
    (hz : integerReal z ∈ minimaFlag5 (fun j => integerReal (w j)) j.succ) :
    (fun i => (z i : ℚ)) ∈ originalFirstFlag5 j w := by
  classical
  let v : Fin 5 → Fin 5 → ℚ := fun j i => (w j i : ℚ)
  have hQ : LinearIndependent ℚ v := SourceFlag241.original_rational_independent241 w hw
  let b := rationalFrameBasis42 v hQ
  have hb (i : Fin 5) : b i=v i := rationalFrameBasis42_apply v hQ i
  have hzero (a : Fin 5) (hj : j.succ≤a) : b.repr (fun i => (z i : ℚ)) a=0 := by
    let f := rationalFrameDual42 v hQ a
    have hk : minimaFlag5 (fun i => integerReal (w i)) j.succ ≤
        LinearMap.ker (allpoint_rational_functional5 f) := by
      apply Submodule.span_le.mpr
      rintro _ ⟨i,rfl⟩
      have hij : (Fin.castLE (by omega) i : Fin 5)≠a := by
        intro he
        have hh := congrArg Fin.val he
        change i.val=a.val at hh
        have hi := i.isLt
        change j.succ.val≤a.val at hj
        omega
      change allpoint_rational_functional5 f (integerReal (w (Fin.castLE (by omega) i)))=0
      have he := allpoint_rational_functional5_cast f (v (Fin.castLE (by omega) i))
      change allpoint_rational_functional5 f (integerReal (w (Fin.castLE (by omega) i)))=_ at he
      rw [he]
      simp [f,rationalFrameDual42_pairing,hij]
    have hh := hk hz
    change allpoint_rational_functional5 f (integerReal z)=0 at hh
    have he := allpoint_rational_functional5_cast f (fun i => (z i : ℚ))
    change allpoint_rational_functional5 f (integerReal z)=_ at he
    rw [he] at hh
    have hf : f (fun i => (z i : ℚ))=0 := by exact_mod_cast hh
    rw [← Basis.dualBasis_apply]
    exact hf
  rw [← b.sum_repr (fun i => (z i : ℚ))]
  apply Submodule.sum_mem
  intro i _
  by_cases hi : i.val<j.val+1
  · apply Submodule.smul_mem
    rw [hb]
    apply Submodule.subset_span
    exact ⟨⟨i.val,hi⟩,rfl⟩
  · rw [hzero i (show j.succ ≤ i from by simpa only [Fin.le_def,Fin.val_succ] using (Nat.le_of_not_gt hi)),zero_smul]
    exact Submodule.zero_mem _

theorem allpoint_source_short_rational5 (A2 A3 : Matrix (Fin 5) (Fin 5) ℚ)
    (L : Matrix (Fin 5) (Fin 5) ℝ) (hL : IsUnit L) (Q : ℝ) (hQ : 1<Q)
    (c0 c2 c3 : Fin 5 → ℝ) (j : Fin 4)
    (u : Fin 5 → originalSourceGroup5 A2 A3 Q c2 c3)
    (hu : (LinearIndependent ℝ (fun i => integerReal (u i : Fin 5 → ℤ)) ∧ ∀ i,coordinateGauge5 (originalCoordinates5 L hL Q hQ c0) (integerReal (u i : Fin 5 → ℤ))=sourceMinima5 A2 A3 L hL Q hQ c0 c2 c3 i))
    (hgap : ∀ i : Fin 5,i<j.succ → sourceMinima5 A2 A3 L hL Q hQ c0 c2 c3 i<
      sourceMinima5 A2 A3 L hL Q hQ c0 c2 c3 j.succ)
    (z : originalSourceGroup5 A2 A3 Q c2 c3)
    (hz : coordinateGauge5 (originalCoordinates5 L hL Q hQ c0) (integerReal (z : Fin 5 → ℤ))<
      sourceMinima5 A2 A3 L hL Q hQ c0 c2 c3 j.succ) :
    (fun i => ((z : Fin 5 → ℤ) i : ℚ)) ∈ originalFirstFlag5 j (fun j => (u j : Fin 5 → ℤ)) := by
  let b := sourceRealBasis5 A2 A3
    (fun i => Nat.ceil (qThreshold 2 Q (c2 i))) (fun i => Nat.ceil (qThreshold 3 Q (c3 i)))
  let e := originalCoordinates5 L hL Q hQ c0
  have hr (j : Fin 5) : sourceMinima5 A2 A3 L hL Q hQ c0 c2 c3 j=successiveRadius5 b e j :=
    sourceMinima5_eq_radius A2 A3 L hL Q hQ c0 c2 c3 j
  have hmem (x : originalSourceGroup5 A2 A3 Q c2 c3) : integerReal (x : Fin 5 → ℤ)∈lattice5Span b := by
    rw [sourceRealBasis5_span]
    exact ⟨(x : Fin 5 → ℤ),x.property,rfl⟩
  apply allpoint_rational_flag5 _ hu.1 j _
  apply lattice5Span_short_mem_minimaFlag b e _ hu.1 (fun i => hmem (u i))
    (fun i => (hu.2 i).trans (hr i)) j.succ
  · intro i hi
    rw [← hr i,← hr j.succ]
    exact hgap i hi
  · exact hmem z
  · rw [← hr j.succ]
    exact hz


theorem allpoint_rank_scalar5 (a b c d e g t q C : ℝ)
    (ha : 0<a) (hb : 0<b) (hc : 0<c) (hd : 0<d) (he : 0<e)
    (ht : 0<t) (hq : 1≤q) (hab : a≤b) (hbc : b≤c) (hcd : c≤d) (hde : d≤e)
    (hag : a≤g) (hgt : g≤t) (hp : C≤a*b*c*d*e) (hcut : t^5*q^20<C) :
    (d*q<e ∧ g<e) ∨ (c*q<d ∧ g<d) ∨ (b*q<c ∧ g<c) ∨ (a*q<b ∧ g<b) := by
  have hq0 : 0<q := lt_of_lt_of_le zero_lt_one hq
  have hbig : t*q^4<e := by
    by_contra hh
    have hele : e≤t*q^4 := le_of_not_gt hh
    have hp1 : a*b*c*d*e≤e^5 := by
      have h1 := mul_le_mul (hab.trans (hbc.trans (hcd.trans hde))) (hbc.trans (hcd.trans hde)) hb.le he.le
      have h2 := mul_le_mul (hcd.trans hde) hde hd.le he.le
      have h3 := mul_le_mul h1 h2 (mul_pos hc hd).le (mul_nonneg he.le he.le)
      have h4 := mul_le_mul_of_nonneg_right h3 he.le
      nlinarith only [h4]
    have hpow := pow_le_pow_left₀ he.le hele 5
    have hh : (t*q^4)^5=t^5*q^20 := by ring
    rw [hh] at hpow
    exact (not_lt_of_ge (hp.trans (hp1.trans hpow))) hcut
  have hq2 : 1≤q^2 := by nlinarith
  have hq3 : 1≤q^3 := by nlinarith [mul_le_mul_of_nonneg_right hq2 hq0.le]
  have hq4 : 1≤q^4 := by nlinarith [sq_nonneg (q^2-1)]
  have hge : g<e := hgt.trans_lt ((by nlinarith : t≤t*q^4).trans_lt hbig)
  by_cases h4 : d*q<e
  · exact Or.inl ⟨h4,hge⟩
  have htd : t*q^3<d := by
    have hh := hbig.trans_le (le_of_not_gt h4)
    have heq : t*q^4=(t*q^3)*q := by ring
    rw [heq] at hh
    exact (mul_lt_mul_right hq0).mp hh
  have hgd : g<d := hgt.trans_lt ((by nlinarith : t≤t*q^3).trans_lt htd)
  by_cases h3 : c*q<d
  · exact Or.inr (Or.inl ⟨h3,hgd⟩)
  have htc : t*q^2<c := by
    have hh := htd.trans_le (le_of_not_gt h3)
    have heq : t*q^3=(t*q^2)*q := by ring
    rw [heq] at hh
    exact (mul_lt_mul_right hq0).mp hh
  have hgc : g<c := hgt.trans_lt ((by nlinarith : t≤t*q^2).trans_lt htc)
  by_cases h2 : b*q<c
  · exact Or.inr (Or.inr (Or.inl ⟨h2,hgc⟩))
  have htb : t*q<b := by
    have hh := htc.trans_le (le_of_not_gt h2)
    have heq : t*q^2=(t*q)*q := by ring
    rw [heq] at hh
    exact (mul_lt_mul_right hq0).mp hh
  have hgb : g<b := hgt.trans_lt ((by nlinarith : t≤t*q).trans_lt htb)
  have haq : a*q<b := (mul_le_mul_of_nonneg_right (hag.trans hgt) hq0.le).trans_lt htb
  exact Or.inr (Or.inr (Or.inr ⟨haq,hgb⟩))
