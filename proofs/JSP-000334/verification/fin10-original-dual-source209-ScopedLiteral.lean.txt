theorem scalar_nonzero_iff (K : Type*) [Field K] (d x y : K)
    (hd : d ≠ 0) (hxy : x = d*y) : x ≠ 0 ↔ y ≠ 0 := by
  subst x
  constructor
  · intro h
    exact (mul_ne_zero_iff.mp h).2
  · intro hy
    exact mul_ne_zero hd hy

theorem linear_row_nonzero_iff (K : Type*) [Field K] (I : Type*)
    (d : K) (x y : I → K) (F : (I → K) →ₗ[K] K)
    (hd : d ≠ 0) (hxy : ∀ i, x i = d*y i) : F x ≠ 0 ↔ F y ≠ 0 := by
  have hv : x = d • y := funext hxy
  have he : F x = d*F y := by rw [hv,map_smul,smul_eq_mul]
  exact scalar_nonzero_iff K d (F x) (F y) hd he

end CompoundFeasibility.M209LiteralDual10

def rowLinear (K : Type*) [Field K] (I : Type*) [Fintype I] (r : I → K) :
    (I → K) →ₗ[K] K where
  toFun x := ∑ i, r i*x i
  map_add' x y := by simp only [Pi.add_apply,mul_add,Finset.sum_add_distrib]
  map_smul' c x := by
    simp only [Pi.smul_apply,smul_eq_mul,Finset.mul_sum,RingHom.id_apply]
    apply Finset.sum_congr rfl
    intro i _
    ring

theorem matrix_row_linear_apply (K : Type*) [Field K] (I : Type*) [Fintype I]
    (B : Matrix I I K) (i : I) (x : I → K) :
    rowLinear K I (fun k => B k i) x = (B.transpose *ᵥ x) i := rfl

def originalRationalDualProduct10 (A2 A3 : Matrix (Fin 10) (Fin 10) ℚ)
    (L : Matrix (Fin 10) (Fin 10) ℝ) (u : Fin 10 → Fin 10 → ℤ)
    (hu : LinearIndependent ℚ (fun j i => (u j i : ℚ))) (s : MixedIndex10) : ℝ :=
  |(L⁻¹.transpose *ᵥ (fun k => (lastDual10 (fun j i => (u j i : ℚ)) hu k : ℝ))) (s 0)| *
    ‖((A2.map (algebraMap ℚ ℚ_[2]))⁻¹.transpose *ᵥ
      (fun k => (lastDual10 (fun j i => (u j i : ℚ)) hu k : ℚ_[2]))) (s 1)‖ *
    ‖((A3.map (algebraMap ℚ ℚ_[3]))⁻¹.transpose *ᵥ
      (fun k => (lastDual10 (fun j i => (u j i : ℚ)) hu k : ℚ_[3]))) (s 2)‖

theorem original_rational_product_nonzero_iff (A2 A3 : Matrix (Fin 10) (Fin 10) ℚ)
    (L : Matrix (Fin 10) (Fin 10) ℝ) (u : Fin 10 → Fin 10 → ℤ)
    (hu : LinearIndependent ℚ (fun j i => (u j i : ℚ))) (s : MixedIndex10) :
    originalRationalDualProduct10 A2 A3 L u hu s ≠ 0 ↔
      mixedProduct10 A2 A3 L (integerLastDual10 u) s ≠ 0 := by
  have hp : originalRationalDualProduct10 A2 A3 L u hu s ≠ 0 ↔
      (L⁻¹.transpose *ᵥ (fun k => (lastDual10 (fun j i => (u j i : ℚ)) hu k : ℝ))) (s 0) ≠ 0 ∧
      ((A2.map (algebraMap ℚ ℚ_[2]))⁻¹.transpose *ᵥ
        (fun k => (lastDual10 (fun j i => (u j i : ℚ)) hu k : ℚ_[2]))) (s 1) ≠ 0 ∧
      ((A3.map (algebraMap ℚ ℚ_[3]))⁻¹.transpose *ᵥ
        (fun k => (lastDual10 (fun j i => (u j i : ℚ)) hu k : ℚ_[3]))) (s 2) ≠ 0 := by
    simp only [originalRationalDualProduct10,mul_ne_zero_iff,abs_ne_zero,norm_ne_zero_iff,and_assoc]
  constructor
  · intro h
    obtain ⟨h0,h2,h3⟩ := hp.mp h
    exact (mixedProduct10_ne_zero_iff A2 A3 L (integerLastDual10 u) s).mpr
      ⟨(original_real_nonzero10 L u hu (s 0)).mpr h0,
       (original_p2_nonzero10 A2 u hu (s 1)).mpr h2,
       (original_p3_nonzero10 A3 u hu (s 2)).mpr h3⟩
  · intro h
    obtain ⟨h0,h2,h3⟩ := (mixedProduct10_ne_zero_iff A2 A3 L (integerLastDual10 u) s).mp h
    exact hp.mpr ⟨(original_real_nonzero10 L u hu (s 0)).mp h0,
      (original_p2_nonzero10 A2 u hu (s 1)).mp h2,
      (original_p3_nonzero10 A3 u hu (s 2)).mp h3⟩

  apply LinearIndependent.of_comp f
  convert hu.restrict_scalars' ℚ using 1

theorem rational_failure_to_cofactor_failure10
    {A2 A3 : Matrix (Fin 10) (Fin 10) ℚ} {L : Matrix (Fin 10) (Fin 10) ℝ}
    {hL : IsUnit L} {delta : ℝ} {S0 : Finset MixedIndex10}
    (U : Set ℝ) (hQU : ∀ q ∈ U, 1 < q)
    (D : (q : U) → OriginalData10 A2 A3 L hL q.val (hQU q.val q.property) delta S0)
    (hfail : ¬ ∃ B : ℝ, ∀ q : U, B < q.val → ∀ s ∈ S0, dataRationalDualProduct10 (D q) s = 0) :
    ¬ ∃ B : ℝ, ∀ q : U, B < q.val → ∀ s ∈ S0,
      mixedProduct10 A2 A3 L (integerLastDual10 (D q).u) s = 0 := by
  rintro ⟨B,hzero⟩
  apply hfail
  refine ⟨B,?_⟩
  intro q hq s hs
  by_contra hn
  exact (data_rational_product_nonzero_iff (D q) s).mp hn (hzero q hq s hs)

theorem rational_counterexamples_unbounded10
    (A2 A3 : Matrix (Fin 10) (Fin 10) ℚ) (L : Matrix (Fin 10) (Fin 10) ℝ)
    (hL : IsUnit L) (delta : ℝ) (S0 : Finset MixedIndex10)
    (hfail : ∀ B : ℝ, ∃ Q : ℝ, ∃ hQ : 1 < Q, B < Q ∧
      ∃ D : OriginalData10 A2 A3 L hL Q hQ delta S0, ∃ s ∈ S0, dataRationalDualProduct10 D s ≠ 0) :
    ∃ U : Set ℝ, ¬ BddAbove U ∧ ∃ hQU : ∀ q ∈ U, 1 < q,
      ∃ D : (q : U) → OriginalData10 A2 A3 L hL q.val (hQU q.val q.property) delta S0,
        (∀ q : U, ∃ s ∈ S0, dataRationalDualProduct10 (D q) s ≠ 0) ∧
        ¬ ∃ B : ℝ, ∀ q : U, B < q.val → ∀ s ∈ S0, dataRationalDualProduct10 (D q) s = 0 := by
  have hcof : ∀ B : ℝ, ∃ Q : ℝ, ∃ hQ : 1 < Q, B < Q ∧
      ∃ D : OriginalData10 A2 A3 L hL Q hQ delta S0, ∃ s ∈ S0,
        mixedProduct10 A2 A3 L (integerLastDual10 D.u) s ≠ 0 := by
    intro B
    obtain ⟨Q,hQ,hBQ,D,s,hs,hn⟩ := hfail B
    exact ⟨Q,hQ,hBQ,D,s,hs,(data_rational_product_nonzero_iff D s).mp hn⟩
  obtain ⟨U,hU,hQU,D,hbad,_hnotzero⟩ := original_counterexamples_unbounded10 A2 A3 L hL delta S0 hcof
  have hrat (q : U) : ∃ s ∈ S0, dataRationalDualProduct10 (D q) s ≠ 0 := by
    obtain ⟨s,hs,hn⟩ := hbad q
    exact ⟨s,hs,(data_rational_product_nonzero_iff (D q) s).mpr hn⟩
  refine ⟨U,hU,hQU,D,hrat,?_⟩
  rintro ⟨B,hzero⟩
  obtain ⟨q,hq,hBq⟩ := not_bddAbove_iff.mp hU B
  obtain ⟨s,hs,hn⟩ := hrat ⟨q,hq⟩
  exact hn (hzero ⟨q,hq⟩ hBq s hs)

end

theorem rational_family_schmidt_inputs10
    (A2 A3 : Matrix (Fin 10) (Fin 10) ℚ) (L : Matrix (Fin 10) (Fin 10) ℝ)
