def Raw_PrimitiveFour (z : Fin 4 → ℤ) : Prop := ∃ c : Fin 4 → ℤ,∑ i,c i*z i=1

def Raw_fourRetainedForm (k i : Fin 4) (y : Fin 4 → ℚ) : ℚ := if i=k then ∑ j,y j else y i

def Raw_localFourHeight (tau : Fin 3) (y : Fin 4 → ℚ) : ℚ :=
  max (rationalPlaceNorm tau (y 0)) (max (rationalPlaceNorm tau (y 1))
    (max (rationalPlaceNorm tau (y 2)) (rationalPlaceNorm tau (y 3))))

def Raw_fourHeight (y : Fin 4 → ℚ) : ℚ := ∏ tau : Fin 3,Raw_localFourHeight tau y

def Raw_archFourHeight (z : Fin 4 → ℤ) : ℚ :=
  max |(z 0 : ℚ)| (max |(z 1 : ℚ)| (max |(z 2 : ℚ)| |(z 3 : ℚ)|))

def Raw_fourRetainedProduct (choice : Fin 3 → Fin 4) (y : Fin 4 → ℚ) : ℚ :=

    (∑ tau : Fin 3, ∑ i : Fin 4, c tau i) = 0

  else ⌈(n : ℝ)*((1-4/(n : ℝ))*c tau i)⌉

noncomputable def Raw_finiteWeightRaw4 (n : ℕ) (z : Fin 3 → Fin 4 → ℤ) (tau : Fin 3) (i : Fin 4) : ℝ :=
  (z tau i : ℝ)/(n : ℝ)

noncomputable def Raw_finiteWeightCandidate4 (n : ℕ) (z : Fin 3 → Fin 4 → ℤ) (tau : Fin 3) (i : Fin 4) : ℝ :=
  if tau = 0 then Raw_finiteWeightRaw4 n z tau i - (∑ t : Fin 3, ∑ j : Fin 4, Raw_finiteWeightRaw4 n z t j)/4
  else Raw_finiteWeightRaw4 n z tau i

noncomputable def Raw_finiteWeightApprox4 (n : ℕ) (c : Fin 3 → Fin 4 → ℝ) : Fin 3 → Fin 4 → ℝ :=
  Raw_finiteWeightCandidate4 n (Raw_finiteWeightIndex4 n c)
