def rationalRealSpan5 (P : Submodule ℚ (Fin 5 → ℚ)) : Submodule ℝ (Fin 5 → ℝ) :=
  Submodule.span ℝ ((fun x : Fin 5 → ℚ => fun i => (x i : ℝ)) '' (P : Set (Fin 5 → ℚ)))

theorem rational_real_independent5 (d : Fin 5) (u : Fin d.val → Fin 5 → ℚ)
    (hu : LinearIndependent ℚ u) : LinearIndependent ℝ (fun j i => (u j i : ℝ)) := by
  classical
  let f := Fintype.linearCombination ℚ u
  obtain ⟨g,hg⟩ := f.exists_leftInverse_of_injective
    (LinearMap.ker_eq_bot.mpr hu.fintypeLinearCombination_injective)
  let U : Matrix (Fin 5) (Fin d.val) ℚ := fun i j => u j i
  let G := LinearMap.toMatrix' g
  have hfmat : LinearMap.toMatrix' f = U := by
    ext i j
    simp [f,U,LinearMap.toMatrix'_apply,Fintype.linearCombination_apply,Finset.sum_apply,Pi.smul_apply,smul_eq_mul,ite_apply]
  have hGU : G * U = 1 := by
    dsimp only [G]
    rw [← hfmat,← LinearMap.toMatrix'_comp,hg,LinearMap.toMatrix'_id]
  let UR := U.map (algebraMap ℚ ℝ)
  let GR := G.map (algebraMap ℚ ℝ)
  have hGR : GR * UR = 1 := by
    dsimp only [GR,UR]
    rw [← Matrix.map_mul,hGU,Matrix.map_one] <;> simp
  have hinj : Function.Injective (Matrix.toLin' UR) := by
    apply Function.LeftInverse.injective (g := Matrix.toLin' GR)
    intro x
    change GR *ᵥ (UR *ᵥ x) = x
    rw [Matrix.mulVec_mulVec,hGR,Matrix.one_mulVec]
  have hli := (Pi.basisFun ℝ (Fin d.val)).linearIndependent.map'
    (Matrix.toLin' UR) (LinearMap.ker_eq_bot.mpr hinj)
  convert hli using 1
  funext j i
  simp [UR,U,Pi.basisFun_apply,Matrix.toLin'_apply,Matrix.mulVec,Pi.single_apply]


theorem rationalRealSpan5_basis_span (P : Submodule ℚ (Fin 5 → ℚ)) (d : Fin 5)
    (b : Basis (Fin d.val) ℚ P) :
    Submodule.span ℝ (Set.range (fun j i => ((b j : Fin 5 → ℚ) i : ℝ))) = rationalRealSpan5 P := by
  let W := Submodule.span ℝ (Set.range (fun j i => ((b j : Fin 5 → ℚ) i : ℝ)))
  apply le_antisymm
  · apply Submodule.span_le.mpr
    rintro _ ⟨j,rfl⟩
    apply Submodule.subset_span
    exact ⟨(b j : Fin 5 → ℚ),(b j).property,rfl⟩
  · apply Submodule.span_le.mpr
    rintro _ ⟨x,hx,rfl⟩
    let y : P := ⟨x,hx⟩
    have heq : (fun i => (x i : ℝ)) =
        ∑ j : Fin d.val, (b.repr y j : ℝ) • (fun i => ((b j : Fin 5 → ℚ) i : ℝ)) := by
      have hQ := congrArg (fun z : P => (z : Fin 5 → ℚ)) (b.sum_repr y)
      ext i
      have hi := congrFun hQ i
      simp only [Submodule.coe_sum,Submodule.coe_smul,Finset.sum_apply,Pi.smul_apply,
        smul_eq_mul] at hi
      change ∑ j, b.repr y j * (b j : Fin 5 → ℚ) i = x i at hi
      simp only [Finset.sum_apply,Pi.smul_apply,smul_eq_mul]
      exact_mod_cast hi.symm
    change (fun i => (x i : ℝ)) ∈ W
    rw [heq]
    exact Submodule.sum_mem W (fun j _ => W.smul_mem _ (Submodule.subset_span ⟨j,rfl⟩))

theorem rationalRealSpan5_finrank (P : Submodule ℚ (Fin 5 → ℚ))
    (hP : Module.finrank ℚ P ≤ 4) :
    Module.finrank ℝ (rationalRealSpan5 P) = Module.finrank ℚ P := by
  let d : Fin 5 := ⟨Module.finrank ℚ P,by omega⟩
  let b : Basis (Fin d.val) ℚ P := Module.finBasis ℚ P
  have hQ : LinearIndependent ℚ (fun j => (b j : Fin 5 → ℚ)) :=
    b.linearIndependent.map' P.subtype P.ker_subtype
  have hR := rational_real_independent5 d (fun j => (b j : Fin 5 → ℚ)) hQ
  rw [← rationalRealSpan5_basis_span P d b,finrank_span_eq_card hR,Fintype.card_fin]


def restrictedLattice5 (P : Submodule ℚ (Fin 5 → ℚ))
    (A₂ A₃ : Matrix (Fin 5) (Fin 5) ℚ) (Q : ℝ) (c₂ c₃ : Fin 5 → ℝ) :
    Submodule ℤ (rationalRealSpan5 P) :=
  (integerRealImage (originalSourceGroup5 A₂ A₃ Q c₂ c₃)).comap
    ((rationalRealSpan5 P).subtype.restrictScalars ℤ)

theorem rational_source_multiple5 (A₂ A₃ : Matrix (Fin 5) (Fin 5) ℚ)
    (Q : ℝ) (c₂ c₃ : Fin 5 → ℝ) (x : Fin 5 → ℚ) :
    ∃ m : ℤ, m ≠ 0 ∧ ∃ w : Fin 5 → ℤ,
      w ∈ originalSourceGroup5 A₂ A₃ Q c₂ c₃ ∧
      integerReal w = (m : ℝ) • (fun i => (x i : ℝ)) := by
  classical
  obtain ⟨D,hD,hm⟩ := clear_matrix_denominators (fun _ i : Fin 5 => x i)
  choose z hz using hm (0 : Fin 5)
  obtain ⟨E,hE,hEmem⟩ := sourceLattice5_multiple A₂ A₃
    (fun i => Nat.ceil (JSP000334.qThreshold 2 Q (c₂ i)))
    (fun i => Nat.ceil (JSP000334.qThreshold 3 Q (c₃ i)))
  let G := originalSourceGroup5 A₂ A₃ Q c₂ c₃
  have hwm : E • z ∈ G := by
    have hrepr : E • z = ∑ j : Fin 5, z j • Pi.single j E := by
      ext i
      simp [Pi.smul_apply,smul_eq_mul,Finset.sum_apply,Pi.single_apply,mul_comm]
    rw [hrepr]
    exact G.sum_mem (fun j _ => G.zsmul_mem (hEmem j) (z j))
  refine ⟨E*D,mul_ne_zero hE hD,E • z,hwm,?_⟩
  ext i
  change ((E * z i : ℤ) : ℝ) = ((E*D : ℤ) : ℝ) * (x i : ℝ)
  have hzr : (z i : ℝ) = (D : ℝ) * (x i : ℝ) := by exact_mod_cast hz i
  push_cast
  rw [hzr]
  ring

instance restrictedLattice5_discrete (P : Submodule ℚ (Fin 5 → ℚ))
    (A₂ A₃ : Matrix (Fin 5) (Fin 5) ℚ) (Q : ℝ) (c₂ c₃ : Fin 5 → ℝ) :
    DiscreteTopology (restrictedLattice5 P A₂ A₃ Q c₂ c₃) := by
  let f : restrictedLattice5 P A₂ A₃ Q c₂ c₃ →
      integerRealImage (originalSourceGroup5 A₂ A₃ Q c₂ c₃) :=
    fun x => ⟨x.val.val,x.property⟩
  apply DiscreteTopology.of_continuous_injective (f := f)
  · exact (continuous_subtype_val.comp continuous_subtype_val).subtype_mk _
  · intro x y h
    apply Subtype.ext
    apply Subtype.ext
    exact congrArg (fun z : integerRealImage (originalSourceGroup5 A₂ A₃ Q c₂ c₃) =>
      (z : Fin 5 → ℝ)) h

instance restrictedLattice5_isZLattice (P : Submodule ℚ (Fin 5 → ℚ))
    (A₂ A₃ : Matrix (Fin 5) (Fin 5) ℚ) (Q : ℝ) (c₂ c₃ : Fin 5 → ℝ) :
    IsZLattice ℝ (restrictedLattice5 P A₂ A₃ Q c₂ c₃) where
  span_top := by
    let S := rationalRealSpan5 P
    let K := Submodule.span ℝ (restrictedLattice5 P A₂ A₃ Q c₂ c₃ : Set S)
    have hSK : S ≤ K.map S.subtype := by
      apply Submodule.span_le.mpr
      rintro _ ⟨x,hx,rfl⟩
      obtain ⟨m,hm,w,hw,heq⟩ := rational_source_multiple5 A₂ A₃ Q c₂ c₃ x
      have hxS : (fun i => (x i : ℝ)) ∈ S := Submodule.subset_span ⟨x,hx,rfl⟩
      let y : S := ⟨integerReal w,heq ▸ S.smul_mem (m : ℝ) hxS⟩
      have hy : y ∈ restrictedLattice5 P A₂ A₃ Q c₂ c₃ := by
        change integerReal w ∈ integerRealImage (originalSourceGroup5 A₂ A₃ Q c₂ c₃)
        exact ⟨w,hw,rfl⟩
      have hmem : (m : ℝ) • (fun i => (x i : ℝ)) ∈ K.map S.subtype := by
        rw [← heq]
        exact ⟨y,Submodule.subset_span hy,rfl⟩
      exact (Submodule.smul_mem_iff (K.map S.subtype) (by exact_mod_cast hm : (m : ℝ) ≠ 0)).mp hmem
    apply Submodule.map_injective_of_injective S.subtype_injective
    apply le_antisymm
    · exact Submodule.map_mono le_top
    · simpa only [Submodule.map_top,Submodule.range_subtype] using hSK

theorem restrictedBody5_gauge (P : Submodule ℚ (Fin 5 → ℚ))
    (L : Matrix (Fin 5) (Fin 5) ℝ) (hL : IsUnit L)
    (Q : ℝ) (hQ : 1 < Q) (c₀ : Fin 5 → ℝ) :
    restrictedBody5 P L Q c₀ = {x | restrictedGauge5 P L hL Q hQ c₀ x ≤ 1} := by

theorem restrictedBody5_compact_convex (P : Submodule ℚ (Fin 5 → ℚ))
    (L : Matrix (Fin 5) (Fin 5) ℝ) (hL : IsUnit L)
    (Q : ℝ) (hQ : 1 < Q) (c₀ : Fin 5 → ℝ) :
    IsCompact (restrictedBody5 P L Q c₀) ∧ Convex ℝ (restrictedBody5 P L Q c₀) := by
  let S := rationalRealSpan5 P
  let e := originalCoordinates5 L hL Q hQ c₀
  have hbody : originalBody5 L Q c₀ = e ⁻¹' Metric.closedBall (0 : Fin 5 → ℝ) 1 := by
    ext x
    rw [originalBody5_mem_iff L hL Q hQ c₀]
    simp only [Set.mem_preimage,Metric.mem_closedBall,dist_zero_right]
    rfl
  have hc : IsCompact (originalBody5 L Q c₀) := by
    rw [hbody]
    exact e.toContinuousLinearEquiv.toHomeomorph.isCompact_preimage.mpr (isCompact_closedBall _ _)
  have hv : Convex ℝ (originalBody5 L Q c₀) := by
    rw [hbody]
    exact (convex_closedBall (0 : Fin 5 → ℝ) 1).linear_preimage e.toLinearMap
  constructor
  · exact S.closed_of_finiteDimensional.isClosedEmbedding_subtypeVal.isProperMap.isCompact_preimage hc
  · exact hv.linear_preimage S.subtype


theorem restrictedBody5_symmetric (P : Submodule ℚ (Fin 5 → ℚ))
    (L : Matrix (Fin 5) (Fin 5) ℝ) (hL : IsUnit L)
    (Q : ℝ) (hQ : 1 < Q) (c₀ : Fin 5 → ℝ) :
    ∀ x ∈ restrictedBody5 P L Q c₀, -x ∈ restrictedBody5 P L Q c₀ := by
  rw [restrictedBody5_gauge P L hL Q hQ c₀]
  intro x hx
  simpa [restrictedGauge5,coordinateGauge5,map_neg] using hx

theorem restrictedBody5_zero_interior (P : Submodule ℚ (Fin 5 → ℚ))
    (L : Matrix (Fin 5) (Fin 5) ℝ) (hL : IsUnit L)
    (Q : ℝ) (hQ : 1 < Q) (c₀ : Fin 5 → ℝ) :
    (0 : rationalRealSpan5 P) ∈ interior (restrictedBody5 P L Q c₀) := by
  rw [restrictedBody5_gauge P L hL Q hQ c₀]
  let e := originalCoordinates5 L hL Q hQ c₀
  have hg : Continuous (restrictedGauge5 P L hL Q hQ c₀) :=
    (e.toContinuousLinearEquiv.continuous.comp continuous_subtype_val).norm
  apply mem_interior_iff_mem_nhds.mpr
  have hopen := (isOpen_lt hg continuous_const).mem_nhds
    (show restrictedGauge5 P L hL Q hQ c₀ 0 < 1 by simp [restrictedGauge5,coordinateGauge5])
  exact Filter.mem_of_superset hopen (fun x hx =>
    (show restrictedGauge5 P L hL Q hQ c₀ x < 1 from hx).le)

theorem source_restricted_prefix_eq5 (P : Submodule ℚ (Fin 5 → ℚ))
    (hP : Module.finrank ℚ P ≤ 4)
    (A₂ A₃ : Matrix (Fin 5) (Fin 5) ℚ) (L : Matrix (Fin 5) (Fin 5) ℝ) (hL : IsUnit L)
    (Q : ℝ) (hQ : 1 < Q) (c₀ c₂ c₃ : Fin 5 → ℝ)
    (u : Fin (Module.finrank ℚ P) → originalSourceGroup5 A₂ A₃ Q c₂ c₃)
    (hu : LinearIndependent ℝ (fun i => integerReal (u i : Fin 5 → ℤ)))
    (huS : ∀ i, integerReal (u i : Fin 5 → ℤ) ∈ rationalRealSpan5 P)
    (hatt : ∀ i, coordinateGauge5 (originalCoordinates5 L hL Q hQ c₀)
      (integerReal (u i : Fin 5 → ℤ)) =
      sourceMinima5 A₂ A₃ L hL Q hQ c₀ c₂ c₃ (ambientPrefixIndex5 P hP i)) :
    ∀ k, restrictedMinima5 P A₂ A₃ L hL Q hQ c₀ c₂ c₃ k =
      sourceMinima5 A₂ A₃ L hL Q hQ c₀ c₂ c₃ (ambientPrefixIndex5 P hP k) := by
  let S := rationalRealSpan5 P
  let v : Fin (Module.finrank ℚ P) → S := fun i => ⟨integerReal (u i : Fin 5 → ℤ),huS i⟩
  have hv : LinearIndependent ℝ v := LinearIndependent.of_comp S.subtype hu
  intro k
  let emb : Fin (k.val+1) → Fin (Module.finrank ℚ P) := Fin.castLE (by omega)
  let w := v ∘ emb
  have hw : sourceMinima5 A₂ A₃ L hL Q hQ c₀ c₂ c₃ (ambientPrefixIndex5 P hP k) ∈
      restrictedRadii5 P A₂ A₃ L hL Q hQ c₀ c₂ c₃ k := by
    refine ⟨w,hv.comp emb (Fin.castLE_injective _),?_,?_⟩
    · intro i
      change integerReal (u (emb i) : Fin 5 → ℤ) ∈
        integerRealImage (originalSourceGroup5 A₂ A₃ Q c₂ c₃)
      exact ⟨(u (emb i) : Fin 5 → ℤ),(u (emb i)).property,rfl⟩
    · intro i
      change coordinateGauge5 (originalCoordinates5 L hL Q hQ c₀)
        (integerReal (u (emb i) : Fin 5 → ℤ)) ≤ _
      rw [hatt]
      exact (sourceMinima5_pos_mono A₂ A₃ L hL Q hQ c₀ c₂ c₃).2
        (show ambientPrefixIndex5 P hP (emb i) ≤ ambientPrefixIndex5 P hP k by
          change i.val ≤ k.val; omega)
  apply le_antisymm
  · exact csInf_le (restrictedRadii5_bddBelow P A₂ A₃ L hL Q hQ c₀ c₂ c₃ k) hw
  · apply le_csInf ⟨_,hw⟩
    rintro r ⟨a,ha,haL,har⟩
    apply csInf_le (sourceMinima5_radius_set_properties A₂ A₃ L hL Q hQ c₀ c₂ c₃
      (ambientPrefixIndex5 P hP k)).2.1
    refine ⟨fun i => (a i : Fin 5 → ℝ),ha.map' S.subtype S.ker_subtype,haL,har⟩

theorem restrictedRadii5_nonempty (P : Submodule ℚ (Fin 5 → ℚ)) (hP : Module.finrank ℚ P ≤ 4)
    (A₂ A₃ : Matrix (Fin 5) (Fin 5) ℚ) (L : Matrix (Fin 5) (Fin 5) ℝ) (hL : IsUnit L)
    (Q : ℝ) (hQ : 1 < Q) (c₀ c₂ c₃ : Fin 5 → ℝ) (k : Fin (Module.finrank ℚ P)) :
    (restrictedRadii5 P A₂ A₃ L hL Q hQ c₀ c₂ c₃ k).Nonempty := by
  let b := restrictedRealBasis5 P hP A₂ A₃ Q c₂ c₃
  let emb : Fin (k.val+1) → Fin (Module.finrank ℚ P) := Fin.castLE (by omega)
  refine ⟨∑ j, restrictedGauge5 P L hL Q hQ c₀ (b j),fun i => b (emb i),
    b.linearIndependent.comp emb (Fin.castLE_injective _),?_,?_⟩
  · intro i
    rw [← restrictedRealBasis5_span P hP A₂ A₃ Q c₂ c₃]
    exact Submodule.subset_span ⟨emb i,rfl⟩
  · intro i
    have hnonneg (j) : 0 ≤ restrictedGauge5 P L hL Q hQ c₀ (b j) := norm_nonneg _
    exact Finset.single_le_sum (fun j _ => hnonneg j) (Finset.mem_univ (emb i))

theorem restrictedMinima5_ge_ambient (P : Submodule ℚ (Fin 5 → ℚ))

theorem restrictedMinima5_pos (P : Submodule ℚ (Fin 5 → ℚ))

theorem restrictedGauge5_ball_compact (P : Submodule ℚ (Fin 5 → ℚ))
    (L : Matrix (Fin 5) (Fin 5) ℝ) (hL : IsUnit L)

  let e := originalCoordinates5 L hL Q hQ c₀
  have heq : {x | restrictedGauge5 P L hL Q hQ c₀ x ≤ r} =
      S.subtype ⁻¹' (e ⁻¹' Metric.closedBall (0 : Fin 5 → ℝ) r) := by
    ext x
    simp [restrictedGauge5,coordinateGauge5,Metric.mem_closedBall,dist_zero_right,e]
  rw [heq]
  exact S.closed_of_finiteDimensional.isClosedEmbedding_subtypeVal.isProperMap.isCompact_preimage
    (e.toContinuousLinearEquiv.toHomeomorph.isCompact_preimage.mpr (isCompact_closedBall _ _))

theorem restrictedGauge5_ball_finite (P : Submodule ℚ (Fin 5 → ℚ))

theorem exists_restricted_shortest_outside5 (P : Submodule ℚ (Fin 5 → ℚ))
    (hP : Module.finrank ℚ P ≤ 4)
    (A₂ A₃ : Matrix (Fin 5) (Fin 5) ℚ) (L : Matrix (Fin 5) (Fin 5) ℝ) (hL : IsUnit L)
    (Q : ℝ) (hQ : 1 < Q) (c₀ c₂ c₃ : Fin 5 → ℝ)
    (W : Submodule ℝ (rationalRealSpan5 P)) (hW : W ≠ ⊤) :
    ∃ x ∈ restrictedLattice5 P A₂ A₃ Q c₂ c₃, x ∉ W ∧
      ∀ y ∈ restrictedLattice5 P A₂ A₃ Q c₂ c₃, y ∉ W →
        restrictedGauge5 P L hL Q hQ c₀ x ≤ restrictedGauge5 P L hL Q hQ c₀ y := by
  classical
  let b := restrictedRealBasis5 P hP A₂ A₃ Q c₂ c₃
  have hex : ∃ i, b i ∉ W := by
    by_contra! h
    apply hW
    apply top_unique
    rw [← b.span_eq]
    exact Submodule.span_le.mpr (by rintro _ ⟨i,rfl⟩; exact h i)
  obtain ⟨i,hi⟩ := hex
  let f := restrictedGauge5 P L hL Q hQ c₀
  let G := restrictedLattice5 P A₂ A₃ Q c₂ c₃
  let T := {x | f x ≤ f (b i)} ∩ (G : Set (rationalRealSpan5 P)) ∩ {x | x ∉ W}
  have hfinite : T.Finite := (restrictedGauge5_ball_finite P hP A₂ A₃ L hL Q hQ c₀ c₂ c₃ _).inter_of_left _
  have hbi : b i ∈ G := by
    change b i ∈ restrictedLattice5 P A₂ A₃ Q c₂ c₃
    rw [← restrictedRealBasis5_span P hP A₂ A₃ Q c₂ c₃]
    exact Submodule.subset_span ⟨i,rfl⟩
  obtain ⟨x,hx,hmin⟩ := Set.exists_min_image T f hfinite ⟨b i,⟨⟨show f (b i) ≤ f (b i) from le_rfl,hbi⟩,hi⟩⟩
  refine ⟨x,hx.1.2,hx.2,?_⟩
  intro y hy hyW
  by_cases hle : f y ≤ f (b i)
  · exact hmin y ⟨⟨hle,hy⟩,hyW⟩
  · exact hx.1.1.trans (le_of_not_ge hle)

def intrinsicPrefix5 {P : Submodule ℚ (Fin 5 → ℚ)} {n : ℕ}
    (v : Fin n → rationalRealSpan5 P) (i : Fin n) : Submodule ℝ (rationalRealSpan5 P) :=

theorem intrinsicPrefix5_snoc_castSucc {P : Submodule ℚ (Fin 5 → ℚ)} {n : ℕ}
    (v : Fin n → rationalRealSpan5 P) (x : rationalRealSpan5 P) (i : Fin n) :
    intrinsicPrefix5 (Fin.snoc v x) i.castSucc = intrinsicPrefix5 v i := by

  apply congrArg (fun u : Fin i.val → rationalRealSpan5 P => Submodule.span ℝ (Set.range u))
  funext j
  have he : (Fin.castLE (by omega) j : Fin (n+1)) =
      (Fin.castLE (Nat.le_of_lt i.isLt) j).castSucc := by apply Fin.ext; rfl
  rw [he]
  simp only [Fin.snoc_castSucc]

theorem intrinsicPrefix5_snoc_last {P : Submodule ℚ (Fin 5 → ℚ)} {n : ℕ}

theorem exists_restricted_greedy_prefix5 (P : Submodule ℚ (Fin 5 → ℚ))
    (hP : Module.finrank ℚ P ≤ 4)
    (A₂ A₃ : Matrix (Fin 5) (Fin 5) ℚ) (L : Matrix (Fin 5) (Fin 5) ℝ) (hL : IsUnit L)
    (Q : ℝ) (hQ : 1 < Q) (c₀ c₂ c₃ : Fin 5 → ℝ) :
    ∀ n ≤ Module.finrank ℚ P, ∃ v : Fin n → rationalRealSpan5 P,
      LinearIndependent ℝ v ∧ (∀ i, v i ∈ restrictedLattice5 P A₂ A₃ Q c₂ c₃) ∧
      ∀ i, ∀ y ∈ restrictedLattice5 P A₂ A₃ Q c₂ c₃, y ∉ intrinsicPrefix5 v i →
        restrictedGauge5 P L hL Q hQ c₀ (v i) ≤ restrictedGauge5 P L hL Q hQ c₀ y := by
  intro n
  induction n with
  | zero =>
    intro _
    exact ⟨Fin.elim0,linearIndependent_empty_type,(fun i => Fin.elim0 i),fun i => Fin.elim0 i⟩
  | succ n ih =>
    intro hn
    obtain ⟨v,hv,hvL,hmin⟩ := ih (by omega)
    let W := Submodule.span ℝ (Set.range v)
    have hW : W ≠ ⊤ := by
      intro h
      have hd := finrank_span_eq_card hv
      simp only [Fintype.card_fin] at hd
      change Module.finrank ℝ W = n at hd
      rw [h,finrank_top,rationalRealSpan5_finrank P hP] at hd
      omega
    obtain ⟨x,hxL,hxW,hxmin⟩ := exists_restricted_shortest_outside5 P hP A₂ A₃ L hL Q hQ c₀ c₂ c₃ W hW
    refine ⟨Fin.snoc v x,linearIndependent_fin_snoc.mpr ⟨hv,hxW⟩,?_,?_⟩
    · intro i
      refine Fin.lastCases ?_ (fun j => ?_) i
      · simpa only [Fin.snoc_last] using hxL
      · simpa only [Fin.snoc_castSucc] using hvL j
    · intro i
      refine Fin.lastCases ?_ (fun j => ?_) i
      · rw [intrinsicPrefix5_snoc_last]
        simpa only [Fin.snoc_last] using hxmin
      · rw [intrinsicPrefix5_snoc_castSucc]
        simpa only [Fin.snoc_castSucc] using hmin j



theorem intrinsicPrefix5_not_mem {P : Submodule ℚ (Fin 5 → ℚ)} {n : ℕ}
    (v : Fin n → rationalRealSpan5 P) (hv : LinearIndependent ℝ v) (i j : Fin n) (hij : i ≤ j) :
    v j ∉ intrinsicPrefix5 v i := by
  have hle : intrinsicPrefix5 v i ≤ Submodule.span ℝ (v '' {t | t < i}) := by
    apply Submodule.span_le.mpr
    rintro _ ⟨a,rfl⟩
    exact Submodule.subset_span ⟨Fin.castLE (by omega) a,by change a.val < i.val; exact a.isLt,rfl⟩
  exact fun h => hv.not_mem_span_image (s := {t | t < i}) (x := j) (not_lt_of_ge hij) (hle h)

theorem intrinsicPrefix5_finrank {P : Submodule ℚ (Fin 5 → ℚ)} {n : ℕ}
    (v : Fin n → rationalRealSpan5 P) (hv : LinearIndependent ℝ v) (i : Fin n) :
    Module.finrank ℝ (intrinsicPrefix5 v i) = i.val := by
  exact (finrank_span_eq_card (hv.comp (Fin.castLE (by omega) : Fin i.val → Fin n)
    (Fin.castLE_injective _))).trans (Fintype.card_fin i.val)

theorem intrinsic_family_escapes5 {P : Submodule ℚ (Fin 5 → ℚ)} {n : ℕ}
    (a : Fin (n+1) → rationalRealSpan5 P) (ha : LinearIndependent ℝ a)
    (W : Submodule ℝ (rationalRealSpan5 P)) (hW : Module.finrank ℝ W = n) :
    ∃ i, a i ∉ W := by
  by_contra! h
  have hle : Submodule.span ℝ (Set.range a) ≤ W :=
    Submodule.span_le.mpr (by rintro _ ⟨i,rfl⟩; exact h i)
  have hd := Submodule.finrank_mono hle
  rw [finrank_span_eq_card ha,Fintype.card_fin,hW] at hd
  omega

theorem exists_restrictedMinima5_attaining (P : Submodule ℚ (Fin 5 → ℚ))
    (hP : Module.finrank ℚ P ≤ 4)
    (A₂ A₃ : Matrix (Fin 5) (Fin 5) ℚ) (L : Matrix (Fin 5) (Fin 5) ℝ) (hL : IsUnit L)
    (Q : ℝ) (hQ : 1 < Q) (c₀ c₂ c₃ : Fin 5 → ℝ) :
    ∃ v : Fin (Module.finrank ℚ P) → rationalRealSpan5 P,
      LinearIndependent ℝ v ∧ (∀ i, v i ∈ restrictedLattice5 P A₂ A₃ Q c₂ c₃) ∧
      (∀ i, 0 < restrictedMinima5 P A₂ A₃ L hL Q hQ c₀ c₂ c₃ i) ∧
      Monotone (restrictedMinima5 P A₂ A₃ L hL Q hQ c₀ c₂ c₃) ∧
      (∀ i, restrictedGauge5 P L hL Q hQ c₀ (v i) =
        restrictedMinima5 P A₂ A₃ L hL Q hQ c₀ c₂ c₃ i) ∧
      ∀ i, ∀ y ∈ restrictedLattice5 P A₂ A₃ Q c₂ c₃, y ∉ intrinsicPrefix5 v i →
        restrictedGauge5 P L hL Q hQ c₀ (v i) ≤ restrictedGauge5 P L hL Q hQ c₀ y := by
  obtain ⟨v,hv,hvL,hmin⟩ := exists_restricted_greedy_prefix5 P hP A₂ A₃ L hL Q hQ c₀ c₂ c₃
    (Module.finrank ℚ P) le_rfl
  let f := restrictedGauge5 P L hL Q hQ c₀
  have hmono : Monotone (fun i => f (v i)) := by
    intro i j hij
    exact hmin i (v j) (hvL j) (intrinsicPrefix5_not_mem v hv i j hij)
  have heq (k) : restrictedMinima5 P A₂ A₃ L hL Q hQ c₀ c₂ c₃ k = f (v k) := by
    apply le_antisymm
    · apply csInf_le (restrictedRadii5_bddBelow P A₂ A₃ L hL Q hQ c₀ c₂ c₃ k)
      let emb : Fin (k.val+1) → Fin (Module.finrank ℚ P) := Fin.castLE (by omega)
      refine ⟨fun i => v (emb i),hv.comp emb (Fin.castLE_injective _),fun i => hvL (emb i),?_⟩
      intro i
      exact hmono (show emb i ≤ k by change i.val ≤ k.val; omega)
    · apply le_csInf (restrictedRadii5_nonempty P hP A₂ A₃ L hL Q hQ c₀ c₂ c₃ k)
      rintro r ⟨a,ha,haL,har⟩
      obtain ⟨i,hi⟩ := intrinsic_family_escapes5 a ha (intrinsicPrefix5 v k) (intrinsicPrefix5_finrank v hv k)
      exact (hmin k (a i) (haL i) hi).trans (har i)
  refine ⟨v,hv,hvL,restrictedMinima5_pos P hP A₂ A₃ L hL Q hQ c₀ c₂ c₃,?_,?_,hmin⟩
  · intro i j hij
    rw [heq i,heq j]
    exact hmono hij
  · intro i
    exact (heq i).symm

def dyadicBasis (d : Fin 5) (a : Fin d.val → ℕ) : Basis (Fin d.val) ℝ (Fin d.val → ℝ) :=
  (Pi.basisFun ℝ (Fin d.val)).isUnitSMul (w := fun i => (2 : ℝ)^a i)
    (fun i => isUnit_iff_ne_zero.mpr (pow_ne_zero _ (by norm_num : (2 : ℝ) ≠ 0)))

theorem dyadic_lattice_coordinates (d : Fin 5) (a : Fin d.val → ℕ)
    (x : Fin d.val → ℝ) (hx : x ∈ Submodule.span ℤ (Set.range (dyadicBasis d a))) :
    ∃ z : Fin d.val → ℤ, ∀ i, x i = (2 : ℝ) ^ a i * (z i : ℝ) := by
  let b := dyadicBasis d a
  let z : Fin d.val → ℤ := fun i => (b.restrictScalars ℤ).repr ⟨x,hx⟩ i
  refine ⟨z,?_⟩
  intro i
  have hr (j) : b.repr x j = (z j : ℝ) :=
    (b.restrictScalars_repr_apply ℤ ⟨x,hx⟩ j).symm
  have h := congrFun (b.sum_repr x) i
  simp only [hr] at h
  simpa [b,dyadicBasis,Basis.isUnitSMul_apply,Finset.sum_apply,Pi.smul_apply,
    smul_eq_mul,Pi.basisFun_apply,Pi.single_apply,mul_comm] using h.symm

theorem dyadic_lattice_avoids (d : Fin 5) (a : Fin d.val → ℕ) (ha : Antitone a)
    (f : (Fin d.val → ℝ) → ℝ)
    (hf : ∀ (r : ℝ) (x : Fin d.val → ℝ), f (r • x) = |r| * f x)
    (mu : Fin d.val → ℝ) (hmu : ∀ i, 1 < (2 : ℝ)^a i * mu i)
    (hlower : ∀ (z : Fin d.val → ℤ) (i : Fin d.val), z i ≠ 0 →
      (∀ j, i < j → z j = 0) → mu i ≤ f (fun j => (z j : ℝ)))
    (x : Fin d.val → ℝ)
    (hx : x ∈ Submodule.span ℤ (Set.range (dyadicBasis d a)))
    (hxf : f x ≤ 1) : x = 0 := by
  classical
  by_contra hne
  obtain ⟨z,hz⟩ := dyadic_lattice_coordinates d a x hx
  let support := Finset.univ.filter (fun i => z i ≠ 0)
  have hs : support.Nonempty := by
    by_contra h
    apply hne
    ext i
    have hzi : z i = 0 := by
      by_contra hzi
      exact h ⟨i,Finset.mem_filter.mpr ⟨Finset.mem_univ i,hzi⟩⟩
    simp [hz i,hzi]
  let k := support.max' hs
  have hk : z k ≠ 0 := (Finset.mem_filter.mp (Finset.max'_mem support hs)).2
  have htail : ∀ j, k < j → z j = 0 := by
    intro j hj
    by_contra h
    have hle : j ≤ k := Finset.le_max' support j (Finset.mem_filter.mpr ⟨Finset.mem_univ j,h⟩)
    exact (not_le_of_gt hj) hle
  let w : Fin d.val → ℤ := fun i => if i ≤ k then (2 : ℤ)^(a i-a k)*z i else 0
  have hwk : w k ≠ 0 := by simpa [w] using hk
  have hw : (fun i => (w i : ℝ)) = ((2 : ℝ)^a k)⁻¹ • x := by
    ext i
    by_cases hi : i ≤ k
    · have hai : a k ≤ a i := ha hi
      simp only [w,if_pos hi,Int.cast_mul,Int.cast_pow,Int.cast_ofNat,Pi.smul_apply,smul_eq_mul]
      rw [hz i,← Nat.sub_add_cancel hai, pow_add]
      field_simp
      ring
    · have hit : k < i := lt_of_not_ge hi
      simp [w,hi,hz i,htail i hit]
  have hlow := hlower w k hwk (fun j hj => by simp [w,not_le_of_gt hj])
  rw [hw,hf,abs_of_pos (inv_pos.mpr (pow_pos (by norm_num : (0 : ℝ) < 2) _))] at hlow
  have hp : (0 : ℝ) < 2^a k := pow_pos (by norm_num) _
  have hle : mu k ≤ ((2 : ℝ)^a k)⁻¹ :=
    hlow.trans (by simpa using mul_le_mul_of_nonneg_left hxf (inv_nonneg.mpr hp.le))
  have hbound : (2 : ℝ)^a k * mu k ≤ 1 := by
    calc
      _ ≤ (2 : ℝ)^a k * ((2 : ℝ)^a k)⁻¹ := mul_le_mul_of_nonneg_left hle hp.le
      _ = 1 := mul_inv_cancel₀ hp.ne'
  exact (not_le_of_gt (hmu k)) hbound


theorem dyadic_fundamental_volume (d : Fin 5) (a : Fin d.val → ℕ) :
    volume (ZSpan.fundamentalDomain (dyadicBasis d a)) =
      ENNReal.ofReal (∏ i, (2 : ℝ)^a i) := by
  have hm : Matrix.of (dyadicBasis d a) = Matrix.diagonal (fun i => (2 : ℝ)^a i) := by
    ext i j
    by_cases h : i = j
    · subst j
      simp [dyadicBasis,Basis.isUnitSMul_apply,Pi.basisFun_apply]
    · simp [dyadicBasis,Basis.isUnitSMul_apply,Pi.basisFun_apply,Pi.single_apply,h]
  rw [ZSpan.volume_fundamentalDomain,hm,Matrix.det_diagonal,abs_of_pos]
  exact Finset.prod_pos (fun _ _ => pow_pos (by norm_num) _)

theorem dyadic_volume_bound (d : Fin 5) (a : Fin d.val → ℕ) (ha : Antitone a)
    (f : (Fin d.val → ℝ) → ℝ)
    (hf : ∀ (r : ℝ) (x : Fin d.val → ℝ), f (r • x) = |r| * f x)
    (mu : Fin d.val → ℝ) (hmu : ∀ i, 1 < (2 : ℝ)^a i * mu i)
    (hlower : ∀ (z : Fin d.val → ℤ) (i : Fin d.val), z i ≠ 0 →
      (∀ j, i < j → z j = 0) → mu i ≤ f (fun j => (z j : ℝ)))
    (hsymm : ∀ x, f (-x) = f x) (hconv : Convex ℝ {x | f x ≤ 1}) :
    volume.real {x | f x ≤ 1} ≤ (∏ i, (2 : ℝ)^a i) * 2^d.val := by
  let b := dyadicBasis d a
  letI : Countable (Submodule.span ℤ (Set.range b)) :=
    inferInstance
  letI : Countable (Submodule.span ℤ (Set.range b)).toAddSubgroup :=
    inferInstanceAs (Countable (Submodule.span ℤ (Set.range b)))
  have hvol : volume {x | f x ≤ 1} ≤

theorem exists_dyadic_exponents (d : Fin 5) (mu : Fin d.val → ℝ)
    (hpos : ∀ i, 0 < mu i) (hone : ∀ i, mu i ≤ 1) (hmono : Monotone mu) :
    ∃ a : Fin d.val → ℕ, Antitone a ∧ ∀ i,
      1 < (2 : ℝ)^a i * mu i ∧ (2 : ℝ)^a i * mu i ≤ 2 := by
  classical
  have hex (i : Fin d.val) : ∃ n : ℕ, 1 < (2 : ℝ)^n * mu i := by
    obtain ⟨n,hn⟩ := pow_unbounded_of_one_lt (mu i)⁻¹ (by norm_num : (1 : ℝ) < 2)
    refine ⟨n,?_⟩
    exact (div_lt_iff₀ (hpos i)).mp (by simpa only [one_div] using hn)
  let a := fun i => Nat.find (hex i)
  refine ⟨a,?_,?_⟩
  · intro i j hij
    apply Nat.find_min' (hex j)
    exact (Nat.find_spec (hex i)).trans_le
      (mul_le_mul_of_nonneg_left (hmono hij) (pow_nonneg (by norm_num) _))
  · intro i
    refine ⟨Nat.find_spec (hex i),?_⟩
    have ha0 : a i ≠ 0 := by
      intro h
      have hg := Nat.find_spec (hex i)
      change 1 < (2 : ℝ)^a i * mu i at hg
      rw [h,pow_zero,one_mul] at hg
      exact (not_lt_of_ge (hone i)) hg
    have hpred : (2 : ℝ)^(a i-1) * mu i ≤ 1 := by
      apply le_of_not_gt
      exact Nat.find_min (hex i) (Nat.sub_lt (Nat.pos_of_ne_zero ha0) (by decide))
    have heq : a i = (a i-1)+1 := by omega
    rw [heq,pow_succ]
    nlinarith

theorem normalized_product_volume_upper (d : Fin 5)
    (f : (Fin d.val → ℝ) → ℝ)
    (hf : ∀ (r : ℝ) (x : Fin d.val → ℝ), f (r • x) = |r| * f x)
    (mu : Fin d.val → ℝ) (hpos : ∀ i, 0 < mu i)
    (hone : ∀ i, mu i ≤ 1) (hmono : Monotone mu)
    (hlower : ∀ (z : Fin d.val → ℤ) (i : Fin d.val), z i ≠ 0 →
      (∀ j, i < j → z j = 0) → mu i ≤ f (fun j => (z j : ℝ)))
    (hsymm : ∀ x, f (-x) = f x) (hconv : Convex ℝ {x | f x ≤ 1}) :
    (∏ i, mu i) * volume.real {x | f x ≤ 1} ≤ 4^d.val := by
  obtain ⟨a,ha,hscale⟩ := exists_dyadic_exponents d mu hpos hone hmono
  have hvol := dyadic_volume_bound d a ha f hf mu (fun i => (hscale i).1) hlower hsymm hconv
  have hp : 0 ≤ ∏ i, mu i := Finset.prod_nonneg (fun i _ => (hpos i).le)
  have hprod : (∏ i, (2 : ℝ)^a i * mu i) ≤ 2^d.val := by
    calc
      _ ≤ ∏ _ : Fin d.val, (2 : ℝ) :=
        Finset.prod_le_prod (fun i _ => mul_nonneg (pow_nonneg (by norm_num) _) (hpos i).le)
          (fun i _ => (hscale i).2)
      _ = _ := by simp
  calc
    _ ≤ (∏ i, mu i) * ((∏ i, (2 : ℝ)^a i) * 2^d.val) := mul_le_mul_of_nonneg_left hvol hp
    _ = (∏ i, (2 : ℝ)^a i * mu i) * 2^d.val := by rw [Finset.prod_mul_distrib]; ring
    _ ≤ 2^d.val * 2^d.val := mul_le_mul_of_nonneg_right hprod (pow_nonneg (by norm_num) _)
    _ = 4^d.val := by rw [← mul_pow]; norm_num


theorem homogeneous_sublevel_dilate (d : Fin 5) (f : (Fin d.val → ℝ) → ℝ)
    (hf : ∀ (r : ℝ) (x : Fin d.val → ℝ), f (r • x) = |r| * f x)
    (t : ℝ) (ht : 0 < t) : {x | f x / t ≤ 1} = t • {x | f x ≤ 1} := by
  ext x
  constructor
  · intro hx
    refine ⟨t⁻¹ • x,?_,smul_inv_smul₀ ht.ne' x⟩
    change f (t⁻¹ • x) ≤ 1
    rw [hf,abs_of_pos (inv_pos.mpr ht)]
    simpa only [div_eq_mul_inv,mul_comm] using hx
  · rintro ⟨y,hy,rfl⟩
    change f (t • y) / t ≤ 1
    rw [hf,abs_of_pos ht,mul_div_cancel_left₀ _ ht.ne']
    exact hy

theorem adapted_product_volume_upper (d : Fin 5)
    (f : (Fin d.val → ℝ) → ℝ)
    (hf : ∀ (r : ℝ) (x : Fin d.val → ℝ), f (r • x) = |r| * f x)
    (mu : Fin d.val → ℝ) (hpos : ∀ i, 0 < mu i) (hmono : Monotone mu)
    (hlower : ∀ (z : Fin d.val → ℤ) (i : Fin d.val), z i ≠ 0 →
      (∀ j, i < j → z j = 0) → mu i ≤ f (fun j => (z j : ℝ)))
    (hsymm : ∀ x, f (-x) = f x) (hconv : Convex ℝ {x | f x ≤ 1}) :
    (∏ i, mu i) * volume.real {x | f x ≤ 1} ≤ 4^d.val := by
  let t := 1 + ∑ i, mu i
  have ht : 0 < t := by
    dsimp [t]
    exact add_pos_of_pos_of_nonneg zero_lt_one (Finset.sum_nonneg (fun i _ => (hpos i).le))
  have hle (i : Fin d.val) : mu i ≤ t := by
    have h := Finset.single_le_sum (fun j _ => (hpos j).le) (Finset.mem_univ i)
    dsimp [t]
    linarith
  have heq := homogeneous_sublevel_dilate d f hf t ht
  have hconv' : Convex ℝ {x | f x / t ≤ 1} := by rw [heq]; exact hconv.smul t
  have h := normalized_product_volume_upper d (fun x => f x / t)
    (fun r x => by dsimp only; rw [hf]; ring) (fun i => mu i/t)
    (fun i => div_pos (hpos i) ht) (fun i => (div_le_one ht).mpr (hle i))
    (fun i j hij => div_le_div_of_nonneg_right (hmono hij) ht.le)
    (fun z i hz htail => div_le_div_of_nonneg_right (hlower z i hz htail) ht.le)
    (fun x => by dsimp only; rw [hsymm]) hconv'
  have hvol : volume.real {x | f x/t ≤ 1} = t^d.val * volume.real {x | f x ≤ 1} := by
    rw [heq,measureReal_def,Measure.addHaar_smul,Module.finrank_pi,Fintype.card_fin,
      abs_of_pos (pow_pos ht _),ENNReal.toReal_mul,ENNReal.toReal_ofReal (pow_nonneg ht.le _)]
    rfl
  rw [hvol,Finset.prod_div_distrib] at h
  simp only [Finset.prod_const,Finset.card_univ,Fintype.card_fin] at h
  have hcancel : (∏ i, mu i) / t^d.val * (t^d.val * volume.real {x | f x ≤ 1}) =
      (∏ i, mu i) * volume.real {x | f x ≤ 1} := by
    field_simp
    ring
  rwa [hcancel] at h
