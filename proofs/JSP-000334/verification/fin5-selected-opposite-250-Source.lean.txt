def rawOriginalBody250 (L : Matrix (Fin 5) (Fin 5) ℝ) (Q : ℝ) (c₀ : Fin 5 → ℝ) : Set (Fin 5 → ℝ) :=
  {x | ∀ s, |(L *ᵥ x) s| ≤ Q ^ c₀ s}

def rawOriginalGroup250 (A₂ A₃ : Matrix (Fin 5) (Fin 5) ℚ) (Q : ℝ) (c₂ c₃ : Fin 5 → ℝ) : AddSubgroup (Fin 5 → ℤ) :=
  (⨅ i, (normBall 2 (Nat.ceil (JSP000334.qThreshold 2 Q (c₂ i)))).comap (row5 A₂ i)) ⊓
  (⨅ i, (normBall 3 (Nat.ceil (JSP000334.qThreshold 3 Q (c₃ i)))).comap (row5 A₃ i))
def rawExponent250 (P : Submodule ℚ (Fin 5 → ℚ)) (hP : Module.finrank ℚ P ≤ 4)
    (A₂ A₃ : Matrix (Fin 5) (Fin 5) ℚ) (L : Matrix (Fin 5) (Fin 5) ℝ)
    (h₂ : A₂.det = 1) (h₃ : A₃.det = 1) (hL : L.det = 1) (c₀ c₂ c₃ : Fin 5 → ℝ) : ℝ :=
    (∑ k : Fin (Module.finrank ℚ P), c₀ (weightOrder5 c₀
      (selectedIndex5 (realRestrictionCoordinates5 P hP L) c₀ (realRestrictionCoordinates5_span P hP L hL) k))) +
    (∑ k : Fin (Module.finrank ℚ P), c₂ (weightOrder5 c₂
      (selectedIndex5 (rationalRowCoordinates5 P A₂) c₂ (rationalRowCoordinates5_span P A₂ h₂) k))) +
    (∑ k : Fin (Module.finrank ℚ P), c₃ (weightOrder5 c₃
      (selectedIndex5 (rationalRowCoordinates5 P A₃) c₃ (rationalRowCoordinates5_span P A₃ h₃) k)))

def rawBody250 {d : ℕ} (P : Submodule ℚ (Fin 5 → ℚ))
    (L : Matrix (Fin 5) (Fin 5) ℝ) (Q : ℝ) (c : Fin 5 → ℝ) (s : Fin d → Fin 5) :
    Set (rationalRealSpan5 P) :=
  {x | ∀ k, |(L *ᵥ (x : Fin 5 → ℝ)) (s k)| ≤ Q^(c (s k))}

def rawLocal250 {d : ℕ} (p : ℕ) [Fact p.Prime]
    (A : Matrix (Fin 5) (Fin 5) ℚ) (Q : ℝ) (c : Fin 5 → ℝ) (s : Fin d → Fin 5) :
    AddSubgroup (Fin 5 → ℤ) :=
  ⨅ k, (normBall p (Nat.ceil (JSP000334.qThreshold p Q (c (s k))))).comap (row5 A (s k))

def rawGroup250 {d : ℕ} (A₂ A₃ : Matrix (Fin 5) (Fin 5) ℚ)
    (Q : ℝ) (c₂ c₃ : Fin 5 → ℝ) (s₂ s₃ : Fin d → Fin 5) : AddSubgroup (Fin 5 → ℤ) :=
  rawLocal250 2 A₂ Q c₂ s₂ ⊓ rawLocal250 3 A₃ Q c₃ s₃

def rawLattice250 {d : ℕ} (P : Submodule ℚ (Fin 5 → ℚ))

    (s₂ s₃ : Fin d → Fin 5) : Submodule ℤ (rationalRealSpan5 P) :=
  (integerRealImage (rawGroup250 A₂ A₃ Q c₂ c₃ s₂ s₃)).comap

def rawOppositePoint250 (P : Submodule ℚ (Fin 5 → ℚ)) (hP : Module.finrank ℚ P ≤ 4)
    (A₂ A₃ : Matrix (Fin 5) (Fin 5) ℚ) (L : Matrix (Fin 5) (Fin 5) ℝ)
    (h₂ : A₂.det = 1) (h₃ : A₃.det = 1) (hL : L.det = 1)
    (c₀ c₂ c₃ : Fin 5 → ℝ) (Q K : ℝ) : Prop :=
  let s₀ := realSelection5 P hP L hL c₀
  let s₂ := finiteSelection5 P A₂ h₂ c₂
  let s₃ := finiteSelection5 P A₃ h₃ c₃
  let G := rawLattice250 P A₂ A₃ Q c₂ c₃ s₂ s₃
  let C := rawBody250 P L Q c₀ s₀
  rawRestrictedBody250 P L Q c₀ ⊆ C ∧
  rawRestrictedLattice250 P A₂ A₃ Q c₂ c₃ ≤ G ∧
  IsCompact C ∧ Convex ℝ C ∧ (0 : rationalRealSpan5 P) ∈ interior C ∧
  0 < (restrictedMeasure5 P).real C ∧ (restrictedMeasure5 P) C < ⊤ ∧
  Module.finrank ℤ G = Module.finrank ℚ P ∧
  0 < ZLattice.covolume G (restrictedMeasure5 P) ∧
  ZLattice.covolume G (restrictedMeasure5 P) ≤
    ZLattice.covolume (rawRestrictedLattice250 P A₂ A₃ Q c₂ c₃) (restrictedMeasure5 P) ∧
  (∀ z : Fin 5 → ℤ, z ∈ rawGroup250 A₂ A₃ Q c₂ c₃ s₂ s₃ ↔
    (∀ k, (padicNorm 2 ((A₂ *ᵥ (fun i => (z i : ℚ))) (s₂ k)) : ℝ) ≤ Q^(c₂ (s₂ k))) ∧
    (∀ k, (padicNorm 3 ((A₃ *ᵥ (fun i => (z i : ℚ))) (s₃ k)) : ℝ) ≤ Q^(c₃ (s₃ k)))) ∧
  (restrictedMeasure5 P).real (rawRestrictedBody250 P L Q c₀) /
    ZLattice.covolume (rawRestrictedLattice250 P A₂ A₃ Q c₂ c₃) (restrictedMeasure5 P) ≤
      (restrictedMeasure5 P).real C / ZLattice.covolume G (restrictedMeasure5 P) ∧
  (restrictedMeasure5 P).real C / ZLattice.covolume G (restrictedMeasure5 P) ≤
    K * Q^(rawExponent250 P hP A₂ A₃ L h₂ h₃ hL c₀ c₂ c₃) ∧
  (restrictedMeasure5 P).real (rawRestrictedBody250 P L Q c₀) /
    ZLattice.covolume (rawRestrictedLattice250 P A₂ A₃ Q c₂ c₃) (restrictedMeasure5 P) ≤
      K * Q^(rawExponent250 P hP A₂ A₃ L h₂ h₃ hL c₀ c₂ c₃)
