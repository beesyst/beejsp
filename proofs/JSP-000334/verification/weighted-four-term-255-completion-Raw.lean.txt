def Two : ℚˣ := Units.mk0 2 (by norm_num)
def Three : ℚˣ := Units.mk0 3 (by norm_num)
def Group : Subgroup ℚˣ := Subgroup.closure ({-1,Two,Three} : Set ℚˣ)
def Nondeg {n : ℕ} (x : Fin n → ℚ) : Prop := ∀ I : Finset (Fin n),I.Nonempty → I≠Finset.univ → ∑ i∈I,x i≠0
def Weighted (A B C D : ℚ) : Set (Group × Group × Group) :=
  {p | A+B*((p.1 : ℚˣ) : ℚ)+C*((p.2.1 : ℚˣ) : ℚ)+D*((p.2.2 : ℚˣ) : ℚ)=0 ∧
    Nondeg ![A,B*((p.1 : ℚˣ) : ℚ),C*((p.2.1 : ℚˣ) : ℚ),D*((p.2.2 : ℚˣ) : ℚ)]}
def Primitive (z : Fin 3 → ℤ) : Prop := ∃ c : Fin 3 → ℤ,∑ i,c i*z i=1
def Height (z : Fin 3 → ℤ) : ℚ := max |(z 0 : ℚ)| (max |(z 1 : ℚ)| |(z 2 : ℚ)|)
def Product (q : ℚ) : ℚ := |q| *padicNorm 2 q*padicNorm 3 q
def FixedProducts (k : Fin 3 → ℚ) (a : ℚ) (y : IntegerTriple) : Prop :=
  Primitive y ∧ (∀ i,Product (y i : ℚ)=Product (k i)) ∧ Product ((y 0 : ℚ)+y 1+y 2)=Product a
def IntegerDiagonal (t : Fin 3 → ℤ) : Matrix (Fin 3) (Fin 3) ℚ := Matrix.diagonal (fun i => (t i : ℚ))
def Diagonal (t : Fin 3 → ℤ) : (Fin 3 → ℚ) →ₗ[ℚ] (Fin 3 → ℚ) := Matrix.toLin' (IntegerDiagonal t)
def K (t : Fin 3 → ℤ) (L A : ℚ) (g : ℕ) : ℚ :=
  Product ((t 0 : ℚ)/g)*Product ((t 1 : ℚ)/g)*Product ((t 2 : ℚ)/g)*Product (L*A/g)
def Pairs (A B C : ℚ) : Set (Group × Group) :=
  {p | A+B*((p.1 : ℚˣ) : ℚ)+C*((p.2 : ℚˣ) : ℚ)=0}
def Raw_weightedFourTermValues3 (p : Group × Group × Group) : Fin 3 → ℚ :=
  ![((p.1 : ℚˣ) : ℚ),((p.2.1 : ℚˣ) : ℚ),((p.2.2 : ℚˣ) : ℚ)]

def Raw_weightedFourTermCoefficientFiber3 (A B C D b0 b1 b2 : ℚ) :
    Set (Group × Group × Group) :=
  {p | p∈Weighted A B C D ∧
    b0*((p.1 : ℚˣ) : ℚ)+b1*((p.2.1 : ℚˣ) : ℚ)+b2*((p.2.2 : ℚˣ) : ℚ)=0}

def Raw_weightedFourTermProperFiber3 (A B C D : ℚ) (P : Submodule ℚ (Fin 3 → ℚ)) :
    Set (Group × Group × Group) :=
  {p | p∈Weighted A B C D ∧ Raw_weightedFourTermValues3 p∈P}
